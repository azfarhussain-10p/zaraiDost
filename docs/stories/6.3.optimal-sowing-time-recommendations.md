**Sowing Window Calculator**

```javascript
// SowingWindowCalculator.js
class SowingWindowCalculator {
  /**
   * Calculate optimal sowing window for crop in region
   */
  async calculateSowingWindow(cropType, region, currentDate = new Date()) {
    // Get base sowing window from crop database
    const baseWindow = CROP_SOWING_WINDOWS[cropType];

    if (!baseWindow) {
      throw new Error(`No sowing window defined for crop: ${cropType}`);
    }

    // Get region-specific adjustments
    const regionalWindow = baseWindow.regions?.[region.toLowerCase()] || {
      start: baseWindow.optimal_start,
      end: baseWindow.optimal_end
    };

    // Calculate actual dates for current year
    const currentYear = currentDate.getFullYear();
    const startDate = this.parseDateString(regionalWindow.start, currentYear);
    const endDate = this.parseDateString(regionalWindow.end, currentYear);
    const extendedEnd = baseWindow.extended_end ?
      this.parseDateString(baseWindow.extended_end, currentYear) : endDate;

    // Check for monsoon-dependent crops
    if (baseWindow.monsoon_dependent) {
      const monsoonOnset = await this.getMonsoonOnsetDate(region);
      // Adjust rice sowing based on monsoon onset
      startDate.setDate(startDate.getDate() + this.calculateMonsoonAdjustment(monsoonOnset));
    }

    return {
      crop_type: cropType,
      region,
      optimal_start_date: startDate,
      optimal_end_date: endDate,
      extended_end_date: extendedEnd,
      soil_temp_min_c: baseWindow.soil_temp_min_c,
      soil_temp_max_c: baseWindow.soil_temp_max_c,
      season: baseWindow.season,
      monsoon_dependent: baseWindow.monsoon_dependent || false
    };
  }

  parseDateString(dateStr, year) {
    // Parse "MM-DD" format
    const [month, day] = dateStr.split('-').map(Number);
    return new Date(year, month - 1, day);
  }

  async getMonsoonOnsetDate(region) {
    // Fetch monsoon onset from Story 6.1
    const monsoonService = new MonsoonForecastService();
    const forecast = await monsoonService.fetchMonsoonForecast();

    return new Date(forecast.onset_date_predicted);
  }

  calculateMonsoonAdjustment(monsoonOnset) {
    // If monsoon early, can sow rice earlier
    // If monsoon late, delay rice sowing
    const normalOnset = new Date('2025-07-01');
    const diffDays = Math.floor((monsoonOnset - normalOnset) / (1000 * 60 * 60 * 24));

    return diffDays;  // Positive = delay, Negative = advance
  }
}

// SowingConditionEvaluator.js
class SowingConditionEvaluator {
  /**
   * Evaluate if current conditions are suitable for sowing
   */
  async evaluateSowingConditions(cropType, fieldId, sowingWindow) {
    const checks = [];

    // 1. Check soil temperature
    const soilTempCheck = await this.checkSoilTemperature(fieldId, sowingWindow);
    checks.push(soilTempCheck);

    // 2. Check soil moisture
    const soilMoistureCheck = await this.checkSoilMoisture(fieldId);
    checks.push(soilMoistureCheck);

    // 3. Check weather forecast (next 7 days)
    const weatherCheck = await this.checkWeatherForecast(fieldId);
    checks.push(weatherCheck);

    // 4. Check drought risk
    const droughtCheck = await this.checkDroughtRisk(fieldId);
    checks.push(droughtCheck);

    // 5. Check flood risk (for monsoon crops)
    if (sowingWindow.monsoon_dependent) {
      const floodCheck = await this.checkFloodRisk(fieldId);
      checks.push(floodCheck);
    }

    // Calculate overall suitability
    const allPassed = checks.every(check => check.status === 'pass');
    const anyFailed = checks.some(check => check.status === 'fail');

    return {
      is_suitable: allPassed,
      recommendation: anyFailed ? 'delay' : (allPassed ? 'proceed' : 'monitor'),
      checks,
      overall_score: this.calculateSuitabilityScore(checks),
      message: this.generateRecommendationMessage(checks)
    };
  }

  async checkSoilTemperature(fieldId, sowingWindow) {
    // Get current soil temperature (from weather data or sensors)
    const weatherService = new WeatherDataService();
    const weather = await weatherService.getCurrentWeather(fieldId);

    // Estimate soil temp from air temp (simplified)
    const soilTemp = weather.temperature_c * 0.8;  // Soil temp ~80% of air temp

    const isInRange = soilTemp >= sowingWindow.soil_temp_min_c &&
                      soilTemp <= (sowingWindow.soil_temp_max_c || 40);

    return {
      parameter: 'soil_temperature',
      status: isInRange ? 'pass' : 'fail',
      current_value: soilTemp,
      required_min: sowingWindow.soil_temp_min_c,
      required_max: sowingWindow.soil_temp_max_c,
      message: isInRange ?
        `Soil temperature (${soilTemp}°C) is suitable` :
        `Soil temperature (${soilTemp}°C) is outside optimal range`
    };
  }

  async checkSoilMoisture(fieldId) {
    // Get soil moisture from Story 4.2
    const soilMoistureService = new SoilMoisturePredictionService();
    const moisture = await soilMoistureService.getCurrentMoisture(fieldId);

    const isAdequate = moisture.moisture_percent >= 60;  // Minimum 60% for sowing

    return {
      parameter: 'soil_moisture',
      status: isAdequate ? 'pass' : 'fail',
      current_value: moisture.moisture_percent,
      required_min: 60,
      message: isAdequate ?
        `Soil moisture (${moisture.moisture_percent}%) is adequate` :
        `Soil moisture (${moisture.moisture_percent}%) is too low. Irrigate before sowing.`
    };
  }

  async checkWeatherForecast(fieldId) {
    // Check 7-day forecast for excessive rain
    const weatherService = new WeatherDataService();
    const forecast = await weatherService.get7DayForecast(fieldId);

    const totalRainfall = forecast.reduce((sum, day) => sum + day.precipitation_amount_mm, 0);
    const heavyRainDays = forecast.filter(day => day.precipitation_amount_mm > 20).length;

    const isSafe = totalRainfall < 50 && heavyRainDays === 0;

    return {
      parameter: 'weather_forecast',
      status: isSafe ? 'pass' : 'warning',
      current_value: totalRainfall,
      message: isSafe ?
        `Weather forecast is favorable (${totalRainfall}mm rainfall expected)` :
        `Heavy rainfall forecasted (${totalRainfall}mm in next 7 days). Consider delaying.`
    };
  }

  async checkDroughtRisk(fieldId) {
    // Check drought risk from Story 6.6
    const droughtService = new DroughtRiskCalculator();
    const field = await db.query('SELECT region FROM fields WHERE id = $1', [fieldId]);
    const droughtRisk = await droughtService.assessDroughtRisk(field[0].region, '3-month');

    const isLowRisk = droughtRisk.drought_risk_level === 'low' ||
                      droughtRisk.drought_risk_level === 'moderate';

    return {
      parameter: 'drought_risk',
      status: isLowRisk ? 'pass' : 'warning',
      current_value: droughtRisk.drought_risk_level,
      message: isLowRisk ?
        `Drought risk is ${droughtRisk.drought_risk_level}` :
        `${droughtRisk.drought_risk_level} drought risk. Ensure irrigation availability.`
    };
  }

  async checkFloodRisk(fieldId) {
    // Check flood risk from Story 6.5
    const field = await db.query('SELECT region, location_gps FROM fields WHERE id = $1', [fieldId]);
    // Placeholder - would check flood risk from Story 6.5
    const floodRisk = 'low';

    return {
      parameter: 'flood_risk',
      status: floodRisk === 'low' ? 'pass' : 'warning',
      current_value: floodRisk,
      message: `Flood risk is ${floodRisk}`
    };
  }

  calculateSuitabilityScore(checks) {
    const passCount = checks.filter(c => c.status === 'pass').length;
    const totalChecks = checks.length;

    return Math.round((passCount / totalChecks) * 100);
  }

  generateRecommendationMessage(checks) {
    const failedChecks = checks.filter(c => c.status === 'fail');
    const warningChecks = checks.filter(c => c.status === 'warning');

    if (failedChecks.length > 0) {
      return `Not suitable for sowing. Issues: ${failedChecks.map(c => c.parameter).join(', ')}`;
    } else if (warningChecks.length > 0) {
      return `Proceed with caution. Monitor: ${warningChecks.map(c => c.parameter).join(', ')}`;
    } else {
      return 'Conditions are optimal for sowing. Proceed as planned.';
    }
  }
}

// LunarCalendarService.js
class LunarCalendarService {
  /**
   * Convert Gregorian date to Hijri (Islamic) calendar
   */
  convertToHijri(gregorianDate) {
    // Use Islamic calendar conversion library
    // Example: https://github.com/batoulapps/adhan-js
    const hijriDate = this.gregorianToHijri(gregorianDate);

    return {
      gregorian_date: gregorianDate,
      hijri_date: hijriDate.format('YYYY-MM-DD'),
      hijri_month_name: hijriDate.monthName,
      hijri_year: hijriDate.year
    };
  }

  /**
   * Calculate moon phase for given date
   */
  calculateMoonPhase(date) {
    // Moon phase calculation
    const year = date.getFullYear();
    const month = date.getMonth() + 1;
    const day = date.getDate();

    // Simplified moon age calculation
    const c = e = jd = b = 0;
    if (month < 3) {
      year--;
      month += 12;
    }

    ++month;
    c = 365.25 * year;
    e = 30.6 * month;
    jd = c + e + day - 694039.09;  // Julian date
    jd /= 29.5305882;  // Moon cycle
    b = parseInt(jd);
    jd -= b;
    b = Math.round(jd * 8);  // 0-7 phase index

    const phases = [
      'new', 'waxing_crescent', 'first_quarter', 'waxing_gibbous',
      'full', 'waning_gibbous', 'last_quarter', 'waning_crescent'
    ];

    const moonAge = jd * 29.5305882;

    return {
      moon_phase: phases[b],
      moon_age_days: Math.round(moonAge),
      illumination_percent: Math.round((1 - Math.cos((jd * 2 * Math.PI))) * 50)
    };
  }

  /**
   * Get planting guidance based on moon phase
   */
  getPlantingGuidance(moonPhase, cropType) {
    // Traditional farming wisdom based on moon phases

        favorability: 'favorable',
        crops: ['leafy_vegetables', 'grain', 'cucumber'],
        guidance: 'Excellent for planting above-ground crops with seeds inside'
      },
      first_quarter: {
        favorability: 'favorable',
        crops: ['beans', 'tomatoes', 'grain'],
        guidance: 'Best for planting crops with seeds inside fruit'
      },
      waxing_gibbous: {
        favorability: 'neutral',
        crops: ['root_crops'],
        guidance: 'Moderate time for planting, energy is still increasing'
      },
      full: {
        favorability: 'unfavorable',
        crops: [],
        guidance: 'Rest period. Focus on weeding and pest control'
      },
      waning_gibbous: {
        favorability: 'favorable',
        crops: ['root_crops', 'tubers'],
        guidance: 'Good for planting root crops and bulbs'
      },
      last_quarter: {
        favorability: 'favorable',
        crops: ['root_crops'],
        guidance: 'Excellent for planting root crops'
      },
      waning_crescent: {
        favorability: 'unfavorable',
        crops: [],
        guidance: 'Rest period. Avoid planting, focus on soil preparation'
      }
    };

    return guidance[moonPhase] || guidance.new;
  }

  /**
   * Integrate lunar calendar with sowing recommendations
   */
  async integrateWithSowingWindow(sowingWindow, gregorianDate) {
    const hijri = this.convertToHijri(gregorianDate);
    const moonPhase = this.calculateMoonPhase(gregorianDate);
    const guidance = this.getPlantingGuidance(moonPhase.moon_phase, sowingWindow.crop_type);

    return {
      ...hijri,
      ...moonPhase,
      planting_favorability: guidance.favorability,
      traditional_guidance: guidance.guidance,
      recommendation: this.combineScienceAndTradition(sowingWindow, guidance)
    };
  }

  combineScienceAndTradition(sowingWindow, lunarGuidance) {

      return 'Scientific conditions are optimal. Traditional guidance is neutral.';
    } else {
      return 'Scientific conditions are optimal, but traditional wisdom suggests waiting for next moon phase.';
    }
  }
}

    const field = await db.query('SELECT * FROM fields WHERE id = $1', [fieldId]);
    const region = field[0].region;

    // 2. Calculate sowing window
    const calculator = new SowingWindowCalculator();
    const sowingWindow = await calculator.calculateSowingWindow(cropType, region);

    // 3. Evaluate current conditions
    const evaluator = new SowingConditionEvaluator();
    const conditions = await evaluator.evaluateSowingConditions(cropType, fieldId, sowingWindow);

    // 4. Get historical success data
    const historical = await this.getHistoricalSuccessRate(cropType, region, sowingWindow);

    // 5. Integrate lunar calendar (if user enabled)
    const userPrefs = await this.getUserPreferences(userId);
    let lunarCalendar = null;
    if (userPrefs.use_lunar_calendar) {
      const lunarService = new LunarCalendarService();
      lunarCalendar = await lunarService.integrateWithSowingWindow(
        sowingWindow,
        new Date()
      );
    }

    // 6. Generate final recommendation
    const recommendation = {
      crop_type: cropType,
      variety,
      field_id: fieldId,
      region,

      // Sowing window
      optimal_start_date: sowingWindow.optimal_start_date,
      optimal_end_date: sowingWindow.optimal_end_date,
      extended_end_date: sowingWindow.extended_end_date,
      days_until_window: this.calculateDaysUntil(sowingWindow.optimal_start_date),

      // Conditions
      current_suitability: conditions.is_suitable,
      suitability_score: conditions.overall_score,
      recommendation_action: conditions.recommendation,
      condition_checks: conditions.checks,

      // Historical data
      historical_success_rate: historical.success_rate,
      optimal_yield_potential: historical.avg_yield,

      // Lunar calendar
      lunar_calendar: lunarCalendar,

      // Confidence and rationale
      confidence_score: this.calculateConfidence(conditions, historical),
      rationale: this.generateRationale(sowingWindow, conditions, historical),

      // Preparation tasks
      preparation_tasks: this.getPreparationTasks(cropType, sowingWindow),

      generated_at: new Date()
    };

    return recommendation;
  }

  calculateDaysUntil(targetDate) {
    const now = new Date();
    const target = new Date(targetDate);
    const diffTime = target - now;
    const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

    return diffDays;
  }

  async getHistoricalSuccessRate(cropType, region, sowingWindow) {
    // Query sowing history for this crop/region
    const history = await db.query(`
      SELECT
        AVG(CASE WHEN outcome = 'success' THEN 1 ELSE 0 END) * 100 as success_rate,
        AVG(final_yield_kg_per_acre) as avg_yield
      FROM sowing_history
      WHERE crop_type = $1
        AND field_id IN (SELECT id FROM fields WHERE region = $2)
        AND actual_sowing_date BETWEEN $3 AND $4
    `, [
      cropType,
      region,
      sowingWindow.optimal_start_date,
      sowingWindow.optimal_end_date
    ]);

    return history[0] || { success_rate: 80, avg_yield: 0 };
  }

  calculateConfidence(conditions, historical) {
    // Combine condition suitability with historical success
    const conditionWeight = 0.6;
    const historicalWeight = 0.4;

    const confidence = (conditions.overall_score * conditionWeight) +
                      (historical.success_rate * historicalWeight);

    return Math.round(confidence);
  }

  generateRationale(sowingWindow, conditions, historical) {
    const rationale = [];

    // Sowing window explanation
    rationale.push(`Optimal sowing window for ${sowingWindow.crop_type} in ${sowingWindow.region} is ${this.formatDate(sowingWindow.optimal_start_date)} to ${this.formatDate(sowingWindow.optimal_end_date)}.`);

    // Condition assessment
    if (conditions.is_suitable) {
      rationale.push('Current conditions are suitable for sowing.');
    } else {
      rationale.push(`Conditions require monitoring: ${conditions.message}`);
    }

    // Historical context
    if (historical.success_rate >= 80) {
      rationale.push(`Historical data shows ${Math.round(historical.success_rate)}% success rate for sowing in this period.`);
    }

    return rationale.join(' ');
  }

  getPreparationTasks(cropType, sowingWindow) {
    const commonTasks = [
      'Plow and level the field',
      'Apply basal fertilizer (NPK)',
      'Ensure seed quality and treatment',
      'Check irrigation system'
    ];

    const cropSpecificTasks = {
      wheat: ['Ensure adequate soil moisture', 'Control weeds'],
      rice: ['Prepare nursery bed', 'Flood the field if transplanting'],
      cotton: ['Deep plowing for root development', 'Apply pre-emergence herbicide']
    };

    return [...commonTasks, ...(cropSpecificTasks[cropType] || [])];
  }

  formatDate(date) {
    return new Date(date).toLocaleDateString('en-US', {
      month: 'short',
      day: 'numeric'
    });
  }

  async getUserPreferences(userId) {
    const prefs = await db.query('SELECT * FROM user_preferences WHERE user_id = $1', [userId]);
    return prefs[0] || { use_lunar_calendar: false };
  }
}
```

CREATE TABLE sowing_history (
