**Matching Algorithm Design**

The farmer matching algorithm uses a weighted scoring system to find similar farmers:

```javascript
// FarmerMatchingService.js
class FarmerMatchingService {
  constructor() {
    this.weights = {
      location: 0.4,      // 40% weight
      crops: 0.35,        // 35% weight
      landSize: 0.15,     // 15% weight
      experience: 0.10    // 10% weight
    };
  }

  async findMatches(farmerId, limit = 20) {
    const farmer = await FarmerRepository.getById(farmerId);

    // Check if user opted out
    if (farmer.matching_opt_out) {
      return [];
    }

    // Get all potential matches (farmers with public/connections-only profiles)
    const candidates = await this.getPotentialMatches(farmerId);

    // Calculate matching scores
    const matches = candidates.map(candidate => ({
      farmer: candidate,
      score: this.calculateMatchScore(farmer, candidate),
      reasons: this.getMatchReasons(farmer, candidate)
    }));

    // Sort by score and return top matches
    return matches
      .sort((a, b) => b.score - a.score)
      .slice(0, limit);
  }

  calculateMatchScore(farmer, candidate) {
    const locationScore = this.calculateLocationScore(farmer, candidate);
    const cropScore = this.calculateCropScore(farmer, candidate);
    const landSizeScore = this.calculateLandSizeScore(farmer, candidate);
    const experienceScore = this.calculateExperienceScore(farmer, candidate);

    return (
      locationScore * this.weights.location +
      cropScore * this.weights.crops +
      landSizeScore * this.weights.landSize +
      experienceScore * this.weights.experience
    );
  }

  calculateLocationScore(farmer, candidate) {
    // Calculate distance using Haversine formula
    const distance = this.haversineDistance(
      farmer.location_lat, farmer.location_lng,
      candidate.location_lat, candidate.location_lng
    );

    // Score: 1.0 if <10km, decreasing to 0 at 50km
    if (distance <= 10) return 1.0;
    if (distance >= 50) return 0.0;
    return 1.0 - ((distance - 10) / 40);
  }

  calculateCropScore(farmer, candidate) {
    const farmerCrops = new Set(farmer.crops_grown || []);
    const candidateCrops = new Set(candidate.crops_grown || []);

    // Calculate Jaccard similarity
    const intersection = new Set(
      [...farmerCrops].filter(x => candidateCrops.has(x))
    );
    const union = new Set([...farmerCrops, ...candidateCrops]);

    if (union.size === 0) return 0;
    return intersection.size / union.size;
  }

  calculateLandSizeScore(farmer, candidate) {
    const farmerSize = farmer.total_land_acres || 0;
    const candidateSize = candidate.total_land_acres || 0;

    // Score based on similarity (±5 acres)
    const difference = Math.abs(farmerSize - candidateSize);
    if (difference <= 5) return 1.0;
    if (difference >= 20) return 0.0;
    return 1.0 - ((difference - 5) / 15);
  }

  calculateExperienceScore(farmer, candidate) {
    const farmerExp = farmer.experience_years || 0;
    const candidateExp = candidate.experience_years || 0;

    // Score based on similar experience levels
    const difference = Math.abs(farmerExp - candidateExp);
    if (difference <= 3) return 1.0;
    if (difference >= 15) return 0.0;
    return 1.0 - ((difference - 3) / 12);
  }

  haversineDistance(lat1, lon1, lat2, lon2) {
    const R = 6371; // Earth's radius in kilometers
    const dLat = this.toRad(lat2 - lat1);
    const dLon = this.toRad(lon2 - lon1);
    const a =
      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
      Math.cos(this.toRad(lat1)) * Math.cos(this.toRad(lat2)) *
      Math.sin(dLon / 2) * Math.sin(dLon / 2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    return R * c;
  }

  toRad(degrees) {
    return degrees * (Math.PI / 180);
  }

  getMatchReasons(farmer, candidate) {
    const reasons = [];

    // Check location
    const distance = this.haversineDistance(
      farmer.location_lat, farmer.location_lng,
      candidate.location_lat, candidate.location_lng
    );
    if (distance < 10) {
      reasons.push({ type: 'location', text: 'Very nearby (< 10km)' });
    } else if (distance < 30) {
      reasons.push({ type: 'location', text: `Nearby (~${Math.round(distance)}km)` });
    }

    // Check crops
    const commonCrops = farmer.crops_grown?.filter(c =>
      candidate.crops_grown?.includes(c)
    ) || [];
    if (commonCrops.length > 0) {
      reasons.push({
        type: 'crops',
        text: `Grows ${commonCrops.join(', ')}`
      });
    }

    // Check land size
    const sizeDiff = Math.abs(
      (farmer.total_land_acres || 0) - (candidate.total_land_acres || 0)
    );
    if (sizeDiff <= 5) {
      reasons.push({
        type: 'landSize',
        text: `Similar land size (~${candidate.total_land_acres} acres)`
      });
    }

    return reasons;
  }
}
```

**Privacy Filtering Service**

```javascript
// PrivacyFilterService.js
class PrivacyFilterService {
  async canViewProfile(viewerId, targetFarmerId) {
    const targetSettings = await this.getPrivacySettings(targetFarmerId);

    // Check global opt-out
    if (!targetSettings.enable_matching) {
      return false;
    }

    // Check profile visibility
    switch (targetSettings.profile_visibility) {
      case 'public':
        return true;
      case 'connections':
        return await this.areConnected(viewerId, targetFarmerId);
      case 'private':
        return viewerId === targetFarmerId; // Only self can view
      default:
        return false;
    }
  }

  async filterProfileData(viewerId, targetProfile) {
    const settings = await this.getPrivacySettings(targetProfile.farmer_id);
    const areConnected = await this.areConnected(viewerId, targetProfile.farmer_id);

    const filtered = { ...targetProfile };

    // Apply field-level privacy
    if (!settings.show_location && !areConnected) {
      filtered.location_lat = null;
      filtered.location_lng = null;
      // Show region only
      filtered.location_display = targetProfile.region;
    } else if (settings.location_precision === 'region') {
      filtered.location_lat = null;
      filtered.location_lng = null;
      filtered.location_display = targetProfile.region;
    } else if (settings.location_precision === 'city') {
      // Fuzzy location to city center
      filtered.location_lat = this.fuzzToCity(targetProfile.location_lat);
      filtered.location_lng = this.fuzzToCity(targetProfile.location_lng);
    }

    if (!settings.show_land_size && !areConnected) {
      filtered.total_land_acres = null;
    }

    if (!settings.show_crops && !areConnected) {
      filtered.crops_grown = [];
    }

    if (!settings.show_experience && !areConnected) {
      filtered.experience_years = null;
    }

    if (!settings.show_phone && !areConnected) {
      filtered.phone = null;
    }

    if (!settings.show_email && !areConnected) {
      filtered.email = null;
    }

    return filtered;
  }

  async areConnected(farmer1Id, farmer2Id) {
    const connection = await db.query(`
      SELECT id FROM farmer_connections
      WHERE ((farmer_id = $1 AND connected_farmer_id = $2)
         OR (farmer_id = $2 AND connected_farmer_id = $1))
        AND status = 'accepted'
    `, [farmer1Id, farmer2Id]);

    return connection.length > 0;
  }

  fuzzToCity(coordinate) {
    // Round to 2 decimal places (approx 1km precision)
    return Math.round(coordinate * 100) / 100;
  }
}
```

**Connection Request Workflow**

```javascript
// ConnectionService.js
class ConnectionService {
  async sendConnectionRequest(fromFarmerId, toFarmerId, message = null) {
    // Check if already connected or pending
    const existing = await this.getConnectionStatus(fromFarmerId, toFarmerId);
    if (existing) {
      throw new Error(`Connection already ${existing.status}`);
    }

    // Check connection limit
    const connectionCount = await this.getConnectionCount(fromFarmerId);
    if (connectionCount >= 500) {
      throw new Error('Connection limit reached (500)');
    }

    // Create connection request
    const connectionId = `conn-${Date.now()}-${Math.random()}`;
    await db.query(`
      INSERT INTO farmer_connections (
        id, farmer_id, connected_farmer_id, status, requested_by
      ) VALUES ($1, $2, $3, 'pending', $2)
    `, [connectionId, fromFarmerId, toFarmerId]);

    // Send notification to target farmer
    await NotificationService.send(toFarmerId, {
      type: 'connection_request',
      title: 'New Connection Request',
      body: 'A farmer wants to connect with you',
      data: { connectionId, fromFarmerId }
    });

    return connectionId;
  }

  async acceptConnectionRequest(connectionId, farmerId) {
    // Verify farmer is the recipient
    const connection = await this.getConnection(connectionId);
    if (connection.connected_farmer_id !== farmerId) {
      throw new Error('Not authorized to accept this request');
    }

    // Update status
    await db.query(`
      UPDATE farmer_connections
      SET status = 'accepted', accepted_at = NOW()
      WHERE id = $1
    `, [connectionId]);

    // Send notification to requester
    await NotificationService.send(connection.farmer_id, {
      type: 'connection_accepted',
      title: 'Connection Request Accepted',
      body: 'Your connection request was accepted',
      data: { connectionId, acceptedBy: farmerId }
    });

    return connection;
  }

  async rejectConnectionRequest(connectionId, farmerId) {
    const connection = await this.getConnection(connectionId);
    if (connection.connected_farmer_id !== farmerId) {
      throw new Error('Not authorized to reject this request');
    }

    await db.query(`
      UPDATE farmer_connections
      SET status = 'rejected'
      WHERE id = $1
    `, [connectionId]);

    return connection;
  }

  async removeConnection(farmerId, connectedFarmerId) {
    await db.query(`
      DELETE FROM farmer_connections
      WHERE ((farmer_id = $1 AND connected_farmer_id = $2)
         OR (farmer_id = $2 AND connected_farmer_id = $1))
        AND status = 'accepted'
    `, [farmerId, connectedFarmerId]);
  }

  async getConnections(farmerId, status = 'accepted') {
    const connections = await db.query(`
      SELECT
        fc.*,
        CASE
          WHEN fc.farmer_id = $1 THEN fc.connected_farmer_id
          ELSE fc.farmer_id
        END as other_farmer_id,
        f.name,
        f.profile_photo_url,
        fpd.region,
        fpd.crops_grown
      FROM farmer_connections fc
      JOIN farmers f ON (
        CASE
          WHEN fc.farmer_id = $1 THEN fc.connected_farmer_id
          ELSE fc.farmer_id
        END = f.id
      )
      LEFT JOIN farmer_profile_details fpd ON f.id = fpd.farmer_id
      WHERE (fc.farmer_id = $1 OR fc.connected_farmer_id = $1)
        AND fc.status = $2
      ORDER BY fc.accepted_at DESC
    `, [farmerId, status]);

    return connections;
  }
}
```

### Error Handling

Common error scenarios:
1. **Connection limit reached**: "You've reached the maximum of 500 connections"
2. **Already connected**: "You're already connected with this farmer"
3. **Profile incomplete**: "Complete your profile to get better matches"
4. **No matches found**: "No farmers found matching your criteria. Try adjusting your profile."
5. **Privacy opt-out**: "This farmer has opted out of matching"
6. **Network error**: "Cannot load suggestions. Check your connection."
7. **Invalid location**: "Location required for matching. Update your profile."
