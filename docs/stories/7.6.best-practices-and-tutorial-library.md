# Story 7.6: Best Practices and Tutorial Library

## Status
Draft

## Story
**As a** farmer wanting to learn new techniques,
**I want** a library of agricultural best practices and tutorials,
**so that** I can improve my farming methods.

## Acceptance Criteria
1. AC1: Tutorial library with text, images, and videos
2. AC2: Topics: modern techniques, organic farming, pest management, soil health, etc.
3. AC3: Step-by-step guides with visual instructions
4. AC4: Tutorials in Urdu, Punjabi, Sindhi
5. AC5: User bookmarking and favorites
6. AC6: Progress tracking for multi-step tutorials
7. AC7: Offline download for video content

**Tutorial Library Schema**

```sql
-- Tutorials
CREATE TABLE tutorials (
  id TEXT PRIMARY KEY,
  title_en TEXT NOT NULL,
  title_ur TEXT,
  title_pa TEXT,
  title_sd TEXT,

  summary_en TEXT,
  summary_ur TEXT,
  summary_pa TEXT,
  summary_sd TEXT,

  content_en TEXT,
  content_ur TEXT,
  content_pa TEXT,
  content_sd TEXT,

  category TEXT NOT NULL, -- irrigation, pest_control, organic, soil_health, etc.
  topics TEXT[], -- Array of related topics
  crop_types TEXT[], -- wheat, rice, cotton, etc.
  difficulty TEXT DEFAULT 'beginner', -- beginner, intermediate, advanced

  thumbnail_url TEXT,
  video_url TEXT,
  video_duration_seconds INTEGER,
  pdf_url TEXT,

  estimated_time_minutes INTEGER,
  prerequisites TEXT[],
  learning_objectives JSONB,

  source TEXT, -- government, ngo, university, expert
  author_name TEXT,
  author_organization TEXT,

  is_published BOOLEAN DEFAULT FALSE,
  is_featured BOOLEAN DEFAULT FALSE,

  view_count INTEGER DEFAULT 0,
  completion_count INTEGER DEFAULT 0,
  bookmark_count INTEGER DEFAULT 0,
  avg_rating REAL,
  rating_count INTEGER DEFAULT 0,

  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  published_at TIMESTAMP
);

CREATE INDEX idx_tutorials_category ON tutorials(category, is_published);
CREATE INDEX idx_tutorials_crops ON tutorials USING gin(crop_types);
CREATE INDEX idx_tutorials_difficulty ON tutorials(difficulty, is_published);
CREATE INDEX idx_tutorials_rating ON tutorials(avg_rating DESC);

-- Tutorial steps
CREATE TABLE tutorial_steps (
  id TEXT PRIMARY KEY,
  tutorial_id TEXT NOT NULL,
  step_number INTEGER NOT NULL,

  title_en TEXT NOT NULL,
  title_ur TEXT,
  title_pa TEXT,
  title_sd TEXT,

  content_en TEXT NOT NULL,
  content_ur TEXT,
  content_pa TEXT,
  content_sd TEXT,

  image_urls JSONB, -- Array of image URLs
  video_url TEXT,
  duration_seconds INTEGER,

  tips_en TEXT,
  tips_ur TEXT,

  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (tutorial_id) REFERENCES tutorials(id) ON DELETE CASCADE
);

CREATE INDEX idx_tutorial_steps_tutorial ON tutorial_steps(tutorial_id, step_number);

-- Tutorial bookmarks
CREATE TABLE tutorial_bookmarks (
  id TEXT PRIMARY KEY,
  tutorial_id TEXT NOT NULL,
  farmer_id TEXT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (tutorial_id) REFERENCES tutorials(id) ON DELETE CASCADE,
  FOREIGN KEY (farmer_id) REFERENCES farmers(id) ON DELETE CASCADE,
  UNIQUE(tutorial_id, farmer_id)
);

-- Tutorial progress
CREATE TABLE tutorial_progress (
  id TEXT PRIMARY KEY,
  tutorial_id TEXT NOT NULL,
  farmer_id TEXT NOT NULL,
  completed_steps INTEGER[] DEFAULT ARRAY[]::INTEGER[],
  current_step INTEGER DEFAULT 1,
  completed BOOLEAN DEFAULT FALSE,
  time_spent_seconds INTEGER DEFAULT 0,
  started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  completed_at TIMESTAMP,
  last_accessed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (tutorial_id) REFERENCES tutorials(id) ON DELETE CASCADE,
  FOREIGN KEY (farmer_id) REFERENCES farmers(id) ON DELETE CASCADE,
  UNIQUE(tutorial_id, farmer_id)
);

CREATE INDEX idx_tutorial_progress_farmer ON tutorial_progress(farmer_id, last_accessed_at DESC);

-- Tutorial ratings
CREATE TABLE tutorial_ratings (
  id TEXT PRIMARY KEY,
  tutorial_id TEXT NOT NULL,
  farmer_id TEXT NOT NULL,
  rating INTEGER NOT NULL CHECK (rating >= 1 AND rating <= 5),
  is_helpful BOOLEAN,
  feedback TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (tutorial_id) REFERENCES tutorials(id) ON DELETE CASCADE,
  FOREIGN KEY (farmer_id) REFERENCES farmers(id) ON DELETE CASCADE,
  UNIQUE(tutorial_id, farmer_id)
);

-- Downloaded videos (for offline access)
CREATE TABLE downloaded_videos (
  id TEXT PRIMARY KEY,
  tutorial_id TEXT NOT NULL,
  farmer_id TEXT NOT NULL,
  video_url TEXT NOT NULL,
  local_file_path TEXT NOT NULL,
  file_size_bytes INTEGER,
  quality TEXT, -- low, medium, high
  downloaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  last_accessed_at TIMESTAMP,
  FOREIGN KEY (tutorial_id) REFERENCES tutorials(id) ON DELETE CASCADE,
  FOREIGN KEY (farmer_id) REFERENCES farmers(id) ON DELETE CASCADE,
  UNIQUE(tutorial_id, farmer_id, quality)
);

CREATE INDEX idx_downloaded_videos_farmer ON downloaded_videos(farmer_id);
```

**Tutorial Service Implementation**

```javascript
// TutorialService.js
class TutorialService {
  async getTutorials(filters = {}, page = 1, limit = 20) {
    const { category, cropType, difficulty, searchTerm, language = 'en', sortBy } = filters;
    const offset = (page - 1) * limit;

    // Dynamic column selection based on language
    const langSuffix = language === 'en' ? 'en' : language === 'ur' ? 'ur' : language === 'pa' ? 'pa' : 'sd';

    let query = `
      SELECT
        id,
        title_${langSuffix} as title,
        summary_${langSuffix} as summary,
        category, difficulty, thumbnail_url, video_duration_seconds,
        estimated_time_minutes, avg_rating, view_count, bookmark_count
      FROM tutorials
      WHERE is_published = TRUE
    `;

    const params = [];
    let paramCount = 0;

    if (category) {
      query += ` AND category = $${++paramCount}`;
      params.push(category);
    }

    if (cropType) {
      query += ` AND $${++paramCount} = ANY(crop_types)`;
      params.push(cropType);
    }

    if (difficulty) {
      query += ` AND difficulty = $${++paramCount}`;
      params.push(difficulty);
    }

    if (searchTerm) {
      query += ` AND (
        title_${langSuffix} ILIKE $${++paramCount}
        OR summary_${langSuffix} ILIKE $${paramCount}
      )`;
      params.push(`%${searchTerm}%`);
    }

    // Sorting
    switch (sortBy) {
      case 'popular':
        query += ` ORDER BY view_count DESC`;
        break;
      case 'rating':
        query += ` ORDER BY avg_rating DESC NULLS LAST`;
        break;
      case 'latest':
        query += ` ORDER BY published_at DESC`;
        break;
      default:
        query += ` ORDER BY is_featured DESC, published_at DESC`;
    }

    query += ` LIMIT $${++paramCount} OFFSET $${++paramCount}`;
    params.push(limit, offset);

    const tutorials = await db.query(query, params);
    return tutorials;
  }

  async getTutorialDetails(tutorialId, farmerId, language = 'en') {
    // Increment view count
    await db.query(`
      UPDATE tutorials SET view_count = view_count + 1 WHERE id = $1
    `, [tutorialId]);

    const langSuffix = language === 'en' ? 'en' : language === 'ur' ? 'ur' : language === 'pa' ? 'pa' : 'sd';

    // Get tutorial with localized content
    let query = `
      SELECT
        id,
        title_${langSuffix} as title,
        summary_${langSuffix} as summary,
        content_${langSuffix} as content,
        category, difficulty, thumbnail_url, video_url, video_duration_seconds,
        pdf_url, estimated_time_minutes, prerequisites, learning_objectives,
        source, author_name, avg_rating, rating_count
    `;

    if (farmerId) {
      query += `,
        EXISTS(SELECT 1 FROM tutorial_bookmarks WHERE tutorial_id = $1 AND farmer_id = $2) as is_bookmarked
      `;
    }

    query += ` FROM tutorials WHERE id = $1 AND is_published = TRUE`;

    const params = farmerId ? [tutorialId, farmerId] : [tutorialId];
    const tutorial = await db.query(query, params);

    if (tutorial.length === 0) {
      throw new Error('Tutorial not found');
    }

    // Get steps
    const steps = await db.query(`
      SELECT
        step_number,
        title_${langSuffix} as title,
        content_${langSuffix} as content,
        tips_${langSuffix} as tips,
        image_urls, video_url, duration_seconds
      FROM tutorial_steps
      WHERE tutorial_id = $1
      ORDER BY step_number ASC
    `, [tutorialId]);

    // Get user progress if logged in
    let progress = null;
    if (farmerId) {
      const progressData = await db.query(`
        SELECT * FROM tutorial_progress
        WHERE tutorial_id = $1 AND farmer_id = $2
      `, [tutorialId, farmerId]);

      if (progressData.length > 0) {
        progress = progressData[0];
      }
    }

    return {
      tutorial: tutorial[0],
      steps,
      progress
    };
  }

  async bookmarkTutorial(farmerId, tutorialId) {
    const bookmarkId = `bookmark-${Date.now()}-${Math.random()}`;

    try {
      await db.query(`
        INSERT INTO tutorial_bookmarks (id, tutorial_id, farmer_id)
        VALUES ($1, $2, $3)
      `, [bookmarkId, tutorialId, farmerId]);

      await db.query(`
        UPDATE tutorials SET bookmark_count = bookmark_count + 1 WHERE id = $1
      `, [tutorialId]);
    } catch (error) {
      if (error.constraint === 'tutorial_bookmarks_tutorial_id_farmer_id_key') {
        // Already bookmarked, remove it
        await db.query(`
          DELETE FROM tutorial_bookmarks
          WHERE tutorial_id = $1 AND farmer_id = $2
        `, [tutorialId, farmerId]);

        await db.query(`
          UPDATE tutorials SET bookmark_count = bookmark_count - 1 WHERE id = $1
        `, [tutorialId]);
      } else {
        throw error;
      }
    }
  }

  async updateProgress(farmerId, tutorialId, stepNumber, timeSpent = 0) {
    // Check if progress exists
    const existing = await db.query(`
      SELECT * FROM tutorial_progress
      WHERE tutorial_id = $1 AND farmer_id = $2
    `, [tutorialId, farmerId]);

    if (existing.length === 0) {
      // Create new progress
      await db.query(`
        INSERT INTO tutorial_progress (
          id, tutorial_id, farmer_id, completed_steps, current_step, time_spent_seconds
        ) VALUES ($1, $2, $3, $4, $5, $6)
      `, [
        `progress-${Date.now()}-${Math.random()}`,
        tutorialId, farmerId, [stepNumber], stepNumber, timeSpent
      ]);
    } else {
      // Update existing progress
      const completedSteps = existing[0].completed_steps || [];
      if (!completedSteps.includes(stepNumber)) {
        completedSteps.push(stepNumber);
      }

      // Check if all steps completed
      const totalSteps = await db.query(`
        SELECT COUNT(*) as count FROM tutorial_steps WHERE tutorial_id = $1
      `, [tutorialId]);

      const completed = completedSteps.length >= totalSteps[0].count;

      await db.query(`
        UPDATE tutorial_progress
        SET completed_steps = $1,
            current_step = $2,
            time_spent_seconds = time_spent_seconds + $3,
            completed = $4,
            completed_at = ${completed ? 'NOW()' : 'NULL'},
            last_accessed_at = NOW()
        WHERE tutorial_id = $5 AND farmer_id = $6
      `, [completedSteps, stepNumber, timeSpent, completed, tutorialId, farmerId]);

      // Update tutorial completion count if completed
      if (completed && !existing[0].completed) {
        await db.query(`
          UPDATE tutorials SET completion_count = completion_count + 1 WHERE id = $1
        `, [tutorialId]);
      }
    }
  }

  async rateTutorial(farmerId, tutorialId, rating, isHelpful, feedback = null) {
    const ratingId = `rating-${Date.now()}-${Math.random()}`;

    await db.query(`
      INSERT INTO tutorial_ratings (id, tutorial_id, farmer_id, rating, is_helpful, feedback)
      VALUES ($1, $2, $3, $4, $5, $6)
      ON CONFLICT (tutorial_id, farmer_id)
      DO UPDATE SET rating = $4, is_helpful = $5, feedback = $6
    `, [ratingId, tutorialId, farmerId, rating, isHelpful, feedback]);

    // Recalculate average rating
    await this.updateTutorialRating(tutorialId);
  }

  async updateTutorialRating(tutorialId) {
    const stats = await db.query(`
      SELECT AVG(rating) as avg_rating, COUNT(*) as rating_count
      FROM tutorial_ratings
      WHERE tutorial_id = $1
    `, [tutorialId]);

    await db.query(`
      UPDATE tutorials SET avg_rating = $1, rating_count = $2 WHERE id = $3
    `, [stats[0].avg_rating, stats[0].rating_count, tutorialId]);
  }

  async downloadVideo(farmerId, tutorialId, videoUrl, quality = 'medium') {
    // Check available storage
    const storageAvailable = await StorageManager.getAvailableSpace();
    if (storageAvailable < 100 * 1024 * 1024) { // Less than 100MB
      throw new Error('Insufficient storage space');
    }

    const downloadId = `download-${Date.now()}-${Math.random()}`;

    // Download video file
    const localPath = await VideoDownloadManager.download(videoUrl, quality, (progress) => {
      // Update download progress
      console.log(`Download progress: ${progress}%`);
    });

    // Get file size
    const fileSize = await FileSystem.getInfoAsync(localPath);

    // Save download record
    await db.query(`
      INSERT INTO downloaded_videos (
        id, tutorial_id, farmer_id, video_url, local_file_path,
        file_size_bytes, quality
      ) VALUES ($1, $2, $3, $4, $5, $6, $7)
    `, [downloadId, tutorialId, farmerId, videoUrl, localPath, fileSize.size, quality]);

    return localPath;
  }

  async getDownloadedVideos(farmerId) {
    const downloads = await db.query(`
      SELECT
        dv.*,
        t.title_en as tutorial_title,
        t.thumbnail_url
      FROM downloaded_videos dv
      JOIN tutorials t ON dv.tutorial_id = t.id
      WHERE dv.farmer_id = $1
      ORDER BY dv.downloaded_at DESC
    `, [farmerId]);

    return downloads;
  }

  async deleteDownload(downloadId) {
    const download = await db.query(`
      SELECT local_file_path FROM downloaded_videos WHERE id = $1
    `, [downloadId]);

    if (download.length > 0) {
      // Delete file from storage
      await FileSystem.deleteAsync(download[0].local_file_path);

      // Delete record
      await db.query(`
        DELETE FROM downloaded_videos WHERE id = $1
      `, [downloadId]);
    }
  }
}
```
