**Elevation-Based Flood Risk Assessment**

```javascript
// ElevationDataService.js
const SRTM = require('srtm-elevation');

class ElevationDataService {
  constructor() {
    this.srtmData = new SRTM();
  }

  /**
   * Get elevation for a GPS coordinate
   */
  async getElevation(latitude, longitude) {
    try {
      // SRTM provides 90m resolution elevation data
      const elevation = await this.srtmData.getElevation(latitude, longitude);

      return {
        latitude,
        longitude,
        elevation_meters: elevation,
        data_source: 'SRTM'
      };
    } catch (error) {
      console.error('SRTM elevation fetch error:', error);
      return null;
    }
  }

  /**
   * Calculate flood risk based on elevation and proximity to water
   */
  async calculateFloodRisk(fieldId, latitude, longitude) {
    const elevation = await this.getElevation(latitude, longitude);

    if (!elevation) {
      return { risk: 'unknown', reason: 'Elevation data unavailable' };
    }

    // Get nearest river/water body
    const nearestRiver = await this.getNearestRiver(latitude, longitude);
    const distanceToRiver = nearestRiver.distance_km;

    // Flood risk classification
    let riskLevel = 'low';
    let riskScore = 0;

    // Elevation-based risk
    if (elevation.elevation_meters < 200) {
      riskScore += 40;  // Very low elevation
    } else if (elevation.elevation_meters < 300) {
      riskScore += 25;  // Low elevation
    } else if (elevation.elevation_meters < 500) {
      riskScore += 10;  // Moderate elevation
    }

    // Distance from river
    if (distanceToRiver < 1) {
      riskScore += 40;  // Within 1km of river
    } else if (distanceToRiver < 5) {
      riskScore += 25;  // Within 5km
    } else if (distanceToRiver < 10) {
      riskScore += 10;  // Within 10km
    }

    // Classify risk level
    if (riskScore >= 60) {
      riskLevel = 'extreme';
    } else if (riskScore >= 40) {
      riskLevel = 'high';
    } else if (riskScore >= 20) {
      riskLevel = 'medium';
    } else {
      riskLevel = 'low';
    }

    // Store in database
    await db.insert('flood_risk_assessments', {
      id: `flood-risk-${Date.now()}`,
      field_id: fieldId,
      latitude,
      longitude,
      elevation_meters: elevation.elevation_meters,
      nearest_river: nearestRiver.name,
      distance_to_river_km: distanceToRiver,
      risk_level: riskLevel,
      risk_score: riskScore,
      assessed_at: new Date()
    });

    return {
      risk: riskLevel,
      elevation: elevation.elevation_meters,
      nearestRiver: nearestRiver.name,
      distanceToRiver,
      riskScore
    };
  }

  async getNearestRiver(latitude, longitude) {
    // Major rivers in Pakistan with approximate coordinates
    const rivers = [
      { name: 'Indus', lat: 25.0, lon: 68.0 },
      { name: 'Ravi', lat: 31.5, lon: 74.3 },
      { name: 'Chenab', lat: 32.5, lon: 74.5 },
      { name: 'Sutlej', lat: 30.8, lon: 73.9 },
      { name: 'Jhelum', lat: 33.2, lon: 73.7 }
    ];

    let nearest = rivers[0];
    let minDistance = this.calculateDistance(latitude, longitude, nearest.lat, nearest.lon);

    for (const river of rivers) {
      const distance = this.calculateDistance(latitude, longitude, river.lat, river.lon);
      if (distance < minDistance) {
        minDistance = distance;
        nearest = river;
      }
    }

    return { ...nearest, distance_km: minDistance };
  }

  calculateDistance(lat1, lon1, lat2, lon2) {
    // Haversine formula for distance calculation
    const R = 6371; // Earth's radius in km
    const dLat = this.toRad(lat2 - lat1);
    const dLon = this.toRad(lon2 - lon1);

    const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
              Math.cos(this.toRad(lat1)) * Math.cos(this.toRad(lat2)) *
              Math.sin(dLon/2) * Math.sin(dLon/2);

    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    return R * c;
  }

  toRad(degrees) {
    return degrees * Math.PI / 180;
  }
}
```

```javascript
// RiverBasinFloodService.js
class RiverBasinFloodService {
  async fetchRiverWaterLevels() {
    // Integrate with IRSA (Indus River System Authority)
    const irsaData = await this.fetchIRSAData();

    // Integrate with WAPDA dam data
    const damData = await this.fetchWAPDADamLevels();

    return { irsaData, damData };
  }

  async fetchIRSAData() {
    // IRSA provides river discharge data
    const url = 'http://www.irsa.gov.pk/api/river-data';

    try {
      const response = await axios.get(url);
      return this.transformIRSAData(response.data);
    } catch (error) {
      console.error('IRSA API error:', error);
      return [];
    }
  }

  async fetchWAPDADamLevels() {
    // WAPDA provides dam reservoir levels and release schedules
    const dams = ['Tarbela', 'Mangla', 'Chashma', 'Taunsa'];
    const damLevels = [];

    for (const dam of dams) {
      try {
        const level = await this.fetchDamLevel(dam);
        damLevels.push(level);
      } catch (error) {
        console.error(`Error fetching ${dam} dam level:`, error);
      }
    }

    return damLevels;
  }

  async generateRiverBasinForecast(riverBasin, district) {
    // Get current water levels
    const currentLevels = await this.fetchRiverWaterLevels();

    // Get seasonal rainfall forecast from Story 6.1
    const seasonalForecast = await this.getSeasonalRainfallForecast(riverBasin);

    // Get upstream dam release schedule
    const damReleases = await this.getUpstreamDamReleases(riverBasin);

    // Calculate flood probability
    const floodProbability = this.calculateFloodProbability(
      currentLevels,
      seasonalForecast,
      damReleases,
      riverBasin
    );

    // Historical flood data
    const historicalFloods = await this.getHistoricalFloods(district);

    const forecast = {
      id: `river-forecast-${Date.now()}`,
      river_basin: riverBasin,
      district,
      current_water_level: currentLevels.irsaData.find(r => r.basin === riverBasin)?.level || 0,
      danger_level: this.getDangerLevel(riverBasin),
      flood_probability_percent: floodProbability,
      flood_risk: this.classifyFloodRisk(floodProbability),
      forecast_period_days: 15,
      expected_peak_date: this.calculatePeakFloodDate(seasonalForecast),
      upstream_rainfall_expected_mm: seasonalForecast.rainfall_mm,
      dam_release_schedule: damReleases,
      recommendation: this.getFloodRecommendation(floodProbability),
      forecast_date: new Date()
    };

    await db.insert('river_basin_forecasts', forecast);

    return forecast;
  }

  calculateFloodProbability(currentLevels, seasonalForecast, damReleases, riverBasin) {
    let probability = 0;

    // Factor 1: Current water level vs danger level (30% weight)
    const currentLevel = currentLevels.irsaData.find(r => r.basin === riverBasin)?.level || 0;
    const dangerLevel = this.getDangerLevel(riverBasin);
    const levelRatio = currentLevel / dangerLevel;

    if (levelRatio > 0.9) probability += 30;
    else if (levelRatio > 0.75) probability += 20;
    else if (levelRatio > 0.5) probability += 10;

    // Factor 2: Seasonal rainfall forecast (40% weight)
    if (seasonalForecast.precipitation_outlook === 'above_normal') {
      probability += 40;
    } else if (seasonalForecast.precipitation_outlook === 'normal') {
      probability += 20;
    } else {
      probability += 5;
    }

    // Factor 3: Dam releases (30% weight)
    const totalRelease = damReleases.reduce((sum, dam) => sum + dam.planned_release_cusecs, 0);
    if (totalRelease > 200000) probability += 30;
    else if (totalRelease > 150000) probability += 20;
    else if (totalRelease > 100000) probability += 10;

    return Math.min(100, probability);
  }

  getDangerLevel(riverBasin) {
    // Danger levels for major rivers (in feet or cusecs)
    const dangerLevels = {
      'Indus': 25,
      'Ravi': 18,
      'Chenab': 20,
      'Sutlej': 17,
      'Jhelum': 19
    };
    return dangerLevels[riverBasin] || 20;
  }

  classifyFloodRisk(probability) {
    if (probability >= 75) return 'extreme';
    if (probability >= 50) return 'high';
    if (probability >= 25) return 'medium';
    return 'low';
  }

  getFloodRecommendation(probability) {
    if (probability >= 75) {
      return 'Extreme flood risk. Consider early harvest and prepare evacuation plan.';
    } else if (probability >= 50) {
      return 'High flood risk. Clear drainage channels and monitor flood alerts daily.';
    } else if (probability >= 25) {
      return 'Moderate flood risk. Prepare flood mitigation measures and monitor forecasts.';
    }
    return 'Low flood risk. Continue normal farming activities with routine monitoring.';
  }
}
```

```javascript
// FloodHistoryService.js
class FloodHistoryService {
  async analyzeFloodFrequency(district, years = 50) {
    // Get historical flood events
    const historicalFloods = await db.query(`
      SELECT * FROM flood_history
      WHERE district = $1
        AND flood_year >= $2
      ORDER BY flood_year DESC
    `, [district, new Date().getFullYear() - years]);

    // Calculate frequency
    const floodYears = historicalFloods.map(f => f.flood_year);
    const totalFloods = floodYears.length;
    const frequencyPercent = (totalFloods / years) * 100;

    // Calculate recurrence intervals
    const recurrenceInterval = years / totalFloods;

    // Identify flood patterns (monsoon-related, dam-release, cyclone)
    const floodCauses = this.categorizeFloodCauses(historicalFloods);

    // Calculate average flood severity
    const avgSeverity = historicalFloods.reduce((sum, f) => {
      const severityScore = { minor: 1, moderate: 2, major: 3, catastrophic: 4 }[f.severity] || 2;
      return sum + severityScore;
    }, 0) / totalFloods;

    // Recent trend analysis (last 10 years vs previous 40 years)
    const recentFloods = historicalFloods.filter(f => f.flood_year >= new Date().getFullYear() - 10);
    const recentFrequency = (recentFloods.length / 10) * 100;
    const trend = recentFrequency > frequencyPercent ? 'increasing' : 'decreasing';

    return {
      district,
      analysis_years: years,
      total_floods: totalFloods,
      flood_frequency_percent: frequencyPercent,
      recurrence_interval_years: recurrenceInterval,
      flood_causes: floodCauses,
      average_severity: avgSeverity,
      trend,
      recent_major_floods: historicalFloods.filter(f => f.severity === 'major' || f.severity === 'catastrophic').slice(0, 5),
      last_flood_year: floodYears[0],
      years_since_last_flood: new Date().getFullYear() - floodYears[0]
    };
  }

  categorizeFloodCauses(historicalFloods) {
    const causes = {
      monsoon: 0,
      dam_release: 0,
      cyclone: 0,
      glacial_melt: 0,
      other: 0
    };

    historicalFloods.forEach(flood => {
      const cause = flood.primary_cause || 'other';
      if (causes[cause] !== undefined) {
        causes[cause]++;
      } else {
        causes.other++;
      }
    });

    return causes;
  }

  async getCurrentSeasonFloodProbability(district) {
    // Combine historical frequency with current season forecast
    const historical = await this.analyzeFloodFrequency(district);
    const seasonalForecast = await this.getSeasonalRainfallForecast(district);

    let probability = historical.flood_frequency_percent;

    // Adjust based on seasonal forecast
    if (seasonalForecast.flood_risk === 'extreme') {
      probability = Math.min(100, probability * 2);
    } else if (seasonalForecast.flood_risk === 'high') {
      probability = Math.min(100, probability * 1.5);
    }

    return {
      base_probability: historical.flood_frequency_percent,
      adjusted_probability: probability,
      historical_context: historical,
      seasonal_factors: seasonalForecast
    };
  }
}
```

```javascript
// FloodMitigationAdvisor.js
class FloodMitigationAdvisor {
  async generateMitigationStrategies(fieldId, floodRisk) {
    const field = await db.query('SELECT * FROM fields WHERE id = $1', [fieldId]);
    const fieldData = field[0];

    const strategies = [];

    // Strategy 1: Drainage improvements
    if (floodRisk >= 'medium') {
      strategies.push({
        strategy: 'drainage_improvement',
        title: 'Improve Field Drainage',
        description: 'Construct or clean drainage channels to prevent waterlogging',
        priority: floodRisk === 'extreme' ? 'critical' : 'high',
        estimated_cost_pkr: this.calculateDrainageCost(fieldData.size_acres),
        implementation_time_days: 7,
        effectiveness_percent: 70,
        steps: [
          'Clear existing drainage channels',
          'Construct new field drains at 50m intervals',
          'Ensure outlets drain to natural watercourse',
          'Install silt traps to prevent clogging'
        ]
      });
    }

    // Strategy 2: Raised bed farming
    if (floodRisk === 'high' || floodRisk === 'extreme') {
      strategies.push({
        strategy: 'raised_beds',
        title: 'Adopt Raised Bed Farming',
        description: 'Plant crops on raised beds to protect from waterlogging',
        priority: 'high',
        estimated_cost_pkr: fieldData.size_acres * 15000,
        implementation_time_days: 14,
        effectiveness_percent: 60,
        steps: [
          'Prepare raised beds 15-20cm high',
          'Create furrows between beds for drainage',
          'Plant crops on top of beds',
          'Maintain bed structure throughout season'
        ],
        suitable_crops: ['wheat', 'cotton', 'vegetables']
      });
    }

    // Strategy 3: Flood-tolerant crop varieties
    strategies.push({
      strategy: 'flood_tolerant_varieties',
      title: 'Plant Flood-Tolerant Crop Varieties',
      description: 'Select crop varieties that can withstand temporary waterlogging',
      priority: 'medium',
      estimated_cost_pkr: 0,  // No additional cost
      implementation_time_days: 0,
      effectiveness_percent: 50,
      crop_recommendations: [
        { crop: 'rice', varieties: ['IRRI-6', 'Basmati 515', 'Super Basmati'], tolerance: 'high' },
        { crop: 'wheat', varieties: ['Faisalabad-2008', 'Punjab-11'], tolerance: 'medium' },
        { crop: 'sugarcane', varieties: ['CPF-246', 'SPF-213'], tolerance: 'high' }
      ]
    });

    // Strategy 4: Crop insurance
    strategies.push({
      strategy: 'crop_insurance',
      title: 'Get Crop Insurance for Flood Protection',
      description: 'Protect against financial losses from flooding',
      priority: floodRisk === 'extreme' ? 'high' : 'medium',
      estimated_cost_pkr: this.calculateInsurancePremium(fieldData.size_acres, floodRisk),
      implementation_time_days: 3,
      coverage_percent: 80,
      link_to_story: '6.8',
      insurance_providers: [
        'State Bank of Pakistan Agricultural Credit Scheme',
        'National Insurance Company Limited',
        'Punjab Crop Insurance Program'
      ]
    });

    // Strategy 5: Bund construction
    if (floodRisk === 'extreme') {
      strategies.push({
        strategy: 'protective_bunds',
        title: 'Construct Protective Bunds',
        description: 'Build earthen bunds around field to prevent flood water entry',
        priority: 'critical',
        estimated_cost_pkr: fieldData.size_acres * 25000,
        implementation_time_days: 21,
        effectiveness_percent: 80,
        steps: [
          'Survey field perimeter and elevation',
          'Construct earthen bunds 1-1.5m high',
          'Compact bund soil to prevent breaches',
          'Install overflow spillways for excess water',
          'Maintain bunds before monsoon season'
        ],
        requires_permission: true
      });
    }

    return strategies;
  }

  calculateDrainageCost(sizeAcres) {
    // Base cost: 10,000 PKR per acre
    return sizeAcres * 10000;
  }

  calculateInsurancePremium(sizeAcres, floodRisk) {
    // Premium based on risk and field size
    const baseRate = {
      low: 500,
      medium: 1000,
      high: 2000,
      extreme: 3500
    };

    return sizeAcres * (baseRate[floodRisk] || 1000);
  }
}
```

```sql
-- Flood risk assessments
CREATE TABLE flood_risk_assessments (
  id TEXT PRIMARY KEY,
  field_id TEXT NOT NULL,

  -- Location data
  latitude REAL NOT NULL,
  longitude REAL NOT NULL,
  elevation_meters REAL,

  -- Proximity to water
  nearest_river TEXT,
  distance_to_river_km REAL,

  -- Risk assessment
  risk_level TEXT NOT NULL,           -- 'low', 'medium', 'high', 'extreme'
  risk_score INTEGER,                 -- 0-100

  -- Contributing factors
  elevation_risk INTEGER,
  proximity_risk INTEGER,
  historical_risk INTEGER,

  -- Mitigation
  mitigation_strategies TEXT,         -- JSON array

  assessed_at DATETIME NOT NULL,
  data_sources TEXT,                  -- JSON array
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (field_id) REFERENCES fields(id)
);

CREATE INDEX idx_flood_risk_field ON flood_risk_assessments(field_id, assessed_at DESC);
CREATE INDEX idx_flood_risk_level ON flood_risk_assessments(risk_level, assessed_at DESC);

-- River basin forecasts
CREATE TABLE river_basin_forecasts (
  id TEXT PRIMARY KEY,
  river_basin TEXT NOT NULL,          -- 'Indus', 'Ravi', 'Chenab', 'Sutlej', 'Jhelum'
  district TEXT,

  -- Current status
  current_water_level REAL,
  danger_level REAL,
  level_percentage REAL,

  -- Forecast
  flood_probability_percent INTEGER NOT NULL,
  flood_risk TEXT NOT NULL,           -- 'low', 'medium', 'high', 'extreme'
  forecast_period_days INTEGER,
  expected_peak_date DATE,

  -- Contributing factors
  upstream_rainfall_expected_mm REAL,
  dam_release_schedule TEXT,          -- JSON array
  monsoon_intensity TEXT,

  -- Recommendations
  recommendation TEXT,
  alert_level TEXT,                   -- 'green', 'yellow', 'orange', 'red'

  forecast_date DATE NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(river_basin, district, forecast_date)
);

CREATE INDEX idx_river_basin_date ON river_basin_forecasts(river_basin, forecast_date DESC);
CREATE INDEX idx_river_basin_risk ON river_basin_forecasts(flood_risk, forecast_date DESC);

-- Flood history
CREATE TABLE flood_history (
  id TEXT PRIMARY KEY,
  province TEXT,
  district TEXT NOT NULL,
  tehsil TEXT,

  flood_year INTEGER NOT NULL,
  flood_month TEXT,
  start_date DATE,
  end_date DATE,
  duration_days INTEGER,

  severity TEXT NOT NULL,             -- 'minor', 'moderate', 'major', 'catastrophic'
  affected_area_sq_km REAL,
  affected_villages INTEGER,
  estimated_crop_loss_percent INTEGER,

  primary_cause TEXT,                 -- 'monsoon', 'dam_release', 'cyclone', 'glacial_melt'
  peak_water_level REAL,

  damages TEXT,                       -- JSON object
  recovery_time_months INTEGER,
  government_relief_provided BOOLEAN,

  notes TEXT,
  data_source TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(district, flood_year, flood_month)
);

CREATE INDEX idx_flood_history_district ON flood_history(district, flood_year DESC);
CREATE INDEX idx_flood_history_year ON flood_history(flood_year DESC);

-- Elevation data cache
CREATE TABLE elevation_data (
  id TEXT PRIMARY KEY,
  latitude REAL NOT NULL,
  longitude REAL NOT NULL,
  elevation_meters REAL NOT NULL,
  data_source TEXT DEFAULT 'SRTM',
  accuracy_meters INTEGER,
  cached_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(latitude, longitude)
);

CREATE INDEX idx_elevation_location ON elevation_data(latitude, longitude);

-- Emergency contacts
CREATE TABLE emergency_contacts (
  id TEXT PRIMARY KEY,

  email TEXT,
  address TEXT,

CREATE INDEX idx_emergency_contacts_district ON emergency_contacts(district, contact_type);
```

### Security Considerations

[Source: architecture/architectural-principles-and-best-practices.md]
- **API Key Security**: NDMA, IRSA, WAPDA API credentials in environment variables
- **Encryption in Transit**: All API calls use HTTPS/TLS
- **Data Privacy**: Flood risk data is location-based; no user PII exposed
- **Emergency Contact Verification**: Regular updates to ensure accuracy
- **GDPR Compliance**: Users can delete their flood assessments

### Performance Requirements

[Source: architecture/non-functional-specifications.md]
- **Flood Risk Assessment**: <3s for elevation-based risk calculation
- **River Basin Forecast**: <5s for multi-source data aggregation
- **Map Rendering**: <2s for flood risk map display
- **Alert Delivery**: <1 minute for critical flood alerts
- **Offline Access**: Cached elevation data and flood maps available offline

**Optimization Strategies**:
- Pre-calculate elevation data for common agricultural regions
- Cache river basin forecasts (6-hour TTL)
- Store historical flood data locally for offline access
- Compress flood risk maps for mobile delivery
- Use Redis for real-time river level caching
- Background sync for NDMA alerts every 30 minutes

### Project Structure Notes

This story provides critical flood risk management capabilities for farmers in Pakistan's flood-prone regions. By combining elevation data, river basin monitoring, and historical flood patterns, the system delivers actionable flood warnings with 3-7 days lead time. Integration with NDMA and IRSA ensures real-time flood monitoring, while mitigation strategies (drainage, raised beds, insurance) empower farmers to protect their livelihoods. The early harvest alert system (integrated with Story 6.7) can save entire harvests when floods are imminent.
