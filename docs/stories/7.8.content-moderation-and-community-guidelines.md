# Story 7.8: Content Moderation and Community Guidelines

## Status
Draft

## Story
**As a** platform administrator,
**I want** effective moderation tools,
**so that** the community remains safe, respectful, and valuable.

## Acceptance Criteria
1. AC1: Community guidelines clearly stated (respectful communication, no spam, no false information)
2. AC2: User reporting system for inappropriate content
3. AC3: Automated keyword filtering for offensive language
4. AC4: Moderator dashboard for reviewing flagged content
5. AC5: User warnings and suspension system for violations
6. AC6: Appeal process for suspended users
7. AC7: Regular moderation reports and analytics

### Architecture Context

**System Architecture**
[Source: architecture/system-overview.md]
- Moderation as separate microservice
- Content filtering at submission time
- Background jobs for expired bans

**Technology Stack**
[Source: architecture/component-definitions.md]

- **Database**: PostgreSQL for moderation logs
- **ML**: Optional NLP for spam detection

### Technical Implementation Details

**Moderation Schema**

```sql
-- Community guidelines
CREATE TABLE community_guidelines (
  id TEXT PRIMARY KEY,
  version TEXT NOT NULL,
  title_en TEXT NOT NULL,
  title_ur TEXT NOT NULL,
  content_en TEXT NOT NULL,
  content_ur TEXT NOT NULL,
  effective_date DATE NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE guideline_acceptances (
  id TEXT PRIMARY KEY,
  farmer_id TEXT NOT NULL,
  guideline_id TEXT NOT NULL,
  accepted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (farmer_id) REFERENCES farmers(id),
  FOREIGN KEY (guideline_id) REFERENCES community_guidelines(id),
  UNIQUE(farmer_id, guideline_id)
);

-- Content reports
CREATE TABLE content_reports (
  id TEXT PRIMARY KEY,
  reporter_id TEXT NOT NULL,
  content_type TEXT NOT NULL, -- forum_post, forum_reply, group_message, story, profile
  content_id TEXT NOT NULL,
  report_category TEXT NOT NULL, -- spam, harassment, misinformation, inappropriate, violence, other
  report_details TEXT,
  status TEXT DEFAULT 'pending', -- pending, under_review, resolved, dismissed
  priority TEXT DEFAULT 'normal', -- critical, high, normal, low
  reviewed_by TEXT,
  reviewed_at TIMESTAMP,
  resolution TEXT, -- approved, content_removed, user_warned, user_banned, false_report
  resolution_notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (reporter_id) REFERENCES farmers(id),
  FOREIGN KEY (reviewed_by) REFERENCES moderators(id)
);

CREATE INDEX idx_reports_status ON content_reports(status, priority DESC, created_at DESC);
CREATE INDEX idx_reports_content ON content_reports(content_type, content_id);

-- Profanity filters
CREATE TABLE profanity_filters (
  id TEXT PRIMARY KEY,
  language TEXT NOT NULL, -- ur, pa, sd, en
  keyword TEXT NOT NULL,
  severity TEXT DEFAULT 'medium', -- low, medium, high
  action TEXT DEFAULT 'flag', -- flag, block
  is_regex BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(language, keyword)
);

CREATE INDEX idx_filters_language ON profanity_filters(language, severity);

-- Moderation actions
CREATE TABLE moderation_actions (
  id TEXT PRIMARY KEY,
  target_user_id TEXT NOT NULL,
  moderator_id TEXT NOT NULL,
  action_type TEXT NOT NULL, -- warning, content_removal, temporary_ban, permanent_ban
  reason TEXT NOT NULL,
  content_type TEXT,
  content_id TEXT,
  duration_days INTEGER, -- For temporary bans
  expires_at TIMESTAMP,
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (target_user_id) REFERENCES farmers(id),
  FOREIGN KEY (moderator_id) REFERENCES moderators(id)
);

CREATE INDEX idx_actions_user ON moderation_actions(target_user_id, created_at DESC);
CREATE INDEX idx_actions_expires ON moderation_actions(expires_at, is_active);

-- User warnings and strikes
CREATE TABLE user_warnings (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  warning_count INTEGER DEFAULT 0,
  last_warning_at TIMESTAMP,
  is_suspended BOOLEAN DEFAULT FALSE,
  suspension_expires_at TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES farmers(id) ON DELETE CASCADE,
  UNIQUE(user_id)
);

-- User appeals
CREATE TABLE user_appeals (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  moderation_action_id TEXT NOT NULL,
  appeal_text TEXT NOT NULL,
  status TEXT DEFAULT 'pending', -- pending, approved, denied, escalated
  reviewed_by TEXT,
  reviewed_at TIMESTAMP,
  decision TEXT,
  decision_notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES farmers(id),
  FOREIGN KEY (moderation_action_id) REFERENCES moderation_actions(id),
  FOREIGN KEY (reviewed_by) REFERENCES moderators(id)
);

-- Moderators
CREATE TABLE moderators (
  id TEXT PRIMARY KEY,
  farmer_id TEXT NOT NULL UNIQUE,
  role TEXT DEFAULT 'moderator', -- moderator, senior_moderator, admin
  permissions JSONB, -- {can_remove: true, can_ban: true, can_review_appeals: false}
  total_actions INTEGER DEFAULT 0,
  accuracy_rate REAL, -- Percentage of correct decisions
  avg_response_time_hours REAL,
  is_active BOOLEAN DEFAULT TRUE,
  appointed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (farmer_id) REFERENCES farmers(id)
);

-- Moderation logs
CREATE TABLE moderation_logs (
  id TEXT PRIMARY KEY,
  moderator_id TEXT NOT NULL,
  action TEXT NOT NULL,
  target_id TEXT,
  details JSONB,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (moderator_id) REFERENCES moderators(id)
);

CREATE INDEX idx_logs_moderator ON moderation_logs(moderator_id, created_at DESC);
CREATE INDEX idx_logs_date ON moderation_logs(created_at DESC);
```

**Moderation Service**

```javascript
// ModerationService.js
class ModerationService {
  async reportContent(reporterId, contentType, contentId, category, details) {
    const reportId = `report-${Date.now()}-${Math.random()}`;

    // Check if user has reported this content before
    const existing = await db.query(`
      SELECT id FROM content_reports
      WHERE reporter_id = $1 AND content_type = $2 AND content_id = $3
    `, [reporterId, contentType, contentId]);

    if (existing.length > 0) {
      throw new Error('You have already reported this content');
    }

    // Determine priority
    const priority = ['violence', 'harassment'].includes(category) ? 'high' : 'normal';

    await db.query(`
      INSERT INTO content_reports (
        id, reporter_id, content_type, content_id,
        report_category, report_details, priority
      ) VALUES ($1, $2, $3, $4, $5, $6, $7)
    `, [reportId, reporterId, contentType, contentId, category, details, priority]);

    // Check auto-hide threshold
    const reportCount = await db.query(`
      SELECT COUNT(*) as count FROM content_reports
      WHERE content_type = $1 AND content_id = $2 AND status = 'pending'
    `, [contentType, contentId]);

    if (reportCount[0].count >= 5) {
      await this.autoHideContent(contentType, contentId);
    }

    // Notify moderators
    await this.notifyModerators(reportId, priority);

    return reportId;
  }

  async autoHideContent(contentType, contentId) {
    const table = this.getTableName(contentType);

    await db.query(`
      UPDATE ${table}
      SET is_moderated = TRUE, moderation_reason = 'Auto-hidden: Multiple reports'
      WHERE id = $1
    `, [contentId]);
  }

  async scanContent(content, language = 'ur') {
    // Get filters for language
    const filters = await db.query(`
      SELECT keyword, severity, action, is_regex
      FROM profanity_filters
      WHERE language = $1
      ORDER BY severity DESC
    `, [language]);

    const violations = [];

    for (const filter of filters) {
      let found = false;

      if (filter.is_regex) {
        const regex = new RegExp(filter.keyword, 'i');
        found = regex.test(content);
      } else {
        found = content.toLowerCase().includes(filter.keyword.toLowerCase());
      }

      if (found) {
        violations.push({
          keyword: filter.keyword,
          severity: filter.severity,
          action: filter.action
        });

        // If high severity with block action, return immediately
        if (filter.severity === 'high' && filter.action === 'block') {
          return {
            allowed: false,
            reason: 'Content contains prohibited language',
            violations
          };
        }
      }
    }

    // If medium severity violations, flag for review
    const hasMediumViolations = violations.some(v => v.severity === 'medium');
    if (hasMediumViolations) {
      return {
        allowed: true,
        flagged: true,
        violations
      };
    }

    return { allowed: true, flagged: false, violations: [] };
  }

  async reviewReport(reportId, moderatorId, resolution, notes) {
    // Get report details
    const report = await db.query(`
      SELECT * FROM content_reports WHERE id = $1
    `, [reportId]);

    if (report.length === 0) {
      throw new Error('Report not found');
    }

    // Update report
    await db.query(`
      UPDATE content_reports
      SET status = 'resolved',
          reviewed_by = $1,
          reviewed_at = NOW(),
          resolution = $2,
          resolution_notes = $3
      WHERE id = $4
    `, [moderatorId, resolution, notes, reportId]);

    // Take action based on resolution
    const rep = report[0];
    if (resolution === 'content_removed') {
      await this.removeContent(rep.content_type, rep.content_id);
    } else if (resolution === 'user_warned') {
      await this.warnUser(rep.target_user_id, moderatorId, `Violated guidelines: ${notes}`);
    } else if (resolution === 'user_banned') {
      await this.banUser(rep.target_user_id, moderatorId, 7, `Serious violation: ${notes}`);
    }

    // Log action
    await this.logAction(moderatorId, 'review_report', reportId, {
      resolution,
      contentType: rep.content_type,
      contentId: rep.content_id
    });

    // Update moderator stats
    await this.updateModeratorStats(moderatorId);
  }

  async warnUser(userId, moderatorId, reason) {
    // Get or create warning record
    const warning = await db.query(`
      SELECT * FROM user_warnings WHERE user_id = $1
    `, [userId]);

    let warningCount = 1;
    if (warning.length > 0) {
      warningCount = warning[0].warning_count + 1;

      await db.query(`
        UPDATE user_warnings
        SET warning_count = $1, last_warning_at = NOW()
        WHERE user_id = $2
      `, [warningCount, userId]);
    } else {
      await db.query(`
        INSERT INTO user_warnings (id, user_id, warning_count, last_warning_at)
        VALUES ($1, $2, 1, NOW())
      `, [`warning-${Date.now()}-${Math.random()}`, userId]);
    }

    // Create moderation action
    const actionId = `action-${Date.now()}-${Math.random()}`;
    await db.query(`
      INSERT INTO moderation_actions (
        id, target_user_id, moderator_id, action_type, reason
      ) VALUES ($1, $2, $3, 'warning', $4)
    `, [actionId, userId, moderatorId, reason]);

    // Notify user
    await NotificationService.send(userId, {
      type: 'moderation_warning',
      title: 'Community Guideline Warning',
      body: `Warning ${warningCount}/3: ${reason}`,
      data: { actionId }
    });

    // Auto-ban if 3 warnings
    if (warningCount >= 3) {
      await this.banUser(userId, moderatorId, 7, 'Three warnings received');
    }

    return actionId;
  }

  async banUser(userId, moderatorId, durationDays, reason) {
    const actionId = `action-${Date.now()}-${Math.random()}`;
    const expiresAt = durationDays === -1 ? null : new Date(Date.now() + durationDays * 24 * 60 * 60 * 1000);

    await db.query(`
      INSERT INTO moderation_actions (
        id, target_user_id, moderator_id,
        action_type, reason, duration_days, expires_at
      ) VALUES ($1, $2, $3, $4, $5, $6, $7)
    `, [
      actionId, userId, moderatorId,
      durationDays === -1 ? 'permanent_ban' : 'temporary_ban',
      reason, durationDays, expiresAt
    ]);

    // Update user warning status
    await db.query(`
      UPDATE user_warnings
      SET is_suspended = TRUE, suspension_expires_at = $1
      WHERE user_id = $2
    `, [expiresAt, userId]);

    // Notify user
    const banType = durationDays === -1 ? 'permanently banned' : `banned for ${durationDays} days`;
    await NotificationService.send(userId, {
      type: 'moderation_ban',
      title: 'Account Suspended',
      body: `Your account has been ${banType}. Reason: ${reason}`,
      data: { actionId }
    });

    return actionId;
  }

  async submitAppeal(userId, moderationActionId, appealText) {
    // Check if already appealed
    const existing = await db.query(`
      SELECT id FROM user_appeals
      WHERE moderation_action_id = $1
    `, [moderationActionId]);

    if (existing.length > 0) {
      throw new Error('Appeal already submitted for this action');
    }

    const appealId = `appeal-${Date.now()}-${Math.random()}`;

    await db.query(`
      INSERT INTO user_appeals (
        id, user_id, moderation_action_id, appeal_text
      ) VALUES ($1, $2, $3, $4)
    `, [appealId, userId, moderationActionId, appealText]);

    // Notify senior moderators
    await this.notifySeniorModerators(appealId);

    return appealId;
  }

  async reviewAppeal(appealId, moderatorId, decision, notes) {
    await db.query(`
      UPDATE user_appeals
      SET status = $1, reviewed_by = $2, reviewed_at = NOW(),
          decision = $1, decision_notes = $3
      WHERE id = $4
    `, [decision, moderatorId, notes, appealId]);

    // If approved, lift ban
    if (decision === 'approved') {
      const appeal = await db.query(`
        SELECT moderation_action_id FROM user_appeals WHERE id = $1
      `, [appealId]);

      await db.query(`
        UPDATE moderation_actions
        SET is_active = FALSE
        WHERE id = $1
      `, [appeal[0].moderation_action_id]);

      // Reinstate user
      const action = await db.query(`
        SELECT target_user_id FROM moderation_actions WHERE id = $1
      `, [appeal[0].moderation_action_id]);

      await db.query(`
        UPDATE user_warnings
        SET is_suspended = FALSE, suspension_expires_at = NULL
        WHERE user_id = $1
      `, [action[0].target_user_id]);
    }
  }

  async getModerationQueue(moderatorId, filters = {}) {
    const { status = 'pending', priority, contentType } = filters;

    let query = `
      SELECT
        cr.*,
        f.name as reporter_name,
        COUNT(*) OVER(PARTITION BY cr.content_type, cr.content_id) as report_count
      FROM content_reports cr
      JOIN farmers f ON cr.reporter_id = f.id
      WHERE cr.status = $1
    `;

    const params = [status];
    let paramCount = 1;

    if (priority) {
      query += ` AND cr.priority = $${++paramCount}`;
      params.push(priority);
    }

    if (contentType) {
      query += ` AND cr.content_type = $${++paramCount}`;
      params.push(contentType);
    }

    query += ` ORDER BY cr.priority DESC, cr.created_at ASC`;

    const queue = await db.query(query, params);
    return queue;
  }

  async getModerationAnalytics(startDate, endDate) {
    const stats = await db.query(`
      SELECT
        COUNT(*) as total_reports,
        COUNT(CASE WHEN status = 'resolved' THEN 1 END) as resolved_reports,
        COUNT(CASE WHEN resolution = 'content_removed' THEN 1 END) as removed_content,
        COUNT(CASE WHEN resolution = 'user_warned' THEN 1 END) as warnings_issued,
        COUNT(CASE WHEN resolution = 'user_banned' THEN 1 END) as bans_issued,
        AVG(EXTRACT(EPOCH FROM (reviewed_at - created_at)) / 3600) as avg_response_hours
      FROM content_reports
      WHERE created_at BETWEEN $1 AND $2
    `, [startDate, endDate]);

    return stats[0];
  }
}
```
