**MSP Integration Service**

```javascript
// MSPService.js
class MSPService {
  async getMSPPrices(season = null) {
    let query = `
      SELECT
        msp.*,
        mp.modal_price_per_kg as current_market_price,
        mp.price_date as market_price_date
      FROM msp_prices msp
      LEFT JOIN mandi_prices mp ON msp.crop_type = mp.crop_type
        AND mp.price_date = CURRENT_DATE
      WHERE msp.effective_from <= CURRENT_DATE
        AND (msp.effective_to IS NULL OR msp.effective_to >= CURRENT_DATE)
    `;

    if (season) {
      query += ` AND msp.season = $1`;
      return await db.query(query, [season]);
    }

    return await db.query(query);
  }

  async compareMSPWithMarket(cropType) {
    const msp = await db.queryOne(`
      SELECT price_per_kg as msp_price
      FROM msp_prices
      WHERE crop_type = $1
        AND effective_from <= CURRENT_DATE
        AND (effective_to IS NULL OR effective_to >= CURRENT_DATE)
    `, [cropType]);

    if (!msp) {
      return {
        has_msp: false,
        message: 'No MSP available for this crop'
      };
    }

    // Get average market price (last 7 days)
    const marketPrice = await db.queryOne(`
      SELECT AVG(modal_price_per_kg) as avg_market_price
      FROM mandi_prices
      WHERE crop_type = $1
        AND price_date >= CURRENT_DATE - INTERVAL '7 days'
    `, [cropType]);

    const avgMarketPrice = marketPrice?.avg_market_price || 0;
    const priceDifference = msp.msp_price - avgMarketPrice;
    const percentDifference = (priceDifference / avgMarketPrice) * 100;

    let recommendation;
    if (priceDifference > 0) {
      recommendation = {
        action: 'sell_to_government',
        message: `MSP is PKR ${Math.round(priceDifference)} higher than market. Sell to government procurement.`,
        benefit: Math.round(priceDifference)
      };
    } else {
      recommendation = {
        action: 'sell_to_market',
        message: `Market price is PKR ${Math.abs(Math.round(priceDifference))} higher than MSP. Sell to market.`,
        benefit: Math.abs(Math.round(priceDifference))
      };
    }

    return {
      has_msp: true,
      msp_price: msp.msp_price,
      market_price: Math.round(avgMarketPrice),
      difference: Math.round(priceDifference),
      percent_difference: Math.round(percentDifference),
      recommendation
    };
  }

  async alertFarmersAboutFavorableMSP() {
    // Find crops where MSP > market price
    const favorableCrops = await db.query(`
      SELECT
        msp.crop_type,
        msp.price_per_kg as msp_price,
        AVG(mp.modal_price_per_kg) as avg_market_price
      FROM msp_prices msp
      JOIN mandi_prices mp ON msp.crop_type = mp.crop_type
      WHERE msp.effective_from <= CURRENT_DATE
        AND (msp.effective_to IS NULL OR msp.effective_to >= CURRENT_DATE)
        AND mp.price_date >= CURRENT_DATE - INTERVAL '7 days'
      GROUP BY msp.crop_type, msp.price_per_kg
      HAVING msp.price_per_kg > AVG(mp.modal_price_per_kg)
    `);

    for (const crop of favorableCrops) {
      // Find farmers growing this crop
      const farmers = await db.query(`
        SELECT DISTINCT f.id, f.name
        FROM farmers f
        JOIN fields fi ON f.id = fi.farmer_id
        JOIN crop_seasons cs ON fi.id = cs.field_id
        WHERE cs.crop_type = $1 AND cs.status = 'active'
      `, [crop.crop_type]);

      const benefit = Math.round(crop.msp_price - crop.avg_market_price);

      for (const farmer of farmers) {
        await notificationService.sendPushNotification(farmer.id, {
          title: 'Government MSP Opportunity!',
          body: `${crop.crop_type} MSP is PKR ${benefit} higher than market. Sell to government procurement.`,
          data: {
            type: 'msp_opportunity',
            crop_type: crop.crop_type
          }
        });
      }
    }
  }
}
```

**Eligibility Checker Service**

```javascript
// EligibilityCheckerService.js
class EligibilityCheckerService {
  async checkEligibility(userId, schemeId) {
    // Get user profile
    const user = await db.queryOne(`
      SELECT
        f.*,
        SUM(fi.size_acres) as total_land_acres
      FROM farmers f
      LEFT JOIN fields fi ON f.id = fi.farmer_id
      WHERE f.id = $1
      GROUP BY f.id
    `, [userId]);

    // Get user's crops
    const userCrops = await db.query(`
      SELECT DISTINCT cs.crop_type
      FROM crop_seasons cs
      JOIN fields fi ON cs.field_id = fi.id
      WHERE fi.farmer_id = $1 AND cs.status IN ('active', 'harvested')
    `, [userId]);

    // Get scheme eligibility criteria
    const criteria = await db.query(`
      SELECT * FROM scheme_eligibility_criteria
      WHERE scheme_id = $1
    `, [schemeId]);

    const eligibilityResults = {
      is_eligible: true,
      met_criteria: [],
      failed_criteria: [],
      suggestions: []
    };

    for (const criterion of criteria) {
      const result = await this.evaluateCriterion(criterion, user, userCrops);

      if (result.met) {
        eligibilityResults.met_criteria.push(result);
      } else {
        eligibilityResults.is_eligible = false;
        eligibilityResults.failed_criteria.push(result);

        if (result.suggestion) {
          eligibilityResults.suggestions.push(result.suggestion);
        }
      }
    }

    return eligibilityResults;
  }

  async evaluateCriterion(criterion, user, userCrops) {
    const result = {
      criterion_name: criterion.criterion_name,
      criterion_description: criterion.description,
      met: false,
      message: '',
      suggestion: null
    };

    switch (criterion.criterion_type) {
      case 'min_land_size':
        const minSize = criterion.value;
        if (user.total_land_acres >= minSize) {
          result.met = true;
          result.message = `You have ${user.total_land_acres} acres (minimum: ${minSize} acres)`;
        } else {
          result.message = `You have ${user.total_land_acres} acres. Minimum required: ${minSize} acres`;
          result.suggestion = `You need ${minSize - user.total_land_acres} more acres to be eligible`;
        }
        break;

      case 'max_land_size':
        const maxSize = criterion.value;
        if (user.total_land_acres <= maxSize) {
          result.met = true;
          result.message = `You have ${user.total_land_acres} acres (maximum: ${maxSize} acres)`;
        } else {
          result.message = `You have ${user.total_land_acres} acres. Maximum allowed: ${maxSize} acres`;
          result.suggestion = `This scheme is for farmers with up to ${maxSize} acres`;
        }
        break;

      case 'crop_type':
        const requiredCrops = JSON.parse(criterion.value);
        const hasCrop = userCrops.some(c => requiredCrops.includes(c.crop_type));
        if (hasCrop) {
          result.met = true;
          result.message = `You are growing eligible crops: ${requiredCrops.join(', ')}`;
        } else {
          result.message = `You must be growing: ${requiredCrops.join(', ')}`;
          result.suggestion = `Plant one of these crops: ${requiredCrops.join(', ')}`;
        }
        break;

      case 'province':
        const requiredProvinces = JSON.parse(criterion.value);
        if (requiredProvinces.includes(user.province) || requiredProvinces.includes('all')) {
          result.met = true;
          result.message = `Available in your province: ${user.province}`;
        } else {
          result.message = `Only available in: ${requiredProvinces.join(', ')}`;
          result.suggestion = `This scheme is not available in ${user.province}`;
        }
        break;

      case 'registration_required':
        // Check if farmer has government registration
        const hasRegistration = user.government_registration_number;
        if (hasRegistration) {
          result.met = true;
          result.message = 'You have government farmer registration';
        } else {
          result.message = 'Government farmer registration required';
          result.suggestion = 'Register at your nearest agricultural extension office';
        }
        break;

      default:
        result.met = true; // Unknown criteria default to met
    }

    return result;
  }

  async getEligibleSchemes(userId) {
    const allSchemes = await new GovernmentSchemesService().getAllActiveSchemes();

    const eligibleSchemes = [];

    for (const scheme of allSchemes) {
      const eligibility = await this.checkEligibility(userId, scheme.id);

      if (eligibility.is_eligible) {
        eligibleSchemes.push({
          ...scheme,
          eligibility: eligibility
        });
      }
    }

    return eligibleSchemes;
  }
}
```

```javascript
// SchemeDeadlineAlertService.js
const cron = require('node-cron');

class SchemeDeadlineAlertService {
  start() {
    // Check deadlines daily at 10 AM
    cron.schedule('0 10 * * *', async () => {
      await this.checkDeadlines();
    });
  }

  async checkDeadlines() {
    const today = new Date();

    // Deadlines in 30 days
    await this.alertDeadlines(30);

    // Deadlines in 7 days
    await this.alertDeadlines(7);

    // Deadlines tomorrow
    await this.alertDeadlines(1);
  }

  async alertDeadlines(daysAhead) {
    const targetDate = new Date();
    targetDate.setDate(targetDate.getDate() + daysAhead);

    const schemes = await db.query(`
      SELECT * FROM government_schemes
      WHERE status = 'active'
        AND application_deadline::date = $1::date
    `, [targetDate.toISOString().split('T')[0]]);

    for (const scheme of schemes) {
      await this.sendDeadlineAlerts(scheme, daysAhead);
    }
  }

  async sendDeadlineAlerts(scheme, daysAhead) {
    // Get all users who might be eligible
    const users = await this.getPotentiallyEligibleUsers(scheme);

    let urgency, message;

    if (daysAhead === 1) {
      urgency = 'high';
      message = `URGENT: ${scheme.name} deadline is TOMORROW! Apply now.`;
    } else if (daysAhead === 7) {
      urgency = 'medium';
      message = `Reminder: ${scheme.name} deadline in 7 days.`;
    } else {
      urgency = 'low';
      message = `${scheme.name} applications close in ${daysAhead} days.`;
    }

    for (const user of users) {
      await notificationService.sendPushNotification(user.id, {
        title: 'Scheme Deadline Alert',
        body: message,
        priority: urgency,
        data: {
          type: 'scheme_deadline',
          scheme_id: scheme.id,
          days_ahead: daysAhead
        }
      });
    }

    console.log(`Sent deadline alerts for ${scheme.name} to ${users.length} users`);
  }

  async getPotentiallyEligibleUsers(scheme) {
    // Simplified - get all farmers (could be refined with eligibility pre-check)
    return await db.query(`SELECT id, name FROM farmers WHERE active = TRUE`);
  }
}
```

```javascript
// ProcurementCentersService.js
class ProcurementCentersService {
  async getNearestCenters(location, cropType = null, maxDistance = 100) {
    let query = `
      SELECT
        pc.*,
        ST_Distance(
          pc.location_gps::geography,
          ST_MakePoint($1, $2)::geography
        ) / 1000 as distance_km
      FROM procurement_centers pc
      WHERE pc.active = TRUE
        AND ST_DWithin(
          pc.location_gps::geography,
          ST_MakePoint($1, $2)::geography,
          $3 * 1000
        )
    `;

    const params = [location.longitude, location.latitude, maxDistance];

    if (cropType) {
      query += ` AND $4 = ANY(pc.crops_purchased)`;
      params.push(cropType);
    }

    query += ` ORDER BY distance_km ASC LIMIT 10`;

    return await db.query(query, params);
  }

  async getCenterDetail(centerId) {
    const center = await db.queryOne(`
      SELECT * FROM procurement_centers WHERE id = $1
    `, [centerId]);

    if (!center) return null;

    // Get procurement schedule
    const schedule = await db.query(`
      SELECT * FROM procurement_schedule
      WHERE center_id = $1
        AND procurement_date >= CURRENT_DATE
      ORDER BY procurement_date ASC
    `, [centerId]);

    return {
      ...center,
      upcoming_schedule: schedule
    };
  }
}
```

### Data Model Requirements

```sql
-- Government schemes
CREATE TABLE government_schemes (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  category TEXT NOT NULL,  -- 'input_subsidy', 'procurement', 'insurance', 'loan', 'training'
  description TEXT,
  benefits TEXT,
  authority TEXT,  -- 'federal', 'provincial'
  applicable_provinces TEXT[],  -- ['punjab', 'sindh', 'kpk', 'balochistan', 'all']
  application_deadline DATE,
  application_start_date DATE,
  status TEXT DEFAULT 'active',  -- 'active', 'expired', 'upcoming'
  priority INTEGER DEFAULT 0,
  official_url TEXT,
  contact_number TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  expiry_date DATE
);

CREATE INDEX idx_government_schemes_status ON government_schemes(status, priority DESC);

-- Scheme eligibility criteria
CREATE TABLE scheme_eligibility_criteria (
  id TEXT PRIMARY KEY,
  scheme_id TEXT NOT NULL,
  criterion_type TEXT NOT NULL,  -- 'min_land_size', 'max_land_size', 'crop_type', 'province', 'registration_required'
  criterion_name TEXT,
  description TEXT,
  value TEXT,  -- JSON or simple value
  FOREIGN KEY (scheme_id) REFERENCES government_schemes(id)
);

-- Scheme required documents
CREATE TABLE scheme_required_documents (
  id TEXT PRIMARY KEY,
  scheme_id TEXT NOT NULL,
  document_name TEXT NOT NULL,
  description TEXT,
  is_mandatory BOOLEAN DEFAULT TRUE,
  FOREIGN KEY (scheme_id) REFERENCES government_schemes(id)
);

-- Scheme application steps
CREATE TABLE scheme_application_steps (
  id TEXT PRIMARY KEY,
  scheme_id TEXT NOT NULL,
  step_number INTEGER NOT NULL,
  step_title TEXT NOT NULL,
  step_description TEXT,
  FOREIGN KEY (scheme_id) REFERENCES government_schemes(id)
);

-- MSP prices
CREATE TABLE msp_prices (
  id TEXT PRIMARY KEY,
  crop_type TEXT NOT NULL,
  season TEXT,  -- 'rabi', 'kharif'
  price_per_kg DECIMAL(10, 2) NOT NULL,
  price_per_maund DECIMAL(10, 2),  -- 40 kg
  effective_from DATE NOT NULL,
  effective_to DATE,
  announced_by TEXT,  -- 'PASSCO', 'Ministry of Agriculture'
  announcement_date DATE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_msp_prices_crop_date ON msp_prices(crop_type, effective_from DESC);

-- Procurement centers
CREATE TABLE procurement_centers (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  location_gps GEOGRAPHY(POINT) NOT NULL,
  district TEXT,
  province TEXT,
  address TEXT,
  contact_number TEXT,
  contact_person TEXT,
  crops_purchased TEXT[],
  operating_hours TEXT,
  active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_procurement_centers_location ON procurement_centers USING GIST(location_gps);

-- Procurement schedule
CREATE TABLE procurement_schedule (
  id TEXT PRIMARY KEY,
  center_id TEXT NOT NULL,
  crop_type TEXT NOT NULL,
  procurement_date DATE NOT NULL,
  start_time TIME,
  end_time TIME,
  quantity_target_tons DECIMAL(10, 2),
  notes TEXT,
  FOREIGN KEY (center_id) REFERENCES procurement_centers(id)
);

-- Extension offices
CREATE TABLE extension_offices (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  district TEXT NOT NULL,
  tehsil TEXT,
  location_gps GEOGRAPHY(POINT),
  address TEXT,
  contact_number TEXT,
  officer_name TEXT,
  officer_designation TEXT,
  services_offered TEXT[],
  office_hours TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_extension_offices_district ON extension_offices(district);
CREATE INDEX idx_extension_offices_location ON extension_offices USING GIST(location_gps);
```
