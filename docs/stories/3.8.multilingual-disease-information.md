```javascript
// i18n/config.js
import i18n from 'i18next';
import { initReactI18next } from 'react-i18next';
import * as Localization from 'expo-localization';

// Import translation files
import en from './locales/en.json';
import ur from './locales/ur.json';
import pa from './locales/pa.json';
import sd from './locales/sd.json';

i18n
  .use(initReactI18next)
  .init({
    compatibilityJSON: 'v3',
    resources: {
      en: { translation: en },
      ur: { translation: ur },
      pa: { translation: pa },
      sd: { translation: sd }
    },
    lng: Localization.locale.split('-')[0], // Device default
    fallbackLng: 'en',
    interpolation: {
      escapeValue: false // React already escapes
    },
    react: {
      useSuspense: false
    }
  });

export default i18n;
```

**Language Service**

```javascript
// services/LanguageService.js
import i18n from '../i18n/config';
import { I18nManager } from 'react-native';

class LanguageService {
  constructor() {
    this.supportedLanguages = ['en', 'ur', 'pa', 'sd'];
    this.rtlLanguages = ['ur', 'sd'];  // Urdu and Sindhi use RTL
  }

  async setLanguage(languageCode) {
    if (!this.supportedLanguages.includes(languageCode)) {
      console.warn(`Language ${languageCode} not supported`);
      return false;
    }

    // Change i18n language
    await i18n.changeLanguage(languageCode);

    // Update RTL if needed
    const isRTL = this.rtlLanguages.includes(languageCode);
    if (I18nManager.isRTL !== isRTL) {
      I18nManager.forceRTL(isRTL);
      // Note: App restart required for RTL change
      return { requiresRestart: true };
    }

    // Save preference to user profile
    await this.saveLanguagePreference(languageCode);

    return { requiresRestart: false };
  }

  async saveLanguagePreference(languageCode) {
    const farmerId = await AuthService.getCurrentFarmerId();
    await db.update('farmers', { id: farmerId }, {
      language_preference: languageCode
    });
  }

  getCurrentLanguage() {
    return i18n.language;
  }

  getDiseaseName(disease) {
    const lang = this.getCurrentLanguage();
    return disease[`name_${lang}`] || disease.name_en;
  }

  getTreatmentName(treatment) {
    const lang = this.getCurrentLanguage();
    return treatment[`name_${lang}`] || treatment.name_en;
  }

  getLocalizedField(entity, fieldName) {
    const lang = this.getCurrentLanguage();
    const localizedField = `${fieldName}_${lang}`;
    return entity[localizedField] || entity[`${fieldName}_en`] || entity[fieldName];
  }

  translateCategory(category, domain = 'diseases') {
    const key = `${domain}.categories.${category}`;
    return i18n.t(key);
  }
}

export default new LanguageService();
```

**Image Labeling with Localized Text**

```javascript
// services/ImageLabelingService.js
import { Canvas } from 'react-native-canvas';

class ImageLabelingService {
  async addDiseaseLabel(imageUri, diseaseName, language) {
    // Load image
    const image = await this.loadImage(imageUri);

    // Create canvas
    const canvas = new Canvas(image.width, image.height);
    const ctx = canvas.getContext('2d');

    // Draw original image
    ctx.drawImage(image, 0, 0);

    // Add text label
    ctx.font = language === 'ur' || language === 'sd'
      ? '24px NotoNastaliqUrdu'  // RTL font
      : '24px Arial';

    ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
    ctx.fillRect(10, 10, 300, 50);

    ctx.fillStyle = 'white';
    ctx.textAlign = language === 'ur' || language === 'sd' ? 'right' : 'left';
    ctx.fillText(diseaseName, language === 'ur' ? 300 : 20, 40);

    // Export labeled image
    const labeledUri = canvas.toDataURL();
    return labeledUri;
  }

  async loadImage(uri) {
    return new Promise((resolve, reject) => {
      const img = new Image();
      img.onload = () => resolve(img);
      img.onerror = reject;
      img.src = uri;
    });
  }
}
```

**Translation Completeness Tracking**

```javascript
// services/TranslationTracking.js
class TranslationTracking {
  async checkCompleteness(entityType, entityId) {
    const languages = ['ur', 'pa', 'sd'];
    const entity = await this.getEntity(entityType, entityId);

    const results = {};

    for (const lang of languages) {
      const requiredFields = this.getRequiredFields(entityType);
      let filledCount = 0;

      requiredFields.forEach(field => {
        const localizedField = `${field}_${lang}`;
        if (entity[localizedField] && entity[localizedField].trim() !== '') {
          filledCount++;
        }
      });

      const completeness = Math.round((filledCount / requiredFields.length) * 100);

      results[lang] = completeness;

      // Update database
      await db.upsert('translation_status', {
        entity_type: entityType,
        entity_id: entityId,
        language_code: lang,
        completeness_pct: completeness,
        last_updated: new Date()
      });
    }

    return results;
  }

  getRequiredFields(entityType) {
    const fieldMap = {
      disease: ['name', 'description'],
      treatment: ['name', 'description', 'instructions', 'precautions']
    };

    return fieldMap[entityType] || [];
  }
}
```

### Pakistani Language Considerations

**Urdu (اردو)**:
- Official language of Pakistan
- Written right-to-left (RTL)
- Uses Nastaliq script
- Formal vs informal variants

**Punjabi (ਪੰਜਾਬੀ / پنجابی)**:
- Most widely spoken language
- Can use Shahmukhi (Arabic script) or Gurmukhi (Indian script)
- Use Shahmukhi for Pakistan

**Sindhi (سنڌي)**:
- Spoken in Sindh province
- Written in Arabic script (RTL)
- Similar to Urdu but distinct
