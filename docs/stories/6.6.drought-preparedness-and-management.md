```javascript
// DroughtMonitoringService.js
class DroughtMonitoringService {
  /**
   * Calculate Standardized Precipitation Index (SPI)
   * SPI measures precipitation deficit over multiple timescales
   */
  async calculateSPI(locationId, timescaleMonths = 3) {
    // Get historical precipitation data
    const historicalRainfall = await this.getHistoricalRainfall(
      locationId,
      timescaleMonths,
      30  // 30 years of data
    );

    // Calculate mean and standard deviation
    const mean = this.calculateMean(historicalRainfall);
    const stdDev = this.calculateStdDev(historicalRainfall, mean);

    // Get current period rainfall
    const currentRainfall = await this.getCurrentPeriodRainfall(locationId, timescaleMonths);

    // Calculate SPI: (current - mean) / stdDev
    const spi = (currentRainfall - mean) / stdDev;

    return {
      spi_value: spi,
      timescale_months: timescaleMonths,
      classification: this.classifySPI(spi),
      drought_category: this.getDroughtCategory(spi),
      current_rainfall_mm: currentRainfall,
      historical_mean_mm: mean,
      rainfall_deficit_mm: mean - currentRainfall,
      rainfall_deficit_percent: ((mean - currentRainfall) / mean) * 100
    };
  }

  classifySPI(spi) {
    if (spi <= -2.0) return 'Extremely Dry';
    if (spi <= -1.5) return 'Severely Dry';
    if (spi <= -1.0) return 'Moderately Dry';
    if (spi <= -0.5) return 'Abnormally Dry';
    if (spi < 0.5) return 'Near Normal';
    if (spi < 1.0) return 'Abnormally Wet';
    if (spi < 1.5) return 'Moderately Wet';
    if (spi < 2.0) return 'Severely Wet';
    return 'Extremely Wet';
  }

  getDroughtCategory(spi) {
    if (spi >= -0.5) return 'none';
    if (spi >= -1.0) return 'mild';
    if (spi >= -1.5) return 'moderate';
    if (spi >= -2.0) return 'severe';
    return 'extreme';
  }

  /**
   * Calculate Palmer Drought Severity Index (PDSI)
   * PDSI considers temperature, precipitation, and soil moisture
   */
  async calculatePDSI(locationId) {
    // Get weather data
    const weatherData = await this.getWeatherData(locationId, 12);  // 12 months

    let pdsiValue = 0;

    for (const month of weatherData) {
      // Calculate potential evapotranspiration (PET)
      const pet = this.calculatePotentialET(month.temperature_avg_c, month.month);

      // Calculate actual evapotranspiration (AET)
      const aet = Math.min(pet, month.rainfall_mm + month.soil_moisture_mm);

      // Calculate precipitation deficit/surplus
      const precipDeficit = pet - month.rainfall_mm;

      // Water balance components
      const recharge = Math.max(0, month.rainfall_mm - pet);
      const runoff = Math.max(0, recharge - month.soil_water_capacity_mm);
      const loss = Math.max(0, pet - month.rainfall_mm);

      // Accumulate PDSI (simplified)
      pdsiValue += (month.rainfall_mm - pet) / 25;  // Normalization factor
    }

    return {
      pdsi_value: pdsiValue,
      classification: this.classifyPDSI(pdsiValue),
      drought_category: this.getPDSIDroughtCategory(pdsiValue)
    };
  }

  classifyPDSI(pdsi) {
    if (pdsi <= -4.0) return 'Extreme Drought';
    if (pdsi <= -3.0) return 'Severe Drought';
    if (pdsi <= -2.0) return 'Moderate Drought';
    if (pdsi <= -1.0) return 'Mild Drought';
    if (pdsi < 1.0) return 'Near Normal';
    if (pdsi < 2.0) return 'Slight Wetness';
    if (pdsi < 3.0) return 'Moderate Wetness';
    if (pdsi < 4.0) return 'Severe Wetness';
    return 'Extreme Wetness';
  }

  getPDSIDroughtCategory(pdsi) {
    if (pdsi >= -1.0) return 'none';
    if (pdsi >= -2.0) return 'mild';
    if (pdsi >= -3.0) return 'moderate';
    if (pdsi >= -4.0) return 'severe';
    return 'extreme';
  }

  calculatePotentialET(temperature, month) {
    // Simplified Thornthwaite equation for PET
    // PET depends on temperature and day length
    const dayLengthFactor = this.getDayLengthFactor(month);
    const pet = 16 * dayLengthFactor * Math.pow((10 * temperature / 365), 1.5);
    return Math.max(0, pet);
  }

  getDayLengthFactor(month) {
    // Average day length factor for Pakistan (approximate)
    const factors = [0.9, 0.95, 1.0, 1.05, 1.1, 1.1, 1.1, 1.05, 1.0, 0.95, 0.9, 0.85];
    return factors[month - 1] || 1.0;
  }

  /**
   * Composite Drought Risk Index
   * Combines SPI, PDSI, and seasonal forecast
   */
  async calculateDroughtRiskIndex(locationId) {
    // Calculate indices
    const spi3 = await this.calculateSPI(locationId, 3);
    const spi6 = await this.calculateSPI(locationId, 6);
    const pdsi = await this.calculatePDSI(locationId);

    // Get seasonal forecast
    const seasonalForecast = await this.getSeasonalForecast(locationId);

    // Composite risk score (0-100)
    let riskScore = 0;

    // SPI-3 contribution (30%)
    riskScore += this.spiToRiskScore(spi3.spi_value) * 0.3;

    // SPI-6 contribution (25%)
    riskScore += this.spiToRiskScore(spi6.spi_value) * 0.25;

    // PDSI contribution (25%)
    riskScore += this.pdsiToRiskScore(pdsi.pdsi_value) * 0.25;

    // Seasonal forecast contribution (20%)
    if (seasonalForecast.precipitation_outlook === 'below_normal') {
      riskScore += 20;
    } else if (seasonalForecast.precipitation_outlook === 'normal') {
      riskScore += 10;
    }

    // Classify risk level
    const riskLevel = this.classifyRiskLevel(riskScore);

    await db.insert('drought_risk_index', {
      id: `drought-risk-${Date.now()}`,
      location_id: locationId,
      risk_score: riskScore,
      risk_level: riskLevel,
      spi_3_month: spi3.spi_value,
      spi_6_month: spi6.spi_value,
      pdsi: pdsi.pdsi_value,
      seasonal_outlook: seasonalForecast.precipitation_outlook,
      calculated_at: new Date()
    });

    return {
      riskScore,
      riskLevel,
      spi3: spi3.classification,
      spi6: spi6.classification,
      pdsi: pdsi.classification,
      seasonalOutlook: seasonalForecast.precipitation_outlook
    };
  }

  spiToRiskScore(spi) {
    // Convert SPI to risk score (0-100)
    if (spi >= 0) return 0;  // No drought
    if (spi >= -0.5) return 10;  // Abnormally dry
    if (spi >= -1.0) return 30;  // Mild drought
    if (spi >= -1.5) return 60;  // Moderate drought
    if (spi >= -2.0) return 85;  // Severe drought
    return 100;  // Extreme drought
  }

  pdsiToRiskScore(pdsi) {
    // Convert PDSI to risk score (0-100)
    if (pdsi >= 0) return 0;
    if (pdsi >= -1.0) return 10;
    if (pdsi >= -2.0) return 30;
    if (pdsi >= -3.0) return 60;
    if (pdsi >= -4.0) return 85;
    return 100;
  }

  classifyRiskLevel(riskScore) {
    if (riskScore >= 75) return 'extreme';
    if (riskScore >= 50) return 'high';
    if (riskScore >= 25) return 'moderate';
    return 'low';
  }
}
```

```javascript
// DroughtEarlyWarningService.js
class DroughtEarlyWarningService {
  /**
   * Generate 30-60 day drought early warning
   */
  async generateEarlyWarning(userId) {
    const userFields = await this.getUserFields(userId);

    const warnings = [];

    for (const field of userFields) {
      // Get current drought index
      const currentIndex = await this.calculateDroughtRiskIndex(field.location_id);

      // Get seasonal forecast for next 60 days
      const forecast = await this.get60DayForecast(field.location_id);

      // Calculate projected drought risk
      const projectedRisk = this.projectDroughtRisk(currentIndex, forecast);

      // Get active crops
      const activeCrops = await this.getActiveCrops(field.id);

      for (const crop of activeCrops) {
        // Identify critical growth stages in next 60 days
        const criticalStages = this.getCriticalGrowthStages(crop, 60);

        if (criticalStages.length > 0 && projectedRisk.riskLevel !== 'low') {
          const warning = {
            id: `drought-warning-${Date.now()}`,
            user_id: userId,
            field_id: field.id,
            crop_season_id: crop.id,
            warning_type: 'drought',
            severity: projectedRisk.riskLevel,
            lead_time_days: projectedRisk.onset_days,
            drought_probability_percent: projectedRisk.probability,
            affected_growth_stages: criticalStages,
            recommendation: this.generateRecommendation(projectedRisk, crop),
            issued_at: new Date()
          };

          warnings.push(warning);

          // Save warning
          await db.insert('drought_warnings', warning);

          // Send alert
          await this.sendDroughtAlert(userId, warning);
        }
      }
    }

    return warnings;
  }

  projectDroughtRisk(currentIndex, forecast) {
    // Project drought risk 30-60 days ahead
    let probability = 0;
    let onsetDays = null;

    // Current drought status (40% weight)
    if (currentIndex.riskLevel === 'extreme') {
      probability += 40;
      onsetDays = 0;  // Already in drought
    } else if (currentIndex.riskLevel === 'high') {
      probability += 30;
      onsetDays = 15;
    } else if (currentIndex.riskLevel === 'moderate') {
      probability += 20;
      onsetDays = 30;
    }

    // Seasonal forecast (60% weight)
    if (forecast.precipitation_outlook === 'below_normal') {
      probability += 50;
      onsetDays = onsetDays || 30;
    } else if (forecast.precipitation_outlook === 'normal') {
      probability += 20;
      onsetDays = onsetDays || 45;
    } else {
      probability += 5;
    }

    // Temperature forecast contribution
    if (forecast.temperature_outlook === 'above_normal') {
      probability += 10;  // Higher temperatures increase drought risk
    }

    const riskLevel = probability >= 60 ? 'high' : (probability >= 40 ? 'moderate' : 'low');

    return {
      probability: Math.min(100, probability),
      riskLevel,
      onset_days: onsetDays,
      rainfall_deficit_expected_percent: forecast.precipitation_anomaly_percent
    };
  }

  getCriticalGrowthStages(crop, daysAhead) {
    // Identify critical growth stages vulnerable to drought
    const criticalStages = [];

    const daysSincePlanting = this.getDaysSince(crop.planting_date);
    const futureDay = daysSincePlanting + daysAhead;

    // Crop-specific critical stages
    const cropStages = {
      wheat: [
        { name: 'tillering', days: [20, 40], vulnerability: 'high' },
        { name: 'flowering', days: [60, 80], vulnerability: 'critical' },
        { name: 'grain_filling', days: [80, 110], vulnerability: 'critical' }
      ],
      rice: [
        { name: 'tillering', days: [20, 40], vulnerability: 'high' },
        { name: 'panicle_initiation', days: [40, 60], vulnerability: 'critical' },
        { name: 'flowering', days: [70, 90], vulnerability: 'critical' }
      ],
      cotton: [
        { name: 'squaring', days: [40, 60], vulnerability: 'high' },
        { name: 'flowering', days: [60, 90], vulnerability: 'critical' },
        { name: 'boll_development', days: [90, 120], vulnerability: 'high' }
      ]
    };

    const stages = cropStages[crop.crop_type] || [];

    for (const stage of stages) {
      if (futureDay >= stage.days[0] && daysSincePlanting <= stage.days[1]) {
        criticalStages.push({
          stage_name: stage.name,
          vulnerability: stage.vulnerability,
          expected_date: this.addDays(crop.planting_date, stage.days[0])
        });
      }
    }

    return criticalStages;
  }

  generateRecommendation(projectedRisk, crop) {
    const recommendations = [];

    if (projectedRisk.riskLevel === 'high') {
      recommendations.push('Prepare supplemental irrigation immediately');
      recommendations.push('Consider switching to drought-tolerant varieties for next season');
      recommendations.push('Apply mulch to conserve soil moisture');
    } else if (projectedRisk.riskLevel === 'moderate') {
      recommendations.push('Monitor soil moisture levels closely');
      recommendations.push('Plan supplemental irrigation schedule');
      recommendations.push('Implement water conservation practices');
    }

    return recommendations.join('. ');
  }

  async sendDroughtAlert(userId, warning) {
    // Send push notification
    await pushNotificationService.send(userId, {
      title: `Drought Warning: ${warning.severity.toUpperCase()}`,
      body: `Drought risk in ${warning.lead_time_days} days. ${warning.recommendation}`,
      data: { warning_id: warning.id }
    });

    // Send SMS for critical warnings
    if (warning.severity === 'high' || warning.severity === 'extreme') {
      await smsService.send(userId, {
        message: `DROUGHT WARNING: High drought risk in ${warning.lead_time_days} days. ${warning.recommendation}`
      });
    }
  }

  getDaysSince(date) {
    const now = new Date();
    const past = new Date(date);
    return Math.floor((now - past) / (1000 * 60 * 60 * 24));
  }

  addDays(date, days) {
    const result = new Date(date);
    result.setDate(result.getDate() + days);
    return result;
  }
}
```

```javascript
      if (priorityOrder[a.priority] !== priorityOrder[b.priority]) {
        return priorityOrder[a.priority] - priorityOrder[b.priority];
      }
      return b.water_savings_percent - a.water_savings_percent;
    });

    // For extreme drought, recommend top 3-4 practices
    if (droughtSeverity === 'extreme') {
      return practices.slice(0, 4);
    }

    return practices;
  }
}
```
