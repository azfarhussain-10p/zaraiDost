```sql
-- Treatments table
CREATE TABLE treatments (
  id TEXT PRIMARY KEY,
  name_en TEXT NOT NULL,
  name_ur TEXT,
  name_pa TEXT,
  name_sd TEXT,
  type TEXT NOT NULL,  -- chemical, organic, cultural, biological
  category TEXT,  -- fungicide, insecticide, fertilizer, practice
  active_ingredient TEXT,
  cost_category TEXT,  -- low, medium, high
  effectiveness_rating REAL,  -- 0.0 to 1.0
  application_method TEXT,  -- spray, soil_drench, foliar, seed_treatment
  dosage_per_acre TEXT,
  timing_guidance TEXT,
  safety_class TEXT,  -- low_risk, moderate_risk, high_risk
  ppe_required TEXT,  -- JSON array
  reentry_interval_hours INTEGER,
  preharvest_interval_days INTEGER,
  description_en TEXT,
  description_ur TEXT,
  instructions_en TEXT,
  instructions_ur TEXT,
  precautions_en TEXT,
  precautions_ur TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Disease-Treatment linkage
CREATE TABLE disease_treatments (
  disease_id TEXT NOT NULL,
  treatment_id TEXT NOT NULL,
  effectiveness_for_disease REAL,  -- 0.0 to 1.0
  severity_applicable TEXT,  -- mild, moderate, severe, all
  PRIMARY KEY (disease_id, treatment_id),
  FOREIGN KEY (disease_id) REFERENCES diseases(id),
  FOREIGN KEY (treatment_id) REFERENCES treatments(id)
);

-- Treatment products (brands available in Pakistan)
CREATE TABLE treatment_products (
  id TEXT PRIMARY KEY,
  treatment_id TEXT NOT NULL,
  brand_name TEXT NOT NULL,
  manufacturer TEXT,
  package_sizes TEXT,  -- JSON array: ["250ml", "500ml", "1L"]
  estimated_cost_pkr INTEGER,
  active_ingredient_concentration TEXT,
  registration_number TEXT,
  FOREIGN KEY (treatment_id) REFERENCES treatments(id)
);

-- Treatment applications (tracking what farmers applied)
CREATE TABLE treatment_applications (
  id TEXT PRIMARY KEY,
  health_check_id TEXT NOT NULL,
  treatment_id TEXT NOT NULL,
  product_id TEXT,
  application_date DATE NOT NULL,
  application_method TEXT,
  dosage_used TEXT,
  field_size_acres REAL,
  cost_pkr INTEGER,
  effectiveness_rating INTEGER,  -- 1-5 stars, farmer feedback
  notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (health_check_id) REFERENCES health_checks(id),
  FOREIGN KEY (treatment_id) REFERENCES treatments(id),
  FOREIGN KEY (product_id) REFERENCES treatment_products(id)
);

CREATE INDEX idx_treatments_type ON treatments(type);
CREATE INDEX idx_treatments_cost ON treatments(cost_category);
CREATE INDEX idx_disease_treatments_disease ON disease_treatments(disease_id);
CREATE INDEX idx_treatment_applications_health_check ON treatment_applications(health_check_id);
```

**Treatment Recommendation Engine**

```javascript
// TreatmentRecommendationEngine.js
class TreatmentRecommendationEngine {
  async getRecommendations(diseaseId, farmerProfile, severity = 'moderate') {
    // Get all applicable treatments for disease
    const diseaseTreatments = await this.getTreatmentsForDisease(diseaseId, severity);

    // Filter by organic preference
    let filteredTreatments = this.filterByOrganicPreference(
      diseaseTreatments,
      farmerProfile.organic_preference
    );

    // Sort by effectiveness and cost
    filteredTreatments = this.sortTreatments(filteredTreatments, farmerProfile.budget_level);

    // Highlight budget-friendly options
    filteredTreatments = this.flagBudgetFriendly(filteredTreatments, farmerProfile.budget_level);

    // Get product availability
    for (const treatment of filteredTreatments) {
      treatment.products = await this.getAvailableProducts(
        treatment.id,
        farmerProfile.location
      );
      treatment.nearestSupplier = await this.getNearestSupplier(
        treatment.id,
        farmerProfile.location
      );
    }

    // Generate AI-enhanced context (if online)
    const aiContext = await this.generateAIRecommendations(diseaseId, severity, farmerProfile);

    return {
      treatments: filteredTreatments,
      aiContext,
      recommendedTreatment: filteredTreatments[0],  // Top recommendation
      alternativeOptions: filteredTreatments.slice(1, 4)
    };
  }

  async getTreatmentsForDisease(diseaseId, severity) {
    const query = `
      SELECT t.*, dt.effectiveness_for_disease
      FROM treatments t
      JOIN disease_treatments dt ON t.id = dt.treatment_id
      WHERE dt.disease_id = $1
        AND (dt.severity_applicable = $2 OR dt.severity_applicable = 'all')
      ORDER BY dt.effectiveness_for_disease DESC, t.cost_category ASC
    `;

    return await db.query(query, [diseaseId, severity]);
  }

  filterByOrganicPreference(treatments, organicPreference) {
    if (organicPreference === 'organic_only') {
      return treatments.filter(t => t.type === 'organic' || t.type === 'cultural');
    }
    // If no preference or mixed, return all but prioritize organic
    return treatments.sort((a, b) => {
      if (a.type === 'organic' && b.type !== 'organic') return -1;
      if (a.type !== 'organic' && b.type === 'organic') return 1;
      return 0;
    });
  }

  sortTreatments(treatments, budgetLevel) {
    return treatments.sort((a, b) => {
      // First by effectiveness
      if (Math.abs(a.effectiveness_rating - b.effectiveness_rating) > 0.1) {
        return b.effectiveness_rating - a.effectiveness_rating;
      }

      // Then by cost for low-budget farmers
      if (budgetLevel === 'low') {
        const costOrder = { low: 1, medium: 2, high: 3 };
        return costOrder[a.cost_category] - costOrder[b.cost_category];
      }

      return 0;
    });
  }

  flagBudgetFriendly(treatments, budgetLevel) {
    return treatments.map(t => ({
      ...t,
      isBudgetFriendly: t.cost_category === 'low' && budgetLevel === 'low'
    }));
  }

  async getAvailableProducts(treatmentId, location) {
    const query = `
      SELECT tp.*, s.name as supplier_name, s.distance_km
      FROM treatment_products tp
      LEFT JOIN supplier_products sp ON tp.id = sp.product_id
      LEFT JOIN suppliers s ON sp.supplier_id = s.id
      WHERE tp.treatment_id = $1
        AND ST_DWithin(s.location, ST_MakePoint($2, $3)::geography, 50000)  -- 50km radius
      ORDER BY s.distance_km ASC
      LIMIT 5
    `;

    return await db.query(query, [treatmentId, location.longitude, location.latitude]);
  }

  async generateAIRecommendations(diseaseId, severity, farmerProfile) {
    // Call LangChain advisory agent from Story 3.3
    const disease = await DiseaseRepository.getById(diseaseId);

    const prompt = `Provide treatment guidance for ${disease.name_en} (severity: ${severity})
    for a farmer in ${farmerProfile.location}.
    Organic preference: ${farmerProfile.organic_preference}.
    Include: urgency, step-by-step plan, prevention tips.`;

    const aiResponse = await AdvisoryAgent.generate(prompt);
    return aiResponse;
  }
}
```

**Application Instructions Generator**

```javascript
// ApplicationInstructionsGenerator.js
class ApplicationInstructionsGenerator {
  generateInstructions(treatment, fieldSizeAcres, cropStage, weather) {
    const dosage = this.calculateDosage(treatment.dosage_per_acre, fieldSizeAcres);

    const instructions = {
      overview: treatment.instructions_en,
      steps: this.generateSteps(treatment, dosage),
      timing: this.generateTimingGuidance(treatment, cropStage, weather),
      safety: this.generateSafetyPrecautions(treatment),
      equipment: this.generateEquipmentNeeds(treatment),
      environmentalPrecautions: this.generateEnvironmentalPrecautions(weather)
    };

    return instructions;
  }

  calculateDosage(dosagePerAcre, fieldSizeAcres) {
    // Parse dosage string like "200-250ml per acre"
    const match = dosagePerAcre.match(/(\d+)-?(\d*)\s*(\w+)/);
    if (!match) return dosagePerAcre;

    const min = parseInt(match[1]);
    const max = match[2] ? parseInt(match[2]) : min;
    const unit = match[3];

    const totalMin = min * fieldSizeAcres;
    const totalMax = max * fieldSizeAcres;

    return `${totalMin}-${totalMax}${unit} for ${fieldSizeAcres} acres`;
  }

  generateSteps(treatment, dosage) {
    const steps = [
      {
        step: 1,
        instruction: `Prepare ${dosage} of ${treatment.name_en}`,
        details: `Measure the required amount based on your field size`
      },
      {
        step: 2,
        instruction: `Mix with water as per product label`,
        details: `Typically dilute in 80-100 liters of water per acre`
      },
      {
        step: 3,
        instruction: `Apply using ${treatment.application_method}`,
        details: `Ensure complete coverage of affected plants`
      },
      {
        step: 4,
        instruction: `Observe re-entry interval of ${treatment.reentry_interval_hours} hours`,
        details: `Do not enter field until this time has passed`
      }
    ];

    return steps;
  }

  generateTimingGuidance(treatment, cropStage, weather) {
    return {
      bestTimeOfDay: 'Early morning or late evening (avoid midday heat)',
      weatherConditions: 'Calm day with no rain expected for 6 hours',
      avoidIf: ['Windy conditions (>10 km/h)', 'Rain forecast', 'Extreme heat (>35°C)'],
      cropStageNote: cropStage === 'flowering'
        ? 'Avoid spray during flowering to protect pollinators'
        : 'Suitable for current crop stage'
    };
  }

  generateSafetyPrecautions(treatment) {
    const basePrecautions = [
      'Read product label completely before use',
      'Store chemicals in original containers',
      'Keep out of reach of children and animals',
      'Dispose of empty containers properly'
    ];

    const ppeSafety = treatment.ppe_required
      ? JSON.parse(treatment.ppe_required).map(ppe => `Wear ${ppe}`)
      : [];

    const chemicalSafety = treatment.safety_class === 'high_risk'
      ? [
        'Highly toxic - use extreme caution',
        'Avoid skin contact and inhalation',
        'Have emergency contact numbers ready'
      ]
      : treatment.safety_class === 'moderate_risk'
      ? [
        'Avoid prolonged exposure',
        'Wash hands and face after use'
      ]
      : [];

    return [...basePrecautions, ...ppeSafety, ...chemicalSafety];
  }

  generateEquipmentNeeds(treatment) {
    const baseEquipment = ['Spray pump or backpack sprayer', 'Measuring cups/cylinders'];

    const ppe = treatment.ppe_required ? JSON.parse(treatment.ppe_required) : [];

    return [...baseEquipment, ...ppe];
  }

  generateEnvironmentalPrecautions(weather) {
    return [
      'Do not spray near water sources',
      'Avoid spray drift to neighboring fields',
      'Follow local environmental regulations',
      weather?.wind_speed > 10 ? 'WARNING: High wind - postpone application' : null
    ].filter(Boolean);
  }
}
```
