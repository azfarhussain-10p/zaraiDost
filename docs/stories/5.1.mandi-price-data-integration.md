**I want** current market prices from nearby mandis (wholesale markets),
**so that** I know the fair price for my produce.

```javascript
// MandiAPIService.js
class MandiAPIService {
  constructor() {
    this.providers = [
      new AgmarknetPakistanAPI(),
      new PunjabAgriAPI(),
      new SindhAgriAPI()
    ];
  }

  async fetchMandiPrices(date = new Date()) {
    const allPrices = [];

    // Try each provider
    for (const provider of this.providers) {
      try {
        const prices = await provider.fetchPrices(date);
        allPrices.push(...prices);
      } catch (error) {
        console.error(`Failed to fetch from ${provider.name}:`, error);
        await this.logAPIFailure(provider.name, error);
      }
    }

    return allPrices;
  }

  async logAPIFailure(provider, error) {
    await db.insert('mandi_sync_log', {
      id: `sync-${Date.now()}`,
      provider_name: provider,
      sync_date: new Date(),
      status: 'failed',
      error_message: error.message,
      created_at: new Date()
    });
  }
}

// Agmarknet Pakistan API implementation
class AgmarknetPakistanAPI {
  constructor() {
    this.name = 'Agmarknet Pakistan';
    this.baseURL = process.env.AGMARKNET_API_URL;
    this.apiKey = process.env.AGMARKNET_API_KEY;
  }

  async fetchPrices(date) {
    try {
      const response = await axios.get(`${this.baseURL}/prices`, {
        params: {
          date: date.toISOString().split('T')[0],
          api_key: this.apiKey
        }
      });

      return this.transformResponse(response.data);
    } catch (error) {
      throw new Error(`Agmarknet API error: ${error.message}`);
    }
  }

  transformResponse(data) {
    // Transform API response to standard format
    return data.prices.map(price => ({
      mandi_id: price.market_id,
      crop_type: this.normalizeCropName(price.commodity),
      variety: price.variety || null,
      min_price_per_kg: this.convertToKg(price.min_price, price.unit),
      max_price_per_kg: this.convertToKg(price.max_price, price.unit),
      modal_price_per_kg: this.convertToKg(price.modal_price, price.unit),
      price_date: new Date(price.arrival_date),
      unit: 'kg',
      source: this.name
    }));
  }

  normalizeCropName(commodity) {
    // Map API crop names to our standard names
    const mapping = {
      'Wheat': 'wheat',
      'Rice (Basmati)': 'rice',
      'Cotton': 'cotton',
      'Sugarcane': 'sugarcane',
      'Maize': 'corn'
    };
    return mapping[commodity] || commodity.toLowerCase();
  }

  convertToKg(price, unit) {
    // Convert prices to per kg base
    const conversions = {
      'quintal': price / 100,     // 100 kg
      'maund': price / 40,        // 40 kg
      'kg': price,
      '40kg': price / 40
    };
    return conversions[unit.toLowerCase()] || price;
  }
}
```

**Price Update Scheduler**

```javascript
// PriceUpdateScheduler.js
const cron = require('node-cron');

class PriceUpdateScheduler {
  constructor() {
    this.mandiAPIService = new MandiAPIService();
    this.priceRepository = new PriceRepository();
  }

  start() {
    // Run daily at 6:00 AM (after mandi reporting)
    cron.schedule('0 6 * * *', async () => {
      console.log('Starting daily mandi price update...');
      await this.updatePrices();
    });

    console.log('Price update scheduler started');
  }

  async updatePrices() {
    try {
      const prices = await this.mandiAPIService.fetchMandiPrices();

      console.log(`Fetched ${prices.length} price records`);

      // Store prices in database
      for (const price of prices) {
        await this.priceRepository.insertPrice(price);
      }

      // Update cache for offline access
      await this.updatePriceCache(prices);

      // Log successful sync
      await this.logSyncSuccess(prices.length);

      console.log('Price update completed successfully');
    } catch (error) {
      console.error('Price update failed:', error);
      await this.logSyncFailure(error);

      // Check if failures exceed threshold
      await this.checkConsecutiveFailures();
    }
  }

  async updatePriceCache(prices) {
    // Group prices by crop and mandi for offline cache
    const cacheData = {};

    prices.forEach(price => {
      const key = `${price.crop_type}_${price.mandi_id}`;
      if (!cacheData[key] || cacheData[key].price_date < price.price_date) {
        cacheData[key] = price;
      }
    });

    // Store in Redis cache
    for (const [key, price] of Object.entries(cacheData)) {
      await redis.setex(`price:${key}`, 86400 * 7, JSON.stringify(price)); // 7 day TTL
    }

    // Also store in SQLite for offline mobile access
    await this.syncToLocalDB(Object.values(cacheData));
  }

  async syncToLocalDB(prices) {
    // Sync to price_cache table for offline mobile access
    await db.transaction(async (trx) => {
      await trx('price_cache').del(); // Clear old cache

      for (const price of prices) {
        await trx('price_cache').insert({
          id: `cache-${price.mandi_id}-${price.crop_type}`,
          mandi_id: price.mandi_id,
          crop_type: price.crop_type,
          min_price_per_kg: price.min_price_per_kg,
          max_price_per_kg: price.max_price_per_kg,
          modal_price_per_kg: price.modal_price_per_kg,
          price_date: price.price_date,
          cached_at: new Date()
        });
      }
    });
  }

  async logSyncSuccess(recordCount) {
    await db.insert('mandi_sync_log', {
      id: `sync-${Date.now()}`,
      provider_name: 'all',
      sync_date: new Date(),
      status: 'success',
      records_fetched: recordCount,
      created_at: new Date()
    });
  }

  async logSyncFailure(error) {
    await db.insert('mandi_sync_log', {
      id: `sync-${Date.now()}`,
      provider_name: 'all',
      sync_date: new Date(),
      status: 'failed',
      error_message: error.message,
      created_at: new Date()
    });
  }

  async checkConsecutiveFailures() {
    const recentLogs = await db.query(`
      SELECT * FROM mandi_sync_log
      WHERE provider_name = 'all'
      ORDER BY sync_date DESC
      LIMIT 2
    `);

    const consecutiveFailures = recentLogs.every(log => log.status === 'failed');

    if (consecutiveFailures) {
      // Send alert to admin
      await this.sendAdminAlert('Mandi price sync has failed for 2+ days');
    }
  }

  async sendAdminAlert(message) {
    // Integration with alert service (email, Slack, etc.)

    // TODO: Implement actual alerting
  }
}
```

**Nearest Mandi Detection**

```javascript
// NearestMandiService.js
class NearestMandiService {
  async findNearestMandis(userLocation, cropType, maxDistance = 200) {
    // Query mandis within maxDistance (km) with prices for crop
    const mandis = await db.query(`
      SELECT
        m.id,
        m.name,
        m.district,
        m.province,
        m.location_gps,
        mp.min_price_per_kg,
        mp.max_price_per_kg,
        mp.modal_price_per_kg,
        mp.price_date,
        ST_Distance(
          m.location_gps::geography,
          ST_MakePoint($1, $2)::geography
        ) / 1000 as distance_km
      FROM mandis m
      LEFT JOIN mandi_prices mp ON m.id = mp.mandi_id
        AND mp.crop_type = $3
        AND mp.price_date = (
          SELECT MAX(price_date) FROM mandi_prices WHERE mandi_id = m.id
        )
      WHERE ST_DWithin(
        m.location_gps::geography,
        ST_MakePoint($1, $2)::geography,
        $4 * 1000
      )
      ORDER BY distance_km ASC
      LIMIT 10
    `, [userLocation.longitude, userLocation.latitude, cropType, maxDistance]);

    // Enrich with travel time estimates
    const enrichedMandis = await Promise.all(
      mandis.map(async (mandi) => ({
        ...mandi,
        travel_time_minutes: await this.estimateTravelTime(userLocation, mandi.location_gps),
        distance_display: `${Math.round(mandi.distance_km)} km`
      }))
    );

    return enrichedMandis;
  }

  async estimateTravelTime(from, to) {
    // Simple estimate: average 40 km/hr travel speed
    const distance = await this.calculateDistance(from, to);
    return Math.round((distance / 40) * 60); // minutes
  }

  async calculateDistance(from, to) {
    // Haversine formula for distance calculation
    const R = 6371; // Earth radius in km
    const dLat = this.toRad(to.latitude - from.latitude);
    const dLon = this.toRad(to.longitude - from.longitude);

    const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
              Math.cos(this.toRad(from.latitude)) * Math.cos(this.toRad(to.latitude)) *
              Math.sin(dLon / 2) * Math.sin(dLon / 2);

    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    return R * c;
  }

  toRad(degrees) {
    return degrees * (Math.PI / 180);
  }

  async getUserFavoriteMandis(userId) {
    return await db.query(`
      SELECT m.*, fm.added_at
      FROM favorite_mandis fm
      JOIN mandis m ON fm.mandi_id = m.id
      WHERE fm.user_id = $1
      ORDER BY fm.added_at DESC
    `, [userId]);
  }

  async addFavoriteMandi(userId, mandiId) {
    await db.insert('favorite_mandis', {
      id: `fav-${Date.now()}`,
      user_id: userId,
      mandi_id: mandiId,
      added_at: new Date()
    });
  }
}
```

```javascript
// PriceUnitConverter.js
class PriceUnitConverter {
  constructor() {
    this.units = {
      kg: { multiplier: 1, label: 'per kg' },
      maund: { multiplier: 40, label: 'per maund (40 kg)' },
      quintal: { multiplier: 100, label: 'per quintal (100 kg)' },
      mann: { multiplier: 40, label: 'per mann' } // Alternative spelling
    };
  }

  convert(pricePerKg, toUnit) {
    const unit = this.units[toUnit.toLowerCase()];
    if (!unit) {
      throw new Error(`Unsupported unit: ${toUnit}`);
    }

    return {
      value: Math.round(pricePerKg * unit.multiplier),
      unit: toUnit,
      label: unit.label,
      per_kg_equivalent: pricePerKg
    };
  }

  formatPrice(pricePerKg, unit, currency = 'PKR') {
    const converted = this.convert(pricePerKg, unit);
    return `${currency} ${converted.value.toLocaleString()} ${converted.label}`;
  }

  getDefaultUnitForCrop(cropType) {
    // Crop-specific pricing conventions in Pakistan
    const conventions = {
      wheat: 'maund',
      rice: 'maund',
      cotton: 'maund',
      sugarcane: 'maund',
      vegetables: 'kg',
      fruits: 'kg'
    };
    return conventions[cropType] || 'kg';
  }
}
```

**Historical Price Trends**

```javascript
// PriceTrendAnalyzer.js
class PriceTrendAnalyzer {
  async getHistoricalPrices(mandiId, cropType, months = 3) {
    const startDate = new Date();
    startDate.setMonth(startDate.getMonth() - months);

    const prices = await db.query(`
      SELECT
        price_date,
        min_price_per_kg,
        max_price_per_kg,
        modal_price_per_kg
      FROM mandi_prices
      WHERE mandi_id = $1
        AND crop_type = $2
        AND price_date >= $3
      ORDER BY price_date ASC
    `, [mandiId, cropType, startDate]);

    return prices;
  }

  async calculateWeeklyAverages(prices) {
    const weeklyData = {};

    prices.forEach(price => {
      const week = this.getWeekKey(price.price_date);

      if (!weeklyData[week]) {
        weeklyData[week] = {
          week,
          prices: [],
          count: 0
        };
      }

      weeklyData[week].prices.push(price.modal_price_per_kg);
      weeklyData[week].count++;
    });

    return Object.values(weeklyData).map(week => ({
      week: week.week,
      average_price: week.prices.reduce((a, b) => a + b, 0) / week.count,
      count: week.count
    }));
  }

  getWeekKey(date) {
    const d = new Date(date);
    const year = d.getFullYear();
    const week = this.getWeekNumber(d);
    return `${year}-W${week}`;
  }

  getWeekNumber(date) {
    const d = new Date(date);
    d.setHours(0, 0, 0, 0);
    d.setDate(d.getDate() + 4 - (d.getDay() || 7));
    const yearStart = new Date(d.getFullYear(), 0, 1);
    return Math.ceil((((d - yearStart) / 86400000) + 1) / 7);
  }

  async calculatePriceChange(currentPrice, historicalPrices) {
    const lastWeekPrice = historicalPrices[historicalPrices.length - 8]; // ~7 days ago
    const lastMonthPrice = historicalPrices[historicalPrices.length - 30];

    const weekChange = lastWeekPrice
      ? ((currentPrice - lastWeekPrice.modal_price_per_kg) / lastWeekPrice.modal_price_per_kg) * 100
      : 0;

    const monthChange = lastMonthPrice
      ? ((currentPrice - lastMonthPrice.modal_price_per_kg) / lastMonthPrice.modal_price_per_kg) * 100
      : 0;

    return {
      week_change_percent: Math.round(weekChange * 10) / 10,
      month_change_percent: Math.round(monthChange * 10) / 10,
      trend: weekChange > 2 ? 'rising' : weekChange < -2 ? 'falling' : 'stable'
    };
  }

  async identifyHighLowPrices(prices) {
    if (!prices.length) return null;

    const sorted = [...prices].sort((a, b) => a.modal_price_per_kg - b.modal_price_per_kg);

    return {
      highest: sorted[sorted.length - 1],
      lowest: sorted[0],
      average: prices.reduce((sum, p) => sum + p.modal_price_per_kg, 0) / prices.length
    };
  }
}
```

```sql
-- Mandis (wholesale markets)
CREATE TABLE mandis (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  district TEXT NOT NULL,
  province TEXT NOT NULL,
  location_gps GEOGRAPHY(POINT),
  address TEXT,
  contact_number TEXT,
  operational_days TEXT, -- e.g., "Mon-Sat"
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_mandis_location ON mandis USING GIST(location_gps);
CREATE INDEX idx_mandis_district ON mandis(district);

-- Mandi prices
CREATE TABLE mandi_prices (
  id TEXT PRIMARY KEY,
  mandi_id TEXT NOT NULL,
  crop_type TEXT NOT NULL,
  variety TEXT,
  min_price_per_kg DECIMAL(10, 2) NOT NULL,
  max_price_per_kg DECIMAL(10, 2) NOT NULL,
  modal_price_per_kg DECIMAL(10, 2) NOT NULL, -- Average/typical price
  unit TEXT DEFAULT 'kg',
  price_date DATE NOT NULL,
  source TEXT, -- API source name
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (mandi_id) REFERENCES mandis(id),
  UNIQUE(mandi_id, crop_type, price_date)
);

CREATE INDEX idx_mandi_prices_crop_date ON mandi_prices(crop_type, price_date DESC);
CREATE INDEX idx_mandi_prices_mandi ON mandi_prices(mandi_id, price_date DESC);

-- Price cache for offline access
CREATE TABLE price_cache (
  id TEXT PRIMARY KEY,
  mandi_id TEXT NOT NULL,
  crop_type TEXT NOT NULL,
  min_price_per_kg DECIMAL(10, 2),
  max_price_per_kg DECIMAL(10, 2),
  modal_price_per_kg DECIMAL(10, 2),
  price_date DATE,
  cached_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_price_cache_mandi_crop ON price_cache(mandi_id, crop_type);

-- Sync log for monitoring
CREATE TABLE mandi_sync_log (
  id TEXT PRIMARY KEY,
  provider_name TEXT NOT NULL,
  sync_date DATE NOT NULL,
  status TEXT NOT NULL, -- 'success', 'failed'
  records_fetched INTEGER,
  error_message TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_mandi_sync_log_date ON mandi_sync_log(sync_date DESC);

-- User favorite mandis
CREATE TABLE favorite_mandis (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  mandi_id TEXT NOT NULL,
  added_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (mandi_id) REFERENCES mandis(id),
  UNIQUE(user_id, mandi_id)
);

CREATE INDEX idx_favorite_mandis_user ON favorite_mandis(user_id);
```
