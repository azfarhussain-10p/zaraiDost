**Climate-Crop Suitability Engine**

```javascript
// ClimateSmartCropSelector.js
class ClimateSmartCropSelector {
  async recommendCrops(userId, fieldId, seasonType) {
    // Get user's field and location data
    const field = await db.queryOne(
      'SELECT * FROM fields WHERE id = $1',
      [fieldId]
    );

    // Get seasonal climate forecast for user's location
    const climateForecast = await this.getSeasonalForecast(
      field.latitude,
      field.longitude,
      seasonType
    );

    // Get crop climate requirements
    const allCrops = await this.getAllCropRequirements(seasonType);

    // Calculate suitability for each crop
    const recommendations = [];
    for (const crop of allCrops) {
      const suitability = await this.calculateSuitability(
        crop,
        climateForecast,
        field
      );

      if (suitability.score >= 50) {  // Only recommend if >50% suitable
        recommendations.push({
          crop_name: crop.name,
          crop_type: crop.type,
          suitability_score: suitability.score,
          suitability_factors: suitability.factors,
          expected_yield_kg_per_acre: suitability.yield,
          risk_level: suitability.risk,
          recommendation_reason: suitability.reason
        });
      }
    }

    // Sort by suitability score
    recommendations.sort((a, b) => b.suitability_score - a.suitability_score);

    // Store recommendations
    await this.saveRecommendations(userId, fieldId, recommendations);

    return recommendations.slice(0, 5);  // Top 5
  }

  async calculateSuitability(crop, climateForecast, field) {
    let score = 100;
    let factors = [];

    // Temperature suitability
    const tempFit = this.checkTemperatureRange(
      crop.optimal_temp_min_c,
      crop.optimal_temp_max_c,
      climateForecast.temperature_forecast_c
    );
    score *= tempFit.score;
    factors.push({
      factor: 'temperature',
      score: tempFit.score * 100,
      description: tempFit.description
    });

    // Rainfall suitability
    const rainfallFit = this.checkRainfallRange(
      crop.rainfall_requirement_mm,
      climateForecast.rainfall_forecast_mm,
      field.irrigation_type
    );
    score *= rainfallFit.score;
    factors.push({
      factor: 'rainfall',
      score: rainfallFit.score * 100,
      description: rainfallFit.description
    });

    // Soil suitability
    const soilFit = this.checkSoilType(
      crop.suitable_soil_types,
      field.soil_type
    );
    score *= soilFit.score;
    factors.push({
      factor: 'soil',
      score: soilFit.score * 100,
      description: soilFit.description
    });

    // Drought resistance
    if (climateForecast.drought_risk === 'high' || climateForecast.drought_risk === 'extreme') {
      const droughtFit = crop.drought_resistance_score / 100;
      score *= droughtFit;
      factors.push({
        factor: 'drought_risk',
        score: droughtFit * 100,
        description: `Drought risk is ${climateForecast.drought_risk}. This crop has ${crop.drought_resistance_score}% drought tolerance.`
      });
    }

    // Flood tolerance
    if (climateForecast.flood_risk === 'high' || climateForecast.flood_risk === 'extreme') {
      const floodFit = crop.flood_tolerance_score / 100;
      score *= floodFit;
      factors.push({
        factor: 'flood_risk',
        score: floodFit * 100,
        description: `Flood risk is ${climateForecast.flood_risk}. This crop has ${crop.flood_tolerance_score}% flood tolerance.`
      });
    }

    // Expected yield calculation
    const expectedYield = this.predictYield(crop, climateForecast, field);

    // Risk level
    const riskLevel = this.calculateRiskLevel(score, climateForecast);

    // Recommendation reason
    const reason = this.generateRecommendationReason(crop, factors, climateForecast);

    return {
      score: Math.round(score),
      factors,
      yield: expectedYield,
      risk: riskLevel,
      reason
    };
  }

  checkTemperatureRange(optimalMin, optimalMax, forecastTemp) {
    if (forecastTemp >= optimalMin && forecastTemp <= optimalMax) {
      return {
        score: 1.0,
        description: `Temperature (${forecastTemp}°C) is optimal`
      };
    }

    const deviation = Math.min(
      Math.abs(forecastTemp - optimalMin),
      Math.abs(forecastTemp - optimalMax)
    );

    if (deviation <= 3) {
      return {
        score: 0.9,
        description: `Temperature (${forecastTemp}°C) is near optimal`
      };
    } else if (deviation <= 5) {
      return {
        score: 0.7,
        description: `Temperature (${forecastTemp}°C) is marginally suitable`
      };
    } else {
      return {
        score: 0.4,
        description: `Temperature (${forecastTemp}°C) is suboptimal`
      };
    }
  }

  checkRainfallRange(requiredRainfall, forecastRainfall, irrigationType) {
    if (irrigationType === 'drip' || irrigationType === 'tube_well') {
      // Irrigated fields are less dependent on rainfall
      return {
        score: 1.0,
        description: 'Irrigation available; rainfall less critical'
      };
    }

    const rainfallRatio = forecastRainfall / requiredRainfall;

    if (rainfallRatio >= 0.8 && rainfallRatio <= 1.2) {
      return {
        score: 1.0,
        description: `Rainfall (${forecastRainfall}mm) is adequate`
      };
    } else if (rainfallRatio >= 0.6 && rainfallRatio <= 1.5) {
      return {
        score: 0.8,
        description: `Rainfall (${forecastRainfall}mm) is acceptable with supplemental irrigation`
      };
    } else if (rainfallRatio < 0.6) {
      return {
        score: 0.5,
        description: `Rainfall (${forecastRainfall}mm) is low; significant irrigation needed`
      };
    } else {
      return {
        score: 0.6,
        description: `Rainfall (${forecastRainfall}mm) is high; ensure drainage`
      };
    }
  }

  checkSoilType(suitableSoils, fieldSoil) {
    if (suitableSoils.includes(fieldSoil)) {
      return {
        score: 1.0,
        description: `${fieldSoil} soil is ideal for this crop`
      };
    } else {
      return {
        score: 0.7,
        description: `${fieldSoil} soil is acceptable but not ideal`
      };
    }
  }

  calculateRiskLevel(suitabilityScore, climateForecast) {
    if (suitabilityScore >= 80 && climateForecast.drought_risk === 'low' && climateForecast.flood_risk === 'low') {
      return 'low';
    } else if (suitabilityScore >= 60) {
      return 'moderate';
    } else if (suitabilityScore >= 40) {
      return 'high';
    } else {
      return 'very_high';
    }
  }

  generateRecommendationReason(crop, factors, climateForecast) {
    const topFactors = factors
      .filter(f => f.score >= 80)
      .map(f => f.factor)
      .join(', ');

    let reason = `${crop.name} is recommended because: `;

    if (topFactors) {
      reason += `excellent fit for ${topFactors}`;
    }

    if (climateForecast.drought_risk === 'high' && crop.drought_resistance_score >= 70) {
      reason += `. Drought-resistant variety suitable for low rainfall forecast`;
    }

    if (climateForecast.flood_risk === 'high' && crop.flood_tolerance_score >= 70) {
      reason += `. Flood-tolerant variety suitable for high rainfall forecast`;
    }

    return reason;
  }
}
```

**Yield Prediction Model**

```javascript
// YieldPredictionService.js
class YieldPredictionService {
  async predictYield(crop, climateForecast, field) {
    // Get historical yield data for this crop in similar conditions
    const historicalData = await db.query(`
      SELECT AVG(yield_kg_per_acre) as avg_yield
      FROM crop_yield_model
      WHERE crop_type = $1
        AND region = $2
        AND temperature_avg_c BETWEEN $3 AND $4
        AND rainfall_mm BETWEEN $5 AND $6
    `, [
      crop.type,
      field.province,
      climateForecast.temperature_forecast_c - 2,
      climateForecast.temperature_forecast_c + 2,
      climateForecast.rainfall_forecast_mm - 100,
      climateForecast.rainfall_forecast_mm + 100
    ]);

    let baseYield = historicalData[0]?.avg_yield || crop.typical_yield_kg_per_acre;

    // Adjust yield based on suitability
    const suitability = await this.calculateSuitability(crop, climateForecast, field);
    const yieldAdjustment = suitability.score / 100;

    const predictedYield = baseYield * yieldAdjustment;

    // Calculate yield range (uncertainty)
    const yieldMin = predictedYield * 0.8;
    const yieldMax = predictedYield * 1.2;

    return {
      predicted_yield_kg: Math.round(predictedYield),
      yield_min_kg: Math.round(yieldMin),
      yield_max_kg: Math.round(yieldMax),
      confidence_percent: 70  // Based on forecast confidence
    };
  }
}
```

```javascript
// ProfitabilityAnalysisService.js
class ProfitabilityAnalysisService {
  async analyzeProfitability(crop, field, yieldPrediction) {
    // Get input costs
    const costs = await this.getCropCosts(crop.type);

    // Get market price
    const marketPrice = await this.getMarketPrice(crop.type, field.province);

    // Calculate revenue
    const revenue = yieldPrediction.predicted_yield_kg * marketPrice.price_per_kg;
    const revenueMin = yieldPrediction.yield_min_kg * marketPrice.price_per_kg;
    const revenueMax = yieldPrediction.yield_max_kg * marketPrice.price_per_kg;

    // Calculate total costs
    const totalCosts = this.calculateTotalCosts(costs, field.size_acres);

    // Calculate profit
    const profit = revenue - totalCosts;
    const profitMin = revenueMin - totalCosts;
    const profitMax = revenueMax - totalCosts;

    // Calculate ROI
    const roi = (profit / totalCosts) * 100;

    return {
      crop_name: crop.name,
      field_size_acres: field.size_acres,

      costs: {
        seed: costs.seed_per_acre * field.size_acres,
        fertilizer: costs.fertilizer_per_acre * field.size_acres,
        pesticide: costs.pesticide_per_acre * field.size_acres,
        labor: costs.labor_per_acre * field.size_acres,
        irrigation: costs.irrigation_per_acre * field.size_acres,
        total: totalCosts
      },

      revenue: {
        predicted: revenue,
        min: revenueMin,
        max: revenueMax,
        market_price_per_kg: marketPrice.price_per_kg
      },

      profit: {
        predicted: profit,
        min: profitMin,
        max: profitMax,
        roi_percent: roi
      },

      break_even_yield_kg: totalCosts / marketPrice.price_per_kg
    };
  }

  async getCropCosts(cropType) {
    const costs = await db.queryOne(
      'SELECT * FROM crop_economics WHERE crop_type = $1',
      [cropType]
    );

    return costs || {
      seed_per_acre: 1000,
      fertilizer_per_acre: 3000,
      pesticide_per_acre: 1500,
      labor_per_acre: 5000,
      irrigation_per_acre: 2000
    };
  }

  async getMarketPrice(cropType, province) {
    // Integration with mandi APIs (Epic 5)
    // Fallback to average price if API unavailable

    const price = await db.queryOne(`
      SELECT AVG(price_per_kg) as avg_price
      FROM market_prices
      WHERE crop_type = $1 AND province = $2
        AND date >= NOW() - INTERVAL '30 days'
    `, [cropType, province]);

    return {
      price_per_kg: price?.avg_price || 50,  // Default fallback
      currency: 'PKR',
      last_updated: new Date()
    };
  }

  calculateTotalCosts(costs, acres) {
    return (
      costs.seed_per_acre +
      costs.fertilizer_per_acre +
      costs.pesticide_per_acre +
      costs.labor_per_acre +
      costs.irrigation_per_acre
    ) * acres;
  }
}
```

```sql
-- Crop climate requirements
CREATE TABLE crop_climate_requirements (
  id TEXT PRIMARY KEY,
  crop_name TEXT NOT NULL,
  crop_type TEXT NOT NULL,              -- 'cereal', 'pulse', 'cash_crop', 'vegetable'
  variety TEXT,

  optimal_temp_min_c REAL,
  optimal_temp_max_c REAL,
  rainfall_requirement_mm REAL,
  suitable_soil_types TEXT,             -- JSON array: ['loamy', 'clay']

  drought_resistance_score INTEGER,     -- 0-100
  flood_tolerance_score INTEGER,        -- 0-100
  heat_stress_tolerance INTEGER,        -- 0-100

  typical_yield_kg_per_acre REAL,
  growing_period_days INTEGER,

  description_en TEXT,
  description_ur TEXT,

  UNIQUE(crop_name, variety)
);

CREATE INDEX idx_crop_climate_type ON crop_climate_requirements(crop_type);

-- Crop economics
CREATE TABLE crop_economics (
  id TEXT PRIMARY KEY,
  crop_type TEXT NOT NULL,
  province TEXT,

  seed_cost_per_acre REAL,
  fertilizer_cost_per_acre REAL,
  pesticide_cost_per_acre REAL,
  labor_cost_per_acre REAL,
  irrigation_cost_per_acre REAL,

  typical_market_price_per_kg REAL,
  price_volatility_percent REAL,

  last_updated DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(crop_type, province)
);

-- Crop yield model (historical)
CREATE TABLE crop_yield_model (
  id TEXT PRIMARY KEY,
  crop_type TEXT NOT NULL,
  region TEXT NOT NULL,
  season_year INTEGER,

  temperature_avg_c REAL,
  rainfall_mm REAL,
  soil_type TEXT,
  irrigation_type TEXT,

  yield_kg_per_acre REAL,

  data_source TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_crop_yield_model_type_region ON crop_yield_model(crop_type, region);

-- Crop recommendations
CREATE TABLE crop_recommendations (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  field_id TEXT,
  recommendation_date DATE NOT NULL,
  season_type TEXT,                     -- 'kharif', 'rabi'

  crop_name TEXT NOT NULL,
  crop_type TEXT NOT NULL,
  suitability_score INTEGER,            -- 0-100
  suitability_factors TEXT,             -- JSON

  expected_yield_kg_per_acre REAL,
  expected_profit_per_acre REAL,
  risk_level TEXT,                      -- 'low', 'moderate', 'high', 'very_high'

  recommendation_reason TEXT,
  is_alternative BOOLEAN DEFAULT FALSE,
  alternative_to TEXT,                  -- Traditional crop name

  FOREIGN KEY (user_id) REFERENCES farmers(id),
  FOREIGN KEY (field_id) REFERENCES fields(id)
);

CREATE INDEX idx_crop_recommendations_user_date ON crop_recommendations(user_id, recommendation_date DESC);

-- Farmer success stories
CREATE TABLE farmer_success_stories (
  id TEXT PRIMARY KEY,
  farmer_name TEXT,
  province TEXT,
  district TEXT,

  traditional_crop TEXT,
  switched_to_crop TEXT,
  switch_reason TEXT,                   -- 'drought', 'flood', 'profitability'

  traditional_yield_kg_per_acre REAL,
  new_yield_kg_per_acre REAL,
  yield_improvement_percent REAL,

  traditional_profit_per_acre REAL,
  new_profit_per_acre REAL,
  profit_improvement_percent REAL,

  testimonial_text TEXT,
  testimonial_video_url TEXT,
  language TEXT DEFAULT 'ur',

  verified BOOLEAN DEFAULT FALSE,
  featured BOOLEAN DEFAULT FALSE,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_success_stories_province_crop ON farmer_success_stories(province, switched_to_crop);
```
