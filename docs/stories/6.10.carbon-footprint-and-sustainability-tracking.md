**Carbon Footprint Calculator (IPCC Guidelines)**

```javascript
// CarbonFootprintCalculator.js
class CarbonFootprintCalculator {
  /**
   * Calculate farm carbon footprint using IPCC 2019 guidelines
   */
  async calculateCarbonFootprint(farmerId, cropSeasonId) {
    // Get farming practices data
    const practices = await this.getFarmingPractices(farmerId, cropSeasonId);

    // Calculate emissions from different sources
    const fertilizerEmissions = this.calculateFertilizerEmissions(practices.fertilizer_use);
    const fuelEmissions = this.calculateFuelEmissions(practices.fuel_consumption);
    const electricityEmissions = this.calculateElectricityEmissions(practices.electricity_use);
    const riceEmissions = this.calculateRiceEmissions(practices.rice_area_acres);
    const livestockEmissions = this.calculateLivestockEmissions(practices.livestock);

    // Calculate carbon sequestration (negative emissions)
    const sequestration = this.calculateCarbonSequestration(practices.conservation_practices);

    // Total emissions
    const totalEmissions = fertilizerEmissions + fuelEmissions + electricityEmissions +
                          riceEmissions + livestockEmissions - sequestration;

    const fieldSize = practices.field_size_acres;
    const emissionsPerAcre = totalEmissions / fieldSize;

    return {
      farm_id: farmerId,
      crop_season_id: cropSeasonId,
      field_size_acres: fieldSize,

      emissions_by_source: {
        fertilizer: fertilizerEmissions,
        fuel: fuelEmissions,
        electricity: electricityEmissions,
        rice_methane: riceEmissions,
        livestock: livestockEmissions
      },

      carbon_sequestration: sequestration,

      total_emissions_kg_co2e: totalEmissions,
      emissions_per_acre_kg_co2e: emissionsPerAcre,

      comparison: {
        pakistan_avg_emissions_per_acre: 2500,  // kg CO2e
        deviation_percent: ((emissionsPerAcre - 2500) / 2500) * 100
      },

      recommendations: this.generateEmissionReductionRecommendations(practices, totalEmissions)
    };
  }

  calculateFertilizerEmissions(fertilizerUse) {
    // IPCC Tier 1 emission factors for synthetic nitrogen fertilizers
    // 1 kg N → 5.87 kg CO2e (includes N2O emissions)
    const nitrogenKg = fertilizerUse.nitrogen_kg || 0;
    const emissionFactor = 5.87;  // kg CO2e per kg N

    return nitrogenKg * emissionFactor;
  }

  calculateFuelEmissions(fuelConsumption) {
    // Diesel combustion: 1 liter → 2.68 kg CO2
    const dieselLiters = fuelConsumption.diesel_liters || 0;
    const emissionFactor = 2.68;

    return dieselLiters * emissionFactor;
  }

  calculateElectricityEmissions(electricityUse) {
    // Pakistan grid electricity: 1 kWh → 0.6 kg CO2 (2023 average)
    const electricityKWh = electricityUse.kwh || 0;
    const emissionFactor = 0.6;

    return electricityKWh * emissionFactor;
  }

  calculateRiceEmissions(riceAreaAcres) {
    // Methane emissions from rice paddies (continuously flooded)
    // IPCC default: 1 hectare → 1,300 kg CH4 per season
    // 1 kg CH4 = 28 kg CO2e (GWP100)
    // 1 acre = 0.405 hectares

    const hectares = riceAreaAcres * 0.405;
    const methanKg = hectares * 1300;
    const co2e = methanKg * 28;

    return co2e;
  }

  calculateLivestockEmissions(livestock) {
    // Enteric fermentation and manure management
    // IPCC defaults for South Asia:
    // 1 dairy cow → 2,000 kg CO2e/year
    // 1 buffalo → 2,100 kg CO2e/year
    // 1 goat/sheep → 20 kg CO2e/year

    const cows = livestock.cows || 0;
    const buffaloes = livestock.buffaloes || 0;
    const goats = livestock.goats || 0;

    return (cows * 2000) + (buffaloes * 2100) + (goats * 20);
  }

  calculateCarbonSequestration(conservationPractices) {
    // Carbon sequestration from conservation practices
    let sequestration = 0;

    // No-till farming: 200 kg CO2/acre/year
    if (conservationPractices.no_till) {
      sequestration += conservationPractices.no_till_acres * 200;
    }

    // Cover crops: 150 kg CO2/acre/year
    if (conservationPractices.cover_crops) {
      sequestration += conservationPractices.cover_crop_acres * 150;
    }

    // Agroforestry: 500 kg CO2/acre/year
    if (conservationPractices.agroforestry) {
      sequestration += conservationPractices.agroforestry_acres * 500;
    }

    return sequestration;
  }

  generateEmissionReductionRecommendations(practices, currentEmissions) {
    const recommendations = [];

    // Recommendation 1: Optimize fertilizer use
    if (practices.fertilizer_use.nitrogen_kg > 100) {  // High nitrogen use
      const potentialReduction = (practices.fertilizer_use.nitrogen_kg - 100) * 5.87;
      recommendations.push({
        strategy: 'optimize_fertilizer',
        title: 'Optimize Nitrogen Fertilizer Use',
        description: 'Reduce synthetic nitrogen by 20% using soil testing and precision application',
        potential_reduction_kg_co2e: potentialReduction * 0.2,
        implementation_difficulty: 'easy'
      });
    }

    // Recommendation 2: Switch to renewable energy
    if (practices.electricity_use.kwh > 1000) {
      const potentialReduction = practices.electricity_use.kwh * 0.6;
      recommendations.push({
        strategy: 'solar_pump',
        title: 'Install Solar Irrigation Pump',
        description: 'Replace electric/diesel pump with solar to eliminate irrigation emissions',
        potential_reduction_kg_co2e: potentialReduction,
        implementation_difficulty: 'medium',
        government_subsidy_available: true
      });
    }

    // Recommendation 3: Conservation tillage
    if (!practices.conservation_practices.no_till) {
      const potentialSequestration = practices.field_size_acres * 200;
      recommendations.push({
        strategy: 'no_till',
        title: 'Adopt No-Till Farming',
        description: 'Sequester 200 kg CO2 per acre annually through conservation tillage',
        potential_reduction_kg_co2e: potentialSequestration,
        implementation_difficulty: 'medium'
      });
    }

    return recommendations;
  }
}
```

**Sustainability Score Calculator**

```javascript
// SustainabilityScoreCalculator.js
class SustainabilityScoreCalculator {
  /**
   * Calculate overall sustainability score (0-100)
   */
  async calculateSustainabilityScore(farmerId) {
    const practices = await this.getFarmingPractices(farmerId);

    // Score components (each out of 20 points)
    const soilHealthScore = this.scoreSoilHealth(practices);
    const waterManagementScore = this.scoreWaterManagement(practices);
    const biodiversityScore = this.scoreBiodiversity(practices);
    const energyScore = this.scoreEnergy(practices);
    const carbonScore = this.scoreCarbonFootprint(practices);

    const totalScore = soilHealthScore + waterManagementScore + biodiversityScore +
                      energyScore + carbonScore;

    return {
      overall_score: totalScore,
      score_breakdown: {
        soil_health: soilHealthScore,
        water_management: waterManagementScore,
        biodiversity_pesticide: biodiversityScore,
        energy_efficiency: energyScore,
        carbon_emissions: carbonScore
      },
      rating: this.getScoreRating(totalScore),
      certification_ready: totalScore >= 70,
      recommendations: this.generateSustainabilityRecommendations(practices, totalScore)
    };
  }

  scoreSoilHealth(practices) {
    let score = 0;

    // No-till or conservation tillage: +10 points
    if (practices.conservation_practices?.no_till) score += 10;

    // Cover crops: +5 points
    if (practices.conservation_practices?.cover_crops) score += 5;

    // Organic matter addition (compost, manure): +5 points
    if (practices.fertilizer_use?.organic_fertilizer) score += 5;

    return Math.min(20, score);
  }

  scoreWaterManagement(practices) {
    let score = 0;

    // Drip irrigation: +10 points
    if (practices.irrigation?.type === 'drip') score += 10;

    // Efficient irrigation scheduling: +5 points
    if (practices.irrigation?.efficiency > 70) score += 5;

    // Rainwater harvesting: +5 points
    if (practices.water_conservation?.rainwater_harvesting) score += 5;

    return Math.min(20, score);
  }

  scoreBiodiversity(practices) {
    let score = 0;

    // No synthetic pesticides (organic): +10 points
    if (practices.pest_management?.type === 'organic') score += 10;
    // IPM (Integrated Pest Management): +7 points
    else if (practices.pest_management?.type === 'IPM') score += 7;

    // Crop rotation: +5 points
    if (practices.crop_rotation?.practices) score += 5;

    // Agroforestry or hedgerows: +5 points
    if (practices.agroforestry?.present) score += 5;

    return Math.min(20, score);
  }

  scoreEnergy(practices) {
    let score = 0;

    // Renewable energy (solar pump): +15 points
    if (practices.energy?.renewable_percent >= 50) score += 15;

    // Energy-efficient equipment: +5 points
    if (practices.energy?.efficient_equipment) score += 5;

    return Math.min(20, score);
  }

  scoreCarbonFootprint(practices) {
    // Lower emissions = higher score
    const emissionsPerAcre = practices.carbon_footprint?.emissions_per_acre || 3000;
    const pakistanAvg = 2500;

    let score = 0;
    if (emissionsPerAcre <= pakistanAvg * 0.5) score = 20;       // 50% below average
    else if (emissionsPerAcre <= pakistanAvg * 0.75) score = 15;  // 25% below average
    else if (emissionsPerAcre <= pakistanAvg) score = 10;         // At or below average
    else if (emissionsPerAcre <= pakistanAvg * 1.25) score = 5;   // Slightly above average
    else score = 0;  // High emissions

    return score;
  }

  getScoreRating(score) {
    if (score >= 90) return { rating: 'Excellent', color: '#4CAF50', icon: '⭐⭐⭐⭐⭐' };
    if (score >= 70) return { rating: 'Good', color: '#8BC34A', icon: '⭐⭐⭐⭐' };
    if (score >= 50) return { rating: 'Fair', color: '#FFC107', icon: '⭐⭐⭐' };
    if (score >= 30) return { rating: 'Poor', color: '#FF9800', icon: '⭐⭐' };
    return { rating: 'Very Poor', color: '#F44336', icon: '⭐' };
  }

  generateSustainabilityRecommendations(practices, currentScore) {
    const recommendations = [];

    if (!practices.conservation_practices?.no_till) {
      recommendations.push('Adopt conservation tillage to improve soil health (+10 points)');
    }

    if (practices.irrigation?.type !== 'drip' && practices.crop_type !== 'rice') {
      recommendations.push('Install drip irrigation for water efficiency (+10 points)');
    }

    if (practices.pest_management?.type === 'synthetic') {
      recommendations.push('Switch to Integrated Pest Management (IPM) for biodiversity (+7 points)');
    }

    if (!practices.energy?.renewable_percent) {
      recommendations.push('Install solar pump for renewable energy (+15 points)');
    }

    return recommendations;
  }
}
```

**Water Footprint Calculator**

```javascript
// WaterFootprintCalculator.js
class WaterFootprintCalculator {
  /**
   * Calculate water footprint (blue, green, grey water)
   */
  async calculateWaterFootprint(cropSeasonId) {
    const cropSeason = await db.query(
      'SELECT * FROM crop_seasons WHERE id = $1',
      [cropSeasonId]
    );

    const field = await db.query(
      'SELECT * FROM fields WHERE id = $1',
      [cropSeason[0].field_id]
    );

    const cropType = cropSeason[0].crop_type;
    const fieldSize = field[0].size_acres;

    // Blue water footprint (irrigation water)
    const irrigationWater = await this.getIrrigationWater(cropSeasonId);
    const blueWater = irrigationWater.total_mm;

    // Green water footprint (rainfall)
    const rainfall = await this.getRainfall(cropSeasonId, field[0].district);
    const greenWater = rainfall.total_mm;

    // Grey water footprint (water pollution from fertilizers/pesticides)
    const greyWater = this.calculateGreyWater(cropType, fieldSize);

    // Total water footprint
    const totalWater = blueWater + greenWater + greyWater;

    // Water use efficiency
    const yield_kg = cropSeason[0].yield_kg_per_acre * fieldSize;
    const waterProductivity = yield_kg / totalWater;  // kg per cubic meter

    return {
      crop_type: cropType,
      field_size_acres: fieldSize,

      blue_water_mm: blueWater,      // Irrigation
      green_water_mm: greenWater,    // Rainfall
      grey_water_mm: greyWater,      // Pollution load

      total_water_footprint_mm: totalWater,
      total_water_cubic_meters: (totalWater * fieldSize * 4047) / 1000,  // Conversion

      water_productivity_kg_per_m3: waterProductivity,

      benchmarks: {
        global_avg_water_footprint: this.getGlobalAvgWaterFootprint(cropType),
        sustainable_threshold: this.getSustainableThreshold(cropType)
      },

      water_efficiency_rating: this.rateWaterEfficiency(totalWater, cropType),
      recommendations: this.generateWaterSavingRecommendations(totalWater, cropType)
    };
  }

  calculateGreyWater(cropType, fieldSize) {
    // Simplified grey water calculation
    // Based on fertilizer and pesticide runoff
    // Typical grey water: 50-100 mm per season
    return 75 * fieldSize;  // mm
  }

  getGlobalAvgWaterFootprint(cropType) {
    // Global average water footprints (mm per growing season)
    const averages = {
      wheat: 1300,
      rice: 2500,
      cotton: 2800,
      corn: 1400,
      sugarcane: 1500
    };
    return averages[cropType] || 1500;
  }

  getSustainableThreshold(cropType) {
    // Sustainable water use thresholds
    const thresholds = {
      wheat: 1000,
      rice: 2000,
      cotton: 2200,
      corn: 1100,
      sugarcane: 1200
    };
    return thresholds[cropType] || 1200;
  }

  rateWaterEfficiency(totalWater, cropType) {
    const sustainableThreshold = this.getSustainableThreshold(cropType);

    if (totalWater <= sustainableThreshold) {
      return { rating: 'Excellent', color: '#4CAF50' };
    } else if (totalWater <= sustainableThreshold * 1.2) {
      return { rating: 'Good', color: '#8BC34A' };
    } else if (totalWater <= sustainableThreshold * 1.5) {
      return { rating: 'Fair', color: '#FFC107' };
    }
    return { rating: 'Poor', color: '#F44336' };
  }

  generateWaterSavingRecommendations(totalWater, cropType) {
    const recommendations = [];

    const sustainableThreshold = this.getSustainableThreshold(cropType);

    if (totalWater > sustainableThreshold * 1.2) {
      recommendations.push({
        strategy: 'drip_irrigation',
        title: 'Switch to Drip Irrigation',
        potential_savings_percent: 40,
        description: 'Reduce water use by 40% while improving yield'
      });

      recommendations.push({
        strategy: 'mulching',
        title: 'Apply Mulch',
        potential_savings_percent: 25,
        description: 'Reduce evaporation and conserve soil moisture'
      });
    }

    return recommendations;
  }
}
```

```sql
  emissions_fertilizer REAL,

  emissions_electricity REAL,
  emissions_rice_methane REAL,
  emissions_livestock REAL,

  carbon_sequestration REAL,

  total_emissions_kg_co2e REAL NOT NULL,
  emissions_per_acre_kg_co2e REAL NOT NULL,

  calculated_at DATETIME NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (farm_id) REFERENCES farmers(id),
  FOREIGN KEY (crop_season_id) REFERENCES crop_seasons(id)
);

CREATE INDEX idx_carbon_footprint_farm ON carbon_footprint_records(farm_id, calculated_at DESC);

-- Sustainability scores
CREATE TABLE sustainability_scores (

  farm_id TEXT NOT NULL,

  soil_health_score INTEGER,
  water_management_score INTEGER,
  biodiversity_score INTEGER,
  energy_score INTEGER,
  carbon_score INTEGER,

  overall_score INTEGER NOT NULL,
  rating TEXT,
  certification_ready BOOLEAN,

  calculated_at DATETIME NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (farm_id) REFERENCES farmers(id)
);

CREATE INDEX idx_sustainability_farm ON sustainability_scores(farm_id, calculated_at DESC);

-- Water footprint records
CREATE TABLE water_footprint_records (
  id TEXT PRIMARY KEY,
  crop_season_id TEXT NOT NULL,

  blue_water_mm REAL,              -- Irrigation
  green_water_mm REAL,             -- Rainfall
  grey_water_mm REAL,              -- Pollution load

  total_water_footprint_mm REAL NOT NULL,
  water_productivity_kg_per_m3 REAL,

  calculated_at DATETIME NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (crop_season_id) REFERENCES crop_seasons(id)
);

CREATE INDEX idx_water_footprint ON water_footprint_records(crop_season_id);

-- Certification pathways
CREATE TABLE certification_pathways (

  benefits TEXT,                     -- JSON array
  requirements TEXT,                 -- JSON object
  process TEXT,                      -- JSON array (steps)
  costs TEXT,                        -- JSON object
  contact TEXT,                      -- JSON object

  active BOOLEAN DEFAULT TRUE,
  last_updated DATE,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Premium market opportunities
CREATE TABLE premium_market_opportunities (

  certification_required TEXT,       -- 'organic', 'sustainable', 'none'

  premium_percent INTEGER,           -- % premium over regular market
  volume_required_kg INTEGER,
  payment_terms TEXT,

  contact_person TEXT,
  phone TEXT,
  email TEXT,
  location TEXT,

  active BOOLEAN DEFAULT TRUE,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_premium_markets_cert ON premium_market_opportunities(certification_required, active);
```

### Project Structure Notes

This story empowers farmers to measure and improve their environmental impact through data-driven sustainability tracking. The IPCC-based carbon footprint calculator provides scientifically accurate emissions measurement, while the sustainability scoring system (0-100) gamifies improvement. Water footprint tracking addresses Pakistan's water scarcity challenges. Certification pathways (PCSIR organic, international standards) open premium market access with 15-30% price premiums, creating financial incentives for sustainable farming. Carbon credit opportunities provide future revenue streams as carbon markets mature.
