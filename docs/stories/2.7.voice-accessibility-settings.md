```javascript
// VoiceSettingsScreen.js
const VoiceSettingsScreen = ({ navigation }) => {
  const [preferences, setPreferences] = useState(defaultVoicePreferences);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    loadPreferences();
  }, []);

  const loadPreferences = async () => {
    const saved = await VoicePreferencesRepository.get();
    setPreferences(saved || defaultVoicePreferences);
    setIsLoading(false);
  };

  const updatePreference = async (path, value) => {
    const updated = { ...preferences };
    setNestedValue(updated, path, value);
    setPreferences(updated);

    await VoicePreferencesRepository.save(updated);
  };

  const handleReset = async () => {
    Alert.alert(
      'Reset Voice Settings',
      'Are you sure you want to reset all voice settings to defaults?',
      [
        { text: 'Cancel', style: 'cancel' },
        {
          text: 'Reset',
          style: 'destructive',
          onPress: async () => {
            setPreferences(defaultVoicePreferences);
            await VoicePreferencesRepository.save(defaultVoicePreferences);
          }
        }
      ]
    );
  };

  return (
    <ScrollView style={styles.container}>
      {/* Voice Input Section */}
      <Section title="Voice Input">
        <ToggleSetting
          label="Enable Voice Input"
          value={preferences.input.enabled}
          onChange={(value) => updatePreference('input.enabled', value)}
        />

        <SelectSetting
          label="Microphone Sensitivity"
          value={preferences.input.microphoneSensitivity}
          options={['Low', 'Medium', 'High']}
          onChange={(value) => updatePreference('input.microphoneSensitivity', value)}
        />

        <ButtonSetting
          label="Test Microphone"
          onPress={() => navigation.navigate('MicrophoneTest')}
        />
      </Section>

      {/* Voice Output Section */}
      <Section title="Voice Output">
        <ToggleSetting
          label="Enable Text-to-Speech"
          value={preferences.output.enabled}
          onChange={(value) => updatePreference('output.enabled', value)}
        />

        <ToggleSetting
          label="Auto-play Responses"
          description="Automatically read AI responses aloud"
          value={preferences.output.autoPlay}
          onChange={(value) => updatePreference('output.autoPlay', value)}
        />

        <SliderSetting
          label="Speech Rate"
          value={preferences.output.speechRate}
          min={0.75}
          max={1.5}
          step={0.25}
          onChange={(value) => updatePreference('output.speechRate', value)}
          preview={(rate) => previewSpeechRate(rate)}
        />

        <SelectSetting
          label="Voice Gender"
          value={preferences.output.voiceGender}
          options={['Female', 'Male', 'Neutral']}
          onChange={(value) => updatePreference('output.voiceGender', value)}
        />

        <ButtonSetting
          label="Select Specific Voice"
          onPress={() => navigation.navigate('VoiceSelector')}
        />
      </Section>

      {/* Advanced Settings */}
      <Section title="Advanced">
        <ToggleSetting
          label="Prefer Offline Mode"
          description="Use on-device speech recognition when possible"
          value={preferences.advanced.preferOfflineMode}
          onChange={(value) => updatePreference('advanced.preferOfflineMode', value)}
        />

        <ToggleSetting
          label="Download Offline Models"
          description="Download speech models for offline use"
          value={preferences.advanced.downloadOfflineModels}
          onChange={(value) => updatePreference('advanced.downloadOfflineModels', value)}
        />

        <ToggleSetting
          label="Enable Voice Commands"
          description="Use voice to navigate the app"
          value={preferences.advanced.enableVoiceCommands}
          onChange={(value) => updatePreference('advanced.enableVoiceCommands', value)}
        />
      </Section>

      {/* Tutorial & Help */}
      <Section title="Help & Tutorial">
        <ButtonSetting
          label="Voice Tutorial"
          description="Learn how to use voice features"
          onPress={() => navigation.navigate('VoiceTutorial')}
          icon="school"
        />

        <ButtonSetting
          label="Voice Commands Reference"
          onPress={() => navigation.navigate('VoiceCommandsHelp')}

        />
```

```javascript
// ToggleSetting.js
const ToggleSetting = ({ label, description, value, onChange, destructive }) => {
  return (
    <View style={styles.settingRow}>
      <View style={styles.settingInfo}>
        <Text style={[styles.label, destructive && styles.destructive]}>
          {label}
        </Text>
        {description && (
          <Text style={styles.description}>{description}</Text>
        )}
      </View>
      <Switch
        value={value}
        onValueChange={onChange}
        trackColor={{ true: destructive ? '#e74c3c' : '#27ae60' }}
      />
    </View>
  );
};

// SliderSetting.js
const SliderSetting = ({ label, value, min, max, step, onChange, preview }) => {
  const [previewValue, setPreviewValue] = useState(value);

  const handleChange = (newValue) => {
    setPreviewValue(newValue);
    onChange(newValue);
  };

  const labels = {
    0.75: 'Slow',
    1.0: 'Normal',
    1.25: 'Fast',
    1.5: 'Very Fast',
  };

  return (
    <View style={styles.settingRow}>
      <Text style={styles.label}>{label}</Text>
      <View style={styles.sliderContainer}>
        <Slider
          value={value}
          minimumValue={min}
          maximumValue={max}
          step={step}
          onValueChange={handleChange}
          style={styles.slider}
        />
        <Text style={styles.sliderValue}>{labels[previewValue]}</Text>
      </View>
      {preview && (
        <TouchableOpacity
          style={styles.previewButton}
          onPress={() => preview(previewValue)}
        >
          <Icon name="play-arrow" size={20} />
          <Text>Preview</Text>
        </TouchableOpacity>
      )}
    </View>
  );
};

// SelectSetting.js
const SelectSetting = ({ label, value, options, onChange }) => {
  const [showPicker, setShowPicker] = useState(false);

  return (
    <View style={styles.settingRow}>
      <Text style={styles.label}>{label}</Text>
      <TouchableOpacity
        style={styles.selectButton}
        onPress={() => setShowPicker(true)}
      >
        <Text style={styles.selectValue}>{value}</Text>
        <Icon name="chevron-right" size={20} />
      </TouchableOpacity>

      {showPicker && (
        <Picker
          selectedValue={value}
          onValueChange={(itemValue) => {
            onChange(itemValue);
            setShowPicker(false);
          }}
        >
          {options.map((option) => (
            <Picker.Item key={option} label={option} value={option.toLowerCase()} />
          ))}
        </Picker>
      )}
    </View>
  );
};
```

```javascript
// VoiceSelectorScreen.js
const VoiceSelectorScreen = () => {
  const [availableVoices, setAvailableVoices] = useState([]);
  const [selectedVoice, setSelectedVoice] = useState(null);
  const language = LanguageManager.getCurrentLanguage();

  useEffect(() => {
    loadAvailableVoices();
  }, [language]);

  const loadAvailableVoices = async () => {
    const voices = await TextToSpeechService.getAvailableVoices(language);
    setAvailableVoices(voices);
  };

  const handleSelectVoice = (voice) => {
    setSelectedVoice(voice);
    VoicePreferencesRepository.setVoice(language, voice.id, voice.gender);
  };

  const previewVoice = async (voice) => {
    const sampleText = 'This is a sample of this voice';
    await TextToSpeechService.speak(sampleText, {
      language,
      voiceId: voice.id,
    });
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Select Voice for {language}</Text>

      <FlatList
        data={availableVoices}
        keyExtractor={(item) => item.id}
        renderItem={({ item }) => (
          <TouchableOpacity
            style={[
              styles.voiceItem,
              selectedVoice?.id === item.id && styles.selected
            ]}
            onPress={() => handleSelectVoice(item)}
          >
            <View style={styles.voiceInfo}>
              <Text style={styles.voiceName}>{item.name}</Text>
              <Text style={styles.voiceDetails}>
                {item.gender} • {item.quality}
              </Text>
            </View>

            <TouchableOpacity
              style={styles.previewButton}
              onPress={() => previewVoice(item)}
            >
              <Icon name="play-arrow" size={24} />
            </TouchableOpacity>

            {selectedVoice?.id === item.id && (
              <Icon name="check-circle" size={24} color="#27ae60" />
            )}
          </TouchableOpacity>
        )}
      />
    </View>
  );
};
```

```javascript
// MicrophoneTestScreen.js
const MicrophoneTestScreen = () => {
  const [isRecording, setIsRecording] = useState(false);
  const [audioLevel, setAudioLevel] = useState(0);
  const [sensitivity, setSensitivity] = useState('medium');

  const startTest = async () => {
    setIsRecording(true);

    // Start recording with level monitoring
    const recording = new Audio.Recording();
    await recording.prepareToRecordAsync({
      ...recordingOptions,
      // Add level metering
    });

    recording.setOnRecordingStatusUpdate((status) => {
      if (status.isRecording) {
        // Update audio level in real-time
        const level = status.metering || 0;
        setAudioLevel(level);
      }
    });

    await recording.startAsync();

    // Auto-stop after 5 seconds
    setTimeout(async () => {
      await recording.stopAndUnloadAsync();
      setIsRecording(false);
    }, 5000);
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Microphone Test</Text>

      <View style={styles.waveformContainer}>
        <Waveform level={audioLevel} />
      </View>

      <View style={styles.controls}>
        <Text style={styles.label}>Sensitivity</Text>
        <Picker
          selectedValue={sensitivity}
          onValueChange={setSensitivity}
        >
          <Picker.Item label="Low" value="low" />
          <Picker.Item label="Medium" value="medium" />
          <Picker.Item label="High" value="high" />
        </Picker>
      </View>

      <TouchableOpacity
        style={styles.testButton}
        onPress={startTest}
        disabled={isRecording}
      >
        <Icon name={isRecording ? 'stop' : 'mic'} size={32} />
        <Text>{isRecording ? 'Recording...' : 'Start Test'}</Text>
      </TouchableOpacity>

      <View style={styles.guidance}>
        <Text style={styles.guidanceText}>
          {audioLevel < 0.2 && 'Speak louder or increase sensitivity'}
          {audioLevel > 0.8 && 'Too loud, reduce sensitivity'}
          {audioLevel >= 0.2 && audioLevel <= 0.8 && 'Good audio level'}
        </Text>
      </View>
    </View>
  );
};
```

### Data Model Requirements

**Voice Preferences Table**

Comprehensive storage for all voice settings:
```sql
CREATE TABLE voice_preferences (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id TEXT,

  -- Voice Input
  input_enabled BOOLEAN DEFAULT 1,
  input_language TEXT DEFAULT 'ur-PK',
  mic_sensitivity TEXT DEFAULT 'medium',  -- 'low', 'medium', 'high'
  noise_threshold REAL DEFAULT 0.3,
  confidence_threshold REAL DEFAULT 0.7,
  recording_max_duration INTEGER DEFAULT 30,
  silence_timeout INTEGER DEFAULT 5,

  -- Voice Output (TTS)
  output_enabled BOOLEAN DEFAULT 1,
  tts_autoplay BOOLEAN DEFAULT 0,
  tts_speech_rate REAL DEFAULT 1.0,
  tts_pitch REAL DEFAULT 1.0,
  tts_voice_gender TEXT DEFAULT 'female',
  tts_voice_id TEXT,
  tts_volume REAL DEFAULT 1.0,

  -- Language-specific voices (JSON)
  language_voices TEXT,  -- {"ur-PK": {"voiceId": "...", "gender": "female"}, ...}

  -- Advanced
  prefer_offline_mode BOOLEAN DEFAULT 0,
  download_offline_models BOOLEAN DEFAULT 1,
  cache_transcriptions BOOLEAN DEFAULT 1,
  enable_voice_commands BOOLEAN DEFAULT 1,
  data_collection_opt_in BOOLEAN DEFAULT 0,

  -- Accessibility
  show_visual_feedback BOOLEAN DEFAULT 1,
  haptic_feedback BOOLEAN DEFAULT 1,
  high_contrast_mode BOOLEAN DEFAULT 0,
  screen_reader_compatible BOOLEAN DEFAULT 1,

  -- Metadata
  tutorial_completed BOOLEAN DEFAULT 0,
  last_updated DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(user_id)
);

CREATE INDEX idx_voice_prefs_user ON voice_preferences(user_id);
```

### Privacy and Data Collection

**Data Collection Opt-In**:
```javascript
const DataCollectionOptIn = ({ value, onChange }) => {
  const [showDetails, setShowDetails] = useState(false);

  return (
    <View>
      <ToggleSetting
        label="Help Improve Voice Recognition"
        description="Allow Zarai Dost to collect voice data for improving accuracy"
        value={value}
        onChange={(enabled) => {
          if (enabled) {
            setShowDetails(true);
          } else {
            onChange(false);
          }
        }}
      />

      {showDetails && (
        <Modal visible={true}>
          <View style={styles.privacyModal}>
            <Text style={styles.title}>Voice Data Collection</Text>
            <ScrollView>
              <Text>
                By enabling this option, you allow Zarai Dost to:
                - Store your voice recordings anonymously
                - Use recordings to improve speech recognition accuracy
                - Analyze common speech patterns in agricultural contexts

                Your data will be:
                - Anonymized (no personal information)
                - Encrypted in transit and at rest
                - Used only for improving this app
                - Deletable at any time

                You can opt out at any time in Settings.
              </Text>
            </ScrollView>
            <View style={styles.buttons}>
              <Button title="Cancel" onPress={() => {
                setShowDetails(false);
                onChange(false);
              }} />
              <Button title="Accept" onPress={() => {
                setShowDetails(false);
                onChange(true);
              }} />
            </View>
          </View>
        </Modal>
      )}
    </View>
  );
};
```
