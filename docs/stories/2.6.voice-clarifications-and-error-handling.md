**Confidence Threshold System**

```javascript
// ConfidenceEvaluator.js
class ConfidenceEvaluator {
  constructor() {
    this.thresholds = {
      HIGH_CONFIDENCE: 0.9,
      MEDIUM_CONFIDENCE: 0.7,
      LOW_CONFIDENCE: 0.5,
    };
  }

  evaluate(transcription) {
    const { confidence, alternatives } = transcription;

    if (confidence >= this.thresholds.HIGH_CONFIDENCE) {
      return {
        level: 'high',
        action: 'accept',
        message: null,
      };
    } else if (confidence >= this.thresholds.MEDIUM_CONFIDENCE) {
      return {
        level: 'medium',
        action: 'confirm',
        message: 'Is this correct?',
        showTranscription: true,
      };
    } else if (confidence >= this.thresholds.LOW_CONFIDENCE) {
      return {
        level: 'low',
        action: 'clarify',
        message: 'Did you say one of these?',
        showAlternatives: true,
        alternatives: alternatives || [],
      };
    } else {
      return {
        level: 'very_low',
        action: 'retry',
        message: "I couldn't understand. Please try again.",
        suggestions: [
          'Speak clearly and slowly',
          'Reduce background noise',
          'Move to a quieter location',
        ],
      };
    }
  }

  // Language-specific thresholds
  getThresholdForLanguage(languageCode) {
    const adjustments = {
      'ur-PK': 0.0,   // Urdu has good model, no adjustment
      'pa-IN': -0.05, // Punjabi may need more lenient threshold
      'sd-IN': -0.1,  // Sindhi even more lenient
    };

    const adjustment = adjustments[languageCode] || 0;

    return {
      HIGH_CONFIDENCE: this.thresholds.HIGH_CONFIDENCE + adjustment,
      MEDIUM_CONFIDENCE: this.thresholds.MEDIUM_CONFIDENCE + adjustment,
      LOW_CONFIDENCE: this.thresholds.LOW_CONFIDENCE + adjustment,
    };
  }
}

export default new ConfidenceEvaluator();
```

```javascript
// ClarificationDialog.js
const ClarificationDialog = ({
  transcription,
  alternatives,
  onSelect,
  onRetry,
  language
}) => {
  const [selectedIndex, setSelectedIndex] = useState(null);

  useEffect(() => {
    // Read clarification prompt via TTS

    TextToSpeechService.speak(prompt, { language });

    alternatives.forEach((alt, index) => {

        TextToSpeechService.speak(
          `${i18n.t('voice.clarification.option')} ${index + 1}: ${alt.text}`,
          { language }
        );
      }, (index + 1) * 2000);
    });
  }, []);

  const handleSelect = (index) => {
    setSelectedIndex(index);
    onSelect(alternatives[index]);
  };

  return (
    <Modal visible={true} animationType="slide">
      <View style={styles.container}>
        <Text style={styles.title}>
          {i18n.t('voice.clarification.title')}
        </Text>

        <Text style={styles.prompt}>
          {i18n.t('voice.clarification.didYouSay')}
        </Text>

        {alternatives.map((alt, index) => (
          <TouchableOpacity
            key={index}
            style={[
              styles.option,
              selectedIndex === index && styles.selected

            <View style={styles.optionContent}>
```

**Retry Manager**

```javascript
// RetryManager.js
class RetryManager {
  constructor() {
    this.maxRetries = 3;
    this.currentAttempt = 0;
    this.retryReasons = [];
  }

  canRetry() {
    return this.currentAttempt < this.maxRetries;
  }

  recordRetry(reason) {
    this.currentAttempt++;
    this.retryReasons.push({
      attempt: this.currentAttempt,
      reason,
      timestamp: Date.now(),
    });
  }

  reset() {
    this.currentAttempt = 0;
    this.retryReasons = [];
  }

  getRetryMessage() {
    const messages = {
      1: 'Please try again',
      2: 'Try speaking more clearly',
      3: 'Last attempt - speak slowly and clearly',
    };

    return messages[this.currentAttempt] || messages[1];
  }

  shouldSuggestTextInput() {
    return this.currentAttempt >= this.maxRetries;
  }

  getSuggestions() {
    const commonIssues = this.retryReasons.map(r => r.reason);

    const suggestions = [];

    if (commonIssues.includes('low_confidence')) {
      suggestions.push('Speak more clearly');
      suggestions.push('Reduce background noise');
    }

    if (commonIssues.includes('too_noisy')) {
      suggestions.push('Move to a quieter location');
      suggestions.push('Use headset microphone');
    }

    if (commonIssues.includes('no_speech')) {
      suggestions.push('Speak louder');
      suggestions.push('Check microphone permissions');
    }

    return suggestions;
  }

  getRetryStatistics() {
    return {
      totalAttempts: this.currentAttempt,
      reasons: this.retryReasons,
      suggestTextInput: this.shouldSuggestTextInput(),
    };
  }
}

export default RetryManager;
```

**Silence Timeout Handler**

```javascript
// SilenceTimeoutHandler.js
class SilenceTimeoutHandler {
  constructor() {
    this.timeoutDuration = 5000;  // 5 seconds
    this.timeoutTimer = null;
    this.onTimeout = null;
  }

  startTimeout(callback) {
    this.onTimeout = callback;
    this.resetTimeout();
  }

  resetTimeout() {
    if (this.timeoutTimer) {
      clearTimeout(this.timeoutTimer);
    }

    this.timeoutTimer = setTimeout(() => {
      this.onTimeout?.();
    }, this.timeoutDuration);
  }

  cancelTimeout() {
    if (this.timeoutTimer) {
      clearTimeout(this.timeoutTimer);
      this.timeoutTimer = null;
    }
  }

  detectSilence(audioLevel) {
    const silenceThreshold = 0.01;  // Very low amplitude = silence

    if (audioLevel < silenceThreshold) {
      // Silence detected, timeout is running
      return true;
    } else {
      // Audio detected, reset timeout
      this.resetTimeout();
      return false;
    }
  }

  setTimeoutDuration(duration) {
    this.timeoutDuration = duration;
  }
}

export default SilenceTimeoutHandler;
```

```javascript
// VoiceErrorHandler.js
class VoiceErrorHandler {
  constructor() {
    this.errorCounts = {};
  }

  handleError(errorType, context = {}) {
    // Log error
    this.logError(errorType, context);

    // Get error details
    const errorInfo = this.getErrorInfo(errorType);

    // Show user-friendly message
    const message = this.getErrorMessage(errorType, context.language);

    // Get recovery action
    const recovery = this.getRecoveryAction(errorType);

    return {
      type: errorType,
      message,
      recovery,
      ...errorInfo,
    };
  }

  getErrorInfo(errorType) {
    const errors = {
      low_confidence: {
        severity: 'medium',
        recoverable: true,
        action: 'clarify',
      },
      no_speech: {
        severity: 'low',
        recoverable: true,
        action: 'retry',
      },
      too_noisy: {
        severity: 'medium',
        recoverable: true,
        action: 'retry_with_guidance',
      },
      silence_timeout: {
        severity: 'low',
        recoverable: true,
        action: 'restart',
      },
      api_error: {
        severity: 'high',
        recoverable: true,
        action: 'fallback_offline',
      },
      microphone_error: {
        severity: 'high',
        recoverable: false,
        action: 'check_permissions',
      },
    };

    return errors[errorType] || {
      severity: 'unknown',
      recoverable: true,
      action: 'retry',
    };
  }

  getErrorMessage(errorType, language = 'ur-PK') {
```
