```python
# price_prediction_service.py
import pandas as pd
import numpy as np
from prophet import Prophet
from statsmodels.tsa.arima.model import ARIMA
from datetime import datetime, timedelta
import pickle

class PricePredictionService:
    def __init__(self):
        self.models = {}  # Cache loaded models
        self.model_type = 'prophet'  # Default to Prophet

    def train_model(self, crop_type, mandi_id):
        """Train prediction model for specific crop and mandi"""
        # Load historical data
        historical_data = self.load_historical_data(crop_type, mandi_id)

        if len(historical_data) < 90:
            raise ValueError(f"Insufficient data: {len(historical_data)} records")

        # Prepare data for Prophet
        df = pd.DataFrame({
            'ds': historical_data['price_date'],
            'y': historical_data['modal_price_per_kg']
        })

        # Initialize Prophet model with seasonality
        model = Prophet(
            yearly_seasonality=True,
            weekly_seasonality=False,
            daily_seasonality=False,
            seasonality_mode='multiplicative',
            interval_width=0.95  # 95% confidence intervals
        )

        # Add custom seasonality for harvest seasons
        model.add_seasonality(
            name='harvest',
            period=365.25,
            fourier_order=5
        )

        # Train model
        model.fit(df)

        # Save model
        model_path = self.save_model(model, crop_type, mandi_id)

        return {
            'model_path': model_path,
            'training_samples': len(df),
            'trained_at': datetime.now()
        }

    def predict_prices(self, crop_type, mandi_id, days=[30, 60, 90]):
        """Generate price predictions for specified horizons"""
        # Load model
        model = self.load_model(crop_type, mandi_id)

        if not model:
            raise ValueError(f"No model found for {crop_type} at {mandi_id}")

        # Create future dataframe
        future = model.make_future_dataframe(periods=max(days))

        # Generate predictions
        forecast = model.predict(future)

        # Extract predictions for specific days
        predictions = []
        today = datetime.now().date()

        for day in days:
            target_date = today + timedelta(days=day)
            forecast_row = forecast[forecast['ds'].dt.date == target_date].iloc[0]

            predictions.append({
                'days_ahead': day,
                'target_date': target_date.isoformat(),
                'predicted_price': round(forecast_row['yhat'], 2),
                'lower_bound': round(forecast_row['yhat_lower'], 2),
                'upper_bound': round(forecast_row['yhat_upper'], 2),
                'confidence_interval_width': round(forecast_row['yhat_upper'] - forecast_row['yhat_lower'], 2)
            })

        return predictions

    def calculate_confidence_level(self, prediction):
        """Calculate confidence level based on prediction interval"""
        interval_width = prediction['confidence_interval_width']
        predicted_price = prediction['predicted_price']

        # Calculate interval as percentage of price
        interval_percent = (interval_width / predicted_price) * 100

        # Classify confidence
        if interval_percent < 15:
            confidence = 'high'
            confidence_score = 85
        elif interval_percent < 30:
            confidence = 'medium'
            confidence_score = 65
        else:
            confidence = 'low'
            confidence_score = 40

        return {
            'level': confidence,
            'score': confidence_score,
            'interval_percent': round(interval_percent, 1)
        }

    def determine_trend(self, predictions):
        """Determine price trend from predictions"""
        if not predictions:
            return 'stable'

        # Compare 30-day and 90-day predictions
        day_30 = next((p for p in predictions if p['days_ahead'] == 30), None)
        day_90 = next((p for p in predictions if p['days_ahead'] == 90), None)

        if not day_30 or not day_90:
            return 'stable'

        # Calculate percent change
        change = ((day_90['predicted_price'] - day_30['predicted_price'])
                  / day_30['predicted_price']) * 100

        if change > 5:
            trend = 'rising'
        elif change < -5:
            trend = 'falling'
        else:
            trend = 'stable'

        return {
            'trend': trend,
            'change_percent': round(change, 1),
            'peak_date': self.find_peak_price_date(predictions)
        }

    def find_peak_price_date(self, predictions):
        """Find date with highest predicted price"""
        if not predictions:
            return None

        peak = max(predictions, key=lambda x: x['predicted_price'])
        return peak['target_date']

    def explain_prediction_factors(self, crop_type, target_date):
        """Generate explanation for price prediction"""
        factors = []

        # Check harvest season
        if self.is_harvest_season(crop_type, target_date):
            factors.append({
                'factor': 'Harvest Season',
                'impact': 'negative',
                'description': 'Prices typically drop during harvest due to oversupply'
            })

        # Check historical seasonality
        month = target_date.month
        seasonal_trend = self.get_seasonal_trend(crop_type, month)
        if seasonal_trend:
            factors.append(seasonal_trend)

        # Check demand periods (festivals, etc.)
        demand_event = self.check_demand_events(target_date)
        if demand_event:
            factors.append(demand_event)

        # Sort by impact
        return factors[:3]  # Top 3 factors

    def is_harvest_season(self, crop_type, date):
        """Check if date falls in harvest season for crop"""
        harvest_periods = {
            'wheat': [(4, 5), (5, 6)],  # April-May
            'rice': [(10, 11), (11, 12)],  # October-November
            'cotton': [(9, 10), (10, 11)],  # September-October
            'sugarcane': [(11, 12), (12, 1), (1, 2)]  # November-February
        }

        periods = harvest_periods.get(crop_type, [])
        month = date.month

        for start, end in periods:
            if start <= month <= end:
                return True

        return False

    def get_seasonal_trend(self, crop_type, month):
        """Get historical seasonal trend for crop and month"""
        # Query historical data for this month
        avg_price = self.get_historical_month_average(crop_type, month)
        overall_avg = self.get_overall_average(crop_type)

        if avg_price > overall_avg * 1.1:
            return {
                'factor': 'Seasonal Demand',
                'impact': 'positive',
                'description': f'Prices historically 10%+ higher in this month'
            }
        elif avg_price < overall_avg * 0.9:
            return {
                'factor': 'Seasonal Oversupply',
                'impact': 'negative',
                'description': f'Prices historically 10%+ lower in this month'
            }

        return None

    def check_demand_events(self, date):
        """Check for demand-increasing events (Eid, Ramadan, etc.)"""
        # Pakistan festival calendar (approximate)
        month = date.month

        # Ramadan (variable, but example)
        if month == 3 or month == 4:
            return {
                'factor': 'Ramadan Demand',
                'impact': 'positive',
                'description': 'Increased food demand during Ramadan month'
            }

        return None

    def load_historical_data(self, crop_type, mandi_id):
        """Load historical price data from database"""
        # PostgreSQL query via psycopg2
        query = """
            SELECT price_date, modal_price_per_kg
            FROM mandi_prices
            WHERE crop_type = %s AND mandi_id = %s
                AND price_date >= NOW() - INTERVAL '3 years'
            ORDER BY price_date ASC
        """
        # Execute query and return DataFrame
        # (Implementation depends on DB connection)
        pass

    def save_model(self, model, crop_type, mandi_id):
        """Save trained model to disk"""
        model_path = f"models/price_prediction/{crop_type}_{mandi_id}.pkl"
        with open(model_path, 'wb') as f:
            pickle.dump(model, f)
        return model_path

    def load_model(self, crop_type, mandi_id):
        """Load trained model from disk"""
        model_key = f"{crop_type}_{mandi_id}"

        if model_key in self.models:
            return self.models[model_key]

        model_path = f"models/price_prediction/{model_key}.pkl"
        try:
            with open(model_path, 'rb') as f:
                model = pickle.load(f)
                self.models[model_key] = model
                return model
        except FileNotFoundError:
            return None
```

**Prediction Accuracy Tracker**

```python
# prediction_accuracy_tracker.py
import psycopg2
from datetime import datetime, timedelta

class PredictionAccuracyTracker:
    def __init__(self, db_connection):
        self.db = db_connection

    def log_prediction(self, crop_type, mandi_id, predictions):
        """Log predictions for later accuracy verification"""
        cursor = self.db.cursor()

        for pred in predictions:
            cursor.execute("""
                INSERT INTO price_predictions (
                    id, crop_type, mandi_id, prediction_date,
                    target_date, days_ahead, predicted_price,
                    lower_bound, upper_bound, confidence_level
                ) VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
            """, (
                f"pred-{datetime.now().timestamp()}",
                crop_type,
                mandi_id,
                datetime.now().date(),
                pred['target_date'],
                pred['days_ahead'],
                pred['predicted_price'],
                pred['lower_bound'],
                pred['upper_bound'],
                pred['confidence']['level']
            ))

        self.db.commit()

    def calculate_accuracy(self, crop_type, mandi_id=None):
        """Calculate prediction accuracy against actual prices"""
        cursor = self.db.cursor()

        # Query predictions that have reached target date
        query = """
            SELECT
                pp.id,
                pp.predicted_price,
                pp.days_ahead,
                mp.modal_price_per_kg as actual_price
            FROM price_predictions pp
            JOIN mandi_prices mp
                ON pp.mandi_id = mp.mandi_id
                AND pp.crop_type = mp.crop_type
                AND pp.target_date = mp.price_date
            WHERE pp.crop_type = %s
                AND pp.target_date <= CURRENT_DATE
                AND pp.accuracy_calculated = FALSE
        """

        params = [crop_type]
        if mandi_id:
            query += " AND pp.mandi_id = %s"
            params.append(mandi_id)

        cursor.execute(query, params)
        results = cursor.fetchall()

        if not results:
            return None

        # Calculate MAPE (Mean Absolute Percentage Error)
        errors = []

        for row in results:
            pred_id, predicted, days_ahead, actual = row

            # Calculate percentage error
            error = abs((predicted - actual) / actual) * 100
            errors.append(error)

            # Update prediction record
            cursor.execute("""
                UPDATE price_predictions
                SET actual_price = %s,
                    absolute_error = %s,
                    percentage_error = %s,
                    accuracy_calculated = TRUE,
                    accuracy_calculated_at = %s
                WHERE id = %s
            """, (actual, abs(predicted - actual), error, datetime.now(), pred_id))

        self.db.commit()

        # Calculate overall metrics
        mape = sum(errors) / len(errors)

        return {
            'mape': round(mape, 2),
            'predictions_evaluated': len(errors),
            'accuracy_percent': round(100 - mape, 2)
        }

    def get_accuracy_report(self, crop_type, time_horizon=None):
        """Get accuracy report for display to users"""
        cursor = self.db.cursor()

        query = """
            SELECT
                days_ahead,
                COUNT(*) as prediction_count,
                AVG(percentage_error) as avg_error,
                MIN(percentage_error) as best_error,
                MAX(percentage_error) as worst_error
            FROM price_predictions
            WHERE crop_type = %s
                AND accuracy_calculated = TRUE
        """

        params = [crop_type]

        if time_horizon:
            query += " AND days_ahead = %s"
            params.append(time_horizon)

        query += " GROUP BY days_ahead ORDER BY days_ahead"

        cursor.execute(query, params)
        results = cursor.fetchall()

        report = []
        for row in results:
            days, count, avg_err, best, worst = row
            report.append({
                'days_ahead': days,
                'prediction_count': count,
                'accuracy_percent': round(100 - avg_err, 1),
                'best_accuracy': round(100 - best, 1),
                'worst_accuracy': round(100 - worst, 1)
            })

        return report
```

```sql
-- Price predictions
CREATE TABLE price_predictions (
  id TEXT PRIMARY KEY,
  crop_type TEXT NOT NULL,
  mandi_id TEXT NOT NULL,
  prediction_date DATE NOT NULL,  -- When prediction was made
  target_date DATE NOT NULL,      -- Date being predicted
  days_ahead INTEGER NOT NULL,    -- 30, 60, or 90
  predicted_price DECIMAL(10, 2) NOT NULL,
  lower_bound DECIMAL(10, 2),
  upper_bound DECIMAL(10, 2),
  confidence_level TEXT,          -- 'high', 'medium', 'low'
  confidence_score INTEGER,       -- 0-100
  trend TEXT,                     -- 'rising', 'stable', 'falling'
  actual_price DECIMAL(10, 2),   -- Filled when target_date arrives
  absolute_error DECIMAL(10, 2),
  percentage_error DECIMAL(10, 2),
  accuracy_calculated BOOLEAN DEFAULT FALSE,
  accuracy_calculated_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (mandi_id) REFERENCES mandis(id)
);

CREATE INDEX idx_price_predictions_crop_target ON price_predictions(crop_type, target_date);
CREATE INDEX idx_price_predictions_accuracy ON price_predictions(accuracy_calculated, target_date);

-- Prediction explanatory factors
CREATE TABLE prediction_factors (
  id TEXT PRIMARY KEY,
  prediction_id TEXT NOT NULL,
  factor_name TEXT NOT NULL,      -- 'Harvest Season', 'Demand Spike', etc.
  impact TEXT NOT NULL,            -- 'positive', 'negative', 'neutral'
  description TEXT,
  rank INTEGER,                    -- Order of importance
  FOREIGN KEY (prediction_id) REFERENCES price_predictions(id)
);

CREATE INDEX idx_prediction_factors_prediction ON prediction_factors(prediction_id);

-- Model training log
CREATE TABLE model_training_log (
  id TEXT PRIMARY KEY,
  crop_type TEXT NOT NULL,
  mandi_id TEXT,
  training_date TIMESTAMP NOT NULL,
  training_samples INTEGER,
  model_version TEXT,
  status TEXT,  -- 'success', 'failed'
  error_message TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_model_training_log_date ON model_training_log(training_date DESC);
```
