**Cost-Benefit Analyzer**

```javascript
// CostBenefitAnalyzer.js
class CostBenefitAnalyzer {
  async analyzePractice(practiceId, fieldId) {
    const practice = await db.query(
      'SELECT * FROM climate_adaptation_practices WHERE id = $1',
      [practiceId]
    );

    const field = await db.query('SELECT * FROM fields WHERE id = $1', [fieldId]);

    if (!practice[0] || !field[0]) {
      throw new Error('Practice or field not found');
    }

    const practiceData = practice[0];
    const fieldData = field[0];

    // Calculate costs
    const implementationCost = this.calculateImplementationCost(practiceData, fieldData);

    // Calculate benefits
    const annualBenefits = this.calculateAnnualBenefits(practiceData, fieldData);

    // Calculate payback period
    const paybackPeriod = implementationCost.total / annualBenefits.total;

    // Calculate 5-year projection
    const fiveYearBenefits = annualBenefits.total * 5;
    const fiveYearCosts = implementationCost.total + (annualBenefits.maintenanceCost * 5);
    const fiveYearNetBenefit = fiveYearBenefits - fiveYearCosts;

    // Calculate ROI
    const roi = (fiveYearNetBenefit / implementationCost.total) * 100;

    return {
      practice_name: practiceData.practice_name,
      field_size_acres: fieldData.size_acres,

      costs: {
        implementation: implementationCost,
        annual_maintenance: annualBenefits.maintenanceCost,
        five_year_total: fiveYearCosts
      },

      benefits: {
        annual: annualBenefits,
        five_year_total: fiveYearBenefits
      },

      financial_metrics: {
        payback_period_years: Math.round(paybackPeriod * 10) / 10,
        roi_percent: Math.round(roi),
        five_year_net_benefit: fiveYearNetBenefit,
        break_even_year: Math.ceil(paybackPeriod)
      },

      comparison_with_status_quo: this.compareWithStatusQuo(
        fiveYearNetBenefit,
        practiceData,
        fieldData
      ),

      recommendation: this.generateRecommendation(paybackPeriod, roi, practiceData)
    };
  }

  calculateImplementationCost(practiceData, fieldData) {
    const costPerAcre = practiceData.cost_benefit?.implementation_cost_per_acre || 5000;
    const totalCost = costPerAcre * fieldData.size_acres;

    // Apply government subsidy if available
    let subsidyAmount = 0;
    if (practiceData.government_subsidy?.available) {
      const subsidyPercent = practiceData.government_subsidy.subsidy_percent / 100;
      subsidyAmount = Math.min(
        totalCost * subsidyPercent,
        practiceData.government_subsidy.max_subsidy_amount
      );
    }

    const netCost = totalCost - subsidyAmount;

    return {
      gross_cost: totalCost,
      government_subsidy: subsidyAmount,
      net_farmer_cost: netCost,
      cost_per_acre: costPerAcre,
      total: netCost
    };
  }

  calculateAnnualBenefits(practiceData, fieldData) {
    const benefits = practiceData.benefits || {};

    // Yield increase benefit
    const avgYield = 1200;  // kg per acre (wheat baseline)
    const marketPrice = 50;  // PKR per kg
    const yieldIncrease = avgYield * (benefits.yield_impact_percent / 100);
    const yieldBenefit = yieldIncrease * marketPrice * fieldData.size_acres;

    // Cost savings
    const savingsPerAcre = practiceData.cost_benefit?.annual_savings_per_acre || 0;
    const costSavings = savingsPerAcre * fieldData.size_acres;

    // Maintenance cost
    const maintenanceCost = practiceData.cost_benefit?.annual_maintenance_cost || 0;

    const totalAnnualBenefit = yieldBenefit + costSavings - maintenanceCost;

    return {
      yield_increase_benefit: yieldBenefit,
      cost_savings: costSavings,
      maintenanceCost: maintenanceCost,
      total: totalAnnualBenefit
    };
  }

  compareWithStatusQuo(fiveYearNetBenefit, practiceData, fieldData) {
    // Estimate status quo 5-year income
    const avgYield = 1200;
    const marketPrice = 50;
    const annualIncome = avgYield * marketPrice * fieldData.size_acres;
    const statusQuo5YearIncome = annualIncome * 5;

    // With climate adaptation
    const yieldIncrease = (practiceData.benefits?.yield_impact_percent || 0) / 100;
    const adapted5YearIncome = annualIncome * (1 + yieldIncrease) * 5;

    const additionalIncome = adapted5YearIncome - statusQuo5YearIncome + fiveYearNetBenefit;

    return {
      status_quo_5_year_income: statusQuo5YearIncome,
      adapted_5_year_income: adapted5YearIncome,
      additional_income: additionalIncome,
      income_increase_percent: (additionalIncome / statusQuo5YearIncome) * 100
    };
  }

  generateRecommendation(paybackPeriod, roi, practiceData) {
    if (paybackPeriod <= 1 && roi > 100) {
      return `Highly recommended! This practice pays for itself in less than 1 year with excellent ROI of ${Math.round(roi)}%.`;
    } else if (paybackPeriod <= 2 && roi > 50) {
      return `Strongly recommended. Payback period of ${Math.round(paybackPeriod * 10) / 10} years with solid ${Math.round(roi)}% ROI.`;
    } else if (paybackPeriod <= 3) {
      return `Good long-term investment. Benefits accumulate over ${Math.round(paybackPeriod)} years.`;
    } else {
      return `Consider if focused on long-term sustainability. Economic benefits may take ${Math.round(paybackPeriod)} years.`;
    }
  }
}
```

### Data Model Requirements

```sql
-- Climate adaptation practices
CREATE TABLE climate_adaptation_practices (
  id TEXT PRIMARY KEY,
  practice_name TEXT NOT NULL,
  category TEXT NOT NULL,              -- 'soil_health', 'water_management', 'crop_diversification', 'agroforestry'
  climate_risks_addressed TEXT,        -- JSON array

  description TEXT NOT NULL,
  implementation TEXT,                 -- JSON object
  benefits TEXT,                       -- JSON object
  cost_benefit TEXT,                   -- JSON object
  suitable_for TEXT,                   -- JSON object

  government_subsidy TEXT,             -- JSON object
  resources TEXT,                      -- JSON object (video_id, guide_url)
  expert_tips TEXT,                    -- JSON array

  adoption_count INTEGER DEFAULT 0,
  avg_satisfaction_rating REAL,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_practices_category ON climate_adaptation_practices(category);
CREATE INDEX idx_practices_risk ON climate_adaptation_practices USING GIN(climate_risks_addressed);

-- Video tutorials
CREATE TABLE video_tutorials (
  id TEXT PRIMARY KEY,
  practice_id TEXT NOT NULL,
  title TEXT NOT NULL,
  description TEXT,

  language TEXT NOT NULL,              -- 'urdu', 'punjabi', 'sindhi', 'pashto'
  duration_seconds INTEGER,
  file_size_mb INTEGER,

  video_url TEXT NOT NULL,
  thumbnail_url TEXT,
  subtitles_url TEXT,

  views INTEGER DEFAULT 0,
  avg_completion_percent INTEGER,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (practice_id) REFERENCES climate_adaptation_practices(id)
);

CREATE INDEX idx_videos_practice ON video_tutorials(practice_id);
CREATE INDEX idx_videos_language ON video_tutorials(language);

-- Success stories
CREATE TABLE farmer_success_stories (
  id TEXT PRIMARY KEY,
  practice_id TEXT NOT NULL,
  farmer_name TEXT NOT NULL,
  district TEXT NOT NULL,
  tehsil TEXT,
  farm_size_acres REAL,

  adoption_year INTEGER NOT NULL,
  years_of_experience INTEGER,

  before_adoption TEXT,                -- JSON object
  after_adoption TEXT,                 -- JSON object
  challenges_faced TEXT,               -- JSON array
  advice_to_others TEXT,

  testimonial_video_url TEXT,
  photos TEXT,                         -- JSON array

  contact_for_learning TEXT,           -- JSON object

  verified BOOLEAN DEFAULT FALSE,
  featured BOOLEAN DEFAULT FALSE,
  views INTEGER DEFAULT 0,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (practice_id) REFERENCES climate_adaptation_practices(id)
);

CREATE INDEX idx_stories_district ON farmer_success_stories(district, practice_id);
CREATE INDEX idx_stories_featured ON farmer_success_stories(featured, verified);

-- Expert forum
CREATE TABLE expert_forum_questions (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  practice_id TEXT,

  question_title TEXT NOT NULL,
  question_body TEXT NOT NULL,

  category TEXT,
  tags TEXT,                           -- JSON array

  status TEXT DEFAULT 'pending',       -- 'pending', 'answered', 'closed'

  expert_answer TEXT,                  -- JSON object
  community_answers TEXT,              -- JSON array

  views INTEGER DEFAULT 0,
  is_featured BOOLEAN DEFAULT FALSE,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (user_id) REFERENCES farmers(id)
);

CREATE INDEX idx_forum_status ON expert_forum_questions(status, created_at DESC);
CREATE INDEX idx_forum_practice ON expert_forum_questions(practice_id);

-- Practice adoption tracker
CREATE TABLE practice_adoptions (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  field_id TEXT NOT NULL,
  practice_id TEXT NOT NULL,

  adoption_status TEXT DEFAULT 'planning',  -- 'planning', 'in_progress', 'adopted'
  adoption_date DATE,
  implementation_notes TEXT,

  feedback TEXT,                       -- Periodic feedback on results
  satisfaction_rating INTEGER,         -- 1-5 stars

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (user_id) REFERENCES farmers(id),
  FOREIGN KEY (field_id) REFERENCES fields(id),
  FOREIGN KEY (practice_id) REFERENCES climate_adaptation_practices(id),

  UNIQUE(user_id, field_id, practice_id)
);

CREATE INDEX idx_adoptions_user ON practice_adoptions(user_id, adoption_status);
CREATE INDEX idx_adoptions_practice ON practice_adoptions(practice_id, adoption_status);
```

### Project Structure Notes

This story empowers farmers with proven climate adaptation strategies backed by cost-benefit analysis and local success stories. The comprehensive practice library, categorized by climate risk, provides actionable guidance on conservation agriculture, crop diversification, and agroforestry. Video tutorials in local languages make complex practices accessible, while success stories from neighboring farmers build trust and motivation. The expert forum connects farmers with agricultural extension services, creating a knowledge-sharing ecosystem for climate-resilient farming.
