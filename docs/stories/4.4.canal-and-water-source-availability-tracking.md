## Story
**As a** farmer dependent on canal irrigation,
**I want** alerts about canal water availability,
**so that** I can plan irrigation when water is actually available.

**Canal Schedule Management**

```javascript
// CanalScheduleService.js
class CanalScheduleService {
  /**
   * Parse canal schedule from irrigation department data
   * Punjab Irrigation Department format example
   */
  parsePunjabCanalSchedule(csvData) {
    // CSV format: Canal Name, Branch, Outlet No, Day Pattern, Start Time, Duration
    const schedules = [];

    csvData.forEach(row => {
      const schedule = {
        id: `canal-${Date.now()}-${Math.random()}`,
        canal_name: row.canal_name,
        branch_name: row.branch_name,
        outlet_number: row.outlet_number,
        schedule_type: this.determineScheduleType(row.day_pattern),
        day_pattern: row.day_pattern,  // e.g., "Mon,Wed,Fri" or "Weekly rotation"
        start_time: row.start_time,    // e.g., "08:00"
        duration_hours: parseFloat(row.duration),
        flow_rate_cusecs: parseFloat(row.flow_rate) || null,
        effective_from: new Date(row.effective_from),
        effective_until: row.effective_until ? new Date(row.effective_until) : null,
        region: row.district,
        data_source: 'Punjab Irrigation Department'
      };

      schedules.push(schedule);
    });

    return schedules;
  }

  determineScheduleType(dayPattern) {
    if (dayPattern.includes('rotation')) return 'rotation';
    if (dayPattern.includes(',')) return 'fixed_days';
    if (dayPattern.includes('daily')) return 'daily';
    return 'custom';
  }

  /**
   * Get canal schedule for specific outlet
   */
  async getCanalScheduleForField(field) {
    if (!field.water_source_canal_outlet) return null;

    const schedule = await db.queryOne(`
      SELECT * FROM canal_schedules
      WHERE outlet_number = $1
        AND (effective_until IS NULL OR effective_until >= NOW())
      ORDER BY effective_from DESC
      LIMIT 1
    `, [field.water_source_canal_outlet]);

    return schedule;
  }

  /**
   * Calculate next available water turns for the week
   */
  async getUpcomingWaterTurns(canalSchedule, startDate = new Date()) {
    const turns = [];

    if (canalSchedule.schedule_type === 'fixed_days') {
      const days = canalSchedule.day_pattern.split(',');  // e.g., "Mon,Wed,Fri"

      for (let i = 0; i < 14; i++) {  // Next 14 days
        const date = new Date(startDate);
        date.setDate(date.getDate() + i);

        const dayName = date.toLocaleDateString('en-US', { weekday: 'short' });

        if (days.includes(dayName)) {
          turns.push({
            date: date,
            start_time: canalSchedule.start_time,
            duration_hours: canalSchedule.duration_hours,
            canal_name: canalSchedule.canal_name
          });
        }
      }
    } else if (canalSchedule.schedule_type === 'rotation') {
      // Weekly rotation logic (simplified)
      // In reality, rotation schedules are more complex
      const rotationDays = 7;  // Example: 7-day rotation

      for (let i = 0; i < 14; i += rotationDays) {
        const date = new Date(startDate);
        date.setDate(date.getDate() + i);

        turns.push({
          date: date,
          start_time: canalSchedule.start_time,
          duration_hours: canalSchedule.duration_hours,
          canal_name: canalSchedule.canal_name
        });
      }
    }

    return turns;
  }

  /**
   * Check if irrigation is scheduled during canal availability
   */
  async validateIrrigationSchedule(irrigationEvent, fieldId) {
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);

    if (field.water_source !== 'canal') {
      return { valid: true, message: 'Field uses non-canal water source' };
    }

    const canalSchedule = await this.getCanalScheduleForField(field);
    if (!canalSchedule) {
      return { valid: false, message: 'No canal schedule available' };
    }

    const upcomingTurns = await this.getUpcomingWaterTurns(canalSchedule);
    const irrigationDate = new Date(irrigationEvent.scheduled_date);

    const matchingTurn = upcomingTurns.find(turn =>
      this.isSameDay(turn.date, irrigationDate)
    );

    if (!matchingTurn) {
      return {
        valid: false,
        message: `Canal not available on ${irrigationDate.toDateString()}`,
        alternative_dates: upcomingTurns.slice(0, 3).map(t => t.date)
      };
    }

    return { valid: true, turn: matchingTurn };
  }

  isSameDay(date1, date2) {
    return date1.toDateString() === date2.toDateString();
  }
}
```

```javascript
// WaterAvailabilityReportService.js
class WaterAvailabilityReportService {
  /**
   * User reports current water availability
   */
  async submitWaterReport(userId, fieldId, status, comments = null) {
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);

    const report = {
      id: `report-${Date.now()}`,
      user_id: userId,
      field_id: fieldId,
      water_source_type: field.water_source,
      canal_outlet: field.water_source_canal_outlet,
      status: status,  // 'flowing', 'stopped', 'low_pressure', 'intermittent'
      location_lat: field.latitude,
      location_lng: field.longitude,
      region: field.district || 'unknown',
      comments: comments,
      upvotes: 0,
      downvotes: 0,
      is_verified: false,
      created_at: new Date()
    };

    await db.insert('water_availability_reports', report);

    // Notify nearby users
    await this.notifyNearbyUsers(report);

    return report;
  }

  /**
   * Get recent reports for a region
   */
  async getRegionalReports(region, waterSourceType = null, hours = 24) {
    const since = new Date(Date.now() - hours * 60 * 60 * 1000);

    let query = `
      SELECT
        war.*,
        f.name as farmer_name,
        COUNT(*) OVER (PARTITION BY war.canal_outlet, war.status) as similar_reports
      FROM water_availability_reports war
      LEFT JOIN farmers f ON war.user_id = f.id
      WHERE war.region = $1
        AND war.created_at >= $2
    `;

    const params = [region, since];

    if (waterSourceType) {
      query += ` AND war.water_source_type = $3`;
      params.push(waterSourceType);
    }

    query += ` ORDER BY war.created_at DESC`;

    const reports = await db.query(query, params);

    return reports;
  }

  /**
   * Aggregate reports to determine current availability status
   */
  async getAggregatedStatus(canalOutlet, hours = 6) {
    const since = new Date(Date.now() - hours * 60 * 60 * 1000);

    const reports = await db.query(`
      SELECT
        status,
        COUNT(*) as count,
        SUM(upvotes - downvotes) as net_votes
      FROM water_availability_reports
      WHERE canal_outlet = $1
        AND created_at >= $2
      GROUP BY status
      ORDER BY count DESC, net_votes DESC
    `, [canalOutlet, since]);

    if (reports.length === 0) {
      return { status: 'unknown', confidence: 0, report_count: 0 };
    }

    const topStatus = reports[0];
    const totalReports = reports.reduce((sum, r) => sum + r.count, 0);

    return {
      status: topStatus.status,
      confidence: (topStatus.count / totalReports) * 100,
      report_count: totalReports,
      last_updated: new Date()
    };
  }

  /**
   * Notify nearby users about water availability changes
   */
  async notifyNearbyUsers(report) {
    // Find users within 5km with same canal outlet
    const nearbyUsers = await db.query(`
      SELECT DISTINCT f.id, f.name
      FROM farmers f
      JOIN fields fd ON f.id = fd.farmer_id
      WHERE fd.water_source_canal_outlet = $1
        AND ST_Distance(
          ST_MakePoint(fd.longitude, fd.latitude)::geography,
          ST_MakePoint($2, $3)::geography
        ) < 5000  -- 5km radius
        AND f.id != $4
    `, [report.canal_outlet, report.location_lng, report.location_lat, report.user_id]);

    for (const user of nearbyUsers) {
      await this.sendWaterAvailabilityNotification(user.id, report);
    }
  }

  async sendWaterAvailabilityNotification(userId, report) {
    const statusMessages = {
      'flowing': 'Canal water is now flowing',
      'stopped': 'Canal water has stopped',
      'low_pressure': 'Low water pressure reported',
      'intermittent': 'Intermittent water flow reported'
    };

    // Send push notification
    // Implementation depends on notification service
  }

  /**
   * Upvote/downvote report for accuracy
   */
  async voteOnReport(reportId, userId, voteType) {
    const field = voteType === 'upvote' ? 'upvotes' : 'downvotes';

    await db.query(`
      UPDATE water_availability_reports
      SET ${field} = ${field} + 1
      WHERE id = $1
    `, [reportId]);

    // Track user vote to prevent duplicate voting
    await db.insert('report_votes', {
      report_id: reportId,
      user_id: userId,
      vote_type: voteType
    });
  }
}
```

**Alternative Irrigation Timing**

```javascript
// WaterAvailabilityOptimizer.js
class WaterAvailabilityOptimizer {
  /**
   * Suggest best irrigation window based on water and weather
   */
  async suggestOptimalIrrigationWindow(fieldId, cropSeasonId, daysAhead = 7) {
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);
    const weatherForecast = await WeatherDataService.get7DayForecast(field.latitude, field.longitude);

    let candidates = [];

    if (field.water_source === 'canal') {
      const canalSchedule = await new CanalScheduleService().getCanalScheduleForField(field);
      if (canalSchedule) {
        const waterTurns = await new CanalScheduleService().getUpcomingWaterTurns(canalSchedule);

        // Prioritize days with canal availability + no rain
        for (const turn of waterTurns) {
          const weather = weatherForecast.find(w =>
            this.isSameDay(new Date(w.forecast_date), turn.date)
          );

          if (weather && weather.rainfall_mm < 5) {  // Less than 5mm rain
            candidates.push({
              date: turn.date,
              water_source: 'canal',
              cost: 0,
              reliability: 0.7,
              score: 90 - weather.precipitation_probability  // High score if low rain probability
            });
          }
        }
      }
    }

    // Tube well is always available (if exists)
    if (field.water_source === 'tube_well' || field.water_source === 'mixed') {
      for (let i = 0; i < daysAhead; i++) {
        const date = this.addDays(new Date(), i);
        const weather = weatherForecast[i];

        if (weather && weather.rainfall_mm < 5) {
          candidates.push({
            date: date,
            water_source: 'tube_well',
            cost: 150 * 4,  // 4 hours irrigation × PKR 150/hour
            reliability: 0.95,
            score: 70 - weather.precipitation_probability  // Lower than canal due to cost
          });
        }
      }
    }

    // Sort by score (best option first)
    candidates.sort((a, b) => b.score - a.score);

    return candidates.slice(0, 3);  // Top 3 options
  }

  isSameDay(date1, date2) {
    return date1.toDateString() === date2.toDateString();
  }

  addDays(date, days) {
    const result = new Date(date);
    result.setDate(result.getDate() + days);
    return result;
  }
}
```
