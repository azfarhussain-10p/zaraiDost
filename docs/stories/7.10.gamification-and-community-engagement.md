**Points Service**

```javascript
// PointsService.js
class PointsService {
  constructor() {
    // Point values for different actions
    this.pointValues = {
      // Forum activity
      'forum_post': 5,
      'forum_reply': 10,
      'helpful_answer': 20, // Upvoted answer
      'accepted_answer': 30,

      // Success stories
      'success_story_submitted': 20,
      'success_story_verified': 50,

      // Community engagement
      'daily_login': 5,
      'profile_completed': 25,
      'connection_made': 3,
      'group_joined': 10,

      // Expert Q&A
      'expert_question_asked': 5,
      'expert_answer_given': 15,

      // Challenges
      'challenge_completed': 100,

      // Penalties
      'content_removed': -50,
      'warning_received': -100
    };

    // Level thresholds
    this.levelThresholds = [
      { level: 'Bronze', number: 1, minPoints: 0 },
      { level: 'Bronze', number: 2, minPoints: 100 },
      { level: 'Bronze', number: 3, minPoints: 250 },
      { level: 'Silver', number: 4, minPoints: 500 },
      { level: 'Silver', number: 5, minPoints: 1000 },
      { level: 'Silver', number: 6, minPoints: 2000 },
      { level: 'Gold', number: 7, minPoints: 5000 },
      { level: 'Gold', number: 8, minPoints: 10000 },
      { level: 'Platinum', number: 9, minPoints: 25000 },
      { level: 'Diamond', number: 10, minPoints: 50000 }
    ];
  }

  async awardPoints(farmerId, actionType, referenceType, referenceId, description = null) {
    const points = this.pointValues[actionType] || 0;

    if (points === 0) {
      console.warn(`Unknown action type: ${actionType}`);
      return null;
    }

    // Check daily limit to prevent gaming
    const todayPoints = await this.getTodayPoints(farmerId);
    const dailyLimit = 500;

    if (todayPoints + points > dailyLimit && points > 0) {
      console.log(`Daily point limit reached for farmer ${farmerId}`);
      return null;
    }

    // Create transaction
    const transactionId = `pts-${Date.now()}-${Math.random()}`;

    await db.query(`
      INSERT INTO points_transactions (
        id, farmer_id, points_delta, action_type,
        reference_type, reference_id, description
      ) VALUES ($1, $2, $3, $4, $5, $6, $7)
    `, [transactionId, farmerId, points, actionType, referenceType, referenceId, description]);

    // Update total points
    await this.updateTotalPoints(farmerId, points);

    // Check for level up
    await this.checkLevelUp(farmerId);

    // Check for badge unlocks
    await this.checkBadgeUnlocks(farmerId);

    // Emit event for real-time updates
    EventBus.emit('points_awarded', {
      farmerId,
      points,
      actionType,
      transactionId
    });

    return {
      transactionId,
      points,
      totalPoints: await this.getTotalPoints(farmerId)
    };
  }

  async updateTotalPoints(farmerId, pointsDelta) {
    // Get or create reputation record
    const reputation = await db.query(`
      SELECT * FROM user_reputation WHERE farmer_id = $1
    `, [farmerId]);

    if (reputation.length === 0) {
      await db.query(`
        INSERT INTO user_reputation (
          id, farmer_id, total_points, level, level_number
        ) VALUES ($1, $2, $3, 'Bronze', 1)
      `, [`rep-${Date.now()}-${Math.random()}`, farmerId, Math.max(0, pointsDelta)]);
    } else {
      await db.query(`
        UPDATE user_reputation
        SET total_points = GREATEST(0, total_points + $1),
            updated_at = NOW()
        WHERE farmer_id = $2
      `, [pointsDelta, farmerId]);
    }

    // Update activity counts based on action
    await this.updateActivityCounts(farmerId, pointsDelta);
  }

  async updateActivityCounts(farmerId, actionType) {
    const countUpdates = {
      'forum_reply': 'total_answers = total_answers + 1',
      'helpful_answer': 'helpful_answers = helpful_answers + 1',
      'success_story_verified': 'success_stories = success_stories + 1',
      'connection_made': 'connections_count = connections_count + 1',
      'daily_login': 'days_active = days_active + 1'
    };

    const updateSql = countUpdates[actionType];
    if (updateSql) {
      await db.query(`
        UPDATE user_reputation
        SET ${updateSql}
        WHERE farmer_id = $1
      `, [farmerId]);
    }

    // Update streak for daily login
    if (actionType === 'daily_login') {
      await this.updateStreak(farmerId);
    }
  }

  async updateStreak(farmerId) {
    const reputation = await db.query(`
      SELECT * FROM user_reputation WHERE farmer_id = $1
    `, [farmerId]);

    if (reputation.length === 0) return;

    const lastActive = reputation[0].last_active_date;
    const today = new Date().toISOString().split('T')[0];
    const yesterday = new Date(Date.now() - 86400000).toISOString().split('T')[0];

    let newStreak = 1;

    if (lastActive === yesterday) {
      // Continue streak
      newStreak = (reputation[0].current_streak_days || 0) + 1;
    } else if (lastActive === today) {
      // Already logged in today
      return;
    } else {
      // Streak broken, reset to 1
      newStreak = 1;
    }

    const longestStreak = Math.max(newStreak, reputation[0].longest_streak_days || 0);

    await db.query(`
      UPDATE user_reputation
      SET current_streak_days = $1,
          longest_streak_days = $2,
          last_active_date = $3
      WHERE farmer_id = $4
    `, [newStreak, longestStreak, today, farmerId]);

    // Check for streak milestones
    if ([7, 30, 100, 365].includes(newStreak)) {
      await this.createMilestone(farmerId, `streak_${newStreak}_days`, newStreak);
    }
  }

  async checkLevelUp(farmerId) {
    const reputation = await db.query(`
      SELECT * FROM user_reputation WHERE farmer_id = $1
    `, [farmerId]);

    if (reputation.length === 0) return;

    const currentPoints = reputation[0].total_points;
    const currentLevel = reputation[0].level_number;

    // Find new level
    let newLevel = this.levelThresholds[0];
    for (const level of this.levelThresholds) {
      if (currentPoints >= level.minPoints) {
        newLevel = level;
      } else {
        break;
      }
    }

    if (newLevel.number > currentLevel) {
      // Level up!
      await db.query(`
        UPDATE user_reputation
        SET level = $1,
            level_number = $2,
            level_up_date = NOW()
        WHERE farmer_id = $3
      `, [newLevel.level, newLevel.number, farmerId]);

      // Notify user
      await NotificationService.send(farmerId, {
        type: 'level_up',
        title: 'Level Up!',
        body: `You've reached ${newLevel.level} Level ${newLevel.number}!`,
        data: { level: newLevel.level, levelNumber: newLevel.number }
      });

      // Award level-up badge if exists
      await this.checkBadgeUnlocks(farmerId);
    }
  }

  async getTodayPoints(farmerId) {
    const today = new Date().toISOString().split('T')[0];

    const result = await db.query(`
      SELECT COALESCE(SUM(points_delta), 0) as total
      FROM points_transactions
      WHERE farmer_id = $1
        AND DATE(created_at) = $2
        AND points_delta > 0
    `, [farmerId, today]);

    return result[0].total;
  }

  async getTotalPoints(farmerId) {
    const reputation = await db.query(`
      SELECT total_points FROM user_reputation WHERE farmer_id = $1
    `, [farmerId]);

    return reputation.length > 0 ? reputation[0].total_points : 0;
  }
}
```

```javascript
// BadgeService.js
class BadgeService {
  async checkBadgeUnlocks(farmerId) {
    // Get farmer's current stats
    const reputation = await db.query(`
      SELECT * FROM user_reputation WHERE farmer_id = $1
    `, [farmerId]);

    if (reputation.length === 0) return [];

    const stats = reputation[0];

    // Get all badges farmer hasn't earned
    const unearnedBadges = await db.query(`
      SELECT b.* FROM badges b
      WHERE b.is_active = TRUE
        AND b.id NOT IN (
          SELECT badge_id FROM user_badges WHERE farmer_id = $1
        )
    `, [farmerId]);

    const newlyEarned = [];

    for (const badge of unearnedBadges) {
      if (this.meetsUnlockCriteria(stats, badge.unlock_criteria)) {
        await this.awardBadge(farmerId, badge.id);
        newlyEarned.push(badge);
      }
    }

    return newlyEarned;
  }

  meetsUnlockCriteria(stats, criteria) {
    const { type, threshold, conditions } = criteria;

    switch (type) {
      case 'answer_count':
        return stats.total_answers >= threshold;

      case 'helpful_answer_count':
        return stats.helpful_answers >= threshold;

      case 'success_story_count':
        return stats.success_stories >= threshold;

      case 'connection_count':
        return stats.connections_count >= threshold;

      case 'points_total':
        return stats.total_points >= threshold;

      case 'level_reached':
        return stats.level_number >= threshold;

      case 'streak_days':
        return stats.current_streak_days >= threshold;

      case 'longest_streak':
        return stats.longest_streak_days >= threshold;

      case 'combined':
        // Multiple conditions (AND logic)
        return conditions.every(cond => this.meetsUnlockCriteria(stats, cond));

      default:
        return false;
    }
  }

  async awardBadge(farmerId, badgeId) {
    const userBadgeId = `ubadge-${Date.now()}-${Math.random()}`;

    await db.query(`
      INSERT INTO user_badges (id, farmer_id, badge_id)
      VALUES ($1, $2, $3)
    `, [userBadgeId, farmerId, badgeId]);

    // Get badge details
    const badge = await db.query(`
      SELECT * FROM badges WHERE id = $1
    `, [badgeId]);

    if (badge.length === 0) return;

    // Award bonus points
    if (badge[0].points_reward > 0) {
      await PointsService.awardPoints(
        farmerId,
        'badge_earned',
        'badge',
        badgeId,
        `Earned badge: ${badge[0].name_en}`
      );
    }

    // Notify user with celebration
    await NotificationService.send(farmerId, {
      type: 'badge_earned',
      title: 'New Badge Earned!',
      body: `You've earned the "${badge[0].name_en}" badge!`,
      data: { badgeId, badgeName: badge[0].name_en, badgeIcon: badge[0].icon_url }
    });

    // Mark for celebration UI
```

```javascript
// LeaderboardService.js
class LeaderboardService {
  async updateLeaderboards() {
    // Global all-time leaderboard
    await this.updateGlobalLeaderboard();

    // Regional leaderboards
    await this.updateRegionalLeaderboards();

    // Monthly leaderboard
    await this.updatePeriodLeaderboard('monthly');

    // Weekly leaderboard
    await this.updatePeriodLeaderboard('weekly');
  }

  async updateGlobalLeaderboard() {
    const top100 = await db.query(`
      SELECT
        ur.farmer_id,
        ur.total_points,
        ur.level,
        ur.level_number,
        f.name,
        fpd.region,
        (
          SELECT COUNT(*) FROM user_badges ub WHERE ub.farmer_id = ur.farmer_id
        ) as badge_count,
        ROW_NUMBER() OVER (ORDER BY ur.total_points DESC) as rank
      FROM user_reputation ur
      JOIN farmers f ON ur.farmer_id = f.id
      LEFT JOIN farmer_profile_details fpd ON f.id = fpd.farmer_id
      JOIN leaderboard_participation lp ON ur.farmer_id = lp.farmer_id
      WHERE lp.opted_in = TRUE
        AND f.is_active = TRUE
      ORDER BY ur.total_points DESC
      LIMIT 100
    `);

    await this.saveLeaderboard('global_all_time', top100);
  }

  async updateRegionalLeaderboards() {
    const regions = ['Punjab', 'Sindh', 'KPK', 'Balochistan'];

    for (const region of regions) {
      const top50 = await db.query(`
        SELECT
          ur.farmer_id,
          ur.total_points,
          ur.level,
          f.name,
          ROW_NUMBER() OVER (ORDER BY ur.total_points DESC) as rank
        FROM user_reputation ur
        JOIN farmers f ON ur.farmer_id = f.id
        JOIN farmer_profile_details fpd ON f.id = fpd.farmer_id
        JOIN leaderboard_participation lp ON ur.farmer_id = lp.farmer_id
        WHERE lp.opted_in = TRUE
          AND f.is_active = TRUE
          AND fpd.region = $1
        ORDER BY ur.total_points DESC
        LIMIT 50
      `, [region]);

      await this.saveLeaderboard(`regional_${region.toLowerCase()}`, top50);
    }
  }

  async updatePeriodLeaderboard(period) {
    const { startDate, endDate } = this.getPeriodDates(period);

    const topUsers = await db.query(`
      SELECT
        pt.farmer_id,
        SUM(pt.points_delta) as period_points,
        f.name,
        ROW_NUMBER() OVER (ORDER BY SUM(pt.points_delta) DESC) as rank
      FROM points_transactions pt
      JOIN farmers f ON pt.farmer_id = f.id
      JOIN leaderboard_participation lp ON pt.farmer_id = lp.farmer_id
      WHERE lp.opted_in = TRUE
        AND f.is_active = TRUE
        AND pt.created_at BETWEEN $1 AND $2
        AND pt.points_delta > 0
      GROUP BY pt.farmer_id, f.name
      ORDER BY period_points DESC
      LIMIT 100
    `, [startDate, endDate]);

    await this.saveLeaderboard(
      `${period}_${startDate.toISOString().split('T')[0]}`,
      topUsers,
      startDate,
      endDate
    );
  }

  getPeriodDates(period) {
    const now = new Date();
    let startDate, endDate;

    if (period === 'weekly') {
      // Start of current week (Monday)
      startDate = new Date(now);
      startDate.setDate(now.getDate() - now.getDay() + 1);
      startDate.setHours(0, 0, 0, 0);

      endDate = new Date(startDate);
      endDate.setDate(startDate.getDate() + 7);
    } else if (period === 'monthly') {
      // Start of current month
      startDate = new Date(now.getFullYear(), now.getMonth(), 1);
      endDate = new Date(now.getFullYear(), now.getMonth() + 1, 1);
    }

    return { startDate, endDate };
  }

  async saveLeaderboard(leaderboardKey, entries, periodStart = null, periodEnd = null) {
    const id = `lb-${Date.now()}-${Math.random()}`;

    await db.query(`
      INSERT INTO leaderboards (
        id, leaderboard_type, leaderboard_key,
        period_start, period_end, top_entries
      ) VALUES ($1, $2, $3, $4, $5, $6)
      ON CONFLICT (leaderboard_key, period_start)
      DO UPDATE SET
        top_entries = $6,
        last_updated = NOW()
    `, [
      id,
      leaderboardKey.split('_')[0],
      leaderboardKey,
      periodStart,
      periodEnd,
      JSON.stringify(entries)
    ]);
  }

  async getUserRank(farmerId, leaderboardKey) {
    // Get full leaderboard
    const leaderboard = await db.query(`
      SELECT top_entries FROM leaderboards
      WHERE leaderboard_key = $1
      ORDER BY last_updated DESC
      LIMIT 1
    `, [leaderboardKey]);

    if (leaderboard.length === 0) return null;

    const entries = JSON.parse(leaderboard[0].top_entries);
    const userEntry = entries.find(e => e.farmer_id === farmerId);

    if (userEntry) {
      return userEntry.rank;
    }

    // User not in top 100, calculate approximate rank
    const totalUsers = await db.query(`
      SELECT COUNT(*) as count
      FROM user_reputation ur
      JOIN leaderboard_participation lp ON ur.farmer_id = lp.farmer_id
      WHERE lp.opted_in = TRUE
    `);

    const userPoints = await db.query(`
      SELECT total_points FROM user_reputation WHERE farmer_id = $1
    `, [farmerId]);

    if (userPoints.length === 0) return null;

    const usersAbove = await db.query(`
      SELECT COUNT(*) as count
      FROM user_reputation ur
      JOIN leaderboard_participation lp ON ur.farmer_id = lp.farmer_id
      WHERE lp.opted_in = TRUE
        AND ur.total_points > $1
    `, [userPoints[0].total_points]);

    return usersAbove[0].count + 1;
  }
}
```

### Cultural Considerations

**Islamic Compliance**:
- Avoid gambling-like mechanics (no randomness, no betting)
- No interest-based rewards
- Focus on skill and effort-based recognition
- Clear value exchange (points for actions)

**Cultural Appropriateness**:
- Use farming metaphors (harvest, growth, seasons)
- Respect community values (helping, sharing, collaboration)
- Modest celebration animations
- Urdu/Punjabi/Sindhi terminology
- Regional themes (Punjab vs Sindh)

**Badge Design**:
- Culturally relevant icons (wheat, tractor, water, sun)
- No human/animal imagery (Islamic considerations)
- Simple, recognizable symbols
- Colors with cultural meaning (green for agriculture)
