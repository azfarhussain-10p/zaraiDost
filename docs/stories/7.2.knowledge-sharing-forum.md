# Story 7.2: Knowledge Sharing Forum

## Status
Draft

## Story
**As a** farmer with questions or experiences,
**I want** a community forum to ask questions and share insights,
**so that** I can learn from and help other farmers.

## Acceptance Criteria
1. AC1: Community forum with topic categories (crops, pests, irrigation, markets, etc.)
2. AC2: Users can post questions, answer questions, share experiences
3. AC3: Rich media support: photos, videos (for showing crop issues, techniques)
4. AC4: Upvoting system for helpful answers
5. AC5: Search and filter posts by topic, crop, region
6. AC6: Notifications when someone answers user's question
7. AC7: Moderation system to flag inappropriate content

**Forum Service Layer**

```javascript
// ForumService.js
class ForumService {
  async createPost(farmerId, postData) {
    const { categoryId, title, content, postType, cropTags, mediaUrls } = postData;

    // Get farmer's region for filtering
    const farmer = await FarmerRepository.getById(farmerId);

    const postId = `post-${Date.now()}-${Math.random()}`;
    const post = await db.query(`
      INSERT INTO forum_posts (
        id, farmer_id, category_id, title, content, post_type,
        region, crop_tags, media_urls
      ) VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)
      RETURNING *
    `, [
      postId, farmerId, categoryId, title, content, postType,
      farmer.region, cropTags, JSON.stringify(mediaUrls)
    ]);

    // Update category post count
    await db.query(`
      UPDATE forum_categories
      SET post_count = post_count + 1
      WHERE id = $1
    `, [categoryId]);

    // Auto-follow own posts
    await this.followPost(farmerId, postId);

    return post[0];
  }

  async getPosts(filters = {}, page = 1, limit = 20) {
    const { categoryId, status, region, cropTag, searchTerm, sortBy } = filters;
    const offset = (page - 1) * limit;

    let query = `
      SELECT
        p.*,
        f.name as author_name,
        f.profile_photo_url as author_photo,
        c.name_en as category_name,
        (SELECT COUNT(*) FROM forum_replies WHERE post_id = p.id) as reply_count
      FROM forum_posts p
      JOIN farmers f ON p.farmer_id = f.id
      JOIN forum_categories c ON p.category_id = c.id
      WHERE p.is_moderated = FALSE
    `;

    const params = [];
    let paramCount = 0;

    if (categoryId) {
      query += ` AND p.category_id = $${++paramCount}`;
      params.push(categoryId);
    }

    if (status) {
      query += ` AND p.status = $${++paramCount}`;
      params.push(status);
    }

    if (region) {
      query += ` AND p.region = $${++paramCount}`;
      params.push(region);
    }

    if (cropTag) {
      query += ` AND $${++paramCount} = ANY(p.crop_tags)`;
      params.push(cropTag);
    }

    if (searchTerm) {
      query += ` AND (
        to_tsvector('english', p.title || ' ' || p.content)
        @@ plainto_tsquery('english', $${++paramCount})
      )`;
      params.push(searchTerm);
    }

    // Sorting
    switch (sortBy) {
      case 'popular':
        query += ` ORDER BY p.upvote_count DESC, p.reply_count DESC`;
        break;
      case 'unanswered':
        query += ` AND p.status = 'open' AND p.reply_count = 0 ORDER BY p.created_at DESC`;
        break;
      case 'latest':
      default:
        query += ` ORDER BY p.last_activity_at DESC`;
    }

    query += ` LIMIT $${++paramCount} OFFSET $${++paramCount}`;
    params.push(limit, offset);

    const posts = await db.query(query, params);
    return posts;
  }

  async getPostDetails(postId, viewerFarmerId = null) {
    // Increment view count
    await db.query(`
      UPDATE forum_posts
      SET view_count = view_count + 1
      WHERE id = $1
    `, [postId]);

    // Get post with author details
    const post = await db.query(`
      SELECT
        p.*,
        f.name as author_name,
        f.profile_photo_url as author_photo,
        f.experience_years as author_experience,
        c.name_en as category_name
      FROM forum_posts p
      JOIN farmers f ON p.farmer_id = f.id
      JOIN forum_categories c ON p.category_id = c.id
      WHERE p.id = $1
    `, [postId]);

    if (post.length === 0) {
      throw new Error('Post not found');
    }

    // Get replies with vote counts
    const replies = await this.getReplies(postId, viewerFarmerId);

    return {
      post: post[0],
      replies
    };
  }

  async getReplies(postId, viewerFarmerId = null) {
    let query = `
      SELECT
        r.*,
        f.name as author_name,
        f.profile_photo_url as author_photo,
        f.experience_years as author_experience,
        r.upvote_count - r.downvote_count as net_votes
    `;

    // Include viewer's vote if logged in
    if (viewerFarmerId) {
      query += `,
        (SELECT vote_type FROM forum_votes
         WHERE farmer_id = $2 AND reply_id = r.id) as viewer_vote
      `;
    }

    query += `
      FROM forum_replies r
      JOIN farmers f ON r.farmer_id = f.id
      WHERE r.post_id = $1 AND r.is_moderated = FALSE
      ORDER BY r.is_best_answer DESC, net_votes DESC, r.created_at ASC
    `;

    const params = viewerFarmerId ? [postId, viewerFarmerId] : [postId];
    const replies = await db.query(query, params);

    return replies;
  }

  async createReply(farmerId, postId, content, mediaUrls = null, parentReplyId = null) {
    const replyId = `reply-${Date.now()}-${Math.random()}`;
    const reply = await db.query(`
      INSERT INTO forum_replies (
        id, post_id, farmer_id, parent_reply_id, content, media_urls
      ) VALUES ($1, $2, $3, $4, $5, $6)
      RETURNING *
    `, [replyId, postId, farmerId, parentReplyId, content, JSON.stringify(mediaUrls)]);

    // Update post reply count and last activity
    await db.query(`
      UPDATE forum_posts
      SET reply_count = reply_count + 1,
          last_activity_at = NOW()
      WHERE id = $1
    `, [postId]);

    // Send notifications to post author and followers
    await this.notifyPostFollowers(postId, farmerId, replyId);

    return reply[0];
  }

  async voteReply(farmerId, replyId, voteType) {
    // Check if user already voted
    const existingVote = await db.query(`
      SELECT vote_type FROM forum_votes
      WHERE farmer_id = $1 AND reply_id = $2
    `, [farmerId, replyId]);

    if (existingVote.length > 0) {
      // Update existing vote
      if (existingVote[0].vote_type === voteType) {
        // Remove vote (toggle)
        await db.query(`
          DELETE FROM forum_votes
          WHERE farmer_id = $1 AND reply_id = $2
        `, [farmerId, replyId]);

        // Decrement vote count
        const column = voteType === 'upvote' ? 'upvote_count' : 'downvote_count';
        await db.query(`
          UPDATE forum_replies
          SET ${column} = ${column} - 1
          WHERE id = $1
        `, [replyId]);
      } else {
        // Change vote type
        await db.query(`
          UPDATE forum_votes
          SET vote_type = $1
          WHERE farmer_id = $2 AND reply_id = $3
        `, [voteType, farmerId, replyId]);

        // Update counts
        const incColumn = voteType === 'upvote' ? 'upvote_count' : 'downvote_count';
        const decColumn = voteType === 'upvote' ? 'downvote_count' : 'upvote_count';
        await db.query(`
          UPDATE forum_replies
          SET ${incColumn} = ${incColumn} + 1,
              ${decColumn} = ${decColumn} - 1
          WHERE id = $1
        `, [replyId]);
      }
    } else {
      // Create new vote
      await db.query(`
        INSERT INTO forum_votes (id, farmer_id, reply_id, vote_type)
        VALUES ($1, $2, $3, $4)
      `, [`vote-${Date.now()}-${Math.random()}`, farmerId, replyId, voteType]);

      // Increment vote count
      const column = voteType === 'upvote' ? 'upvote_count' : 'downvote_count';
      await db.query(`
        UPDATE forum_replies
        SET ${column} = ${column} + 1
        WHERE id = $1
      `, [replyId]);

      // Notify reply author on upvotes (threshold: 5)
      if (voteType === 'upvote') {
        const reply = await db.query(`
          SELECT farmer_id, upvote_count FROM forum_replies WHERE id = $1
        `, [replyId]);

        if (reply[0].upvote_count >= 5) {
          await NotificationService.send(reply[0].farmer_id, {
            type: 'reply_upvoted',
            title: 'Your answer was helpful!',
            body: `Your answer received ${reply[0].upvote_count} upvotes`,
            data: { replyId }
          });
        }
      }
    }
  }

  async markBestAnswer(postId, replyId, postAuthorId) {
    // Verify caller is post author
    const post = await db.query(`
      SELECT farmer_id FROM forum_posts WHERE id = $1
    `, [postId]);

    if (post[0].farmer_id !== postAuthorId) {
      throw new Error('Only post author can mark best answer');
    }

    // Clear previous best answer
    await db.query(`
      UPDATE forum_replies
      SET is_best_answer = FALSE
      WHERE post_id = $1
    `, [postId]);

    // Set new best answer
    await db.query(`
      UPDATE forum_replies
      SET is_best_answer = TRUE
      WHERE id = $1
    `, [replyId]);

    // Update post status and best answer reference
    await db.query(`
      UPDATE forum_posts
      SET status = 'answered', best_answer_reply_id = $1
      WHERE id = $2
    `, [replyId, postId]);

    // Notify reply author
    const reply = await db.query(`
      SELECT farmer_id FROM forum_replies WHERE id = $1
    `, [replyId]);

    await NotificationService.send(reply[0].farmer_id, {
      type: 'best_answer',
      title: 'Your answer was marked as best!',
      body: 'The post author found your answer most helpful',
      data: { postId, replyId }
    });
  }

  async followPost(farmerId, postId) {
    await db.query(`
      INSERT INTO forum_post_follows (id, farmer_id, post_id)
      VALUES ($1, $2, $3)
      ON CONFLICT (farmer_id, post_id) DO NOTHING
    `, [`follow-${Date.now()}-${Math.random()}`, farmerId, postId]);
  }

  async notifyPostFollowers(postId, replierFarmerId, replyId) {
    // Get all followers except the replier
    const followers = await db.query(`
      SELECT f.farmer_id, p.title
      FROM forum_post_follows f
      JOIN forum_posts p ON f.post_id = p.id
      WHERE f.post_id = $1 AND f.farmer_id != $2
    `, [postId, replierFarmerId]);

    // Send notifications
    for (const follower of followers) {
      await NotificationService.send(follower.farmer_id, {
        type: 'post_reply',
        title: 'New answer on post you follow',
        body: `Someone replied to "${follower.title}"`,
        data: { postId, replyId }
      });
    }
  }

  async reportContent(reporterFarmerId, contentType, contentId, reason, details) {
    const reportId = `report-${Date.now()}-${Math.random()}`;
    await db.query(`
      INSERT INTO forum_content_reports (
        id, reporter_farmer_id, content_type, content_id,
        report_reason, report_details
      ) VALUES ($1, $2, $3, $4, $5, $6)
    `, [reportId, reporterFarmerId, contentType, contentId, reason, details]);

    // Check if content has multiple reports (auto-hide threshold: 5)
    const reportCount = await db.query(`
      SELECT COUNT(*) as count
      FROM forum_content_reports
      WHERE content_type = $1 AND content_id = $2 AND status = 'pending'
    `, [contentType, contentId]);

    if (reportCount[0].count >= 5) {
      // Auto-hide content
      const table = contentType === 'post' ? 'forum_posts' : 'forum_replies';
      await db.query(`
        UPDATE ${table}
        SET is_moderated = TRUE, moderation_reason = 'Multiple reports (auto-moderated)'
        WHERE id = $1
      `, [contentId]);
    }

    return reportId;
  }

  async searchPosts(searchTerm, filters = {}, page = 1, limit = 20) {
    const { categoryId, cropTag, region } = filters;
    const offset = (page - 1) * limit;

    let query = `
      SELECT
        p.*,
        f.name as author_name,
        c.name_en as category_name,
        ts_rank(to_tsvector('english', p.title || ' ' || p.content),
                plainto_tsquery('english', $1)) as rank
      FROM forum_posts p
      JOIN farmers f ON p.farmer_id = f.id
      JOIN forum_categories c ON p.category_id = c.id
      WHERE to_tsvector('english', p.title || ' ' || p.content)
            @@ plainto_tsquery('english', $1)
        AND p.is_moderated = FALSE
    `;

    const params = [searchTerm];
    let paramCount = 1;

    if (categoryId) {
      query += ` AND p.category_id = $${++paramCount}`;
      params.push(categoryId);
    }

    if (cropTag) {
      query += ` AND $${++paramCount} = ANY(p.crop_tags)`;
      params.push(cropTag);
    }

    if (region) {
      query += ` AND p.region = $${++paramCount}`;
      params.push(region);
    }

    query += ` ORDER BY rank DESC, p.last_activity_at DESC`;
    query += ` LIMIT $${++paramCount} OFFSET $${++paramCount}`;
    params.push(limit, offset);

    const results = await db.query(query, params);
    return results;
  }
}
```
