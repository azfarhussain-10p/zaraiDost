### Technical Implementation Details

**Text-to-Speech Libraries**

Option 1: `react-native-tts` (Community package)
- Native TTS for iOS and Android
- Supports multiple languages
- Good offline support
- Voice selection capability

Option 2: `expo-speech` (Expo framework)
- Simple API for basic TTS
- Good for managed Expo projects
- Limited customization

Option 3: Google Cloud Text-to-Speech API
- High-quality voices
- Requires internet connection
- Cost per character
- Good for online TTS enhancement

**Recommended**: Use `react-native-tts` for offline-first approach with optional Google Cloud TTS enhancement for online mode.

**TextToSpeechService Implementation**

```javascript
// TextToSpeechService.js
import Tts from 'react-native-tts';

class TextToSpeechService {
  constructor() {
    this.isInitialized = false;
    this.currentLanguage = 'ur-PK';
    this.speechRate = 1.0;
  }

  async initialize() {
    try {
      // Initialize TTS engine
      Tts.setDefaultLanguage(this.currentLanguage);
      Tts.setDefaultRate(this.speechRate);

      // Get available voices
      const voices = await Tts.voices();
      this.availableVoices = voices;

      // Set up event listeners
      Tts.addEventListener('tts-start', this.onStart);
      Tts.addEventListener('tts-finish', this.onFinish);
      Tts.addEventListener('tts-cancel', this.onCancel);
      Tts.addEventListener('tts-progress', this.onProgress);

      this.isInitialized = true;
      return true;
    } catch (error) {
      console.error('TTS initialization failed:', error);
      return false;
    }
  }

  async speak(text, options = {}) {
    if (!this.isInitialized) {
      await this.initialize();
    }

    const {
      language = this.currentLanguage,
      rate = this.speechRate,
      pitch = 1.0,
    } = options;

    try {
      // Set language-specific voice
      const voice = this.getVoiceForLanguage(language);
      if (voice) {
        await Tts.setDefaultVoice(voice.id);
      }

      await Tts.setDefaultRate(rate);
      await Tts.setDefaultPitch(pitch);

      // Clean text for TTS
      const cleanedText = this.cleanTextForTTS(text);

      // Speak
      await Tts.speak(cleanedText);
    } catch (error) {
      console.error('TTS speak failed:', error);
      throw error;
    }
  }

  async pause() {
    await Tts.stop();
  }

  async resume() {
    // Note: react-native-tts doesn't support true pause/resume
    // Need to track position and restart from there
  }

  async stop() {
    await Tts.stop();
  }

  async replay(text, options) {
    await this.stop();
    await this.speak(text, options);
  }

  getVoiceForLanguage(languageCode) {
    // Find best voice for language
    const voices = this.availableVoices?.filter(voice =>
      voice.language.startsWith(languageCode.substring(0, 2))
    );

    if (voices && voices.length > 0) {
      // Prefer female voice if available
      const femaleVoice = voices.find(v => v.name.toLowerCase().includes('female'));
      return femaleVoice || voices[0];
    }

    return null;
  }

  cleanTextForTTS(text) {
    // Remove markdown formatting
    let cleaned = text.replace(/[*_~`#]/g, '');

    // Handle numbers (spell out in local language if needed)
    // For now, leave as-is for TTS engine to handle

    // Remove excessive whitespace
    cleaned = cleaned.replace(/\s+/g, ' ').trim();

    return cleaned;
  }

  async getAvailableVoices(languageCode) {
    if (!this.isInitialized) {
      await this.initialize();
    }

    return this.availableVoices?.filter(voice =>
      voice.language.startsWith(languageCode.substring(0, 2))
    ) || [];
  }

  onStart = (event) => {
    console.log('TTS started');
  };

  onFinish = (event) => {
    console.log('TTS finished');
  };

  onCancel = (event) => {
    console.log('TTS cancelled');
  };

  onProgress = (event) => {
    // event contains: {location, length}
    console.log('TTS progress:', event);
  };

  cleanup() {
    Tts.removeEventListener('tts-start', this.onStart);
    Tts.removeEventListener('tts-finish', this.onFinish);
    Tts.removeEventListener('tts-cancel', this.onCancel);
    Tts.removeEventListener('tts-progress', this.onProgress);
  }
}

export default new TextToSpeechService();
```

**TTS Player Component**

```javascript
// TTSPlayer.js
const TTSPlayer = ({ text, language, autoPlay = false, onComplete }) => {
  const [isPlaying, setIsPlaying] = useState(false);
  const [hasStarted, setHasStarted] = useState(false);
  const [progress, setProgress] = useState(0);

  useEffect(() => {
    if (autoPlay && !hasStarted) {
      handlePlay();
    }
  }, [autoPlay]);

  const handlePlay = async () => {
    setIsPlaying(true);
    setHasStarted(true);

    try {
      await TextToSpeechService.speak(text, { language });
      setIsPlaying(false);
      onComplete?.();
    } catch (error) {
      console.error('TTS playback error:', error);
      setIsPlaying(false);
    }
  };

  const handlePause = async () => {
    await TextToSpeechService.pause();
    setIsPlaying(false);
  };

  const handleReplay = async () => {
    await TextToSpeechService.replay(text, { language });
  };

  return (
    <View style={styles.player}>
      <TouchableOpacity onPress={isPlaying ? handlePause : handlePlay}>
        <Icon name={isPlaying ? 'pause' : 'play-arrow'} size={32} />
      </TouchableOpacity>

      <TouchableOpacity onPress={handleReplay}>
        <Icon name="replay" size={28} />
      </TouchableOpacity>

      <View style={styles.progressBar}>
        <View style={[styles.progress, { width: `${progress}%` }]} />
      </View>
    </View>
  );
};
```

**Auto-Play Implementation**

```javascript
// AutoPlayManager.js
class AutoPlayManager {
  constructor() {
    this.enabled = false;
    this.delay = 500;  // ms before auto-play starts
  }

  async initialize() {
    // Load auto-play preference from storage
    const autoPlayEnabled = await AsyncStorage.getItem('tts_autoplay');
    this.enabled = autoPlayEnabled === 'true';
  }

  async setEnabled(enabled) {
    this.enabled = enabled;
    await AsyncStorage.setItem('tts_autoplay', enabled.toString());
  }

  shouldAutoPlay() {
    return this.enabled;
  }

  async playWithDelay(text, language) {
    if (!this.enabled) return;

    // Give user brief moment to cancel
    await new Promise(resolve => setTimeout(resolve, this.delay));

    await TextToSpeechService.speak(text, { language });
  }
}

export default new AutoPlayManager();
```

**Text Cleaning for TTS**

```javascript
// TTSTextCleaner.js
const cleanTextForTTS = (text) => {
  // Remove markdown formatting
  let cleaned = text;

  // Remove bold/italic/strikethrough
  cleaned = cleaned.replace(/\*\*(.+?)\*\*/g, '$1');  // **bold**
  cleaned = cleaned.replace(/\*(.+?)\*/g, '$1');      // *italic*
  cleaned = cleaned.replace(/~~(.+?)~~/g, '$1');      // ~~strikethrough~~

  // Remove headers
  cleaned = cleaned.replace(/^#+\s+/gm, '');

  // Remove code blocks
  cleaned = cleaned.replace(/```[\s\S]*?```/g, '');
  cleaned = cleaned.replace(/`(.+?)`/g, '$1');

  // Remove links but keep text
  cleaned = cleaned.replace(/\[(.+?)\]\(.+?\)/g, '$1');

  // Remove bullet points, keep text
  cleaned = cleaned.replace(/^[\-\*]\s+/gm, '');

  // Handle numbers - TTS engine will handle pronunciation
  // But could add language-specific number formatting here

  // Remove excessive whitespace
  cleaned = cleaned.replace(/\s+/g, ' ').trim();

  // Split into sentences for better pacing (optional)
  const sentences = cleaned.split(/[۔.!?]+/).filter(s => s.trim());

  return sentences.join('. ');
};
```

### Error Handling

TTS error scenarios:
1. **TTS engine not initialized**: Retry initialization, fallback to silent mode
2. **Voice not available**: Prompt user to download, or use fallback language
3. **Text too long**: Split into chunks, play sequentially
4. **Playback interrupted**: Allow user to resume or replay
5. **Device in silent mode**: Visual indicator that audio won't play
6. **Low volume**: Detect and suggest increasing volume
