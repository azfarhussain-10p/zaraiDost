**Degradation Service**

```javascript
// DegradationService.js
class DegradationService {
  constructor() {
    this.degradationRates = null;
  }

  async loadDegradationRates() {
    if (this.degradationRates) return this.degradationRates;

    // Load from database
    this.degradationRates = await db.query(`
      SELECT * FROM crop_degradation_rates
    `);

    return this.degradationRates;
  }

  async calculateDegradation(cropType, storageType, daysStored, temperature = null) {
    await this.loadDegradationRates();

    // Find degradation rate for crop and storage type
    const rate = this.degradationRates.find(
      r => r.crop_type === cropType && r.storage_type === storageType
    );

    if (!rate) {
      // Default degradation rate
      return this.getDefaultDegradation(cropType, daysStored);
    }

    // Calculate degradation percentage
    let degradationPercent;

    if (rate.degradation_model === 'linear') {
      // Linear degradation: % per day
      degradationPercent = rate.degradation_rate_per_day * daysStored;
    } else if (rate.degradation_model === 'exponential') {
      // Exponential degradation: accelerates over time
      degradationPercent = 100 * (1 - Math.exp(-rate.degradation_coefficient * daysStored));
    } else {
      // S-curve degradation: slow, then fast, then plateaus
      degradationPercent = 100 / (1 + Math.exp(-rate.degradation_coefficient * (daysStored - rate.inflection_point)));
    }

    // Adjust for temperature (if applicable and provided)
    if (temperature && storageType !== 'cold_storage') {
      const tempAdjustment = this.calculateTemperatureAdjustment(temperature, rate.optimal_temperature);
      degradationPercent *= tempAdjustment;
    }

    // Cap at 100%
    degradationPercent = Math.min(degradationPercent, 100);

    return {
      loss_percent: Math.round(degradationPercent * 10) / 10,
      days_stored: daysStored,
      storage_type: storageType,
      model: rate.degradation_model,
      warning_level: this.getWarningLevel(degradationPercent),
      days_until_critical: this.calculateDaysUntilCritical(rate, degradationPercent, daysStored)
    };
  }

  calculateTemperatureAdjustment(actualTemp, optimalTemp) {
    // Degradation accelerates with temperature deviation
    const tempDiff = Math.abs(actualTemp - optimalTemp);

    if (tempDiff === 0) return 1.0;
    if (tempDiff <= 5) return 1.2;
    if (tempDiff <= 10) return 1.5;
    return 2.0; // Double degradation rate for large temp deviations
  }

  getWarningLevel(degradationPercent) {
    if (degradationPercent < 5) return 'none';
    if (degradationPercent < 10) return 'low';
    if (degradationPercent < 25) return 'medium';
    if (degradationPercent < 50) return 'high';
    return 'critical';
  }

  calculateDaysUntilCritical(rate, currentPercent, currentDays) {
    const criticalThreshold = 50; // 50% degradation = critical

    if (currentPercent >= criticalThreshold) return 0;

    // Reverse calculation to find days to reach critical
    if (rate.degradation_model === 'linear') {
      const daysToAdd = (criticalThreshold - currentPercent) / rate.degradation_rate_per_day;
      return Math.ceil(daysToAdd);
    } else {
      // Approximate for non-linear models
      return Math.ceil((criticalThreshold - currentPercent) / 2); // Rough estimate
    }
  }

  getDefaultDegradation(cropType, daysStored) {
    // Default degradation rates if not in database
    const defaultRates = {
      'vegetables': 2.0, // 2% per day
      'fruits': 1.5,
      'wheat': 0.1,
      'rice': 0.15,
      'cotton': 0.2
    };

    const category = this.getCropCategory(cropType);
    const rate = defaultRates[category] || 0.5;

    return {
      loss_percent: Math.min(rate * daysStored, 100),
      days_stored: daysStored,
      storage_type: 'unknown',
      model: 'linear',
      warning_level: 'medium'
    };
  }

  getCropCategory(cropType) {
    const categories = {
      tomatoes: 'vegetables',
      potatoes: 'vegetables',
      onions: 'vegetables',
      mangoes: 'fruits',
      citrus: 'fruits',
      wheat: 'wheat',
      rice: 'rice',
      cotton: 'cotton'
    };
    return categories[cropType] || 'vegetables';
  }

  async generateDegradationTimeline(cropType, storageType, maxDays = 90) {
    const timeline = [];

    for (let day = 0; day <= maxDays; day += 5) { // Every 5 days
      const degradation = await this.calculateDegradation(cropType, storageType, day);
      timeline.push({
        day,
        degradation_percent: degradation.loss_percent,
        warning_level: degradation.warning_level
      });
    }

    return timeline;
  }

  calculateEffectiveQuantity(originalQuantity, degradationPercent) {
    const effectiveQuantity = originalQuantity * (1 - degradationPercent / 100);
    const lossQuantity = originalQuantity - effectiveQuantity;

    return {
      effective_quantity_kg: Math.round(effectiveQuantity * 10) / 10,
      loss_quantity_kg: Math.round(lossQuantity * 10) / 10,
      loss_percent: degradationPercent
    };
  }

  estimatePriceDiscount(degradationPercent) {
    // Price discount increases faster than degradation
    // Slight degradation can significantly reduce price
    if (degradationPercent < 5) return 0;
    if (degradationPercent < 10) return 10; // 10% price reduction
    if (degradationPercent < 25) return 25;
    if (degradationPercent < 50) return 50;
    return 75; // Severely degraded = 75% price reduction
  }
}
```

**Storage Cost Service**

```javascript
// StorageCostService.js
class StorageCostService {
  async calculateCost(storageType, daysStored, quantityKg, region = 'punjab') {
    // Get storage cost rates
    const rates = await this.getStorageRates(storageType, region);

    if (!rates) {
      return this.getDefaultCost(storageType, daysStored, quantityKg);
    }

    let totalCost = 0;

    // Calculate based on pricing model
    if (rates.pricing_model === 'per_kg_per_day') {
      totalCost = rates.rate * quantityKg * daysStored;
    } else if (rates.pricing_model === 'flat_rate_per_day') {
      totalCost = rates.rate * daysStored;
    } else if (rates.pricing_model === 'tiered') {
      // Tiered pricing (cheaper per kg for larger quantities)
      const tierRate = this.getTieredRate(rates, quantityKg);
      totalCost = tierRate * quantityKg * daysStored;
    }

    // Add utility costs (electricity, etc.)
    if (storageType === 'cold_storage') {
      const utilityCost = this.calculateUtilityCost(quantityKg, daysStored);
      totalCost += utilityCost;
    }

    return {
      total_cost: Math.round(totalCost),
      daily_cost: Math.round(totalCost / (daysStored || 1)),
      storage_type: storageType,
      quantity_kg: quantityKg,
      days_stored: daysStored
    };
  }

  async getStorageRates(storageType, region) {
    const rate = await db.queryOne(`
      SELECT * FROM storage_cost_rates
      WHERE storage_type = $1 AND region = $2
    `, [storageType, region]);

    return rate;
  }

  getDefaultCost(storageType, daysStored, quantityKg) {
    // Default storage costs (PKR)
    const defaultRates = {
      'cold_storage': 2.5, // Per kg per day
      'warehouse': 1.0,
      'home': 0.2
    };

    const rate = defaultRates[storageType] || 0.5;
    const totalCost = rate * quantityKg * daysStored;

    return {
      total_cost: Math.round(totalCost),
      daily_cost: Math.round(totalCost / (daysStored || 1)),
      storage_type: storageType,
      quantity_kg: quantityKg,
      days_stored: daysStored
    };
  }

  getTieredRate(rates, quantityKg) {
    // Tiered pricing example
    if (quantityKg < 100) return rates.rate; // Base rate
    if (quantityKg < 500) return rates.rate * 0.9; // 10% discount
    if (quantityKg < 1000) return rates.rate * 0.8; // 20% discount
    return rates.rate * 0.7; // 30% discount for 1000+ kg
  }

  calculateUtilityCost(quantityKg, daysStored) {
    // Electricity cost for cold storage (approximate)
    const electricityPerKgPerDay = 0.5; // PKR
    return electricityPerKgPerDay * quantityKg * daysStored;
  }

  async generateCostProjection(storageType, quantityKg, region, days = [30, 60, 90]) {
    const projections = [];

    for (const day of days) {
      const cost = await this.calculateCost(storageType, day, quantityKg, region);
      projections.push({
        days: day,
        total_cost: cost.total_cost,
        daily_cost: cost.daily_cost
      });
    }

    return projections;
  }
}
```

**Break-Even Analyzer**

```javascript
// BreakEvenAnalyzer.js
class BreakEvenAnalyzer {
  constructor() {
    this.degradationService = new DegradationService();
    this.storageCostService = new StorageCostService();
  }

  async calculateBreakEven(cropType, storageType, quantityKg, currentPrice, pricePredictions) {
    const breakEvenPoints = [];

    // For each day, calculate if waiting is profitable
    for (let day = 1; day <= 90; day++) {
      // Get predicted price for this day
      const predictedPrice = this.interpolatePrice(day, pricePredictions, currentPrice);

      // Calculate storage cost
      const storageCost = await this.storageCostService.calculateCost(
        storageType,
        day,
        quantityKg
      );

      // Calculate degradation
      const degradation = await this.degradationService.calculateDegradation(
        cropType,
        storageType,
        day
      );

      // Calculate effective quantity after degradation
      const effectiveQuantity = quantityKg * (1 - degradation.loss_percent / 100);

      // Revenue now vs later
      const revenueNow = quantityKg * currentPrice;
      const revenueLater = effectiveQuantity * predictedPrice - storageCost.total_cost;

      // Profit difference
      const profitDifference = revenueLater - revenueNow;

      breakEvenPoints.push({
        day,
        predicted_price: predictedPrice,
        storage_cost: storageCost.total_cost,
        degradation_percent: degradation.loss_percent,
        effective_quantity: effectiveQuantity,
        profit_difference: profitDifference,
        is_break_even: Math.abs(profitDifference) < 100 // Within 100 PKR = break-even
      });
    }

    // Find first break-even day
    const breakEvenDay = breakEvenPoints.find(p => p.profit_difference <= 0);

    return {
      break_even_day: breakEvenDay ? breakEvenDay.day : null,
      timeline: breakEvenPoints,
      recommendation: this.generateBreakEvenRecommendation(breakEvenDay, breakEvenPoints)
    };
  }

  interpolatePrice(day, predictions, currentPrice) {
    // Simple linear interpolation between prediction points
    // (Same logic as in Story 5.3)
    if (day === 0) return currentPrice;

    const pred30 = predictions.find(p => p.days_ahead === 30);
    const pred60 = predictions.find(p => p.days_ahead === 60);
    const pred90 = predictions.find(p => p.days_ahead === 90);

    if (day <= 30) {
      const ratio = day / 30;
      return currentPrice + (pred30.predicted_price - currentPrice) * ratio;
    } else if (day <= 60) {
      const ratio = (day - 30) / 30;
      return pred30.predicted_price + (pred60.predicted_price - pred30.predicted_price) * ratio;
    } else {
      const ratio = (day - 60) / 30;
      return pred60.predicted_price + (pred90.predicted_price - pred60.predicted_price) * ratio;
    }
  }

  generateBreakEvenRecommendation(breakEvenDay, timeline) {
    if (!breakEvenDay) {
      return {
        message: 'Storage costs do not outweigh price gains within 90 days',
        action: 'wait',
        safe_window_days: 90
      };
    }

    if (breakEvenDay.day < 14) {
      return {
        message: `Sell within ${breakEvenDay.day} days - storage costs will exceed price gains`,
        action: 'sell_soon',
        safe_window_days: breakEvenDay.day
      };
    } else {
      return {
        message: `You have ${breakEvenDay.day} days before storage costs exceed gains`,
        action: 'monitor',
        safe_window_days: breakEvenDay.day
      };
    }
  }
}
```

```javascript
// PerishabilityAlertService.js
const cron = require('node-cron');

class PerishabilityAlertService {
  constructor() {
    this.degradationService = new DegradationService();
    this.notificationService = new NotificationService();
  }

  start() {
    // Check perishability daily at 8 AM
    cron.schedule('0 8 * * *', async () => {
      await this.checkPerishableCrops();
    });
  }

  async checkPerishableCrops() {
    // Get crops currently in storage
    const storedCrops = await db.query(`
      SELECT
        cs.*,
        f.user_id,
        f.name as farmer_name,
        sf.storage_type,
        CURRENT_DATE - cs.harvest_date as days_stored
      FROM crop_seasons cs
      JOIN fields f ON cs.field_id = f.id
      LEFT JOIN storage_facilities sf ON f.storage_facility_id = sf.id
      WHERE cs.status = 'harvested'
        AND cs.harvest_date IS NOT NULL
        AND cs.sold_date IS NULL
    `);

    for (const crop of storedCrops) {
      const degradation = await this.degradationService.calculateDegradation(
        crop.crop_type,
        crop.storage_type || 'home',
        crop.days_stored
      );

      // Send alerts based on degradation level
      if (degradation.warning_level === 'critical' && !crop.critical_alert_sent) {
        await this.sendCriticalAlert(crop, degradation);
        await this.markAlertSent(crop.id, 'critical');
      } else if (degradation.warning_level === 'high' && !crop.high_alert_sent) {
        await this.sendHighAlert(crop, degradation);
        await this.markAlertSent(crop.id, 'high');
      } else if (degradation.warning_level === 'medium' && !crop.medium_alert_sent) {
        await this.sendMediumAlert(crop, degradation);
        await this.markAlertSent(crop.id, 'medium');
      }
    }
  }

  async sendCriticalAlert(crop, degradation) {
    await this.notificationService.sendPushNotification(crop.user_id, {
      title: 'URGENT: Severe Crop Degradation!',
      body: `Your ${crop.crop_type} has degraded ${Math.round(degradation.loss_percent)}%. Sell immediately to avoid total loss!`,
      priority: 'high',
      data: {
        type: 'perishability_critical',
        crop_season_id: crop.id
      }
    });

    console.log(`CRITICAL alert sent to ${crop.farmer_name} for ${crop.crop_type}`);
  }

  async sendHighAlert(crop, degradation) {
    await this.notificationService.sendPushNotification(crop.user_id, {
      title: 'High Crop Degradation Warning',
      body: `Your ${crop.crop_type} has degraded ${Math.round(degradation.loss_percent)}%. Sell within ${degradation.days_until_critical} days.`,
      data: {
        type: 'perishability_high',
        crop_season_id: crop.id
      }
    });
  }

  async sendMediumAlert(crop, degradation) {
    await this.notificationService.sendPushNotification(crop.user_id, {
      title: 'Crop Degradation Notice',
      body: `Your ${crop.crop_type} has degraded ${Math.round(degradation.loss_percent)}%. Consider selling soon.`,
      data: {
        type: 'perishability_medium',
        crop_season_id: crop.id
      }
    });
  }

  async markAlertSent(cropSeasonId, level) {
    const field = `${level}_alert_sent`;
    await db.update('crop_seasons', { id: cropSeasonId }, {
      [field]: true,
      [`${field}_at`]: new Date()
    });
  }
}
```

```sql
-- Crop degradation rates
CREATE TABLE crop_degradation_rates (
  id TEXT PRIMARY KEY,
  crop_type TEXT NOT NULL,
  storage_type TEXT NOT NULL,  -- 'cold_storage', 'warehouse', 'home'
  degradation_model TEXT,      -- 'linear', 'exponential', 's_curve'
  degradation_rate_per_day DECIMAL(5, 3), -- For linear model
  degradation_coefficient DECIMAL(5, 3),  -- For exponential/s-curve
  inflection_point INTEGER,    -- For s-curve (days)
  optimal_temperature DECIMAL(5, 2), -- Celsius
  shelf_life_days INTEGER,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(crop_type, storage_type)
);

CREATE TABLE storage_facilities (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,

  location_gps GEOGRAPHY(POINT),
  temperature_celsius DECIMAL(5, 2),
  humidity_percent DECIMAL(5, 2),
  cost_per_kg_per_day DECIMAL(10, 2),

  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_storage_facilities_user ON storage_facilities(user_id);
```
