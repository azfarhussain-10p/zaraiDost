# Story 6.7: Early Harvest Alerts

## Status

## Story
**As a** farmer with maturing crops,
**I want** alerts to harvest early if extreme weather threatens,
**so that** I can salvage my crop before damage.

5. AC5: Post-harvest handling advice for premature harvest
6. AC6: Market impact considered (price changes due to early harvest)
7. AC7: User can confirm harvest completion to stop alerts

**Crop Maturity Assessment**

```javascript
// CropMaturityAssessor.js
class CropMaturityAssessor {
  /**
   * Calculate crop maturity percentage
   */
  async assessCropMaturity(cropSeasonId) {
    const cropSeason = await db.query(
      'SELECT * FROM crop_seasons WHERE id = $1',
      [cropSeasonId]
    );

    if (!cropSeason[0]) {
      throw new Error('Crop season not found');
    }

    const crop = cropSeason[0];

    // Calculate days since planting
    const daysSincePlanting = this.getDaysSince(crop.planting_date);

    // Get typical maturity period for crop
    const maturityPeriod = this.getMaturityPeriod(crop.crop_type, crop.variety);

    // Calculate maturity percentage
    const maturityPercent = Math.min(100, (daysSincePlanting / maturityPeriod.days) * 100);

    // Calculate Growth Degree Days (GDD) for more accuracy
    const gdd = await this.calculateGDD(crop.planting_date, crop.field_id);
    const gddMaturityPercent = Math.min(100, (gdd / maturityPeriod.gdd_required) * 100);

    // Average both methods
    const finalMaturityPercent = (maturityPercent + gddMaturityPercent) / 2;

    // Determine if ready for early harvest (>70% mature)
    const readyForEarlyHarvest = finalMaturityPercent >= 70;

    return {
      cropSeasonId,
      cropType: crop.crop_type,
      variety: crop.variety,
      plantingDate: crop.planting_date,
      daysSincePlanting,
      maturityPercent: Math.round(finalMaturityPercent),
      readyForEarlyHarvest,
      expectedHarvestDate: crop.expected_harvest_date,
      daysToFullMaturity: Math.max(0, maturityPeriod.days - daysSincePlanting),
      maturityStage: this.getMaturityStage(finalMaturityPercent),
      visualIndicators: this.getVisualIndicators(crop.crop_type, finalMaturityPercent)
    };
  }

  getMaturityPeriod(cropType, variety) {
    // Typical maturity periods for Pakistani crops
    const maturityPeriods = {
      wheat: { days: 120, gdd_required: 2000 },
      rice: { days: 150, gdd_required: 2500 },
      cotton: { days: 180, gdd_required: 3000 },
      sugarcane: { days: 365, gdd_required: 5000 },
      corn: { days: 90, gdd_required: 1400 }
    };

    return maturityPeriods[cropType] || { days: 120, gdd_required: 2000 };
  }

  async calculateGDD(plantingDate, fieldId) {
    // Growth Degree Days: accumulated heat units
    const weatherData = await this.getWeatherDataSincePlanting(plantingDate, fieldId);

    let gdd = 0;
    const baseTemp = 10;  // Base temperature for most crops (°C)

    for (const day of weatherData) {
      const avgTemp = (day.max_temp + day.min_temp) / 2;
      const dailyGDD = Math.max(0, avgTemp - baseTemp);
      gdd += dailyGDD;
    }

    return gdd;
  }

  getMaturityStage(maturityPercent) {
    if (maturityPercent < 50) return 'early_development';
    if (maturityPercent < 70) return 'developing';
    if (maturityPercent < 85) return 'maturing';
    if (maturityPercent < 95) return 'near_harvest';
    return 'fully_mature';
  }

  getVisualIndicators(cropType, maturityPercent) {
    // Crop-specific visual maturity indicators
    const indicators = {
      wheat: {
        '70%': 'Grains hard but still contain some moisture; heads turning yellow',
        '80%': 'Grains fully hard; heads completely yellow',
        '90%': 'Grains very hard and dry; heads bent over',
        '100%': 'Grains shattering from heads; risk of losses'
      },
      rice: {
        '70%': 'Grains in dough stage; some yellowing of panicles',
        '80%': 'Grains hardening; 80% of panicles yellow',
        '90%': 'Grains fully hard; all panicles yellow',
        '100%': 'Grains dry and brittle; risk of shattering'
      },
      cotton: {
        '70%': 'Bolls opening; fiber visible',
        '80%': '60-70% bolls open; fiber expanding',
        '90%': '80-90% bolls open; fiber fully developed',
        '100%': 'All bolls open; risk of weather damage to fiber'
      }
    };

    const cropIndicators = indicators[cropType] || indicators.wheat;

    // Find closest maturity level
    const levels = ['70%', '80%', '90%', '100%'];
    for (const level of levels) {
      if (maturityPercent <= parseInt(level)) {
        return cropIndicators[level];
      }
    }

    return cropIndicators['100%'];
  }

  getDaysSince(date) {
    const now = new Date();
    const past = new Date(date);
    return Math.floor((now - past) / (1000 * 60 * 60 * 24));
  }
}
```

**Trade-Off Analysis Engine**

```javascript
// TradeOffAnalysisEngine.js
class TradeOffAnalysisEngine {
  /**
   * Analyze trade-off between early harvest and waiting
   */
  async analyzeTradeOff(cropSeasonId, weatherEventId) {
    // Get crop maturity
    const maturity = await cropMaturityAssessor.assessCropMaturity(cropSeasonId);

    // Get weather event details
    const weatherEvent = await db.query(
      'SELECT * FROM extreme_weather_alerts WHERE id = $1',
      [weatherEventId]
    );

    if (!weatherEvent[0]) {
      throw new Error('Weather event not found');
    }

    const event = weatherEvent[0];

    // Calculate yield loss from early harvest
    const earlyHarvestYieldLoss = this.calculateEarlyHarvestYieldLoss(maturity.maturityPercent);

    // Calculate expected damage from weather event
    const weatherDamageRisk = this.calculateWeatherDamageRisk(event, maturity.cropType);

    // Get crop season data for financial calculations
    const cropSeason = await db.query(
      'SELECT * FROM crop_seasons WHERE id = $1',
      [cropSeasonId]
    );

    const field = await db.query(
      'SELECT * FROM fields WHERE id = $1',
      [cropSeason[0].field_id]
    );

    // Get market price
    const marketPrice = await this.getCurrentMarketPrice(maturity.cropType);

    // Financial calculations
    const normalYield = this.getExpectedYield(maturity.cropType, field[0].size_acres);
    const earlyHarvestYield = normalYield * (1 - earlyHarvestYieldLoss / 100);
    const weatherDamagedYield = normalYield * (1 - weatherDamageRisk.expectedDamagePercent / 100);

    const earlyHarvestRevenue = earlyHarvestYield * marketPrice;
    const weatherDamagedRevenue = weatherDamagedYield * marketPrice * weatherDamageRisk.probability;
    const normalRevenue = normalYield * marketPrice;

    // Decision logic
    const expectedValueEarlyHarvest = earlyHarvestRevenue;
    const expectedValueWait = (weatherDamagedRevenue * (weatherDamageRisk.probability / 100)) +
                              (normalRevenue * (1 - weatherDamageRisk.probability / 100));

    const recommendation = expectedValueEarlyHarvest > expectedValueWait ? 'early_harvest' : 'wait';

    const confidenceScore = Math.abs(expectedValueEarlyHarvest - expectedValueWait) / normalRevenue * 100;

    return {
      recommendation,
      confidenceScore: Math.min(100, confidenceScore),

      earlyHarvestOption: {
        yieldLossPercent: earlyHarvestYieldLoss,
        expectedYieldKg: earlyHarvestYield,
        expectedRevenuePkr: earlyHarvestRevenue,
        guaranteedRevenue: true
      },

      waitOption: {
        weatherDamageProbability: weatherDamageRisk.probability,
        expectedDamagePercent: weatherDamageRisk.expectedDamagePercent,
        bestCaseYieldKg: normalYield,
        worstCaseYieldKg: weatherDamagedYield,
        expectedRevenuePkr: expectedValueWait
      },

      financialImpact: {
        normalRevenuePkr: normalRevenue,
        revenueAtRisk: normalRevenue - weatherDamagedRevenue,
        potentialSavings: Math.max(0, earlyHarvestRevenue - weatherDamagedRevenue)
      },

      reasoning: this.generateReasoning(recommendation, maturity, weatherDamageRisk, confidenceScore)
    };
  }

  calculateEarlyHarvestYieldLoss(maturityPercent) {
    // Yield loss increases exponentially as maturity decreases
    if (maturityPercent >= 95) return 2;   // Minimal loss
    if (maturityPercent >= 90) return 5;
    if (maturityPercent >= 85) return 10;
    if (maturityPercent >= 80) return 15;
    if (maturityPercent >= 75) return 22;
    if (maturityPercent >= 70) return 30;  // Maximum acceptable loss
    return 50;  // Too early, very high loss
  }

  calculateWeatherDamageRisk(weatherEvent, cropType) {
    // Calculate expected damage based on weather event type and crop
    let expectedDamagePercent = 0;
    let probability = 0;

    if (weatherEvent.alert_type === 'heavy_rainfall') {
      // Wheat/rice at harvest stage: high lodging and grain damage risk
      if (cropType === 'wheat' || cropType === 'rice') {
        expectedDamagePercent = weatherEvent.severity === 'warning' ? 60 : 40;
        probability = weatherEvent.confidence || 70;
      } else if (cropType === 'cotton') {
        expectedDamagePercent = weatherEvent.severity === 'warning' ? 40 : 25;
        probability = weatherEvent.confidence || 70;
      }
    } else if (weatherEvent.alert_type === 'hail') {
      // Hail causes severe damage to most crops
      expectedDamagePercent = weatherEvent.severity === 'warning' ? 70 : 50;
      probability = weatherEvent.confidence || 60;
    } else if (weatherEvent.alert_type === 'high_wind') {
      // Wind causes lodging in grain crops
      if (cropType === 'wheat' || cropType === 'rice') {
        expectedDamagePercent = weatherEvent.severity === 'warning' ? 50 : 30;
        probability = weatherEvent.confidence || 65;
      }
    }

    return {
      expectedDamagePercent,
      probability,
      severity: weatherEvent.severity
    };
  }

  getExpectedYield(cropType, sizeAcres) {
    // Average yields in Pakistan (kg per acre)
    const avgYields = {
      wheat: 1200,
      rice: 1800,
      cotton: 800,  // kg of seed cotton
      sugarcane: 30000,
      corn: 2000
    };

    return (avgYields[cropType] || 1000) * sizeAcres;
  }

  async getCurrentMarketPrice(cropType) {
    // Get current market price from Story 2.3
    const prices = await db.query(`
      SELECT price_per_kg FROM mandi_prices
      WHERE crop_type = $1
      ORDER BY date DESC
      LIMIT 1
    `, [cropType]);

    return prices[0]?.price_per_kg || 50;  // Default price
  }

  generateReasoning(recommendation, maturity, weatherDamageRisk, confidenceScore) {
    if (recommendation === 'early_harvest') {
      return `Your crop is ${maturity.maturityPercent}% mature and ready for early harvest. ` +
             `With ${weatherDamageRisk.probability}% probability of ${weatherDamageRisk.expectedDamagePercent}% weather damage, ` +
             `early harvest is the safer option. You'll lose ~${this.calculateEarlyHarvestYieldLoss(maturity.maturityPercent)}% yield, ` +
             `but this is better than risking ${weatherDamageRisk.expectedDamagePercent}% damage. ` +
             `Confidence: ${Math.round(confidenceScore)}%.`;
    } else {
      return `Your crop is only ${maturity.maturityPercent}% mature. ` +
             `Early harvest would cause ${this.calculateEarlyHarvestYieldLoss(maturity.maturityPercent)}% yield loss, ` +
             `which is higher than the expected weather damage (${weatherDamageRisk.expectedDamagePercent}% × ${weatherDamageRisk.probability}% = ` +
             `${Math.round(weatherDamageRisk.expectedDamagePercent * weatherDamageRisk.probability / 100)}%). ` +
             `Waiting is the better option. Monitor weather forecasts closely. ` +
             `Confidence: ${Math.round(confidenceScore)}%.`;
    }
  }
}
```

**Harvest Timing Optimizer**

```javascript
// HarvestTimingOptimizer.js
class HarvestTimingOptimizer {
  /**
   * Calculate optimal harvest dates considering weather and maturity
   */
  async calculateOptimalHarvestTiming(cropSeasonId, weatherEventId) {
    const maturity = await cropMaturityAssessor.assessCropMaturity(cropSeasonId);
    const weatherEvent = await db.query(
      'SELECT * FROM extreme_weather_alerts WHERE id = $1',
      [weatherEventId]
    );

    const event = weatherEvent[0];

    // Calculate earliest safe harvest date (70% maturity)
    const daysToMinimumMaturity = this.calculateDaysToMaturity(maturity, 70);
    const earliestHarvestDate = this.addDays(new Date(), daysToMinimumMaturity);

    // Calculate optimal harvest window (3-5 days before weather event)
    const weatherEventDate = new Date(event.event_start_time);
    const optimalWindowStart = this.addDays(weatherEventDate, -5);
    const optimalWindowEnd = this.addDays(weatherEventDate, -3);

    // Check if crop will be mature enough
    const matureEnough = maturity.maturityPercent >= 70 ||
                        new Date() >= earliestHarvestDate;

    let recommendation = '';
    let recommendedDate = null;

    if (matureEnough && new Date() <= optimalWindowStart) {
      // Crop is mature and we have time
      recommendedDate = optimalWindowStart;
      recommendation = `Start harvest on ${this.formatDate(optimalWindowStart)} (3-5 days before weather event)`;
    } else if (matureEnough && new Date() > optimalWindowStart) {
      // Crop is mature but weather event is soon
      recommendedDate = new Date();
      recommendation = `Start harvest immediately. Weather event in ${this.getDaysUntil(weatherEventDate)} days`;
    } else {
      // Crop not mature enough
      recommendedDate = earliestHarvestDate;
      recommendation = `Earliest safe harvest: ${this.formatDate(earliestHarvestDate)} (70% maturity). ` +
                      `Monitor maturity and weather closely`;
    }

    // Estimate harvest duration
    const cropSeason = await db.query('SELECT * FROM crop_seasons WHERE id = $1', [cropSeasonId]);
    const field = await db.query('SELECT * FROM fields WHERE id = $1', [cropSeason[0].field_id]);
    const harvestDurationDays = this.estimateHarvestDuration(field[0].size_acres);

    return {
      recommendedDate,
      recommendation,
      earliestHarvestDate,
      optimalWindowStart,
      optimalWindowEnd,
      weatherEventDate,
      daysUntilWeatherEvent: this.getDaysUntil(weatherEventDate),
      harvestDurationDays,
      urgency: this.calculateUrgency(recommendedDate, weatherEventDate, harvestDurationDays),
      logistics: {
        requiredDays: harvestDurationDays,
        recommendedCrewSize: Math.ceil(field[0].size_acres / 3),  // 1 person per 3 acres per day
        machineryNeeded: field[0].size_acres > 10 ? 'harvester' : 'manual'
      }
    };
  }

  calculateDaysToMaturity(maturity, targetMaturityPercent) {
    if (maturity.maturityPercent >= targetMaturityPercent) {
      return 0;
    }

    const remainingMaturity = targetMaturityPercent - maturity.maturityPercent;
    const maturityRate = maturity.maturityPercent / maturity.daysSincePlanting;
    const daysNeeded = remainingMaturity / maturityRate;

    return Math.ceil(daysNeeded);
  }

  estimateHarvestDuration(sizeAcres) {
    // Manual harvesting: ~3 acres per day with 3-person crew
    // Mechanical harvesting: ~20 acres per day
    if (sizeAcres <= 10) {
      return Math.ceil(sizeAcres / 3);
    } else {
      return Math.ceil(sizeAcres / 20);  // Assuming mechanical harvester
    }
  }

  calculateUrgency(recommendedDate, weatherEventDate, harvestDurationDays) {
    const daysAvailable = this.getDaysUntil(weatherEventDate) - harvestDurationDays;

    if (daysAvailable < 0) {
      return 'critical';  // Not enough time to finish before weather
    } else if (daysAvailable < 2) {
      return 'high';
    } else if (daysAvailable < 5) {
      return 'medium';
    }
    return 'low';
  }

  addDays(date, days) {
    const result = new Date(date);
    result.setDate(result.getDate() + days);
    return result;
  }

  getDaysUntil(date) {
    const now = new Date();
    const target = new Date(date);
    return Math.ceil((target - now) / (1000 * 60 * 60 * 24));
  }

  formatDate(date) {
    return new Date(date).toLocaleDateString('en-PK', {
      weekday: 'long',
      year: 'numeric',
      month: 'long',
      day: 'numeric'
    });
  }
}
```

```sql
CREATE TABLE early_harvest_alerts (

  recommendation TEXT NOT NULL,       -- 'early_harvest', 'wait'
  confidence_score INTEGER,

  weather_damage_probability INTEGER,
  expected_weather_damage_percent INTEGER,

  -- Harvest timing
  recommended_harvest_date DATE,
  earliest_harvest_date DATE,
  optimal_window_start DATE,
  optimal_window_end DATE,
  weather_event_date DATE,
  days_until_weather_event INTEGER,
  urgency TEXT,                       -- 'low', 'medium', 'high', 'critical'

  -- Financial impact
  expected_revenue_early_harvest INTEGER,
  expected_revenue_wait INTEGER,
  potential_savings INTEGER,

  -- Status
  is_acknowledged BOOLEAN DEFAULT FALSE,
  harvest_status TEXT DEFAULT 'pending',
  alert_sent_at DATETIME NOT NULL,
  acknowledged_at DATETIME,
  harvest_started_at DATETIME,
  harvest_completed_at DATETIME,

);
```

### Project Structure Notes

This story provides critical harvest timing guidance to farmers facing extreme weather threats. By combining crop maturity assessment with weather forecasts and financial trade-off analysis, the system delivers data-driven recommendations on whether to harvest early or wait. The 3-7 day early harvest window allows farmers to mobilize labor and machinery, potentially saving entire harvests from weather damage. Post-harvest handling advice ensures premature crops are properly dried and stored, minimizing quality losses.
