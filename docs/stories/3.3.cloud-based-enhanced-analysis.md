**Cloud Vision Service Architecture**

```javascript
// CloudVisionService.js - Abstract vision provider selection
class CloudVisionService {
  constructor() {
    this.providers = {
      gpt4: new GPT4VisionProvider(),
      gemini: new GeminiVisionProvider(),
      claude: new ClaudeVisionProvider()  // Future
    };
    this.activeProvider = 'gpt4';  // Default, configurable
  }

  async analyzeImage(imageUrl, context) {
    const provider = this.providers[this.activeProvider];

    try {
      const result = await provider.analyze(imageUrl, context);
      return result;
    } catch (error) {
      // Fallback to alternative provider
      console.error(`${this.activeProvider} failed, trying alternative`);
      const fallbackProvider = this.getFallbackProvider();
      return await fallbackProvider.analyze(imageUrl, context);
    }
  }

  async analyzeMultipleImages(imageUrls, context) {
    const provider = this.providers[this.activeProvider];
    return await provider.analyzeMultiple(imageUrls, context);
  }
}
```

**GPT-4 Vision Integration**

```javascript
// GPT4VisionProvider.js
import { ChatOpenAI } from "@langchain/openai";

class GPT4VisionProvider {
  constructor() {
    this.model = new ChatOpenAI({
      modelName: "gpt-4-vision-preview",
      maxTokens: 1000,
      temperature: 0.2,  // Low temperature for consistent medical diagnosis
    });
  }

  async analyze(imageUrl, context) {
    const prompt = this.buildPrompt(context);

    const response = await this.model.invoke([
      {
        role: "user",
        content: [
          {
            type: "text",
            text: prompt
          },
          {
            type: "image_url",
            image_url: {
              url: imageUrl,
              detail: "high"  // High detail for medical analysis
            }
          }
        ]
      }
    ]);

    return this.parseResponse(response);
  }

  async analyzeMultiple(imageUrls, context) {
    const prompt = this.buildMultiImagePrompt(context);

    const imageContent = imageUrls.map(url => ({
      type: "image_url",
      image_url: { url, detail: "high" }
    }));

    const response = await this.model.invoke([
      {
        role: "user",
        content: [
          { type: "text", text: prompt },
          ...imageContent
        ]
      }
    ]);

    return this.parseResponse(response);
  }

  buildPrompt(context) {
    return `You are an expert agricultural pathologist specializing in Pakistani crop diseases.

Analyze this crop image and identify any diseases, pests, or nutrient deficiencies.

Context:
- Crop Type: ${context.cropType}
- Location: ${context.location}
- Farmer Description: ${context.description || 'None'}
- On-device Prediction: ${context.onDevicePrediction || 'None'}

Provide:
1. Primary diagnosis with confidence level (0-100%)
2. Top 3 possible diseases/issues with confidence
3. Severity assessment (mild/moderate/severe)
4. Key visual indicators observed
5. Reasoning for diagnosis

Format response as JSON.`;
  }

  buildMultiImagePrompt(context) {
    return `You are an expert agricultural pathologist specializing in Pakistani crop diseases.

Analyze these ${context.imageCount} images of the same crop issue and provide a comprehensive diagnosis.

Context:
- Crop Type: ${context.cropType}
- Location: ${context.location}
- On-device Prediction: ${context.onDevicePrediction || 'None'}

Analyze all images together to:
1. Identify the disease/pest/deficiency
2. Assess severity across the crop
3. Note any progression or variation visible
4. Provide confidence level for diagnosis

Format response as JSON with top 3 diagnoses.`;
  }

  parseResponse(response) {
    const content = response.content;

    try {
      const parsed = JSON.parse(content);
      return {
        provider: 'gpt4-vision',
        predictions: parsed.diagnoses || parsed.top_diagnoses,
        reasoning: parsed.reasoning,
        severity: parsed.severity,
        confidence: parsed.confidence,
        visualIndicators: parsed.visual_indicators
      };
    } catch (error) {
      // Fallback to text parsing if JSON invalid
      return this.parseTextResponse(content);
    }
  }
}
```

**Gemini Vision Integration**

```javascript
// GeminiVisionProvider.js
import { GoogleGenerativeAI } from "@google/generative-ai";

class GeminiVisionProvider {
  constructor() {
    const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);
    this.model = genAI.getGenerativeModel({ model: "gemini-pro-vision" });
  }

  async analyze(imageUrl, context) {
    // Download image as base64
    const imageBase64 = await this.downloadImageAsBase64(imageUrl);

    const prompt = this.buildPrompt(context);

    const result = await this.model.generateContent([
      prompt,
      {
        inlineData: {
          data: imageBase64,
          mimeType: "image/jpeg"
        }
      }
    ]);

    const response = await result.response;
    return this.parseResponse(response.text());
  }

  async analyzeMultiple(imageUrls, context) {
    const images = await Promise.all(
      imageUrls.map(url => this.downloadImageAsBase64(url))
    );

    const prompt = this.buildMultiImagePrompt(context);

    const parts = [
      prompt,
      ...images.map(img => ({
        inlineData: { data: img, mimeType: "image/jpeg" }
      }))
    ];

    const result = await this.model.generateContent(parts);
    const response = await result.response;
    return this.parseResponse(response.text());
  }

  // Similar prompt building and parsing methods as GPT-4 Vision
}
```

**LangChain Multi-Agent Workflow**

```python
# backend/services/disease_analysis_workflow.py
from langchain.agents import AgentExecutor, create_openai_tools_agent
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate

class DiseaseAnalysisWorkflow:
    """Multi-agent workflow for comprehensive disease analysis"""

    def __init__(self):
        self.vision_agent = self._create_vision_agent()
        self.advisory_agent = self._create_advisory_agent()

    def _create_vision_agent(self):
        """Agent specialized in image analysis"""
        llm = ChatOpenAI(model="gpt-4-vision-preview", temperature=0.2)

        prompt = ChatPromptTemplate.from_messages([
            ("system", "You are a plant pathology vision expert. Analyze crop images for diseases, pests, and deficiencies."),
            ("user", "{input}")
        ])

        return create_openai_tools_agent(llm, [], prompt)

    def _create_advisory_agent(self):
        """Agent specialized in agricultural advisory"""
        llm = ChatOpenAI(model="gpt-4", temperature=0.3)

        prompt = ChatPromptTemplate.from_messages([
            ("system", "You are an agricultural advisor for Pakistani farmers. Provide actionable treatment recommendations."),
            ("user", "{input}")
        ])

        return create_openai_tools_agent(llm, [], prompt)

    async def analyze(self, image_urls, context):
        """Run complete analysis workflow"""

        # Step 1: Vision analysis
        vision_result = await self.vision_agent.invoke({
            "input": self._build_vision_prompt(image_urls, context)
        })

        # Step 2: Advisory refinement
        advisory_result = await self.advisory_agent.invoke({
            "input": self._build_advisory_prompt(vision_result, context)
        })

        # Step 3: Combine results
        final_result = self._combine_results(vision_result, advisory_result)

        return final_result

    def _build_vision_prompt(self, image_urls, context):
        return f"""Analyze these crop images:

Crop: {context['crop_type']}
Location: {context['location']}
Images: {len(image_urls)}
On-device prediction: {context.get('on_device_prediction', 'None')}

Identify diseases, pests, or deficiencies with confidence scores."""

    def _build_advisory_prompt(self, vision_result, context):
        return f"""Based on this disease diagnosis:

{vision_result}

Provide:
1. Treatment recommendations (chemical, organic, cultural)
2. Severity assessment
3. Urgency of action
4. Prevention measures

Consider Pakistani farming context and local product availability."""
```

**Backend API Endpoint**

```javascript
// backend/routes/health-analysis.js
router.post('/api/health-checks/:id/cloud-analysis', authenticate, async (req, res) => {
  const { id } = req.params;
  const startTime = Date.now();

  try {
    // Get health check with images
    const healthCheck = await HealthCheck.findById(id);
    if (!healthCheck) {
      return res.status(404).json({ error: 'Health check not found' });
    }

    // Get S3 image URLs
    const images = await healthCheck.getImages();
    const imageUrls = images.map(img => img.remote_url);

    if (imageUrls.length === 0) {
      return res.status(400).json({ error: 'No images available for analysis' });
    }

    // Build context
    const context = {
      cropType: healthCheck.crop_type,
      location: `${healthCheck.location_latitude}, ${healthCheck.location_longitude}`,
      description: healthCheck.description,
      onDevicePrediction: healthCheck.on_device_result?.topPrediction,
      imageCount: imageUrls.length
    };

    // Run cloud analysis via AI Wrapper
    const cloudResult = await CloudVisionService.analyzeMultipleImages(imageUrls, context);

    // Calculate analysis time
    const analysisTime = Date.now() - startTime;

    // Check 10 second timeout (AC5)
    if (analysisTime > 10000) {
      console.warn(`Cloud analysis exceeded 10s: ${analysisTime}ms`);
    }

    // Save cloud result
    healthCheck.cloud_result = cloudResult;
    healthCheck.cloud_analysis_time_ms = analysisTime;
    healthCheck.cloud_model_version = cloudResult.provider;
    healthCheck.status = 'cloud_completed';
    await healthCheck.save();

    // Compare with on-device result
    const comparison = compareResults(
      healthCheck.on_device_result,
      cloudResult
    );

    res.json({
      cloudResult,
      comparison,
      analysisTime
    });

  } catch (error) {
    console.error('Cloud analysis failed:', error);
    res.status(500).json({
      error: 'Cloud analysis failed',
      fallbackToOnDevice: true
    });
  }
});
```

**Result Comparison Logic**

```javascript
// services/ResultComparison.js
const compareResults = (onDeviceResult, cloudResult) => {
  if (!onDeviceResult || !cloudResult) {
    return { agreement: null, differences: [] };
  }

  const onDeviceTop = onDeviceResult.topPrediction || onDeviceResult.predictions?.[0];
  const cloudTop = cloudResult.predictions?.[0];

  // Check if top predictions match
  const topDiseaseMatch = onDeviceTop.diseaseId === cloudTop.diseaseId;

  // Calculate confidence difference
  const confidenceDelta = Math.abs(
    cloudTop.confidence - onDeviceTop.confidence
  );

  // Agreement score
  let agreementScore = 0;
  if (topDiseaseMatch) {
    agreementScore = 100 - (confidenceDelta * 100);
  } else {
    // Check if cloud's top is in on-device's top 3
    const inTopThree = onDeviceResult.predictions
      ?.slice(0, 3)
      .some(pred => pred.diseaseId === cloudTop.diseaseId);
    agreementScore = inTopThree ? 50 : 0;
  }

  const isDifferent = !topDiseaseMatch || confidenceDelta > 0.2;

  return {
    topDiseaseMatch,
    agreementScore,
    confidenceDelta,
    isDifferent,
    cloudMoreConfident: cloudTop.confidence > onDeviceTop.confidence,
    recommendation: isDifferent
      ? 'Cloud analysis suggests different diagnosis - review recommended'
      : 'Cloud analysis confirms on-device result',
    shouldNotifyUser: isDifferent,
    onDeviceTop,
    cloudTop
  };
};
```

### Error Handling

Common error scenarios:
1. **No internet**: Queue for later, use on-device result
2. **Cloud timeout (>10s)**: Cancel request, use on-device result
3. **API rate limit**: Queue for retry after cooldown
4. **API error (500)**: Retry once, then use on-device result
5. **Invalid API key**: Alert admin, disable cloud analysis
6. **Image not in S3**: Upload image first, then analyze
7. **Quota exceeded**: Notify admin, disable cloud for new requests
