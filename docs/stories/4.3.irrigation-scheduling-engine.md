### Technical Implementation Details

**Irrigation Scheduling Algorithm**

The scheduling engine uses a soil moisture depletion approach:

```javascript
// IrrigationSchedulingEngine.js
class IrrigationSchedulingEngine {
  constructor(field, cropSeason, irrigationMethod) {
    this.field = field;
    this.cropSeason = cropSeason;
    this.irrigationMethod = irrigationMethod;
    this.moistureModel = new SoilMoisturePredictionModel(field, cropSeason, field.soil_type);
    this.soilProps = new SoilPropertiesService().getProperties(field.soil_type);
  }

  /**
   * Generate irrigation schedule for next 30 days
   * @param {Array} weatherForecast - 30-day weather forecast
   * @returns {Array} Irrigation schedule events
   */
  async generateSchedule(weatherForecast) {
    const schedule = [];
    let currentMoisture = await this.moistureModel.calculateCurrentMoisture();

    // Get crop water requirements
    const threshold = new MoistureThresholdService().getCriticalThreshold(
      this.cropSeason.crop_type,
      this.field.soil_type
    );

    const fieldCapacityMm = this.soilProps.field_capacity * this.soilProps.root_zone_depth_mm;
    const criticalMoistureMm = (threshold.critical_percent / 100) * fieldCapacityMm;

    for (let day = 0; day < 30; day++) {
      const weather = weatherForecast[day];

      // Calculate daily water balance
      const daysSincePlanting = this.getDaysSincePlanting() + day;
      const Kc = new CropCoefficientService().getKcByDays(this.cropSeason.crop_type, daysSincePlanting);
      const ETc = Kc * weather.reference_et_mm;
      const effectiveRainfall = this.calculateEffectiveRainfall(weather.rainfall_mm || 0);

      currentMoisture = currentMoisture + effectiveRainfall - ETc;

      // Check if irrigation needed
      if (currentMoisture <= criticalMoistureMm) {
        // Schedule irrigation to refill to field capacity
        const irrigationAmount = fieldCapacityMm - currentMoisture;
        const grossIrrigation = this.calculateGrossIrrigation(irrigationAmount);

        const event = {
          id: `irr-${Date.now()}-${day}`,
          field_id: this.field.id,
          crop_season_id: this.cropSeason.id,
          scheduled_date: this.addDays(new Date(), day),
          irrigation_method: this.irrigationMethod.method_type,

          // Water amounts
          net_irrigation_mm: irrigationAmount,
          gross_irrigation_mm: grossIrrigation,
          water_volume_liters: this.calculateWaterVolume(grossIrrigation),

          // Duration
          duration_hours: this.calculateDuration(grossIrrigation),

          // Metadata
          trigger_reason: 'moisture_depletion',
          soil_moisture_before_percent: (currentMoisture / fieldCapacityMm) * 100,
          soil_moisture_after_percent: 100,  // Refill to field capacity

          status: 'scheduled',
          is_weather_dependent: true,
          confidence_level: this.calculateConfidence(day, weather)
        };

        schedule.push(event);

        // Update moisture after irrigation
        currentMoisture = fieldCapacityMm;
      }

      // Prevent moisture from exceeding field capacity (percolation)
      if (currentMoisture > fieldCapacityMm) {
        currentMoisture = fieldCapacityMm;
      }
    }

    return schedule;
  }

  /**
   * Calculate gross irrigation accounting for application efficiency
   */
  calculateGrossIrrigation(netIrrigation) {
    const efficiency = this.irrigationMethod.application_efficiency;
    return netIrrigation / efficiency;
  }

  /**
   * Convert irrigation depth (mm) to water volume (liters)
   */
  calculateWaterVolume(depthMm) {
    const fieldAreaM2 = this.field.size_acres * 4046.86;  // Acres to m²
    const volumeM3 = (depthMm / 1000) * fieldAreaM2;
    const volumeLiters = volumeM3 * 1000;
    return volumeLiters;
  }

  /**
   * Calculate irrigation duration based on application rate
   */
  calculateDuration(grossIrrigationMm) {
    const applicationRate = this.irrigationMethod.application_rate_mm_hr;
    const durationHours = grossIrrigationMm / applicationRate;
    return durationHours;
  }

  /**
   * Calculate schedule confidence based on weather forecast certainty
   */
  calculateConfidence(daysAhead, weather) {
    // Confidence decreases with forecast distance
    let confidence = 100 - (daysAhead * 3);  // 3% decrease per day

    // Decrease confidence if high precipitation probability
    if (weather.precipitation_probability > 50) {
      confidence -= 20;
    }

    return Math.max(50, Math.min(100, confidence));  // 50-100% range
  }

  calculateEffectiveRainfall(rainfall) {
    if (rainfall <= 0) return 0;
    if (rainfall < 50) {
      return rainfall * (125 - 0.2 * rainfall) / 125;
    } else {
      return 125 / 2.5;
    }
  }

  getDaysSincePlanting() {
    const plantingDate = new Date(this.cropSeason.planting_date);
    const now = new Date();
    return Math.floor((now - plantingDate) / (1000 * 60 * 60 * 24));
  }

  addDays(date, days) {
    const result = new Date(date);
    result.setDate(result.getDate() + days);
    return result;
  }

  /**
   * Check for upcoming rain and adjust schedule
   */
  async adjustForWeather(scheduleEvents, weatherForecast) {
    const adjustedSchedule = [];

    for (const event of scheduleEvents) {
      const eventDate = new Date(event.scheduled_date);
      const weather = weatherForecast.find(w =>
        this.isSameDay(new Date(w.forecast_date), eventDate)
      );

      // Delay irrigation if significant rain expected
      if (weather && weather.rainfall_mm > 10) {
        event.status = 'postponed';
        event.postpone_reason = `Expected ${weather.rainfall_mm}mm rain`;
        event.original_date = event.scheduled_date;
        event.scheduled_date = this.addDays(eventDate, 2);  // Postpone 2 days
      }

      adjustedSchedule.push(event);
    }

    return adjustedSchedule;
  }

  isSameDay(date1, date2) {
    return date1.toDateString() === date2.toDateString();
  }
}
```

```javascript
class IrrigationMethodService {
  getMethod(methodType) {
    return IRRIGATION_METHODS[methodType] || IRRIGATION_METHODS.flood;
  }

  recommendMethod(cropType, soilType, fieldSize) {
    // Simple recommendation logic
    if (cropType === 'rice') return IRRIGATION_METHODS.flood;
    if (fieldSize < 2) return IRRIGATION_METHODS.drip;  // Small fields
    if (soilType === 'sandy') return IRRIGATION_METHODS.drip;  // Fast drainage
    if (cropType === 'wheat' || cropType === 'corn') return IRRIGATION_METHODS.sprinkler;
    return IRRIGATION_METHODS.furrow;  // Default for row crops
  }
}
```

**Schedule Modification**

```javascript
// IrrigationScheduleService.js
class IrrigationScheduleService {
  /**
   * Postpone scheduled irrigation
   */
  async postponeIrrigation(eventId, newDate, reason) {
    const event = await db.queryOne(
      'SELECT * FROM irrigation_schedule WHERE id = $1',
      [eventId]
    );

    if (!event) throw new Error('Irrigation event not found');

    // Update event
    await db.update('irrigation_schedule', { id: eventId }, {
      original_date: event.scheduled_date,
      scheduled_date: newDate,
      status: 'postponed',
      postpone_reason: reason,
      updated_at: new Date()
    });

    // Log modification
    await this.logScheduleModification(eventId, 'postponed', { newDate, reason });

    // Recalculate subsequent schedule
    await this.recalculateSchedule(event.field_id, event.crop_season_id);

    return { success: true };
  }

  /**
   * Mark irrigation as completed
   */
  async markIrrigationComplete(eventId, actualAmount, completionDate) {
    await db.update('irrigation_schedule', { id: eventId }, {
      status: 'completed',
      actual_water_liters: actualAmount,
      completion_date: completionDate,
      updated_at: new Date()
    });

    // Update soil moisture model with actual irrigation
    const event = await db.queryOne('SELECT * FROM irrigation_schedule WHERE id = $1', [eventId]);
    await this.moistureModel.updateWithIrrigation(actualAmount, completionDate);

    // Log completion
    await this.logScheduleModification(eventId, 'completed', { actualAmount, completionDate });

    return { success: true };
  }

  /**
   * Skip irrigation with reason
   */
  async skipIrrigation(eventId, reason) {
    await db.update('irrigation_schedule', { id: eventId }, {
      status: 'skipped',
      skip_reason: reason,
      updated_at: new Date()
    });

    await this.logScheduleModification(eventId, 'skipped', { reason });

    // Recalculate schedule if needed
    const event = await db.queryOne('SELECT * FROM irrigation_schedule WHERE id = $1', [eventId]);
    await this.recalculateSchedule(event.field_id, event.crop_season_id);

    return { success: true };
  }

  /**
   * Recalculate schedule after modifications
   */
  async recalculateSchedule(fieldId, cropSeasonId) {
    // Get current moisture state
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);
    const cropSeason = await db.queryOne('SELECT * FROM crop_seasons WHERE id = $1', [cropSeasonId]);
    const irrigationMethod = await this.getFieldIrrigationMethod(fieldId);

    // Get weather forecast
    const weatherForecast = await WeatherDataService.get30DayForecast(field.latitude, field.longitude);

    // Generate new schedule
    const engine = new IrrigationSchedulingEngine(field, cropSeason, irrigationMethod);
    const newSchedule = await engine.generateSchedule(weatherForecast);

    // Remove old future events
    await db.query(`
      DELETE FROM irrigation_schedule
      WHERE field_id = $1 AND crop_season_id = $2
        AND scheduled_date > NOW()
        AND status = 'scheduled'
    `, [fieldId, cropSeasonId]);

    // Insert new schedule
    for (const event of newSchedule) {
      await db.insert('irrigation_schedule', event);
    }

    return newSchedule;
  }

  async logScheduleModification(eventId, modificationType, details) {
    await db.insert('irrigation_schedule_log', {
      id: `log-${Date.now()}`,
      irrigation_event_id: eventId,
      modification_type: modificationType,
      modification_details: JSON.stringify(details),
      created_at: new Date()
    });
  }

  async getFieldIrrigationMethod(fieldId) {
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);
    const methodType = field.irrigation_method || 'flood';
    return new IrrigationMethodService().getMethod(methodType);
  }
}
```

### Data Model Requirements

**Irrigation Schedule Table**
```sql
CREATE TABLE irrigation_schedule (
  id TEXT PRIMARY KEY,
  field_id TEXT NOT NULL,
  crop_season_id TEXT NOT NULL,

  scheduled_date DATE NOT NULL,
  original_date DATE,                    -- If postponed
  completion_date DATE,

  -- Irrigation method
  irrigation_method TEXT NOT NULL,       -- 'drip', 'sprinkler', 'flood', 'furrow'

  -- Water amounts
  net_irrigation_mm REAL NOT NULL,       -- Net water requirement
  gross_irrigation_mm REAL NOT NULL,     -- Accounting for efficiency
  water_volume_liters REAL NOT NULL,

  -- Duration
  duration_hours REAL NOT NULL,

  -- Actual values (when completed)
  actual_water_liters REAL,
  actual_duration_hours REAL,

  -- Status
  status TEXT DEFAULT 'scheduled',       -- 'scheduled', 'postponed', 'completed', 'skipped', 'cancelled'
  postpone_reason TEXT,
  skip_reason TEXT,

  -- Metadata
  trigger_reason TEXT,                   -- 'moisture_depletion', 'growth_stage', 'manual'
  soil_moisture_before_percent REAL,
  soil_moisture_after_percent REAL,
  is_weather_dependent BOOLEAN DEFAULT 1,
  confidence_level REAL,                 -- 0-100

  -- Notifications
  reminder_sent BOOLEAN DEFAULT 0,
  reminder_sent_at DATETIME,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (field_id) REFERENCES fields(id),
  FOREIGN KEY (crop_season_id) REFERENCES crop_seasons(id)
);

CREATE INDEX idx_irrigation_schedule_field_date ON irrigation_schedule(field_id, scheduled_date);
CREATE INDEX idx_irrigation_schedule_status ON irrigation_schedule(status, scheduled_date);
CREATE INDEX idx_irrigation_schedule_reminders ON irrigation_schedule(scheduled_date, reminder_sent) WHERE status = 'scheduled';
```
