**Market Dashboard Service**

```javascript
// MarketDashboardService.js
class MarketDashboardService {
  constructor() {
    this.priceService = new MandiPriceService();
    this.predictionService = new PricePredictionService();
    this.recommendationService = new SellingTimeOptimizer();
    this.supplyDemandService = new SupplyForecastService();
  }

  async getDashboardData(userId) {
    // Get user's monitored crops
    const monitoredCrops = await this.getMonitoredCrops(userId);

    // Get user's location for mandi proximity
    const userProfile = await this.getUserProfile(userId);

    // Aggregate all market data
    const dashboardData = {
      user_crops: monitoredCrops,
      price_overview: await this.getPriceOverview(monitoredCrops, userProfile.location),
      predictions: await this.getPredictions(monitoredCrops),
      recommendations: await this.getRecommendations(userId, monitoredCrops),
      favorite_mandis: await this.getFavoriteMandis(userId),
      active_alerts: await this.getActiveAlerts(userId),
      market_news: await this.getMarketNews(monitoredCrops, 3),
      supply_demand_status: await this.getSupplyDemandStatus(monitoredCrops),
      export_insights: await this.getExportInsights(monitoredCrops),
      last_updated: new Date()
    };

    // Cache dashboard data
    await this.cacheDashboard(userId, dashboardData);

    return dashboardData;
  }

  async getMonitoredCrops(userId) {
    // Get crops from user's fields
    const activeCrops = await db.query(`
      SELECT DISTINCT cs.crop_type
      FROM crop_seasons cs
      JOIN fields f ON cs.field_id = f.id
      WHERE f.farmer_id = $1
        AND cs.status IN ('active', 'harvested')
    `, [userId]);

    // Get additional monitored crops from user preferences
    const preferences = await db.query(`
      SELECT crop_type FROM user_monitored_crops WHERE user_id = $1
    `, [userId]);

    // Combine and deduplicate
    const allCrops = [...new Set([
      ...activeCrops.map(c => c.crop_type),
      ...preferences.map(p => p.crop_type)
    ])];

    return allCrops;
  }

  async getPriceOverview(crops, userLocation) {
    const priceData = [];

    for (const crop of crops) {
      // Get current price at nearest mandi
      const nearestMandi = await this.getNearestMandiPrice(crop, userLocation);

      if (!nearestMandi) continue;

      // Get price trend (last 7 days)
      const trend = await this.getPriceTrend(crop, nearestMandi.mandi_id, 7);

      priceData.push({
        crop_type: crop,
        current_price: nearestMandi.modal_price_per_kg,
        mandi_name: nearestMandi.mandi_name,
        price_date: nearestMandi.price_date,
        trend: trend.direction, // 'up', 'down', 'stable'
        change_percent: trend.change_percent,
        change_amount: trend.change_amount
      });
    }

    return priceData;
  }

  async getNearestMandiPrice(cropType, userLocation) {
    const result = await db.queryOne(`
      SELECT
        mp.*,
        m.name as mandi_name,
        ST_Distance(
          m.location_gps::geography,
          ST_MakePoint($2, $3)::geography
        ) / 1000 as distance_km
      FROM mandi_prices mp
      JOIN mandis m ON mp.mandi_id = m.id
      WHERE mp.crop_type = $1
        AND mp.price_date = (
          SELECT MAX(price_date) FROM mandi_prices WHERE crop_type = $1
        )
      ORDER BY distance_km ASC
      LIMIT 1
    `, [cropType, userLocation.longitude, userLocation.latitude]);

    return result;
  }

  async getPriceTrend(cropType, mandiId, days) {
    const prices = await db.query(`
      SELECT modal_price_per_kg, price_date
      FROM mandi_prices
      WHERE crop_type = $1
        AND mandi_id = $2
        AND price_date >= CURRENT_DATE - INTERVAL '${days} days'
      ORDER BY price_date DESC
    `, [cropType, mandiId]);

    if (prices.length < 2) {
      return { direction: 'stable', change_percent: 0, change_amount: 0 };
    }

    const latest = prices[0].modal_price_per_kg;
    const oldest = prices[prices.length - 1].modal_price_per_kg;
    const change = latest - oldest;
    const changePercent = (change / oldest) * 100;

    let direction;
    if (changePercent > 2) direction = 'up';
    else if (changePercent < -2) direction = 'down';
    else direction = 'stable';

    return {
      direction,
      change_percent: Math.round(changePercent * 10) / 10,
      change_amount: Math.round(change)
    };
  }

  async getPredictions(crops) {
    const predictions = [];

    for (const crop of crops) {
      // Get 30-day prediction for crop
      const prediction = await db.queryOne(`
        SELECT *
        FROM price_predictions
        WHERE crop_type = $1
          AND days_ahead = 30
        ORDER BY prediction_date DESC
        LIMIT 1
      `, [crop]);

      if (prediction) {
        predictions.push({
          crop_type: crop,
          predicted_price_30d: prediction.predicted_price,
          confidence: prediction.confidence_level,
          trend: prediction.trend
        });
      }
    }

    return predictions;
  }

  async getRecommendations(userId, crops) {
    const recommendations = [];

    for (const crop of crops) {
      // Get active crop seasons for this crop
      const cropSeasons = await db.query(`
        SELECT cs.id
        FROM crop_seasons cs
        JOIN fields f ON cs.field_id = f.id
        WHERE f.farmer_id = $1
          AND cs.crop_type = $2
          AND cs.status IN ('active', 'harvested')
      `, [userId, crop]);

      for (const season of cropSeasons) {
        // Get selling recommendation
        const rec = await db.queryOne(`
          SELECT *
          FROM selling_recommendations
          WHERE crop_season_id = $1
          ORDER BY created_at DESC
          LIMIT 1
        `, [season.id]);

        if (rec) {
          recommendations.push({
            crop_type: crop,
            crop_season_id: season.id,
            recommendation: rec.recommendation_action,
            message: rec.recommendation_message,
            optimal_date: rec.optimal_date
          });
        }
      }
    }

    return recommendations;
  }

  async getFavoriteMandis(userId) {
    return await db.query(`
      SELECT
        m.*,
        fm.added_at,
        COUNT(DISTINCT mp.crop_type) as crops_available
      FROM favorite_mandis fm
      JOIN mandis m ON fm.mandi_id = m.id
      LEFT JOIN mandi_prices mp ON m.id = mp.mandi_id
        AND mp.price_date = CURRENT_DATE
      WHERE fm.user_id = $1
      GROUP BY m.id, fm.added_at
      ORDER BY fm.added_at DESC
      LIMIT 5
    `, [userId]);
  }

  async getActiveAlerts(userId) {
    return await db.query(`
      SELECT * FROM price_alerts
      WHERE user_id = $1
        AND active = TRUE
      ORDER BY created_at DESC
    `, [userId]);
  }

  async getMarketNews(crops, limit = 3) {
    return await db.query(`
      SELECT * FROM market_news
      WHERE crop_type = ANY($1) OR crop_type IS NULL
      ORDER BY published_at DESC
      LIMIT $2
    `, [crops, limit]);
  }

  async getSupplyDemandStatus(crops) {
    const status = [];

    for (const crop of crops) {
      const supplyForecast = await db.queryOne(`
        SELECT supply_index, status
        FROM supply_forecasts
        WHERE crop_type = $1
        ORDER BY created_at DESC
        LIMIT 1
      `, [crop]);

      if (supplyForecast) {
        status.push({
          crop_type: crop,
          supply_index: supplyForecast.supply_index,
          supply_status: supplyForecast.status
        });
      }
    }

    return status;
  }

  async getExportInsights(crops) {
    const insights = [];

    for (const crop of crops) {
      const exportData = await db.queryOne(`
        SELECT *
        FROM export_statistics
        WHERE crop_type = $1
        ORDER BY year DESC
        LIMIT 1
      `, [crop]);

      if (exportData) {
        insights.push({
          crop_type: crop,
          export_volume: exportData.export_volume_tons,
          export_value: exportData.export_value_usd,
          primary_destinations: exportData.primary_destinations
        });
      }
    }

    return insights;
  }

  async cacheDashboard(userId, data) {
    // Cache in Redis for 1 hour
    await redis.setex(`dashboard:${userId}`, 3600, JSON.stringify(data));
  }
}
```

**Price Alert Service**

```javascript
// PriceAlertService.js
const cron = require('node-cron');

class PriceAlertService {
  constructor() {
    this.notificationService = new NotificationService();
  }

  start() {
    // Check price alerts daily at 9 AM (after price updates)
    cron.schedule('0 9 * * *', async () => {
      await this.checkPriceAlerts();
    });
  }

  async createAlert(userId, alertData) {
    const alert = {
      id: `alert-${Date.now()}`,
      user_id: userId,
      crop_type: alertData.cropType,
      mandi_id: alertData.mandiId,
      condition: alertData.condition, // 'above', 'below'
      target_price: alertData.targetPrice,
      active: true,
      created_at: new Date()
    };

    await db.insert('price_alerts', alert);
    return alert;
  }

  async checkPriceAlerts() {
    // Get all active alerts
    const alerts = await db.query(`
      SELECT * FROM price_alerts WHERE active = TRUE
    `);

    for (const alert of alerts) {
      await this.evaluateAlert(alert);
    }
  }

  async evaluateAlert(alert) {
    // Get current price
    const currentPrice = await db.queryOne(`
      SELECT modal_price_per_kg
      FROM mandi_prices
      WHERE crop_type = $1
        AND mandi_id = $2
        AND price_date = CURRENT_DATE
    `, [alert.crop_type, alert.mandi_id]);

    if (!currentPrice) return;

    let triggered = false;

    if (alert.condition === 'above' && currentPrice.modal_price_per_kg >= alert.target_price) {
      triggered = true;
    } else if (alert.condition === 'below' && currentPrice.modal_price_per_kg <= alert.target_price) {
      triggered = true;
    }

    if (triggered) {
      await this.triggerAlert(alert, currentPrice.modal_price_per_kg);
    }
  }

  async triggerAlert(alert, currentPrice) {
    // Get mandi name
    const mandi = await db.queryOne('SELECT name FROM mandis WHERE id = $1', [alert.mandi_id]);

    // Send notification
    await this.notificationService.sendPushNotification(alert.user_id, {
      title: 'Price Alert!',
      body: `${alert.crop_type} price at ${mandi.name} is PKR ${currentPrice} (target: ${alert.target_price})`,
      priority: 'high',
      data: {
        type: 'price_alert',
        alert_id: alert.id,
        crop_type: alert.crop_type
      }
    });

    // Mark alert as triggered
    await db.update('price_alerts', { id: alert.id }, {
      last_triggered_at: new Date(),
      trigger_count: db.raw('trigger_count + 1')
    });

    console.log(`Price alert triggered for user ${alert.user_id}: ${alert.crop_type} @ ${currentPrice}`);
  }

  async deleteAlert(alertId) {
    await db.update('price_alerts', { id: alertId }, {
      active: false,
      deleted_at: new Date()
    });
  }
}
```

**Seasonal Calendar Analyzer**

```javascript
// SeasonalCalendarService.js
class SeasonalCalendarService {
  async generateSeasonalCalendar(cropType, years = 5) {
    // Get historical prices by month
    const monthlyData = await db.query(`
      SELECT
        EXTRACT(MONTH FROM price_date) as month,
        AVG(modal_price_per_kg) as avg_price,
        MIN(modal_price_per_kg) as min_price,
        MAX(modal_price_per_kg) as max_price,
        STDDEV(modal_price_per_kg) as price_volatility
      FROM mandi_prices
      WHERE crop_type = $1
        AND price_date >= CURRENT_DATE - INTERVAL '${years} years'
      GROUP BY EXTRACT(MONTH FROM price_date)
      ORDER BY month
    `, [cropType]);

    // Calculate overall average
    const overallAvg = monthlyData.reduce((sum, m) => sum + m.avg_price, 0) / monthlyData.length;

    // Classify each month
    const calendar = monthlyData.map(month => {
      const priceIndex = (month.avg_price / overallAvg) * 100;

      let classification;
      if (priceIndex > 120) classification = 'peak';
      else if (priceIndex > 110) classification = 'high';
      else if (priceIndex < 80) classification = 'trough';
      else if (priceIndex < 90) classification = 'low';
      else classification = 'normal';

      return {
        month: month.month,
        month_name: this.getMonthName(month.month),
        avg_price: Math.round(month.avg_price),
        min_price: Math.round(month.min_price),
        max_price: Math.round(month.max_price),
        price_index: Math.round(priceIndex),
        classification: classification,
        volatility: Math.round(month.price_volatility || 0)
      };
    });

    return {
      crop_type: cropType,
      years_analyzed: years,
      overall_avg_price: Math.round(overallAvg),
      monthly_calendar: calendar,
      peak_months: calendar.filter(m => m.classification === 'peak').map(m => m.month_name),
      trough_months: calendar.filter(m => m.classification === 'trough').map(m => m.month_name)
    };
  }

  getMonthName(monthNumber) {
    const months = [
      'January', 'February', 'March', 'April', 'May', 'June',
      'July', 'August', 'September', 'October', 'November', 'December'
    ];
    return months[monthNumber - 1];
  }

  async getCurrentYearComparison(cropType) {
    // Overlay current year prices on historical pattern
    const currentYearPrices = await db.query(`
      SELECT
        EXTRACT(MONTH FROM price_date) as month,
        AVG(modal_price_per_kg) as avg_price
      FROM mandi_prices
      WHERE crop_type = $1
        AND EXTRACT(YEAR FROM price_date) = EXTRACT(YEAR FROM CURRENT_DATE)
      GROUP BY EXTRACT(MONTH FROM price_date)
      ORDER BY month
    `, [cropType]);

    const historicalCalendar = await this.generateSeasonalCalendar(cropType, 5);

    return {
      historical: historicalCalendar.monthly_calendar,
      current_year: currentYearPrices.map(m => ({
        month: m.month,
        avg_price: Math.round(m.avg_price)
      }))
    };
  }
}
```

### Data Model Requirements

```sql
-- User monitored crops
CREATE TABLE user_monitored_crops (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  crop_type TEXT NOT NULL,
  added_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(user_id, crop_type)
);

-- Price alerts
CREATE TABLE price_alerts (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  crop_type TEXT NOT NULL,
  mandi_id TEXT,
  condition TEXT NOT NULL,  -- 'above', 'below'
  target_price DECIMAL(10, 2) NOT NULL,
  active BOOLEAN DEFAULT TRUE,
  last_triggered_at TIMESTAMP,
  trigger_count INTEGER DEFAULT 0,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  deleted_at TIMESTAMP,
  FOREIGN KEY (mandi_id) REFERENCES mandis(id)
);

CREATE INDEX idx_price_alerts_user ON price_alerts(user_id, active);
CREATE INDEX idx_price_alerts_crop ON price_alerts(crop_type, active);

-- Market news
CREATE TABLE market_news (
  id TEXT PRIMARY KEY,
  title TEXT NOT NULL,
  summary TEXT,
  content TEXT,
  source TEXT,
  source_url TEXT,
  crop_type TEXT,  -- NULL for general news
  category TEXT,   -- 'prices', 'policy', 'weather', 'technology'
  published_at TIMESTAMP NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_market_news_crop_date ON market_news(crop_type, published_at DESC);
CREATE INDEX idx_market_news_published ON market_news(published_at DESC);

-- Supply forecasts (cached)
CREATE TABLE supply_forecasts (
  id TEXT PRIMARY KEY,
  crop_type TEXT NOT NULL,
  season TEXT,
  year INTEGER,
  supply_index INTEGER,
  status TEXT,  -- 'oversupply', 'normal', 'shortage'
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_supply_forecasts_crop ON supply_forecasts(crop_type, created_at DESC);
```
