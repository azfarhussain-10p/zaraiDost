```javascript
const NOAA_CFS_BASE_URL = 'https://www.cpc.ncep.noaa.gov/products/CFSv2';

async function fetchCFSv2Forecast(latitude, longitude, leadMonths = 3) {
  // CFS v2 provides 9-month seasonal forecasts
  // Lead months: 3 (3-month) or 6 (6-month) forecast

  const response = await axios.get(`${NOAA_CFS_BASE_URL}/forecast`, {
    params: {
      lat: latitude,
      lon: longitude,
      lead: leadMonths,
      format: 'json'
    }
  });

  return transformCFSv2Response(response.data);
}
```

**El Niño/La Niña (ENSO) Tracking**

```javascript
// ENSOTrackingService.js
class ENSOTrackingService {
  async fetchENSOIndex() {
    // NOAA Oceanic Niño Index (ONI)
    const response = await axios.get(
      'https://www.cpc.ncep.noaa.gov/data/indices/oni.ascii.txt'
    );

    const oniData = this.parseONIData(response.data);

    // Store in database
    await db.insert('enso_forecasts', {
      id: `enso-${Date.now()}`,
      month: oniData.currentMonth,
      oni_value: oniData.value,
      phase: this.classifyENSOPhase(oniData.value),
      pakistan_impact: this.predictPakistanImpact(oniData.value),
      forecast_confidence: oniData.confidence,
      created_at: new Date()
    });

    return oniData;
  }

  classifyENSOPhase(oniValue) {
    // ONI classification thresholds
    if (oniValue >= 0.5) return 'El Niño';
    if (oniValue <= -0.5) return 'La Niña';
    return 'Neutral';
  }

  predictPakistanImpact(oniValue) {
    // Pakistan-specific ENSO impacts
    if (oniValue >= 0.5) {
      // El Niño: Typically drier winter, warmer summer
      return {
        phase: 'El Niño',
        winter_rainfall: 'below_normal',
        summer_rainfall: 'near_normal',
        monsoon_impact: 'weaker',
        temperature: 'above_normal',
        drought_risk: 'elevated',
        description_en: 'El Niño typically brings drier winters and weaker monsoons to Pakistan',
        description_ur: 'El Niño se sardi mein kam barish aur kamzor monsoon hota hai'
      };
    } else if (oniValue <= -0.5) {
      // La Niña: Typically wetter winter, stronger monsoon
      return {
        phase: 'La Niña',
        winter_rainfall: 'above_normal',
        summer_rainfall: 'above_normal',
        monsoon_impact: 'stronger',
        temperature: 'near_normal',
        flood_risk: 'elevated',
        description_en: 'La Niña typically brings wetter winters and stronger monsoons to Pakistan',
        description_ur: 'La Niña se sardi mein zyada barish aur tez monsoon hota hai'
      };
    } else {
      return {
        phase: 'Neutral',
        winter_rainfall: 'near_normal',
        summer_rainfall: 'near_normal',
        monsoon_impact: 'normal',
        temperature: 'near_normal',
        description_en: 'Neutral conditions: near-normal climate patterns expected',
        description_ur: 'Normal halat: mamuli mausam ki tawaqo hai'
      };
    }
  }
}
```

**Drought Risk Assessment**

```javascript
// DroughtRiskService.js
class DroughtRiskService {
  async calculateSPI(locationId, months = 3) {
    // Standardized Precipitation Index
    // months: accumulation period (3, 6, 12, 24 months)

    const historicalRainfall = await this.getHistoricalRainfall(locationId, months);
    const mean = this.calculateMean(historicalRainfall);
    const stdDev = this.calculateStdDev(historicalRainfall, mean);

    const currentRainfall = historicalRainfall[historicalRainfall.length - 1];
    const spi = (currentRainfall - mean) / stdDev;

    return {
      spi_value: spi,
      classification: this.classifySPI(spi),
      drought_category: this.getDroughtCategory(spi),
      description: this.getSPIDescription(spi)
    };
  }

  classifySPI(spi) {
    if (spi <= -2.0) return 'Extremely Dry';
    if (spi <= -1.5) return 'Severely Dry';
    if (spi <= -1.0) return 'Moderately Dry';
    if (spi <= -0.5) return 'Abnormally Dry';
    if (spi < 0.5) return 'Near Normal';
    if (spi < 1.0) return 'Abnormally Wet';
    if (spi < 1.5) return 'Moderately Wet';
    if (spi < 2.0) return 'Severely Wet';
    return 'Extremely Wet';
  }

  getDroughtCategory(spi) {
    if (spi >= -0.5) return 'none';
    if (spi >= -1.0) return 'mild';
    if (spi >= -1.5) return 'moderate';
    if (spi >= -2.0) return 'severe';
    return 'extreme';
  }

  async calculatePDSI(locationId) {
    // Palmer Drought Severity Index
    // More complex calculation considering temperature and soil moisture

    const weatherData = await this.getHistoricalWeatherData(locationId);

    // Simplified PDSI calculation
    // Actual implementation requires soil moisture modeling
    const precipitation = weatherData.rainfall_mm;
    const temperature = weatherData.temperature_avg_c;
    const potentialET = this.calculatePotentialET(temperature);

    const moistureAnomaly = precipitation - potentialET;
    const pdsi = this.normalizePDSI(moistureAnomaly);

    return {
      pdsi_value: pdsi,
      classification: this.classifyPDSI(pdsi),
      description: this.getPDSIDescription(pdsi)
    };
  }

  classifyPDSI(pdsi) {
    if (pdsi <= -4.0) return 'Extreme Drought';
    if (pdsi <= -3.0) return 'Severe Drought';
    if (pdsi <= -2.0) return 'Moderate Drought';
    if (pdsi <= -1.0) return 'Mild Drought';
    if (pdsi < 1.0) return 'Near Normal';
    if (pdsi < 2.0) return 'Slight Wetness';
    if (pdsi < 3.0) return 'Moderate Wetness';
    if (pdsi < 4.0) return 'Severe Wetness';
    return 'Extreme Wetness';
  }

  async assessRegionalDroughtRisk(province, forecastMonths = 3) {
    // Calculate drought risk for upcoming season
    const seasonalForecast = await this.getSeasonalForecast(province, forecastMonths);
    const historicalNormal = await this.getHistoricalNormal(province, forecastMonths);

    const rainfallDeficit = ((seasonalForecast.rainfall - historicalNormal.rainfall) / historicalNormal.rainfall) * 100;

    let droughtRisk = 'low';
    if (rainfallDeficit < -30) droughtRisk = 'extreme';
    else if (rainfallDeficit < -20) droughtRisk = 'high';
    else if (rainfallDeficit < -10) droughtRisk = 'moderate';

    await db.insert('drought_risk_forecasts', {
      id: `drought-${Date.now()}`,
      province,
      forecast_period_months: forecastMonths,
      rainfall_deficit_percent: rainfallDeficit,
      drought_risk: droughtRisk,
      recommendation: this.getDroughtRecommendation(droughtRisk),
      forecast_date: new Date()
    });

    return { province, droughtRisk, rainfallDeficit };
  }

  getDroughtRecommendation(droughtRisk) {
    const recommendations = {
      extreme: 'Switch to drought-resistant crops (millet, sorghum). Invest in drip irrigation.',
      high: 'Consider drought-tolerant varieties. Implement water conservation practices.',
      moderate: 'Monitor soil moisture closely. Prepare supplemental irrigation.',
      low: 'Normal crop selection. Maintain standard irrigation practices.'
    };
    return recommendations[droughtRisk] || recommendations.low;
  }
}
```

```sql
-- Seasonal forecasts
CREATE TABLE seasonal_forecasts (
  id TEXT PRIMARY KEY,
  forecast_type TEXT NOT NULL,          -- 'seasonal'
  forecast_period TEXT NOT NULL,        -- '3-month', '6-month'
  issue_date DATE NOT NULL,
  valid_from DATE NOT NULL,
  valid_to DATE NOT NULL,

  region TEXT,
  district TEXT,
  latitude REAL,
  longitude REAL,

  temperature_outlook TEXT,             -- 'above_normal', 'normal', 'below_normal'
  temperature_anomaly_c REAL,
  temperature_confidence INTEGER,

  precipitation_outlook TEXT,
  precipitation_anomaly_percent REAL,
  precipitation_confidence INTEGER,

  monsoon_onset_date DATE,
  monsoon_intensity TEXT,
  monsoon_total_rainfall_mm REAL,

  enso_phase TEXT,
  enso_oni_value REAL,
  enso_impact TEXT,

  drought_risk TEXT,
  flood_risk TEXT,

  primary_source TEXT,
  secondary_sources TEXT,               -- JSON array
  forecast_confidence_overall INTEGER,

  historical_10yr_avg_rainfall_mm REAL,
  historical_10yr_avg_temp_c REAL,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  last_updated DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(region, district, valid_from, forecast_period)
);

CREATE INDEX idx_seasonal_forecasts_region_date ON seasonal_forecasts(region, valid_from DESC);
CREATE INDEX idx_seasonal_forecasts_valid ON seasonal_forecasts(valid_from, valid_to);

-- ENSO tracking
CREATE TABLE enso_forecasts (
  id TEXT PRIMARY KEY,
  month DATE NOT NULL,
  oni_value REAL NOT NULL,              -- Oceanic Niño Index
  phase TEXT NOT NULL,                  -- 'El Niño', 'La Niña', 'Neutral'
  pakistan_impact TEXT,                 -- JSON with regional impacts
  forecast_confidence INTEGER,
  data_source TEXT DEFAULT 'NOAA',
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(month)
);

CREATE INDEX idx_enso_forecasts_month ON enso_forecasts(month DESC);

-- Drought risk forecasts
CREATE TABLE drought_risk_forecasts (
  id TEXT PRIMARY KEY,
  province TEXT NOT NULL,
  district TEXT,
  forecast_period_months INTEGER,       -- 3, 6, 12
  forecast_date DATE NOT NULL,

  spi_value REAL,                       -- Standardized Precipitation Index
  spi_classification TEXT,
  pdsi_value REAL,                      -- Palmer Drought Severity Index
  pdsi_classification TEXT,

  rainfall_deficit_percent REAL,
  drought_risk TEXT NOT NULL,           -- 'low', 'moderate', 'high', 'extreme'
  drought_category TEXT,                -- 'none', 'mild', 'moderate', 'severe', 'extreme'

  recommendation TEXT,
  early_warning_days INTEGER,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(province, district, forecast_date, forecast_period_months)
);

CREATE INDEX idx_drought_risk_province_date ON drought_risk_forecasts(province, forecast_date DESC);

-- Flood risk forecasts
CREATE TABLE flood_risk_forecasts (
  id TEXT PRIMARY KEY,
  province TEXT NOT NULL,
  district TEXT,
  river_basin TEXT,                     -- 'Indus', 'Ravi', 'Chenab', 'Sutlej', 'Jhelum'
  forecast_date DATE NOT NULL,
  forecast_period_months INTEGER,

  seasonal_rainfall_expected_mm REAL,
  rainfall_anomaly_percent REAL,
  flood_risk TEXT NOT NULL,             -- 'low', 'medium', 'high', 'extreme'
  flood_probability_percent INTEGER,

  affected_areas TEXT,                  -- JSON array of districts/tehsils
  elevation_threshold_m REAL,           -- Areas below this elevation at risk

  recommendation TEXT,
  early_warning_days INTEGER,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(province, district, forecast_date, forecast_period_months)
);

CREATE INDEX idx_flood_risk_province_date ON flood_risk_forecasts(province, forecast_date DESC);
CREATE INDEX idx_flood_risk_basin ON flood_risk_forecasts(river_basin, forecast_date DESC);

-- Climate history (10+ years)
CREATE TABLE climate_history (
  id TEXT PRIMARY KEY,
  location_id TEXT NOT NULL,
  year INTEGER NOT NULL,
  month INTEGER NOT NULL,

  avg_temperature_c REAL,
  max_temperature_c REAL,
  min_temperature_c REAL,
  total_precipitation_mm REAL,

  monsoon_onset_date DATE,
  monsoon_total_rainfall_mm REAL,

  enso_phase TEXT,

  drought_occurred BOOLEAN DEFAULT FALSE,
  flood_occurred BOOLEAN DEFAULT FALSE,

  data_source TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  UNIQUE(location_id, year, month)
);

CREATE INDEX idx_climate_history_location_year ON climate_history(location_id, year DESC, month DESC);
```
