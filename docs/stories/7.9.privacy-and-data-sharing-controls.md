**Privacy Enforcement Service**

```javascript
// PrivacyEnforcementService.js
class PrivacyEnforcementService {
  /**
   * Enforce privacy settings when accessing user data
   */
  async enforcePrivacy(viewerId, targetFarmerId, dataType) {
    // Get target's privacy settings
    const settings = await this.getPrivacySettings(targetFarmerId);

    // Check if feature is opted out
    if (!this.isFeatureEnabled(settings, dataType)) {
      throw new Error('User has opted out of this feature');
    }

    // Check visibility permissions
    const canView = await this.canViewData(viewerId, targetFarmerId, settings, dataType);
    if (!canView) {
      throw new Error('Privacy settings prevent access');
    }

    // Log access for transparency
    await this.logDataAccess(viewerId, targetFarmerId, dataType);

    return settings;
  }

  async filterUserData(viewerId, userData, privacySettings) {
    const filtered = { ...userData };

    // Apply visibility rules
    const relationship = await this.getUserRelationship(viewerId, userData.id);

    // Profile visibility
    if (privacySettings.profile_visibility === 'private' && viewerId !== userData.id) {
      return null; // No access
    }

    if (privacySettings.profile_visibility === 'connections' && relationship !== 'connection') {
      return null; // Connections only
    }

    // Location privacy
    if (!privacySettings.show_location || relationship === 'stranger') {
      filtered.location_lat = null;
      filtered.location_lng = null;
    } else {
      filtered.location_lat = this.fuzzLocation(
        userData.location_lat,
        privacySettings.location_precision
      );
      filtered.location_lng = this.fuzzLocation(
        userData.location_lng,
        privacySettings.location_precision
      );
    }

    // Contact information
    if (!privacySettings.show_phone || relationship === 'stranger') {
      filtered.phone = null;
    }

    if (!privacySettings.show_email || relationship === 'stranger') {
      filtered.email = null;
    }

    if (!privacySettings.show_address) {
      filtered.address = null;
    }

    // Activity visibility
    if (!privacySettings.show_posts) {
      filtered.recent_posts = [];
    }

    if (!privacySettings.show_connections && viewerId !== userData.id) {
      filtered.connections = [];
    }

    return filtered;
  }

  fuzzLocation(coordinate, precision) {
    if (!coordinate) return null;

    switch (precision) {
      case 'exact':
        return coordinate;
      case 'city':
        // Round to 2 decimal places (~1km precision)
        return Math.round(coordinate * 100) / 100;
      case 'region':
        // Round to 1 decimal place (~10km precision)
        return Math.round(coordinate * 10) / 10;
      case 'hidden':
        return null;
      default:
        return Math.round(coordinate * 10) / 10;
    }
  }

  async getUserRelationship(viewerId, targetId) {
    if (viewerId === targetId) return 'self';

    // Check if connected
    const connection = await db.query(`
      SELECT id FROM farmer_connections
      WHERE ((farmer_id = $1 AND connected_farmer_id = $2)
         OR (farmer_id = $2 AND connected_farmer_id = $1))
        AND status = 'accepted'
    `, [viewerId, targetId]);

    if (connection.length > 0) return 'connection';

    // Check if in same group
    const sharedGroup = await db.query(`
      SELECT gm1.group_id
      FROM group_members gm1
      JOIN group_members gm2 ON gm1.group_id = gm2.group_id
      WHERE gm1.farmer_id = $1 AND gm2.farmer_id = $2
    `, [viewerId, targetId]);

    if (sharedGroup.length > 0) return 'group_member';

    return 'stranger';
  }

  async logDataAccess(accessorId, accessedId, accessType, resourceType, resourceId) {
    const logId = `access-${Date.now()}-${Math.random()}`;

    await db.query(`
      INSERT INTO data_access_log (
        id, accessed_farmer_id, accessor_farmer_id,
        access_type, resource_type, resource_id,
        ip_address, user_agent
      ) VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
    `, [
      logId, accessedId, accessorId,
      accessType, resourceType, resourceId,
      this.getRequestIP(), this.getRequestUserAgent()
    ]);
  }

  isFeatureEnabled(settings, feature) {
    const featureMap = {
      'matching': settings.enable_matching,
      'groups': settings.enable_groups,
      'forum': settings.enable_forum,
      'expert_qa': settings.enable_expert_qa
    };

    return featureMap[feature] !== false;
  }

  async canViewData(viewerId, targetId, settings, dataType) {
    // Self can always view own data
    if (viewerId === targetId) return true;

    // Check profile visibility
    if (settings.profile_visibility === 'private') return false;

    if (settings.profile_visibility === 'connections') {
      const relationship = await this.getUserRelationship(viewerId, targetId);
      return relationship === 'connection';
    }

    return true; // Public
  }
}

// AnonymousPostingService.js
class AnonymousPostingService {
  async getOrCreateAnonymousProfile(farmerId) {
    // Check if user already has anonymous profile
    let profile = await db.query(`
      SELECT * FROM anonymous_profiles
      WHERE farmer_id = $1 AND is_active = TRUE
    `, [farmerId]);

    if (profile.length > 0) {
      return profile[0];
    }

    // Create new anonymous profile
    const anonymousUsername = await this.generateAnonymousUsername();
    const anonymousAvatar = this.selectRandomAvatar();

    const profileId = `anon-${Date.now()}-${Math.random()}`;

    await db.query(`
      INSERT INTO anonymous_profiles (
        id, farmer_id, anonymous_username, anonymous_avatar
      ) VALUES ($1, $2, $3, $4)
    `, [profileId, farmerId, anonymousUsername, anonymousAvatar]);

    return {
      id: profileId,
      farmer_id: farmerId,
      anonymous_username: anonymousUsername,
      anonymous_avatar: anonymousAvatar
    };
  }

  async generateAnonymousUsername() {
    const adjectives = ['Helpful', 'Wise', 'Green', 'Smart', 'Expert', 'Skilled'];
    const nouns = ['Farmer', 'Grower', 'Cultivator', 'Agriculturist'];

    let username;
    let exists = true;

    while (exists) {
      const adj = adjectives[Math.floor(Math.random() * adjectives.length)];
      const noun = nouns[Math.floor(Math.random() * nouns.length)];
      const num = Math.floor(Math.random() * 9999);

      username = `${adj}${noun}${num}`;

      // Check if username already exists
      const check = await db.query(`
        SELECT id FROM anonymous_profiles WHERE anonymous_username = $1
      `, [username]);

      exists = check.length > 0;
    }

    return username;
  }

  selectRandomAvatar() {
    const avatars = [
      'avatar_wheat', 'avatar_rice', 'avatar_cotton',
      'avatar_corn', 'avatar_tree', 'avatar_tractor'
    ];
    return avatars[Math.floor(Math.random() * avatars.length)];
  }

  async createAnonymousPost(farmerId, postData) {
    const profile = await this.getOrCreateAnonymousProfile(farmerId);

    // Create post with anonymous author
    const post = {
      ...postData,
      author_id: farmerId, // Encrypted/hidden
      display_author_id: profile.id,
      display_author_name: profile.anonymous_username,
      is_anonymous: true
    };

    return post;
  }

  async claimAnonymousPost(farmerId, postId) {
    // Verify post belongs to this user
    const post = await db.query(`
      SELECT * FROM forum_posts WHERE id = $1 AND author_id = $2
    `, [postId, farmerId]);

    if (post.length === 0) {
      throw new Error('Post not found or not owned by user');
    }

    // Remove anonymity
    await db.query(`
      UPDATE forum_posts
      SET is_anonymous = FALSE,
          display_author_id = author_id,
          display_author_name = NULL
      WHERE id = $1
    `, [postId]);
  }
}

// DataExportService.js
const archiver = require('archiver');
const fs = require('fs');

class DataExportService {
  async requestDataExport(farmerId, exportType = 'full') {
    const requestId = `export-${Date.now()}-${Math.random()}`;
    const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000); // 7 days

    await db.query(`
      INSERT INTO data_export_requests (
        id, farmer_id, export_type, expires_at
      ) VALUES ($1, $2, $3, $4)
    `, [requestId, farmerId, exportType, expiresAt]);

    // Trigger background job to generate export
    await this.scheduleExportGeneration(requestId);

    return requestId;
  }

  async generateDataExport(exportRequestId) {
    const request = await db.query(`
      SELECT * FROM data_export_requests WHERE id = $1
    `, [exportRequestId]);

    if (request.length === 0) {
      throw new Error('Export request not found');
    }

    const farmerId = request[0].farmer_id;

    // Update status
    await db.query(`
      UPDATE data_export_requests SET status = 'processing' WHERE id = $1
    `, [exportRequestId]);

    try {
      // Collect all user data
      const data = await this.collectUserData(farmerId, request[0].export_type);

      // Generate ZIP archive
      const { filePath, fileSize } = await this.createArchive(exportRequestId, data);

      // Generate download URL (presigned S3 URL or similar)
      const downloadUrl = await this.generateDownloadUrl(filePath);

      // Update request with file info
      await db.query(`
        UPDATE data_export_requests
        SET status = 'completed',
            file_path = $1,
            file_size_bytes = $2,
            download_url = $3,
            completed_at = NOW()
        WHERE id = $4
      `, [filePath, fileSize, downloadUrl, exportRequestId]);

      // Notify user
      await NotificationService.send(farmerId, {
        type: 'data_export_ready',
        title: 'Your Data Export is Ready',
        body: 'Download your data (expires in 7 days)',
        data: { exportRequestId, downloadUrl }
      });

      return downloadUrl;

    } catch (error) {
      await db.query(`
        UPDATE data_export_requests SET status = 'failed' WHERE id = $1
      `, [exportRequestId]);
      throw error;
    }
  }

  async collectUserData(farmerId, exportType) {
    const data = {};

    // Profile data
    data.profile = await db.query(`
      SELECT * FROM farmers WHERE id = $1
    `, [farmerId]);

    data.profile_details = await db.query(`
      SELECT * FROM farmer_profile_details WHERE farmer_id = $1
    `, [farmerId]);

    if (exportType === 'full' || exportType === 'activity_only') {
      // Forum activity
      data.forum_posts = await db.query(`
        SELECT * FROM forum_posts WHERE author_id = $1
      `, [farmerId]);

      data.forum_replies = await db.query(`
        SELECT * FROM forum_replies WHERE author_id = $1
      `, [farmerId]);

      // Connections
      data.connections = await db.query(`
        SELECT * FROM farmer_connections
        WHERE farmer_id = $1 OR connected_farmer_id = $1
      `, [farmerId]);

      // Messages
      data.messages = await db.query(`
        SELECT * FROM direct_messages
        WHERE sender_id = $1 OR receiver_id = $1
      `, [farmerId]);

      // Group memberships
      data.group_memberships = await db.query(`
        SELECT * FROM group_members WHERE farmer_id = $1
      `, [farmerId]);

      // Success stories
      data.success_stories = await db.query(`
        SELECT * FROM success_stories WHERE farmer_id = $1
      `, [farmerId]);

      // Expert consultations
      data.expert_consultations = await db.query(`
        SELECT * FROM expert_consultations WHERE farmer_id = $1
      `, [farmerId]);
    }

    // Privacy settings
    data.privacy_settings = await db.query(`
      SELECT * FROM privacy_settings WHERE farmer_id = $1
    `, [farmerId]);

    // Consent records
    data.consent_records = await db.query(`
      SELECT * FROM data_consent_records WHERE farmer_id = $1
    `, [farmerId]);

    // Access log (last 90 days)
    data.access_log = await db.query(`
      SELECT * FROM data_access_log
      WHERE accessed_farmer_id = $1
        AND access_timestamp > NOW() - INTERVAL '90 days'
      ORDER BY access_timestamp DESC
    `, [farmerId]);

    return data;
  }

  async createArchive(exportRequestId, data) {
    const filePath = `/tmp/export-${exportRequestId}.zip`;
    const output = fs.createWriteStream(filePath);
    const archive = archiver('zip', { zlib: { level: 9 } });

    archive.pipe(output);

    // Add JSON files for each data type
    for (const [key, value] of Object.entries(data)) {
      archive.append(JSON.stringify(value, null, 2), { name: `${key}.json` });
    }

    // Add README
    const readme = this.generateReadme();
    archive.append(readme, { name: 'README.txt' });

    await archive.finalize();

    const stats = fs.statSync(filePath);

    return {
      filePath,
      fileSize: stats.size
    };
  }

  generateReadme() {
    return `
Zarai Dost Data Export
======================

This archive contains all your personal data from Zarai Dost.

Contents:
- profile.json: Your farmer profile information
- profile_details.json: Extended profile details
- forum_posts.json: Your forum posts
- forum_replies.json: Your forum comments
- connections.json: Your farmer connections
- messages.json: Your direct messages
- group_memberships.json: Groups you've joined
- success_stories.json: Success stories you've shared
- privacy_settings.json: Your privacy settings
- consent_records.json: Your data consent history
- access_log.json: Recent access to your data

This export was generated on ${new Date().toISOString()}
Export expires 7 days from generation date.

For questions, contact support@zaraidost.pk
    `;
  }
}

    const gracePeriodEnds = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000); // 30 days

    await db.query(`
      INSERT INTO data_deletion_requests (
        id, farmer_id, deletion_type, status, grace_period_ends
      ) VALUES ($1, $2, $3, 'grace_period', $4)
    `, [requestId, farmerId, deletionType, gracePeriodEnds]);

    // Disable account immediately
    await db.query(`
      UPDATE farmers SET is_active = FALSE WHERE id = $1
    `, [farmerId]);

    // Schedule deletion after grace period
    await this.scheduleDeletion(requestId, gracePeriodEnds);

    // Notify user
    await NotificationService.send(farmerId, {
      type: 'deletion_scheduled',
      title: 'Account Deletion Scheduled',
      body: `Your account will be deleted on ${gracePeriodEnds.toLocaleDateString()}. You can cancel within 30 days.`,
      data: { requestId }
    });

    return requestId;
  }

  async cancelDeletionRequest(farmerId, requestId) {
    // Verify request belongs to user
    const request = await db.query(`
      SELECT * FROM data_deletion_requests
      WHERE id = $1 AND farmer_id = $2 AND status = 'grace_period'
    `, [requestId, farmerId]);

    if (request.length === 0) {
      throw new Error('Deletion request not found or cannot be cancelled');
    }

    // Cancel deletion
    await db.query(`
      UPDATE data_deletion_requests
      SET status = 'cancelled', cancelled_at = NOW()
      WHERE id = $1
    `, [requestId]);

    // Reactivate account
    await db.query(`
      UPDATE farmers SET is_active = TRUE WHERE id = $1
    `, [farmerId]);

    return true;
  }

  async executeDataDeletion(requestId) {
    const request = await db.query(`
      SELECT * FROM data_deletion_requests WHERE id = $1
    `, [requestId]);

    if (request.length === 0) {
      throw new Error('Deletion request not found');
    }

    const farmerId = request[0].farmer_id;
    const deletionType = request[0].deletion_type;

    // Update status
    await db.query(`
      UPDATE data_deletion_requests SET status = 'processing' WHERE id = $1
    `, [requestId]);

    try {
      if (deletionType === 'full') {
        await this.deleteAllUserData(farmerId);
      } else if (deletionType === 'community_only') {
        await this.deleteCommunityData(farmerId);
      } else if (deletionType === 'account_only') {
        await this.deleteAccountData(farmerId);
      }

      // Mark as completed
      await db.query(`
        UPDATE data_deletion_requests
        SET status = 'completed', completed_at = NOW()
        WHERE id = $1
      `, [requestId]);

      return true;

    } catch (error) {
      await db.query(`
        UPDATE data_deletion_requests SET status = 'failed' WHERE id = $1
      `, [requestId]);
      throw error;
    }
  }

  async deleteAllUserData(farmerId) {
    // Delete or anonymize content
    await this.anonymizeForumPosts(farmerId);
    await this.deleteMessages(farmerId);
    await this.deleteConnections(farmerId);
    await this.deleteGroupMemberships(farmerId);
    await this.deleteSuccessStories(farmerId);
    await this.deleteProfileData(farmerId);

    // Keep moderation records for 90 days (legal requirement)
    // These will be auto-purged by scheduled job

    // Delete main account (soft delete)
    await db.query(`
      UPDATE farmers
      SET is_deleted = TRUE,
          deleted_at = NOW(),
          email = NULL,
          phone = NULL,
          name = 'Deleted User'
      WHERE id = $1
    `, [farmerId]);
  }

  async anonymizeForumPosts(farmerId) {
    // Replace author with "Deleted User" instead of deleting
    // This preserves community value while removing personal connection
    await db.query(`
      UPDATE forum_posts
      SET author_id = NULL,
          display_author_name = 'Deleted User',
          is_anonymous = TRUE
      WHERE author_id = $1
    `, [farmerId]);

    await db.query(`
      UPDATE forum_replies
      SET author_id = NULL,
          display_author_name = 'Deleted User',
          is_anonymous = TRUE
      WHERE author_id = $1
    `, [farmerId]);
  }

  async deleteMessages(farmerId) {
    await db.query(`
      DELETE FROM direct_messages
      WHERE sender_id = $1 OR receiver_id = $1
    `, [farmerId]);

    await db.query(`
      DELETE FROM message_threads
      WHERE farmer1_id = $1 OR farmer2_id = $1
    `, [farmerId]);
  }

  async deleteConnections(farmerId) {
    await db.query(`
      DELETE FROM farmer_connections
      WHERE farmer_id = $1 OR connected_farmer_id = $1
    `, [farmerId]);
  }

  async deleteGroupMemberships(farmerId) {
    await db.query(`
      DELETE FROM group_members WHERE farmer_id = $1
    `, [farmerId]);
  }

  async deleteSuccessStories(farmerId) {
    await db.query(`
      DELETE FROM success_stories WHERE farmer_id = $1
    `, [farmerId]);
  }

  async deleteProfileData(farmerId) {
    await db.query(`
      DELETE FROM farmer_profile_details WHERE farmer_id = $1
    `, [farmerId]);

    await db.query(`
      DELETE FROM privacy_settings WHERE farmer_id = $1
    `, [farmerId]);
  }
}
```

### Security Considerations

[Source: architecture/architectural-principles-and-best-practices.md]
- **Encryption**: All personal data encrypted at rest (AES-256) and in transit (TLS 1.3)
- **Access Control**: Privacy middleware enforces settings on every API call
- **Audit Trail**: Comprehensive logging of all data access
- **GDPR Compliance**: Full implementation of GDPR rights (access, portability, deletion)
- **Data Minimization**: Collect only necessary data, delete when no longer needed
- **Consent Management**: Granular, versioned consent with withdrawal options

**Additional Security Measures**:
- Anonymous profile linking encrypted in database
- Export files encrypted and time-limited
- Deletion grace period to prevent malicious account deletion
- Multi-factor authentication for sensitive privacy actions
- Regular privacy audits and compliance reviews

[Source: architecture/non-functional-specifications.md]
- **Privacy Enforcement**: <100ms overhead per API call
- **Data Export Generation**: <5 minutes for average user

**Performance Optimization**:
- Cache privacy settings in Redis (5-minute TTL)

- Indexed queries for access log retrieval
