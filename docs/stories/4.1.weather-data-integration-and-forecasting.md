**Evapotranspiration Calculation**

Using simplified Penman-Monteith equation for reference evapotranspiration (ETo):

```javascript
// EvapotranspirationCalculator.js
class EvapotranspirationCalculator {
  /**
   * Calculate reference evapotranspiration using simplified Penman-Monteith
   * @param {Object} weatherData - Daily weather data
   * @returns {number} Reference ET in mm/day
   */
  calculateReferenceET(weatherData) {
    const {
      temperature_high_c: Tmax,
      temperature_low_c: Tmin,
      humidity_percent: RH,
      wind_speed_kmh: windSpeed,
      latitude,
      date
    } = weatherData;

    // Calculate mean temperature
    const Tmean = (Tmax + Tmin) / 2;

    // Solar radiation estimation (MJ/m²/day) based on latitude and date
    const Ra = this.calculateExtraterrestrialRadiation(latitude, date);
    const Rs = Ra * 0.5;  // Simplified: assume 50% of Ra reaches surface

    // Net radiation (Rn)
    const Rn = 0.77 * Rs - 0.00006 * Math.pow(Tmean + 273, 4);

    // Soil heat flux (G) - negligible for daily calculation
    const G = 0;

    // Saturation vapor pressure (es)
    const es = (this.saturationVaporPressure(Tmax) + this.saturationVaporPressure(Tmin)) / 2;

    // Actual vapor pressure (ea)
    const ea = es * (RH / 100);

    // Vapor pressure deficit
    const vpd = es - ea;

    // Slope of vapor pressure curve
    const delta = 4098 * es / Math.pow(Tmean + 237.3, 2);

    // Psychrometric constant
    const gamma = 0.665;  // kPa/°C at standard pressure

    // Wind speed adjustment (convert km/h to m/s and adjust for 2m height)
    const u2 = (windSpeed * 1000 / 3600) * (4.87 / Math.log(67.8 * 10 - 5.42));

    // FAO Penman-Monteith equation (simplified)
    const numerator = 0.408 * delta * (Rn - G) + gamma * (900 / (Tmean + 273)) * u2 * vpd;
    const denominator = delta + gamma * (1 + 0.34 * u2);

    const ETo = numerator / denominator;

    // Return ET in mm/day (ensure non-negative)
    return Math.max(0, ETo);
  }

  saturationVaporPressure(temp) {
    // Tetens equation
    return 0.6108 * Math.exp((17.27 * temp) / (temp + 237.3));
  }

  calculateExtraterrestrialRadiation(latitude, date) {
    // Simplified Ra calculation
    // Returns MJ/m²/day
    const dayOfYear = this.getDayOfYear(date);
    const latRad = latitude * Math.PI / 180;

    // Solar declination
    const delta = 0.409 * Math.sin(2 * Math.PI * dayOfYear / 365 - 1.39);

    // Sunset hour angle
    const ws = Math.acos(-Math.tan(latRad) * Math.tan(delta));

    // Inverse relative distance Earth-Sun
    const dr = 1 + 0.033 * Math.cos(2 * Math.PI * dayOfYear / 365);

    // Solar constant
    const Gsc = 0.0820;  // MJ/m²/min

    // Extraterrestrial radiation
    const Ra = (24 * 60 / Math.PI) * Gsc * dr *
               (ws * Math.sin(latRad) * Math.sin(delta) +
                Math.cos(latRad) * Math.cos(delta) * Math.sin(ws));

    return Ra;
  }

  getDayOfYear(date) {
    const start = new Date(date.getFullYear(), 0, 0);
    const diff = date - start;
    const oneDay = 1000 * 60 * 60 * 24;
    return Math.floor(diff / oneDay);
  }
}
```

**Weather Alert System**

```javascript
// WeatherAlertService.js
class WeatherAlertService {
  async checkForAlerts(weatherForecast, userId) {
    const alerts = [];

    // Check precipitation threshold (>10mm)
    if (weatherForecast.precipitation_amount_mm > 10) {
      alerts.push({
        id: `alert-${Date.now()}`,
        user_id: userId,
        alert_type: 'heavy_precipitation',
        severity: 'warning',
        title: 'Heavy Rain Alert',
        message: `${weatherForecast.precipitation_amount_mm}mm rainfall expected on ${weatherForecast.forecast_date}`,
        forecast_date: weatherForecast.forecast_date,
        created_at: new Date()
      });
    }

    // Check extreme temperature (>40°C or <0°C)
    if (weatherForecast.temperature_high_c > 40) {
      alerts.push({
        alert_type: 'extreme_heat',
        severity: 'critical',
        title: 'Extreme Heat Warning',
        message: `Temperature expected to reach ${weatherForecast.temperature_high_c}°C`
      });
    }

    // Check high wind speed (>50 km/h)
    if (weatherForecast.wind_gust_kmh > 50) {
      alerts.push({
        alert_type: 'high_wind',
        severity: 'warning',
        title: 'High Wind Alert',
        message: `Wind gusts up to ${weatherForecast.wind_gust_kmh} km/h expected`
      });
    }

    // Store alerts and send notifications
    for (const alert of alerts) {
      await this.saveAlert(alert);
      await this.sendPushNotification(userId, alert);
    }

    return alerts;
  }

}
```

```sql
  user_id TEXT NOT NULL,
  location_id TEXT,
```
