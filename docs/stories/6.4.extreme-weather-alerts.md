**Extreme Weather Monitor Service**

```javascript
// ExtremeWeatherMonitorService.js
class ExtremeWeatherMonitorService {
  constructor() {
    this.checkInterval = 15 * 60 * 1000;  // 15 minutes
    this.thresholds = EXTREME_WEATHER_THRESHOLDS;
  }

  /**
   * Start continuous monitoring for extreme weather
   */
  startMonitoring() {
    // Check immediately on start
    this.checkExtremeWeather();

    // Then check every 15 minutes
    setInterval(() => {
      this.checkExtremeWeather();
    }, this.checkInterval);

    // Also listen for real-time webhooks from weather services
    this.subscribeToWeatherWebhooks();
  }

  async checkExtremeWeather() {
    try {
      // 1. Fetch severe weather alerts from multiple sources
      const pmdAlerts = await this.fetchPMDSevereWeatherAlerts();
      const ibmAlerts = await this.fetchIBMSevereWeatherAlerts();
      const noaaAlerts = await this.fetchNOAAGlobalAlerts();

      // 2. Combine and deduplicate alerts
      const allAlerts = [...pmdAlerts, ...ibmAlerts, ...noaaAlerts];
      const uniqueAlerts = this.deduplicateAlerts(allAlerts);

      // 3. Classify alerts by severity
      const classifiedAlerts = uniqueAlerts.map(alert =>
        this.classifyAlert(alert)
      );

      // 4. Store new alerts in database
      for (const alert of classifiedAlerts) {
        await this.storeAlert(alert);
      }

      // 5. Generate user-specific alerts for affected farmers
      await this.generateUserAlerts(classifiedAlerts);

      console.log(`Processed ${classifiedAlerts.length} extreme weather alerts`);
    } catch (error) {
      console.error('Error checking extreme weather:', error);
    }
  }

  async fetchPMDSevereWeatherAlerts() {
    // Fetch from Pakistan Meteorological Department
    const url = 'https://www.pmd.gov.pk/api/severe-weather-alerts';

    try {
      const response = await axios.get(url);
      return this.transformPMDAlerts(response.data);
    } catch (error) {
      console.error('PMD API error:', error);
      return [];
    }
  }

  async fetchIBMSevereWeatherAlerts() {
    // Fetch severe weather alerts from IBM Weather API
    const url = `${IBM_WEATHER_BASE_URL}/alerts/pakistan`;

    try {
      const response = await axios.get(url, {
        params: { apiKey: process.env.IBM_WEATHER_API_KEY }
      });
      return this.transformIBMAlerts(response.data);
    } catch (error) {
      console.error('IBM Weather API error:', error);
      return [];
    }
  }

  classifyAlert(alert) {
    // Classify severity based on thresholds
    let severity = 'advisory';

    if (alert.alert_type === 'heatwave') {
      const temp = alert.forecast_value;
      if (temp >= this.thresholds.heatwave.warning) {
        severity = 'warning';
      } else if (temp >= this.thresholds.heatwave.watch) {
        severity = 'watch';
      }
    } else if (alert.alert_type === 'heavy_rainfall') {
      const rain = alert.forecast_value;
      if (rain >= this.thresholds.heavy_rainfall.warning) {
        severity = 'warning';
      } else if (rain >= this.thresholds.heavy_rainfall.watch) {
        severity = 'watch';
      }
    } else if (alert.alert_type === 'high_wind') {
      const wind = alert.forecast_value;
      if (wind >= this.thresholds.high_wind.warning) {
        severity = 'warning';
      } else if (wind >= this.thresholds.high_wind.watch) {
        severity = 'watch';
      }
    }

    return { ...alert, severity };
  }

  deduplicateAlerts(alerts) {
    // Remove duplicate alerts from multiple sources
    const unique = [];
    const seen = new Set();

    for (const alert of alerts) {
      const key = `${alert.alert_type}_${alert.region}_${alert.event_start_time}`;
      if (!seen.has(key)) {
        seen.add(key);
        unique.push(alert);
      }
    }

    return unique;
  }

  async storeAlert(alert) {
    // Store alert in database
    await db.insert('extreme_weather_alerts', alert);
  }

  async generateUserAlerts(extremeWeatherAlerts) {
    // For each extreme weather alert, find affected farmers
    for (const alert of extremeWeatherAlerts) {
      const affectedUsers = await this.findAffectedUsers(alert);

      for (const user of affectedUsers) {
        await this.createUserAlert(user, alert);
      }
    }
  }

  async findAffectedUsers(alert) {
    // Find users with fields in affected area
    const affectedArea = alert.affected_area_gps;

    const users = await db.query(`
      SELECT DISTINCT f.farmer_id as user_id
      FROM fields f
      WHERE ST_DWithin(
        f.location_gps::geography,
        ST_MakePoint($1, $2)::geography,
        $3
      )
    `, [
      affectedArea.center.lon,
      affectedArea.center.lat,
      affectedArea.radius_km * 1000  // Convert to meters
    ]);

    return users;
  }

  async createUserAlert(user, extremeWeatherAlert) {
    const impactAssessor = new CropImpactAssessor();
    const actionEngine = new ProtectiveActionEngine();

    // Assess impact on user's fields
    const impactAssessment = await impactAssessor.assessImpact(
      user.user_id,
      extremeWeatherAlert
    );

    // Generate protective actions
    const actions = await actionEngine.generateActions(
      extremeWeatherAlert,
      impactAssessment
    );

    // Create personalized user alert
    const userAlert = {
      id: `user-alert-${Date.now()}-${Math.random()}`,
      user_id: user.user_id,
      extreme_weather_alert_id: extremeWeatherAlert.id,
      alert_type: extremeWeatherAlert.alert_type,
      severity: extremeWeatherAlert.severity,
      title: this.generateAlertTitle(extremeWeatherAlert),
      message: this.generateAlertMessage(extremeWeatherAlert, impactAssessment),
      affected_fields: impactAssessment.affected_fields,
      recommended_actions: actions,
      created_at: new Date()
    };

    await db.insert('user_extreme_weather_alerts', userAlert);

    // Deliver alert via multiple channels
    await this.deliverAlert(userAlert);
  }

  generateAlertTitle(alert) {
    const titles = {
      heatwave: {
        warning: 'Extreme Heatwave Warning',
        watch: 'Heatwave Watch',
        advisory: 'Hot Weather Advisory'
      },
      heavy_rainfall: {
        warning: 'Extreme Rainfall Warning',
        watch: 'Heavy Rainfall Watch',
        advisory: 'Rainfall Advisory'
      },
      hail: {
        warning: 'Severe Hailstorm Warning',
        watch: 'Hailstorm Watch',
        advisory: 'Hail Possible'
      },
      high_wind: {
        warning: 'Severe Storm Warning',
        watch: 'High Wind Watch',
        advisory: 'Windy Conditions Expected'
      },
      frost: {
        warning: 'Hard Freeze Warning',
        watch: 'Frost Watch',
        advisory: 'Frost Advisory'
      }
    };

    return titles[alert.alert_type]?.[alert.severity] || 'Weather Alert';
  }

  generateAlertMessage(alert, impact) {
    const startDate = new Date(alert.event_start_time).toLocaleDateString();
    const leadHours = alert.lead_time_hours;

    let message = `${alert.description} Expected on ${startDate} (in ${leadHours} hours). `;

    if (impact.affected_fields.length > 0) {
      message += `${impact.affected_fields.length} of your fields may be affected. `;
    }

    return message;
  }

  async deliverAlert(userAlert) {
    const deliveryService = new AlertDeliveryService();

    // 1. Push notification (highest priority)
    await deliveryService.sendPushNotification(userAlert);

    // 2. SMS for critical alerts
    if (userAlert.severity === 'warning') {
      await deliveryService.sendSMS(userAlert);
    }

    // 3. In-app notification
    await deliveryService.createInAppNotification(userAlert);
  }

  subscribeToWeatherWebhooks() {
    // Set up webhook listeners for real-time alerts
    // Implementation depends on weather service webhook support
  }
}
```

```javascript
// CropImpactAssessor.js
class CropImpactAssessor {
  async assessImpact(userId, extremeWeatherAlert) {
    // Get user's fields in affected area
    const fields = await this.getAffectedFields(userId, extremeWeatherAlert);

    const affectedFields = [];

    for (const field of fields) {
      // Get active crops in field
      const activeCrops = await db.query(`
        SELECT * FROM crop_seasons
        WHERE field_id = $1 AND status = 'active'
      `, [field.id]);

      for (const crop of activeCrops) {
        const vulnerability = this.assessCropVulnerability(
          crop.crop_type,
          crop.growth_stage,
          extremeWeatherAlert.alert_type
        );

        const damageRisk = this.calculateDamageRisk(
          vulnerability,
          extremeWeatherAlert.severity,
          extremeWeatherAlert.forecast_value
        );

        affectedFields.push({
          field_id: field.id,
          field_name: field.name,
          crop_type: crop.crop_type,
          growth_stage: crop.growth_stage,
          vulnerability,
          estimated_damage_risk: damageRisk
        });
      }
    }

    return { affected_fields: affectedFields };
  }

  async getAffectedFields(userId, alert) {
    const affectedArea = alert.affected_area_gps;

    const fields = await db.query(`
      SELECT * FROM fields
      WHERE farmer_id = $1
        AND ST_DWithin(
          location_gps::geography,
          ST_MakePoint($2, $3)::geography,
          $4
        )
    `, [
      userId,
      affectedArea.center.lon,
      affectedArea.center.lat,
      affectedArea.radius_km * 1000
    ]);

    return fields;
  }

  assessCropVulnerability(cropType, growthStage, weatherType) {
    const vulnerability = CROP_VULNERABILITY[weatherType]?.[cropType];

    if (!vulnerability) return 'low';

    // Check growth stage specific vulnerability
    if (vulnerability[growthStage]) {
      return vulnerability[growthStage];
    }

    // Fall back to general vulnerability
    return vulnerability.all || 'medium';
  }

  calculateDamageRisk(vulnerability, severity, forecastValue) {
    // Calculate percentage damage risk
    const vulnerabilityScores = {
      'very_high': 90,
      'high': 70,
      'medium': 40,
      'low': 15
    };

    const severityMultipliers = {
      'warning': 1.2,
      'watch': 1.0,
      'advisory': 0.7
    };

    const baseRisk = vulnerabilityScores[vulnerability] || 40;
    const multiplier = severityMultipliers[severity] || 1.0;

    const risk = Math.min(100, Math.round(baseRisk * multiplier));

    return risk;
  }
}
```

```javascript
        },
        {
          action: 'Light smudge fires (if safe and legal)',
          priority: 'medium',
          urgency: '12 hours',
          effectiveness: 50
        }
      ]
    };

    return actions[weatherType] || [];
  }

  getCropSpecificActions(cropType, growthStage, weatherType, vulnerability) {
    // Crop-specific protective measures
    const actions = [];

    if (weatherType === 'heatwave' && growthStage === 'flowering') {
      actions.push({
        action: `Apply foliar spray to reduce heat stress in ${cropType}`,
        priority: 'high',
        urgency: '24 hours',
        effectiveness: 65
      });
    }

    if (weatherType === 'heavy_rainfall' && growthStage === 'harvest') {
      actions.push({
        action: `Expedite harvest of ${cropType} to avoid grain damage`,
        priority: 'critical',
        urgency: '6 hours',
        effectiveness: 90
      });
    }

    return actions;
  }

  prioritizeActions(actions, alert) {
    // Remove duplicates
    const unique = this.deduplicateActions(actions);

    // Sort by priority and effectiveness
    const priorityOrder = { critical: 0, high: 1, medium: 2, low: 3 };

    unique.sort((a, b) => {
      if (priorityOrder[a.priority] !== priorityOrder[b.priority]) {
        return priorityOrder[a.priority] - priorityOrder[b.priority];
      }
      return b.effectiveness - a.effectiveness;
    });

    // Return top 5 actions
    return unique.slice(0, 5);
  }

  deduplicateActions(actions) {
    const seen = new Set();
    return actions.filter(action => {
      if (seen.has(action.action)) {
        return false;
      }
      seen.add(action.action);
      return true;
    });
  }
}
```
