# Story 3.9: Confidence and Accuracy Feedback Loop

## Status

## Story
**As a** farmer receiving a diagnosis,
**I want** to confirm if the diagnosis was correct,
**so that** the AI can improve over time.

## Acceptance Criteria
1. AC1: User can mark diagnosis as correct/incorrect/unsure
2. AC2: If incorrect, user can select correct issue from list

4. AC4: Aggregate accuracy metrics visible to development team
5. AC5: High-confidence incorrect predictions flagged for review
6. AC6: User receives acknowledgment when feedback improves model
7. AC7: Opt-in to share images for model training

```sql
-- Prediction feedback table
CREATE TABLE prediction_feedback (
  id TEXT PRIMARY KEY,
  health_check_id TEXT NOT NULL,
  prediction_source TEXT NOT NULL,  -- on_device, cloud
  predicted_disease_id TEXT NOT NULL,
  predicted_confidence REAL NOT NULL,
  model_version TEXT NOT NULL,
  feedback_type TEXT NOT NULL,  -- correct, incorrect, unsure
  actual_disease_id TEXT,  -- If user provides correction
  user_explanation TEXT,
  feedback_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  farmer_id TEXT NOT NULL,
  is_flagged_for_review BOOLEAN DEFAULT FALSE,
  expert_reviewed BOOLEAN DEFAULT FALSE,
  expert_verdict TEXT,  -- confirmed, overridden, disputed
  expert_notes TEXT,
  expert_reviewer_id TEXT,
  expert_review_timestamp TIMESTAMP,
  FOREIGN KEY (health_check_id) REFERENCES health_checks(id),
  FOREIGN KEY (predicted_disease_id) REFERENCES diseases(id),
  FOREIGN KEY (actual_disease_id) REFERENCES diseases(id),
  FOREIGN KEY (farmer_id) REFERENCES farmers(id)
);

CREATE INDEX idx_feedback_health_check ON prediction_feedback(health_check_id);
CREATE INDEX idx_feedback_flagged ON prediction_feedback(is_flagged_for_review) WHERE is_flagged_for_review = TRUE;
CREATE INDEX idx_feedback_type ON prediction_feedback(feedback_type);
CREATE INDEX idx_feedback_timestamp ON prediction_feedback(feedback_timestamp DESC);

-- User consent tracking
CREATE TABLE user_data_consent (
  farmer_id TEXT PRIMARY KEY,
  consent_feedback BOOLEAN DEFAULT FALSE,
  consent_image_sharing BOOLEAN DEFAULT FALSE,
  consent_model_training BOOLEAN DEFAULT FALSE,
  consent_timestamp TIMESTAMP,
  consent_version TEXT,  -- Track which consent version was agreed to
  opt_out_timestamp TIMESTAMP,
  FOREIGN KEY (farmer_id) REFERENCES farmers(id)
);

-- Model accuracy metrics (aggregated)
CREATE TABLE model_accuracy_metrics (
  id TEXT PRIMARY KEY,
  model_version TEXT NOT NULL,
  metric_date DATE NOT NULL,
  total_predictions INTEGER DEFAULT 0,
  total_feedback INTEGER DEFAULT 0,
  correct_count INTEGER DEFAULT 0,
  incorrect_count INTEGER DEFAULT 0,
  accuracy_pct REAL,
  precision_pct REAL,
  recall_pct REAL,
  crop_type TEXT,
  disease_id TEXT,
  confidence_range TEXT,  -- low, medium, high
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_accuracy_metrics_model_date ON model_accuracy_metrics(model_version, metric_date DESC);
```

**Feedback Collection Service**

```javascript
// FeedbackService.js
class FeedbackService {
  async submitFeedback(healthCheckId, feedbackData) {
    const { feedbackType, actualDiseaseId, explanation } = feedbackData;

    // Get original prediction
    const healthCheck = await HealthCheckRepository.getById(healthCheckId);
    const prediction = JSON.parse(healthCheck.on_device_result);

    // Create feedback record
    const feedback = {
      id: `feedback-${Date.now()}-${Math.random()}`,
      health_check_id: healthCheckId,
      prediction_source: 'on_device',
      predicted_disease_id: prediction.topPrediction.diseaseId,
      predicted_confidence: prediction.topPrediction.confidence,
      model_version: prediction.modelVersion,
      feedback_type: feedbackType,
      actual_disease_id: actualDiseaseId,
      user_explanation: explanation,
      farmer_id: healthCheck.farmer_id,
      feedback_timestamp: new Date()
    };

    // Flag high-confidence incorrect predictions
    if (feedbackType === 'incorrect' && prediction.topPrediction.confidence > 0.8) {
      feedback.is_flagged_for_review = true;
    }

    // Save to local database
    await db.insert('prediction_feedback', feedback);

    // Sync to backend when online
    await this.syncFeedbackToBackend(feedback.id);

    // Update accuracy metrics
    await this.updateAccuracyMetrics(feedback);

    // Send acknowledgment
    await this.sendFeedbackAcknowledgment(healthCheck.farmer_id);

    return feedback;
  }

  async syncFeedbackToBackend(feedbackId) {
    const isOnline = await NetworkMonitor.isConnected();
    if (!isOnline) {
      // Queue for later sync
      await SyncService.queueForSync('prediction_feedback', feedbackId);
      return;
    }

    const feedback = await db.query(
      'SELECT * FROM prediction_feedback WHERE id = $1',
      [feedbackId]
    );

    // Check user consent
    const consent = await this.checkUserConsent(feedback[0].farmer_id);
    if (!consent.consent_feedback) {
      console.log('User has not consented to feedback sharing');
      return;
    }

    // Submit to backend
    const payload = {
      ...feedback[0],
      include_image: consent.consent_image_sharing
    };

    await api.post('/api/feedback/predictions', payload);
  }

  async checkUserConsent(farmerId) {
    const consent = await db.query(
      'SELECT * FROM user_data_consent WHERE farmer_id = $1',
      [farmerId]
    );

    if (!consent[0]) {
      // No consent recorded, prompt user
      return { consent_feedback: false, consent_image_sharing: false };
    }

    return consent[0];
  }

  async sendFeedbackAcknowledgment(farmerId) {
    // Send thank-you notification
    await NotificationService.send(farmerId, {
      title: 'Thank You!',
      body: 'Your feedback helps improve Zarai Dost for all farmers.',
      type: 'feedback_acknowledgment'
    });
  }

  async updateAccuracyMetrics(feedback) {
    // Update aggregate accuracy metrics
    const { model_version, feedback_type, crop_type } = feedback;

    await db.query(`
      INSERT INTO model_accuracy_metrics (
        id, model_version, metric_date, total_predictions, total_feedback,
        correct_count, incorrect_count
      )
      VALUES (
        $1, $2, CURRENT_DATE, 1, 1,
        $3, $4
      )
      ON CONFLICT (model_version, metric_date, crop_type) DO UPDATE SET
        total_predictions = model_accuracy_metrics.total_predictions + 1,
        total_feedback = model_accuracy_metrics.total_feedback + 1,
        correct_count = model_accuracy_metrics.correct_count + $3,
        incorrect_count = model_accuracy_metrics.incorrect_count + $4
    `, [
      `metric-${Date.now()}`,
      model_version,
      feedback_type === 'correct' ? 1 : 0,
      feedback_type === 'incorrect' ? 1 : 0
    ]);
  }
}
```

**Accuracy Metrics Calculator**

```javascript
// AccuracyMetricsCalculator.js
class AccuracyMetricsCalculator {
  async calculateModelAccuracy(modelVersion, dateRange = 'month') {
    const startDate = this.getStartDate(dateRange);

    const metrics = await db.query(`
      SELECT
        COUNT(*) as total_feedback,
        SUM(CASE WHEN feedback_type = 'correct' THEN 1 ELSE 0 END) as correct_count,
        SUM(CASE WHEN feedback_type = 'incorrect' THEN 1 ELSE 0 END) as incorrect_count,
        SUM(CASE WHEN feedback_type = 'unsure' THEN 1 ELSE 0 END) as unsure_count,
        AVG(predicted_confidence) as avg_confidence
      FROM prediction_feedback
      WHERE model_version = $1 AND feedback_timestamp >= $2
    `, [modelVersion, startDate]);

    const { total_feedback, correct_count, incorrect_count } = metrics[0];

    const accuracy = total_feedback > 0
      ? (correct_count / total_feedback) * 100
      : null;

    return {
      modelVersion,
      totalFeedback: total_feedback,
      correctCount: correct_count,
      incorrectCount: incorrect_count,
      accuracy: accuracy ? accuracy.toFixed(2) : null,
      avgConfidence: metrics[0].avg_confidence
    };
  }

  async calculateConfusionMatrix(modelVersion) {
    // Get all incorrect feedback with corrections
    const confusions = await db.query(`
      SELECT
        predicted_disease_id,
        actual_disease_id,
        COUNT(*) as confusion_count
      FROM prediction_feedback
      WHERE model_version = $1
        AND feedback_type = 'incorrect'
        AND actual_disease_id IS NOT NULL
      GROUP BY predicted_disease_id, actual_disease_id
      ORDER BY confusion_count DESC
      LIMIT 20
    `, [modelVersion]);

    return confusions;
  }

  async calculateAccuracyByDisease(modelVersion) {
    const diseaseMetrics = await db.query(`
      SELECT
        d.id,
        d.name_en,
        COUNT(*) as total_predictions,
        SUM(CASE WHEN pf.feedback_type = 'correct' THEN 1 ELSE 0 END) as correct_count,
        SUM(CASE WHEN pf.feedback_type = 'correct' THEN 1 ELSE 0 END)::float / COUNT(*) * 100 as accuracy_pct
      FROM prediction_feedback pf
      JOIN diseases d ON pf.predicted_disease_id = d.id
      WHERE pf.model_version = $1
      GROUP BY d.id, d.name_en
      HAVING COUNT(*) >= 10
      ORDER BY accuracy_pct ASC
    `, [modelVersion]);

    return diseaseMetrics;
  }

  async identifyProblematicClasses(modelVersion) {
    // Find disease classes with lowest accuracy
    const problematic = await this.calculateAccuracyByDisease(modelVersion);

    return problematic.filter(d => d.accuracy_pct < 70);
  }

  getStartDate(dateRange) {
    const now = new Date();
    switch (dateRange) {
      case 'week': return new Date(now.setDate(now.getDate() - 7));
      case 'month': return new Date(now.setMonth(now.getMonth() - 1));
      case 'quarter': return new Date(now.setMonth(now.getMonth() - 3));
      case 'year': return new Date(now.setFullYear(now.getFullYear() - 1));
      default: return new Date(now.setMonth(now.getMonth() - 1));
    }
  }
}
```

```javascript
// components/feedback/FeedbackPrompt.js
const FeedbackPrompt = ({ healthCheck, prediction }) => {
  const [feedback, setFeedback] = useState(null);
  const [showCorrection, setShowCorrection] = useState(false);
  const [selectedDisease, setSelectedDisease] = useState(null);

  const handleFeedback = async (type) => {
    setFeedback(type);

    if (type === 'incorrect') {
      setShowCorrection(true);
    } else {
      await submitFeedback(type);
    }
  };

  const submitFeedback = async (type) => {
    await FeedbackService.submitFeedback(healthCheck.id, {
      feedbackType: type,
      actualDiseaseId: selectedDisease,
      explanation: null
    });

    showSuccessMessage('Thank you for your feedback!');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.question}>Was this diagnosis correct?</Text>

      <View style={styles.buttonRow}>
        <Button
          title="Correct"
          icon="check-circle"
          onPress={() => handleFeedback('correct')}
          color="#4CAF50"
        />
        <Button
          title="Incorrect"
          icon="x-circle"
          onPress={() => handleFeedback('incorrect')}
          color="#F44336"
        />
        <Button
          title="Not Sure"
          icon="help-circle"
          onPress={() => handleFeedback('unsure')}
          color="#FF9800"
        />
      </View>

      {showCorrection && (
        <CorrectionSelector
          diseases={availableDiseases}
          onSelect={(disease) => {
            setSelectedDisease(disease.id);
            submitFeedback('incorrect');
          }}
        />
      )}
    </View>
  );
};
```

```javascript
const PrivacyConsentScreen = () => {
  const [consents, setConsents] = useState({
    feedback: false,

    modelTraining: false
```
