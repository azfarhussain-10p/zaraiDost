**Sentinel-2 Satellite Integration**

```python
# SatelliteImageryService.py
import requests
from datetime import datetime, timedelta
import numpy as np
from sentinelhub import SHConfig, SentinelHubRequest, DataCollection, MimeType, BBox, CRS, bbox_to_dimensions

class SatelliteImageryService:
    def __init__(self):
        self.config = SHConfig()
        self.config.sh_client_id = os.getenv('SENTINEL_HUB_CLIENT_ID')
        self.config.sh_client_secret = os.getenv('SENTINEL_HUB_CLIENT_SECRET')

    def get_field_imagery(self, field_lat, field_lng, field_boundary, date=None):
        """
        Retrieve Sentinel-2 imagery for a field
        Args:
            field_lat, field_lng: Field center coordinates
            field_boundary: List of (lat, lng) points defining field polygon
            date: Date for imagery (defaults to most recent)
        """
        # Define bounding box
        bbox = self.calculate_bbox(field_boundary)
        bbox_coords = BBox(bbox=bbox, crs=CRS.WGS84)

        # Set time range (last 30 days if no date specified)
        if date:
            time_interval = (date, date + timedelta(days=1))
        else:
            end_date = datetime.now()
            start_date = end_date - timedelta(days=30)
            time_interval = (start_date.strftime('%Y-%m-%d'), end_date.strftime('%Y-%m-%d'))

        # Request RGB and NIR bands
        evalscript = """
        //VERSION=3
        function setup() {
            return {
                input: [{
                    bands: ["B02", "B03", "B04", "B08", "SCL"],  // Blue, Green, Red, NIR, Scene Classification
                    units: "DN"
                }],
                output: {
                    bands: 5
                }
            };
        }
        function evaluatePixel(sample) {
            return [sample.B04, sample.B03, sample.B02, sample.B08, sample.SCL];
        }
        """

        request = SentinelHubRequest(
            evalscript=evalscript,
            input_data=[
                SentinelHubRequest.input_data(
                    data_collection=DataCollection.SENTINEL2_L2A,
                    time_interval=time_interval,
                    maxcc=0.2  # Max 20% cloud cover
                )
            ],
            responses=[
                SentinelHubRequest.output_response('default', MimeType.TIFF)
            ],
            bbox=bbox_coords,
            size=bbox_to_dimensions(bbox_coords, resolution=10),  # 10m resolution
            config=self.config
        )

        imagery = request.get_data()[0]  # Get most recent cloud-free image

        return {
            'red': imagery[:,:,0],
            'green': imagery[:,:,1],
            'blue': imagery[:,:,2],
            'nir': imagery[:,:,3],
            'cloud_mask': imagery[:,:,4],
            'bbox': bbox,
            'date': self.extract_image_date(request)
        }

    def calculate_bbox(self, field_boundary):
        """Calculate bounding box from field polygon"""
        lats = [point[0] for point in field_boundary]
        lngs = [point[1] for point in field_boundary]

        return [min(lngs), min(lats), max(lngs), max(lats)]  # [min_lng, min_lat, max_lng, max_lat]

    def calculate_ndvi(self, imagery):
        """
        Calculate NDVI from red and NIR bands
        NDVI = (NIR - Red) / (NIR + Red)
        """
        red = imagery['red'].astype(float)
        nir = imagery['nir'].astype(float)

        # Avoid division by zero
        denominator = nir + red
        denominator[denominator == 0] = 0.0001

        ndvi = (nir - red) / denominator

        # Clip to valid range [-1, 1]
        ndvi = np.clip(ndvi, -1, 1)

        return ndvi

    def calculate_ndwi(self, imagery):
        """
        Calculate NDWI (water/moisture index)
        NDWI = (Green - NIR) / (Green + NIR)
        """
        green = imagery['green'].astype(float)
        nir = imagery['nir'].astype(float)

        denominator = green + nir
        denominator[denominator == 0] = 0.0001

        ndwi = (green - nir) / denominator

        return np.clip(ndwi, -1, 1)

    def classify_moisture_zones(self, ndvi, ndwi):
        """
        Classify field into moisture zones using NDVI and NDWI
        Returns: zone_map with values 0=low, 1=medium, 2=high
        """
        # Combine NDVI (vegetation health) and NDWI (moisture)
        # High NDVI + High NDWI = Adequate moisture (2)
        # Medium values = Medium moisture (1)
        # Low values = Low moisture (0)

        zone_map = np.zeros(ndvi.shape, dtype=int)

        # High moisture: NDVI > 0.6 AND NDWI > 0
        zone_map[(ndvi > 0.6) & (ndwi > 0)] = 2

        # Medium moisture: NDVI 0.3-0.6 OR NDWI -0.2-0
        zone_map[((ndvi >= 0.3) & (ndvi <= 0.6)) | ((ndwi >= -0.2) & (ndwi <= 0))] = 1

        # Low moisture: NDVI < 0.3 AND NDWI < -0.2
        zone_map[(ndvi < 0.3) & (ndwi < -0.2)] = 0

        return zone_map

    def calculate_zone_statistics(self, zone_map, field_size_acres):
        """Calculate area and percentage for each zone"""
        total_pixels = zone_map.size
        unique, counts = np.unique(zone_map, return_counts=True)

        zones = {}
        for zone_level, count in zip(unique, counts):
            percentage = (count / total_pixels) * 100
            area_acres = field_size_acres * (percentage / 100)

            zones[zone_level] = {
                'level': ['Low', 'Medium', 'High'][zone_level],
                'pixels': int(count),
                'percentage': round(percentage, 1),
                'area_acres': round(area_acres, 2)
            }

        return zones
```

// ZoneIrrigationService.js
class ZoneIrrigationService {
  async generateZoneRecommendations(fieldId, zoneMap, zoneStats) {
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);

    const recommendations = [];

    for (const [zoneLevel, stats] of Object.entries(zoneStats)) {
      let waterRequirement, priority;

      if (stats.level === 'Low') {
        waterRequirement = 50;  // mm
        priority = 1;  // Irrigate first
      } else if (stats.level === 'Medium') {
        waterRequirement = 30;  // mm
        priority = 2;
      } else {  // High
        waterRequirement = 0;  // No irrigation needed
        priority = 3;
      }

      if (waterRequirement > 0) {
        // Convert to liters for this zone
        const areaM2 = stats.area_acres * 4046.86;
        const waterLiters = (waterRequirement / 1000) * areaM2 * 1000;

        recommendations.push({
          zone_level: stats.level,
          zone_percentage: stats.percentage,
          zone_area_acres: stats.area_acres,
          water_requirement_mm: waterRequirement,
          water_requirement_liters: Math.round(waterLiters),
          priority: priority,
          instructions: this.getZoneInstructions(stats.level, waterRequirement)
        });
      }
    }

    // Sort by priority
    recommendations.sort((a, b) => a.priority - b.priority);

    return recommendations;
  }

  getZoneInstructions(level, waterMm) {
    if (level === 'Low') {
      return `Critical: Irrigate immediately with ${waterMm}mm. Focus on low-moisture areas first.`;
    } else if (level === 'Medium') {
      return `Moderate: Irrigate with ${waterMm}mm within 2-3 days.`;
    } else {
      return `Adequate: No irrigation needed currently.`;
    }
  }
}

CREATE TABLE satellite_imagery_cache (
  id TEXT PRIMARY KEY,
  field_id TEXT NOT NULL,
  imagery_date DATE NOT NULL,
  satellite_source TEXT,            -- 'sentinel-2', 'planet', 'landsat'

  -- Image data (stored in S3)
  image_url TEXT,
  ndvi_raster_url TEXT,
  ndwi_raster_url TEXT,
  zone_map_url TEXT,

  -- Statistics
  avg_ndvi REAL,
  avg_ndwi REAL,
  cloud_cover_percent REAL,

  -- Zone statistics (JSON)
  zone_statistics TEXT,

  resolution_meters REAL,
  bbox TEXT,                        -- JSON: [min_lng, min_lat, max_lng, max_lat]

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (field_id) REFERENCES fields(id),
  UNIQUE(field_id, imagery_date)
);

CREATE INDEX idx_satellite_imagery_field_date ON satellite_imagery_cache(field_id, imagery_date DESC);
