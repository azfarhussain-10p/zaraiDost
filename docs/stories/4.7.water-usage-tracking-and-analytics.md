**Water Usage Logging**

```javascript
// WaterUsageService.js
class WaterUsageService {
  async logWaterUsage(userId, irrigationEventId, actualUsage) {
    const event = await db.queryOne(
      'SELECT * FROM irrigation_schedule WHERE id = $1',
      [irrigationEventId]
    );

    const usageRecord = {
      id: `usage-${Date.now()}`,
      user_id: userId,
      field_id: event.field_id,
      crop_season_id: event.crop_season_id,
      irrigation_event_id: irrigationEventId,

      // Actual usage
      actual_water_liters: actualUsage.liters,
      actual_duration_hours: actualUsage.hours,
      water_source: actualUsage.source,

      // Recommended (from schedule)
      recommended_water_liters: event.water_volume_liters,
      recommended_duration_hours: event.duration_hours,

      // Variance
      variance_liters: actualUsage.liters - event.water_volume_liters,
      variance_percent: ((actualUsage.liters - event.water_volume_liters) / event.water_volume_liters) * 100,

      irrigation_date: actualUsage.date || new Date(),
      notes: actualUsage.notes,
      created_at: new Date()
    };

    await db.insert('water_usage_log', usageRecord);

    // Update irrigation event status
    await db.update('irrigation_schedule', { id: irrigationEventId }, {
      status: 'completed',
      actual_water_liters: actualUsage.liters,
      completion_date: usageRecord.irrigation_date
    });

    return usageRecord;
  }

  convertToLiters(amount, unit, fieldSize, irrigationMethod) {
    if (unit === 'liters') return amount;

    if (unit === 'hours') {
      // Convert hours to liters based on irrigation method flow rate
      const flowRates = {
        drip: 500,      // liters/hour/acre
        sprinkler: 2000,
        flood: 5000,
        furrow: 3000
      };
      const flowRate = flowRates[irrigationMethod] || 2000;
      return amount * flowRate * fieldSize;
    }

    if (unit === 'mm') {
      // Convert water depth to volume
      const fieldAreaM2 = fieldSize * 4046.86;  // Acres to m²
      const volumeM3 = (amount / 1000) * fieldAreaM2;
      return volumeM3 * 1000;  // Convert to liters
    }

    return amount;  // Default: assume liters
  }
}
```

**Water Savings Calculation**

```javascript
// WaterSavingsAnalytics.js
class WaterSavingsAnalytics {
  async calculateSavingsVsBaseline(fieldId, cropSeasonId) {
    const cropSeason = await db.queryOne(
      'SELECT * FROM crop_seasons WHERE id = $1',
      [cropSeasonId]
    );
    const field = await db.queryOne(
      'SELECT * FROM fields WHERE id = $1',
      [fieldId]
    );

    // Get actual usage
    const actualUsage = await db.queryOne(`
      SELECT SUM(actual_water_liters) as total_used
      FROM water_usage_log
      WHERE field_id = $1 AND crop_season_id = $2
    `, [fieldId, cropSeasonId]);

    // Calculate traditional baseline
    const baseline = this.getTraditionalBaseline(
      cropSeason.crop_type,
      field.size_acres,
      cropSeason.planting_date,
      cropSeason.expected_harvest_date
    );

    const savingsLiters = baseline.total_water - (actualUsage.total_used || 0);
    const savingsPercent = (savingsLiters / baseline.total_water) * 100;

    // Calculate cost savings (if tube well)
    const costSavings = this.calculateCostSavings(savingsLiters, field.water_source);

    return {
      actual_usage_liters: actualUsage.total_used || 0,
      baseline_usage_liters: baseline.total_water,
      savings_liters: savingsLiters,
      savings_percent: savingsPercent,
      cost_savings_pkr: costSavings,
      co2_saved_kg: this.calculateCO2Savings(savingsLiters, field.water_source)
    };
  }

  getTraditionalBaseline(cropType, fieldSizeAcres, plantingDate, harvestDate) {
    // Traditional irrigation: fixed schedule every 7-10 days
    const growthPeriodDays = this.getDaysBetween(plantingDate, harvestDate);
    const irrigationFrequency = 8;  // days (traditional schedule)
    const numberOfIrrigations = Math.floor(growthPeriodDays / irrigationFrequency);

    // Traditional water amount per irrigation (mm)
    const traditionalDepth = {
      wheat: 75,
      rice: 100,
      cotton: 75,
      sugarcane: 100,
      corn: 75
    };

    const depthMm = traditionalDepth[cropType] || 75;
    const fieldAreaM2 = fieldSizeAcres * 4046.86;
    const litersPerIrrigation = (depthMm / 1000) * fieldAreaM2 * 1000;
    const totalWater = litersPerIrrigation * numberOfIrrigations;

    return {
      total_water: totalWater,
      num_irrigations: numberOfIrrigations,
      liters_per_irrigation: litersPerIrrigation
    };
  }

  calculateCostSavings(waterSavedLiters, waterSource) {
    if (waterSource !== 'tube_well') return 0;

    // Tube well electricity cost estimation
    const hoursOfPumping = waterSavedLiters / 5000;  // Assume 5000 L/hour pump rate
    const costPerHour = 150;  // PKR
    return hoursOfPumping * costPerHour;
  }

  calculateCO2Savings(waterSavedLiters, waterSource) {
    if (waterSource !== 'tube_well') return 0;

    // CO2 emissions from electricity generation in Pakistan
    const hoursOfPumping = waterSavedLiters / 5000;
    const kWhPerHour = 5;  // Pump power consumption
    const co2PerKWh = 0.5;  // kg CO2 per kWh (Pakistan grid)
    return hoursOfPumping * kWhPerHour * co2PerKWh;
  }

  getDaysBetween(date1, date2) {
    const diff = new Date(date2) - new Date(date1);
    return Math.floor(diff / (1000 * 60 * 60 * 24));
  }
}
```

**Water Efficiency Score**

```javascript
// WaterEfficiencyScorer.js
class WaterEfficiencyScorer {
  async calculateFieldEfficiencyScore(fieldId, cropSeasonId) {
    const scores = {
      adherence: await this.calculateAdherenceScore(fieldId, cropSeasonId),
      productivity: await this.calculateProductivityScore(fieldId, cropSeasonId),
      methodEfficiency: await this.calculateMethodEfficiencyScore(fieldId),
      timingOptimization: await this.calculateTimingScore(fieldId, cropSeasonId)
    };

    // Weighted average
    const totalScore = (
      scores.adherence * 0.3 +
      scores.productivity * 0.4 +
      scores.methodEfficiency * 0.2 +
      scores.timingOptimization * 0.1
    );

    return {
      overall_score: Math.round(totalScore),
      breakdown: scores,
      grade: this.getGrade(totalScore),
      recommendations: this.getRecommendations(scores)
    };
  }

  async calculateAdherenceScore(fieldId, cropSeasonId) {
    // How closely farmer followed recommendations
    const usage = await db.query(`
      SELECT
        ABS(variance_percent) as variance
      FROM water_usage_log
      WHERE field_id = $1 AND crop_season_id = $2
    `, [fieldId, cropSeasonId]);

    const avgVariance = usage.reduce((sum, u) => sum + u.variance, 0) / usage.length;
    const score = Math.max(0, 100 - avgVariance);

    return score;
  }

  async calculateProductivityScore(fieldId, cropSeasonId) {
    // Crop yield per unit water (water productivity)
    const cropSeason = await db.queryOne(
      'SELECT * FROM crop_seasons WHERE id = $1',
      [cropSeasonId]
    );

    if (!cropSeason.yield_kg_per_acre) return 50;  // Default if no yield data

    const usage = await db.queryOne(`
      SELECT SUM(actual_water_liters) as total_water
      FROM water_usage_log
      WHERE field_id = $1 AND crop_season_id = $2
    `, [fieldId, cropSeasonId]);

    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);
    const waterPerAcre = usage.total_water / field.size_acres;
    const waterProductivity = cropSeason.yield_kg_per_acre / (waterPerAcre / 1000);  // kg/m³

    // Compare to benchmark water productivity
    const benchmarks = {
      wheat: 1.0,   // kg/m³
      rice: 0.6,
      cotton: 0.4,
      sugarcane: 8.0,
      corn: 1.5
    };

    const benchmark = benchmarks[cropSeason.crop_type] || 1.0;
    const score = Math.min(100, (waterProductivity / benchmark) * 100);

    return score;
  }

  async calculateMethodEfficiencyScore(fieldId) {
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);

    const methodScores = {
      drip: 90,
      sprinkler: 75,
      furrow: 65,
      flood: 60
    };

    return methodScores[field.irrigation_method] || 60;
  }

  async calculateTimingScore(fieldId, cropSeasonId) {
    // How well-timed irrigations were (based on soil moisture)
    // Simplified: assume 70 if user follows schedule
    return 70;
  }

  getGrade(score) {
    if (score >= 90) return 'A';
    if (score >= 80) return 'B';
    if (score >= 70) return 'C';
    if (score >= 60) return 'D';
    return 'F';
  }

  getRecommendations(scores) {
    const recs = [];

    if (scores.adherence < 70) {
      recs.push('Follow irrigation schedule more closely');
    }
    if (scores.productivity < 60) {
      recs.push('Consider crop varieties with better water efficiency');
    }
    if (scores.methodEfficiency < 75) {
      recs.push('Upgrade to drip or sprinkler irrigation');
    }

    return recs;
  }
}
```

### Data Model Requirements

```sql
CREATE TABLE water_usage_log (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  field_id TEXT NOT NULL,
  crop_season_id TEXT NOT NULL,
  irrigation_event_id TEXT,

  -- Actual usage
  actual_water_liters REAL NOT NULL,
  actual_duration_hours REAL,
  water_source TEXT,                    -- 'canal', 'tube_well', 'mixed'

  -- Recommended (from schedule)
  recommended_water_liters REAL,
  recommended_duration_hours REAL,

  -- Variance
  variance_liters REAL,
  variance_percent REAL,

  irrigation_date DATE NOT NULL,
  notes TEXT,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (field_id) REFERENCES fields(id),
  FOREIGN KEY (crop_season_id) REFERENCES crop_seasons(id)
);

CREATE INDEX idx_water_usage_field_date ON water_usage_log(field_id, irrigation_date DESC);
CREATE INDEX idx_water_usage_crop_season ON water_usage_log(crop_season_id);
```
