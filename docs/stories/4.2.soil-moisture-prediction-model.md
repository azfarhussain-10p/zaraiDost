### Technical Implementation Details

**Soil Water Balance Model**

The soil moisture prediction uses the FAO-56 water balance approach:

```
SMt+1 = SMt + Rain + Irrigation - ETc - Percolation - Runoff

Where:
- SMt = Soil moisture at time t (mm)
- Rain = Rainfall (mm)
- Irrigation = Applied irrigation water (mm)
- ETc = Crop evapotranspiration (mm)
- Percolation = Deep drainage below root zone (mm)
- Runoff = Surface runoff (mm)
```

**Crop Coefficient (Kc) Database**

Crop coefficients vary by growth stage (from FAO-56):

```javascript
// CropCoefficientDatabase.js
const CROP_COEFFICIENTS = {
  wheat: {
    initial: 0.3,      // 0-15 days after planting
    development: 0.7,  // 15-60 days
    mid_season: 1.15,  // 60-90 days
    late_season: 0.4   // 90-120 days (maturity)
  },
  rice: {
    initial: 1.05,     // Flooded conditions
    development: 1.10,
    mid_season: 1.20,
    late_season: 0.95
  },
  cotton: {
    initial: 0.35,
    development: 0.7,
    mid_season: 1.15,
    late_season: 0.7
  },
  sugarcane: {
    initial: 0.4,
    development: 0.8,
    mid_season: 1.25,
    late_season: 0.75
  },
  corn: {
    initial: 0.3,
    development: 0.8,
    mid_season: 1.15,
    late_season: 0.6
  }
};

class CropCoefficientService {
  getKc(cropType, growthStage) {
    const coefficients = CROP_COEFFICIENTS[cropType];
    if (!coefficients) return 1.0;  // Default Kc

    const stageMap = {
      'germination': 'initial',
      'vegetative': 'development',
      'flowering': 'mid_season',
      'maturity': 'late_season',
      'harvest': 'late_season'
    };

    const stage = stageMap[growthStage] || 'mid_season';
    return coefficients[stage];
  }

  // Interpolate Kc based on days since planting
  getKcByDays(cropType, daysSincePlanting) {
    const coefficients = CROP_COEFFICIENTS[cropType];
    if (!coefficients) return 1.0;

    // Simplified linear interpolation between stages
    if (daysSincePlanting < 15) {
      return coefficients.initial;
    } else if (daysSincePlanting < 60) {
      const progress = (daysSincePlanting - 15) / 45;
      return coefficients.initial + progress * (coefficients.development - coefficients.initial);
    } else if (daysSincePlanting < 90) {
      const progress = (daysSincePlanting - 60) / 30;
      return coefficients.development + progress * (coefficients.mid_season - coefficients.development);
    } else if (daysSincePlanting < 120) {
      const progress = (daysSincePlanting - 90) / 30;
      return coefficients.mid_season + progress * (coefficients.late_season - coefficients.mid_season);
    } else {
      return coefficients.late_season;
    }
  }
}
```

**Soil Water Holding Capacity**

Different soil types hold different amounts of water:

```javascript
// SoilPropertiesDatabase.js
const SOIL_PROPERTIES = {
  sandy: {
    field_capacity: 0.10,        // 10% volumetric water content
    wilting_point: 0.05,         // 5%
    saturation: 0.40,            // 40%
    infiltration_rate_mm_hr: 30, // Fast infiltration
    root_zone_depth_mm: 600      // Typical root depth for crops
  },
  loamy: {
    field_capacity: 0.25,        // 25%
    wilting_point: 0.12,         // 12%
    saturation: 0.45,            // 45%
    infiltration_rate_mm_hr: 15,
    root_zone_depth_mm: 900
  },
  clay: {
    field_capacity: 0.35,        // 35%
    wilting_point: 0.20,         // 20%
    saturation: 0.50,            // 50%
    infiltration_rate_mm_hr: 5,  // Slow infiltration
    root_zone_depth_mm: 900
  },
  silty: {
    field_capacity: 0.30,
    wilting_point: 0.15,
    saturation: 0.48,
    infiltration_rate_mm_hr: 10,
    root_zone_depth_mm: 900
  }
};

class SoilPropertiesService {
  getProperties(soilType) {
    return SOIL_PROPERTIES[soilType] || SOIL_PROPERTIES.loamy;
  }

  // Total available water (TAW) in root zone
  getTotalAvailableWater(soilType) {
    const props = this.getProperties(soilType);
    const TAW = (props.field_capacity - props.wilting_point) * props.root_zone_depth_mm;
    return TAW;  // mm
  }

  // Readily available water (RAW) - water available before stress
  getReadilyAvailableWater(soilType, cropType) {
    const TAW = this.getTotalAvailableWater(soilType);

    // Depletion fraction (p) varies by crop
    const depletionFraction = {
      wheat: 0.55,
      rice: 0.20,   // Rice is sensitive to water stress
      cotton: 0.65,
      sugarcane: 0.65,
      corn: 0.55
    };

    const p = depletionFraction[cropType] || 0.50;
    const RAW = p * TAW;
    return RAW;  // mm
  }
}
```

**Soil Moisture Prediction Model**

```javascript
// SoilMoisturePredictionModel.js
class SoilMoisturePredictionModel {
  constructor(field, cropSeason, soilType) {
    this.field = field;
    this.cropSeason = cropSeason;
    this.soilProps = new SoilPropertiesService().getProperties(soilType);
    this.kcService = new CropCoefficientService();
  }

  /**
   * Predict soil moisture for the next 7 days
   * @param {Object} currentMoisture - Starting moisture level (mm)
   * @param {Array} weatherForecast - 7-day weather forecast
   * @param {Array} plannedIrrigation - Planned irrigation events
   * @returns {Array} Daily moisture predictions
   */
  async predict7DayMoisture(currentMoisture, weatherForecast, plannedIrrigation = []) {
    let moisture = currentMoisture;  // mm of water in root zone
    const predictions = [];

    for (let day = 0; day < 7; day++) {
      const weather = weatherForecast[day];
      const irrigation = this.getIrrigationForDay(plannedIrrigation, day);

      // Calculate crop ET
      const daysSincePlanting = this.getDaysSincePlanting() + day;
      const Kc = this.kcService.getKcByDays(this.cropSeason.crop_type, daysSincePlanting);
      const ETc = Kc * weather.reference_et_mm;

      // Water inputs
      const rainfall = weather.rainfall_mm || 0;
      const effectiveRainfall = this.calculateEffectiveRainfall(rainfall);

      // Water balance
      moisture = moisture + effectiveRainfall + irrigation - ETc;

      // Calculate percolation (drainage below field capacity)
      const fieldCapacity = this.soilProps.field_capacity * this.soilProps.root_zone_depth_mm;
      let percolation = 0;
      if (moisture > fieldCapacity) {
        percolation = moisture - fieldCapacity;
        moisture = fieldCapacity;
      }

      // Ensure moisture doesn't go below zero
      moisture = Math.max(0, moisture);

      // Convert to percentage
      const moisturePercent = (moisture / fieldCapacity) * 100;

      predictions.push({
        date: this.addDays(new Date(), day),
        moisture_mm: moisture,
        moisture_percent: moisturePercent,
        etc_mm: ETc,
        rainfall_mm: effectiveRainfall,
        irrigation_mm: irrigation,
        percolation_mm: percolation,
        kc: Kc
      });
    }

    return predictions;
  }

  calculateEffectiveRainfall(rainfall) {
    // Not all rainfall infiltrates; some runs off
    // USDA SCS method: effective rainfall for irrigation
    if (rainfall <= 0) return 0;
    if (rainfall < 50) {
      return rainfall * (125 - 0.2 * rainfall) / 125;
    } else {
      return 125 / 2.5;  // ~50mm max effective rainfall
    }
  }

  getIrrigationForDay(plannedIrrigation, dayOffset) {
    const targetDate = this.addDays(new Date(), dayOffset);
    const event = plannedIrrigation.find(irr =>
      this.isSameDay(new Date(irr.date), targetDate)
    );
    return event ? event.amount_mm : 0;
  }

  getDaysSincePlanting() {
    const plantingDate = new Date(this.cropSeason.planting_date);
    const now = new Date();
    return Math.floor((now - plantingDate) / (1000 * 60 * 60 * 24));
  }

  addDays(date, days) {
    const result = new Date(date);
    result.setDate(result.getDate() + days);
    return result;
  }

  isSameDay(date1, date2) {
    return date1.toDateString() === date2.toDateString();
  }

  /**
   * Calculate current moisture level from historical data
   */
  async calculateCurrentMoisture() {
    // Get last known moisture state
    const lastPrediction = await db.queryOne(`
      SELECT moisture_mm FROM soil_moisture_predictions
      WHERE field_id = $1 AND crop_season_id = $2
      ORDER BY prediction_date DESC
      LIMIT 1
    `, [this.field.id, this.cropSeason.id]);

    if (lastPrediction) {
      return lastPrediction.moisture_mm;
    }

    // No history: assume field capacity
    return this.soilProps.field_capacity * this.soilProps.root_zone_depth_mm;
  }

  /**
   * Update prediction with actual moisture observation
   * Uses Kalman filter for combining prediction and observation
   */
  async updateWithObservation(observedMoisturePercent, observationDate) {
    const fieldCapacity = this.soilProps.field_capacity * this.soilProps.root_zone_depth_mm;
    const observedMoisture = (observedMoisturePercent / 100) * fieldCapacity;

    // Get prediction for same date
    const prediction = await db.queryOne(`
      SELECT moisture_mm FROM soil_moisture_predictions
      WHERE field_id = $1 AND crop_season_id = $2 AND prediction_date = $3
    `, [this.field.id, this.cropSeason.id, observationDate]);

    if (!prediction) {
      // No prediction: use observation directly
      return observedMoisture;
    }

    // Simple Kalman filter: weighted average
    const predictionUncertainty = 0.15;  // 15% uncertainty in prediction
    const observationUncertainty = 0.10; // 10% uncertainty in observation

    const kalmanGain = predictionUncertainty / (predictionUncertainty + observationUncertainty);
    const updatedMoisture = prediction.moisture_mm + kalmanGain * (observedMoisture - prediction.moisture_mm);

    // Store observation
    await db.insert('soil_moisture_observations', {
      id: `obs-${Date.now()}`,
      field_id: this.field.id,
      crop_season_id: this.cropSeason.id,
      observation_date: observationDate,
      observed_moisture_percent: observedMoisturePercent,
      observed_moisture_mm: observedMoisture,
      predicted_moisture_mm: prediction.moisture_mm,
      updated_moisture_mm: updatedMoisture
    });

    return updatedMoisture;
  }

  /**
   * Calculate prediction accuracy
   */
  async calculateAccuracy() {
    const observations = await db.query(`
      SELECT
        observed_moisture_mm,
        predicted_moisture_mm
      FROM soil_moisture_observations
      WHERE field_id = $1 AND crop_season_id = $2
      ORDER BY observation_date DESC
      LIMIT 30
    `, [this.field.id, this.cropSeason.id]);

    if (observations.length === 0) return null;

    // Calculate RMSE (Root Mean Square Error)
    let sumSquaredError = 0;
    let sumAbsError = 0;

    observations.forEach(obs => {
      const error = obs.observed_moisture_mm - obs.predicted_moisture_mm;
      sumSquaredError += error * error;
      sumAbsError += Math.abs(error);
    });

    const rmse = Math.sqrt(sumSquaredError / observations.length);
    const mae = sumAbsError / observations.length;

    // Convert to percentage accuracy
    const fieldCapacity = this.soilProps.field_capacity * this.soilProps.root_zone_depth_mm;
    const accuracy = 100 - (rmse / fieldCapacity * 100);

    return {
      rmse,
      mae,
      accuracy_percent: Math.max(0, accuracy),
      sample_size: observations.length
    };
  }
}
```

**Moisture Threshold Alerts**

```javascript
// MoistureThresholdService.js
class MoistureThresholdService {
  /**
   * Get critical moisture threshold for crop
   */
  getCriticalThreshold(cropType, soilType) {
    const soilProps = new SoilPropertiesService().getProperties(soilType);
    const TAW = (soilProps.field_capacity - soilProps.wilting_point) * soilProps.root_zone_depth_mm;

    // Crop-specific depletion fractions
    const depletionFractions = {
      wheat: 0.55,
      rice: 0.20,   // Rice needs frequent irrigation
      cotton: 0.65,
      sugarcane: 0.65,
      corn: 0.55
    };

    const p = depletionFractions[cropType] || 0.50;
    const RAW = p * TAW;

    // Critical threshold: when RAW is depleted
    const criticalMoisture = soilProps.wilting_point * soilProps.root_zone_depth_mm + RAW;
    const criticalPercent = (criticalMoisture / (soilProps.field_capacity * soilProps.root_zone_depth_mm)) * 100;

    return {
      critical_moisture_mm: criticalMoisture,
      critical_percent: criticalPercent,
      raw_mm: RAW,
      taw_mm: TAW
    };
  }

  /**
   * Check if moisture forecast triggers alert
   */
  async checkMoistureAlerts(fieldId, cropSeasonId, moistureForecast) {
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);
    const cropSeason = await db.queryOne('SELECT * FROM crop_seasons WHERE id = $1', [cropSeasonId]);

    const threshold = this.getCriticalThreshold(cropSeason.crop_type, field.soil_type);
    const alerts = [];

    moistureForecast.forEach(prediction => {
      if (prediction.moisture_percent < threshold.critical_percent) {
        alerts.push({
          id: `moisture-alert-${Date.now()}`,
          field_id: fieldId,
          crop_season_id: cropSeasonId,
          alert_type: 'critical_moisture',
          alert_date: prediction.date,
          moisture_percent: prediction.moisture_percent,
          threshold_percent: threshold.critical_percent,
          message: `Critical moisture level: ${prediction.moisture_percent.toFixed(1)}% (threshold: ${threshold.critical_percent.toFixed(1)}%)`,
          created_at: new Date()
        });
      }
    });

    // Save alerts
    for (const alert of alerts) {
      await db.insert('moisture_alerts', alert);
      // Send push notification
      await this.sendMoistureAlert(alert);
    }

    return alerts;

CREATE TABLE soil_moisture_predictions (
  id TEXT PRIMARY KEY,
  field_id TEXT NOT NULL,
  crop_season_id TEXT NOT NULL,
  prediction_date DATE NOT NULL,

  -- Moisture levels
  moisture_mm REAL NOT NULL,             -- Water in root zone (mm)
  moisture_percent REAL NOT NULL,        -- % of field capacity

  -- Water balance components
  etc_mm REAL,                           -- Crop ET (mm)
  rainfall_mm REAL DEFAULT 0,
  irrigation_mm REAL DEFAULT 0,
  percolation_mm REAL DEFAULT 0,

  -- Crop coefficient
  kc REAL,

  -- Prediction metadata
  confidence_level REAL,                 -- 0-1 scale
  is_forecast BOOLEAN DEFAULT 1,         -- TRUE for future predictions

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (field_id) REFERENCES fields(id),
  FOREIGN KEY (crop_season_id) REFERENCES crop_seasons(id),
  UNIQUE(field_id, crop_season_id, prediction_date)
);

CREATE INDEX idx_moisture_predictions_field_date ON soil_moisture_predictions(field_id, prediction_date DESC);

  crop_season_id TEXT NOT NULL,
  alert_type TEXT NOT NULL,              -- 'critical_moisture', 'irrigation_needed'
  alert_date DATE NOT NULL,

  moisture_percent REAL,
  threshold_percent REAL,
  days_until_critical INTEGER,

  message TEXT,
  is_acknowledged BOOLEAN DEFAULT 0,
  acknowledged_at DATETIME,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (field_id) REFERENCES fields(id),
  FOREIGN KEY (crop_season_id) REFERENCES crop_seasons(id)
);

CREATE INDEX idx_moisture_alerts_field ON moisture_alerts(field_id, created_at DESC);
CREATE INDEX idx_moisture_alerts_active ON moisture_alerts(alert_date) WHERE is_acknowledged = 0;
```

### Security Considerations

[Source: architecture/architectural-principles-and-best-practices.md]
- **Data Privacy**: Moisture predictions linked to user's fields (not shared)
- **GDPR Compliance**: User can delete moisture history and observations
- **Model Security**: Prediction algorithms run locally on device (no external ML API)

### Performance Requirements

[Source: architecture/non-functional-specifications.md]
- **Prediction Speed**: 7-day moisture forecast calculation <2s
- **UI Responsiveness**: Moisture gauge updates instantly
- **Model Accuracy**: >75% prediction accuracy (AC5 requirement)

**Optimization Strategies**:
- Cache crop coefficients and soil properties in memory
- Pre-calculate field capacity and wilting point values
- Batch process moisture predictions for all fields overnight
- Store daily predictions to avoid recalculation
- Use indexed queries for fast moisture history retrieval
