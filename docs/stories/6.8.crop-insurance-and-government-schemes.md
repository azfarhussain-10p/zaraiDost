**Insurance Eligibility Checker**

```javascript
// InsuranceEligibilityChecker.js
class InsuranceEligibilityChecker {
  async checkEligibility(userId, fieldId, cropType) {
    // Get user and field data
    const user = await db.query('SELECT * FROM farmers WHERE id = $1', [userId]);
    const field = await db.query('SELECT * FROM fields WHERE id = $1', [fieldId]);

    if (!field[0]) {
      throw new Error('Field not found');
    }

    const userData = user[0];
    const fieldData = field[0];

    // Get all insurance schemes
    const allSchemes = await db.query('SELECT * FROM insurance_schemes WHERE active = TRUE');

    const eligibleSchemes = [];
    const ineligibleSchemes = [];

    for (const scheme of allSchemes) {
      const eligibility = this.assessEligibility(userData, fieldData, cropType, scheme);

      if (eligibility.eligible) {
        eligibleSchemes.push({
          ...scheme,
          eligibility_details: eligibility
        });
      } else {
        ineligibleSchemes.push({
          ...scheme,
          eligibility_details: eligibility
        });
      }
    }

    return {
      eligible_schemes: eligibleSchemes,
      ineligible_schemes: ineligibleSchemes,
      total_eligible: eligibleSchemes.length
    };
  }

  assessEligibility(userData, fieldData, cropType, scheme) {
    const reasons = [];
    let eligible = true;

    // Check crop coverage
    if (!scheme.crops_covered.includes(cropType)) {
      eligible = false;
      reasons.push(`Crop type "${cropType}" not covered by this scheme`);
    }

    // Check farm size
    if (scheme.eligibility.min_farm_size_acres && fieldData.size_acres < scheme.eligibility.min_farm_size_acres) {
      eligible = false;
      reasons.push(`Farm size ${fieldData.size_acres} acres is below minimum ${scheme.eligibility.min_farm_size_acres} acres`);
    }

    if (scheme.eligibility.max_farm_size_acres && fieldData.size_acres > scheme.eligibility.max_farm_size_acres) {
      eligible = false;
      reasons.push(`Farm size ${fieldData.size_acres} acres exceeds maximum ${scheme.eligibility.max_farm_size_acres} acres`);
    }

    // Check region
    if (scheme.eligibility.regions && !scheme.eligibility.regions.includes(fieldData.province)) {
      eligible = false;
      reasons.push(`Scheme not available in ${fieldData.province}`);
    }

    // Check farmer registration (optional check)
    if (scheme.eligibility.farmer_registration_required && !userData.is_registered) {
      eligible = false;
      reasons.push('Farmer registration with agriculture department required');
    }

    // Check land ownership (if required)
    if (scheme.eligibility.land_ownership_required && userData.land_ownership_status !== 'owner') {
      eligible = false;
      reasons.push('Land ownership required (tenants not eligible)');
    }

    return {
      eligible,
      reasons: eligible ? ['All eligibility criteria met'] : reasons
    };
  }
}
```

**Insurance Premium Calculator**

```javascript
// InsurancePremiumCalculator.js
class InsurancePremiumCalculator {
  async calculatePremium(fieldId, cropType, schemeId, coverageLevel = 100) {
    // Get field data
    const field = await db.query('SELECT * FROM fields WHERE id = $1', [fieldId]);
    const fieldData = field[0];

    // Get insurance scheme
    const scheme = await db.query('SELECT * FROM insurance_schemes WHERE id = $1', [schemeId]);
    const schemeData = scheme[0];

    // Calculate sum insured (crop value at expected harvest)
    const sumInsured = await this.calculateSumInsured(fieldData, cropType);

    // Apply coverage level adjustment
    const adjustedSumInsured = sumInsured * (coverageLevel / 100);

    // Calculate base premium
    const basePremiumRate = schemeData.premium_structure.base_rate_percent / 100;
    const basePremium = adjustedSumInsured * basePremiumRate;

    // Apply risk zone multiplier
    const riskZone = await this.assessRiskZone(fieldData);
    const riskMultiplier = schemeData.premium_structure.risk_zone_multipliers[riskZone] || 1.0;
    const riskAdjustedPremium = basePremium * riskMultiplier;

    // Apply government subsidy
    const subsidyPercent = schemeData.premium_structure.government_subsidy_percent / 100;
    const subsidyAmount = riskAdjustedPremium * subsidyPercent;
    const netPremium = riskAdjustedPremium - subsidyAmount;

    // Calculate per-acre premium
    const premiumPerAcre = netPremium / fieldData.size_acres;

    return {
      field_size_acres: fieldData.size_acres,
      crop_type: cropType,
      coverage_level_percent: coverageLevel,
      sum_insured: adjustedSumInsured,

      premium_breakdown: {
        base_premium: basePremium,
        risk_zone: riskZone,
        risk_adjusted_premium: riskAdjustedPremium,
        government_subsidy: subsidyAmount,
        net_farmer_premium: netPremium
      },

      premium_per_acre: premiumPerAcre,
      total_premium: netPremium,

      payment_options: this.getPaymentOptions(netPremium),

      recommendation: this.generateRecommendation(coverageLevel, netPremium, adjustedSumInsured)
    };
  }

  async calculateSumInsured(fieldData, cropType) {
    // Expected yield per acre
    const expectedYield = await this.getExpectedYield(cropType, fieldData.district);

    // Market price
    const marketPrice = await this.getMarketPrice(cropType);

    // Calculate crop value at harvest
    const totalYield = expectedYield * fieldData.size_acres;
    const cropValue = totalYield * marketPrice;

    return cropValue;
  }

  async assessRiskZone(fieldData) {
    // Check flood risk from Story 6.5
    const floodRisk = await db.query(`
      SELECT risk_level FROM flood_risk_assessments
      WHERE field_id = $1
      ORDER BY assessed_at DESC
      LIMIT 1
    `, [fieldData.id]);

    // Check drought risk from Story 6.6
    const droughtRisk = await db.query(`
      SELECT risk_level FROM drought_risk_index
      WHERE district = $1
      ORDER BY calculated_at DESC
      LIMIT 1
    `, [fieldData.district]);

    // Combine risks
    const floodRiskLevel = floodRisk[0]?.risk_level || 'low';
    const droughtRiskLevel = droughtRisk[0]?.risk_level || 'low';

    // Determine overall risk zone
    if (floodRiskLevel === 'extreme' || droughtRiskLevel === 'extreme') {
      return 'extreme';
    } else if (floodRiskLevel === 'high' || droughtRiskLevel === 'high') {
      return 'high';
    } else if (floodRiskLevel === 'medium' || droughtRiskLevel === 'medium') {
      return 'medium';
    }
    return 'low';
  }

  async getExpectedYield(cropType, district) {
    // District-specific average yields (kg per acre)
    const avgYields = {
      wheat: 1200,
      rice: 1800,
      cotton: 800,
      sugarcane: 30000,
      corn: 2000
    };

    return avgYields[cropType] || 1000;
  }

  async getMarketPrice(cropType) {
    // Get current market price from Story 2.3
    const prices = await db.query(`
      SELECT price_per_kg FROM mandi_prices
      WHERE crop_type = $1
      ORDER BY date DESC
      LIMIT 1
    `, [cropType]);

    return prices[0]?.price_per_kg || 50;
  }

  getPaymentOptions(totalPremium) {
    return [
      {
        option: 'lump_sum',
        description: 'Pay full premium at the time of application',
        amount: totalPremium,
        discount_percent: 0
      },
      {
        option: 'two_installments',
        description: 'Pay in 2 equal installments (at sowing and mid-season)',
        installment_amount: totalPremium / 2,
        discount_percent: 0
      }
    ];
  }

  generateRecommendation(coverageLevel, netPremium, sumInsured) {
    const premiumPercentOfSumInsured = (netPremium / sumInsured) * 100;

    if (coverageLevel < 75) {
      return `Consider increasing coverage to 75-100% for better protection. Current coverage may not fully compensate for major losses.`;
    } else if (premiumPercentOfSumInsured < 2) {
      return `Excellent value! Premium is only ${premiumPercentOfSumInsured.toFixed(2)}% of sum insured. Highly recommended.`;
    } else if (premiumPercentOfSumInsured < 3) {
      return `Good value for comprehensive coverage. Premium is reasonable at ${premiumPercentOfSumInsured.toFixed(2)}% of sum insured.`;
    }
    return `Premium is ${premiumPercentOfSumInsured.toFixed(2)}% of sum insured. Consider if risk warrants this cost.`;
  }
}
```

**Deadline Alert System**

```javascript
// DeadlineAlertService.js
class DeadlineAlertService {
  async scheduleInsuranceDeadlineAlerts() {
    // Run daily to check upcoming deadlines
    const today = new Date();

    // Get all active insurance deadlines
    const deadlines = await db.query(`
      SELECT * FROM insurance_deadlines
      WHERE deadline_date >= $1
        AND deadline_date <= $2
        AND alert_sent = FALSE
    `, [today, this.addDays(today, 30)]);

    for (const deadline of deadlines) {
      const daysUntilDeadline = this.getDaysUntil(deadline.deadline_date);

      // Send alert at 30, 15, 7, and 1 day intervals
      if ([30, 15, 7, 1].includes(daysUntilDeadline)) {
        await this.sendDeadlineAlert(deadline, daysUntilDeadline);
      }
    }
  }

  async sendDeadlineAlert(deadline, daysRemaining) {
    // Get affected users (farmers with matching crops in deadline's region)
    const users = await this.getAffectedUsers(deadline);

    for (const user of users) {
      const message = this.generateDeadlineMessage(deadline, daysRemaining);

      // Send push notification
      await pushNotificationService.send(user.id, {
        title: `Insurance Deadline: ${daysRemaining} Days Left`,
        body: message,
        data: { deadline_id: deadline.id, type: 'insurance_deadline' }
      });

      // Send SMS for 7-day and 1-day reminders
      if (daysRemaining <= 7) {
        await smsService.send(user.id, { message });
      }
    }

    // Mark alert as sent
    await db.update('insurance_deadlines', { id: deadline.id }, {
      alert_sent: true,
      alert_sent_at: new Date()
    });
  }

  generateDeadlineMessage(deadline, daysRemaining) {
    if (deadline.deadline_type === 'application') {
      return `Reminder: ${daysRemaining} days left to apply for ${deadline.scheme_name} crop insurance. ` +
             `Apply before ${this.formatDate(deadline.deadline_date)}.`;
    } else if (deadline.deadline_type === 'claim') {
      return `Important: ${daysRemaining} days left to file insurance claim for ${deadline.related_event}. ` +
             `File before ${this.formatDate(deadline.deadline_date)}.`;
    }
    return `Deadline reminder: ${deadline.description}. ${daysRemaining} days remaining.`;
  }

  getDaysUntil(date) {
    const now = new Date();
    const target = new Date(date);
    return Math.ceil((target - now) / (1000 * 60 * 60 * 24));
  }

  addDays(date, days) {
    const result = new Date(date);
    result.setDate(result.getDate() + days);
    return result;
  }

  formatDate(date) {
    return new Date(date).toLocaleDateString('en-PK', {
      year: 'numeric',
      month: 'long',
      day: 'numeric'
    });
  }

  async getAffectedUsers(deadline) {
    // Find users with active crop seasons matching the deadline's criteria
    return await db.query(`
      SELECT DISTINCT f.id, f.phone_number
      FROM farmers f
      JOIN fields fld ON f.id = fld.farmer_id
      JOIN crop_seasons cs ON fld.id = cs.field_id
      WHERE cs.crop_type = ANY($1)
        AND cs.status = 'active'
        AND fld.province = $2
    `, [deadline.applicable_crops, deadline.region]);
  }
}
```

```sql
  alert_intervals TEXT,               -- JSON: [30, 15, 7, 1] days
  alert_sent BOOLEAN DEFAULT FALSE,
  alert_sent_at DATETIME,

  related_event TEXT,                 -- For claim deadlines

  active BOOLEAN DEFAULT TRUE,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (scheme_id) REFERENCES insurance_schemes(id)
);

CREATE INDEX idx_insurance_deadlines_date ON insurance_deadlines(deadline_date, active);
CREATE INDEX idx_insurance_deadlines_alert ON insurance_deadlines(alert_sent, deadline_date) WHERE alert_sent = FALSE;

-- Farmer insurance policies (optional tracking)
CREATE TABLE farmer_insurance_policies (
  id TEXT PRIMARY KEY,
  farmer_id TEXT NOT NULL,
  field_id TEXT NOT NULL,
  crop_season_id TEXT,
  scheme_id TEXT NOT NULL,

  policy_number TEXT UNIQUE,
  application_date DATE NOT NULL,
  approval_date DATE,
  policy_start_date DATE,
  policy_end_date DATE,

  sum_insured INTEGER,
  premium_amount INTEGER,
  subsidy_amount INTEGER,
  net_premium_paid INTEGER,
  coverage_level_percent INTEGER,

  status TEXT DEFAULT 'pending',      -- 'pending', 'approved', 'active', 'expired', 'claimed'

  claim_filed BOOLEAN DEFAULT FALSE,
  claim_id TEXT,

  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (farmer_id) REFERENCES farmers(id),
  FOREIGN KEY (field_id) REFERENCES fields(id),
  FOREIGN KEY (scheme_id) REFERENCES insurance_schemes(id)
);

CREATE INDEX idx_farmer_policies ON farmer_insurance_policies(farmer_id, status);
CREATE INDEX idx_farmer_policies_field ON farmer_insurance_policies(field_id, policy_end_date);
```

### Security Considerations

[Source: architecture/architectural-principles-and-best-practices.md]
- **Data Privacy**: Insurance policy data is sensitive; enforce strict user isolation
- **Encryption**: Encrypt premium calculations and policy information in transit/rest
- **GDPR Compliance**: Allow users to delete their insurance inquiries and calculations
- **No Financial Transactions**: App provides information only; no payment processing

### Performance Requirements

[Source: architecture/non-functional-specifications.md]
- **Eligibility Check**: <2s for eligibility assessment
- **Premium Calculation**: <2s for premium computation
- **Scheme Search**: <1s for insurance scheme retrieval
- **Cache Access**: <500ms for cached scheme data

**Optimization Strategies**:
- Cache insurance schemes locally (updated monthly)
- Pre-calculate premiums for common coverage levels
- Offline premium calculator using cached data
- Background job for deadline alert generation (runs daily)
- Redis caching for scheme data (30-day TTL)

This story provides comprehensive crop insurance information to financially protect farmers against climate risks. By cataloging government and private insurance schemes, assessing eligibility, and calculating subsidized premiums, the system empowers farmers to make informed insurance decisions. Integration with flood/drought risk assessments ensures accurate premium calculations. Deadline alerts prevent missed application opportunities, while government compensation scheme information ensures disaster-affected farmers access available relief.
