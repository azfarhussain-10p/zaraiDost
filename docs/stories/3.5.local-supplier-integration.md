**Supplier Database Schema**

```sql
-- Suppliers table with geospatial data
CREATE EXTENSION IF NOT EXISTS postgis;

CREATE TABLE suppliers (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  type TEXT,  -- shop, wholesaler, cooperative, online
  phone TEXT,
  whatsapp_number TEXT,
  email TEXT,
  address TEXT NOT NULL,
  city TEXT,
  province TEXT,
  location GEOGRAPHY(POINT, 4326),  -- PostGIS geography type
  business_hours TEXT,
  is_verified BOOLEAN DEFAULT FALSE,
  verified_at TIMESTAMP,
  rating REAL,  -- 0.0 to 5.0
  total_ratings INTEGER DEFAULT 0,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Spatial index for fast location queries
CREATE INDEX idx_suppliers_location ON suppliers USING GIST(location);
CREATE INDEX idx_suppliers_city ON suppliers(city);

-- Supplier products with availability
CREATE TABLE supplier_products (
  supplier_id TEXT NOT NULL,
  product_id TEXT NOT NULL,
  availability_status TEXT DEFAULT 'unknown',  -- in_stock, low_stock, out_of_stock, unknown
  last_updated TIMESTAMP,
  updated_by TEXT,  -- farmer_id who updated or 'admin'
  estimated_price_pkr INTEGER,
  PRIMARY KEY (supplier_id, product_id),
  FOREIGN KEY (supplier_id) REFERENCES suppliers(id),
  FOREIGN KEY (product_id) REFERENCES treatment_products(id)
);

CREATE INDEX idx_supplier_products_availability ON supplier_products(availability_status);

-- Farmer favorites
CREATE TABLE farmer_favorite_suppliers (
  farmer_id TEXT NOT NULL,
  supplier_id TEXT NOT NULL,
  notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (farmer_id, supplier_id),
  FOREIGN KEY (farmer_id) REFERENCES farmers(id),
  FOREIGN KEY (supplier_id) REFERENCES suppliers(id)
);

-- Product alternatives (equivalent products)
CREATE TABLE product_alternatives (
  product_id TEXT NOT NULL,
  alternative_product_id TEXT NOT NULL,
  equivalence_type TEXT,  -- same_ingredient, similar_effect, generic_brand
  PRIMARY KEY (product_id, alternative_product_id),
  FOREIGN KEY (product_id) REFERENCES treatment_products(id),
  FOREIGN KEY (alternative_product_id) REFERENCES treatment_products(id)
);
```

**Geospatial Supplier Search**

```javascript
// SupplierSearchService.js
class SupplierSearchService {
  async findNearestSuppliers(farmerLocation, productId, radiusKm = 50) {
    // PostGIS query to find suppliers within radius, sorted by distance
    const query = `
      SELECT
        s.*,
        sp.availability_status,
        sp.estimated_price_pkr,
        ST_Distance(s.location, ST_MakePoint($1, $2)::geography) / 1000 AS distance_km
      FROM suppliers s
      JOIN supplier_products sp ON s.id = sp.supplier_id
      WHERE sp.product_id = $3
        AND ST_DWithin(s.location, ST_MakePoint($1, $2)::geography, $4)
      ORDER BY distance_km ASC
      LIMIT 20
    `;

    const suppliers = await db.query(query, [
      farmerLocation.longitude,
      farmerLocation.latitude,
      productId,
      radiusKm * 1000  // Convert km to meters
    ]);

    // If no suppliers found, expand radius
    if (suppliers.length === 0 && radiusKm < 200) {
      return this.findNearestSuppliers(farmerLocation, productId, radiusKm * 2);
    }

    return suppliers;
  }

  async getSuppliersForTreatment(treatmentId, farmerLocation) {
    // Get all products for this treatment
    const products = await db.query(
      'SELECT id FROM treatment_products WHERE treatment_id = $1',
      [treatmentId]
    );

    // Find suppliers for each product
    const allSuppliers = await Promise.all(
      products.map(p => this.findNearestSuppliers(farmerLocation, p.id))
    );

    // Deduplicate and merge
    const uniqueSuppliers = this.deduplicateSuppliers(allSuppliers.flat());

    return uniqueSuppliers;
  }

  deduplicateSuppliers(suppliers) {
    const seen = new Set();
    return suppliers.filter(s => {
      if (seen.has(s.id)) return false;
      seen.add(s.id);
      return true;
    });
  }

  async getAlternativeProducts(productId, farmerLocation) {
    // Find alternative products
    const alternatives = await db.query(`
      SELECT tp.*, pa.equivalence_type
      FROM treatment_products tp
      JOIN product_alternatives pa ON tp.id = pa.alternative_product_id
      WHERE pa.product_id = $1
    `, [productId]);

    // Find suppliers for alternatives
    for (const alt of alternatives) {
      alt.suppliers = await this.findNearestSuppliers(farmerLocation, alt.id, 50);
    }

    return alternatives.filter(alt => alt.suppliers.length > 0);
  }
}
```

**Availability Status Management**

```javascript
// AvailabilityTracker.js
class AvailabilityTracker {
  async updateAvailability(supplierId, productId, status, farmerId) {
    const now = new Date();

    await db.query(`
      UPDATE supplier_products
      SET availability_status = $1,
          last_updated = $2,
          updated_by = $3
      WHERE supplier_id = $4 AND product_id = $5
    `, [status, now, farmerId, supplierId, productId]);

    // Track contribution for gamification (future)
    await this.recordContribution(farmerId, 'availability_update');
  }

  async getAvailabilityConfidence(supplierId, productId) {
    const result = await db.query(`
      SELECT last_updated, updated_by
      FROM supplier_products
      WHERE supplier_id = $1 AND product_id = $2
    `, [supplierId, productId]);

    if (!result[0]) return { confidence: 'unknown', daysOld: null };

    const lastUpdated = new Date(result[0].last_updated);
    const daysOld = Math.floor((Date.now() - lastUpdated) / (1000 * 60 * 60 * 24));

    let confidence;
    if (daysOld <= 7) confidence = 'high';
    else if (daysOld <= 30) confidence = 'medium';
    else confidence = 'low';

    return { confidence, daysOld, lastUpdatedBy: result[0].updated_by };
  }

  async recordContribution(farmerId, contributionType) {
    // Track farmer contributions for data quality
    await db.query(`
      INSERT INTO farmer_contributions (farmer_id, contribution_type, created_at)
      VALUES ($1, $2, NOW())
    `, [farmerId, contributionType]);
  }
}
```

**Contact Integration**

```javascript
// SupplierContactService.js
import { Linking, Platform } from 'react-native';

class SupplierContactService {
  async callSupplier(phoneNumber) {
    const formattedNumber = this.formatPakistaniNumber(phoneNumber);
    const url = `tel:${formattedNumber}`;

    try {
      const canOpen = await Linking.canOpenURL(url);
      if (canOpen) {
        await Linking.openURL(url);
        // Track contact attempt
        this.trackContact('phone_call', formattedNumber);
      } else {
        throw new Error('Cannot make phone calls on this device');
      }
    } catch (error) {
      console.error('Failed to call supplier:', error);
      // Show error to user
    }
  }

  async whatsappSupplier(whatsappNumber, message) {
    const formattedNumber = this.formatPakistaniNumber(whatsappNumber);
    // Remove leading '+' for WhatsApp URL
    const cleanNumber = formattedNumber.replace('+', '');
    const encodedMessage = encodeURIComponent(message);

    const url = `whatsapp://send?phone=${cleanNumber}&text=${encodedMessage}`;

    try {
      const canOpen = await Linking.canOpenURL(url);
      if (canOpen) {
        await Linking.openURL(url);
        this.trackContact('whatsapp', whatsappNumber);
      } else {
        // WhatsApp not installed
        throw new Error('WhatsApp not installed');
      }
    } catch (error) {
      // Fallback to regular SMS
      this.smsSupplier(formattedNumber, message);
    }
  }

  async smsSupplier(phoneNumber, message) {
    const formattedNumber = this.formatPakistaniNumber(phoneNumber);
    const url = Platform.OS === 'ios'
      ? `sms:${formattedNumber}&body=${encodeURIComponent(message)}`
      : `sms:${formattedNumber}?body=${encodeURIComponent(message)}`;

    await Linking.openURL(url);
    this.trackContact('sms', phoneNumber);
  }

  async getDirections(supplierLocation, supplierName) {
    const { latitude, longitude } = supplierLocation;
    const label = encodeURIComponent(supplierName);

    const url = Platform.select({
      ios: `maps://app?daddr=${latitude},${longitude}&q=${label}`,
      android: `google.navigation:q=${latitude},${longitude}&label=${label}`
    });

    const webFallback = `https://www.google.com/maps/search/?api=1&query=${latitude},${longitude}`;

    try {
      const canOpen = await Linking.canOpenURL(url);
      if (canOpen) {
        await Linking.openURL(url);
      } else {
        await Linking.openURL(webFallback);
      }
    } catch (error) {
      await Linking.openURL(webFallback);
    }
  }

  formatPakistaniNumber(phoneNumber) {
    // Convert Pakistani numbers to international format
    // Input: "03001234567" or "0300-1234567"
    // Output: "+923001234567"

    let cleaned = phoneNumber.replace(/\D/g, '');  // Remove non-digits

    if (cleaned.startsWith('92')) {
      return `+${cleaned}`;
    } else if (cleaned.startsWith('0')) {
      return `+92${cleaned.substring(1)}`;
    } else {
      return `+92${cleaned}`;
    }
  }

  trackContact(method, phoneNumber) {
    // Track for analytics
    MetricsService.log('supplier_contact', {
      method,
      timestamp: Date.now()
    });
  }
}
```
