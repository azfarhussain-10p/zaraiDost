**Transport Cost Service**

```javascript
// TransportCostService.js
const axios = require('axios');

class TransportCostService {
  constructor() {
    this.googleMapsApiKey = process.env.GOOGLE_MAPS_API_KEY;
    this.distanceCache = new Map();
  }

  async calculateTransportCost(fromLocation, toLocation, quantityKg, vehicleType = 'pickup') {
    // Get distance from Google Maps
    const distance = await this.getDistance(fromLocation, toLocation);

    // Get current fuel prices
    const fuelPrice = await this.getCurrentFuelPrice();

    // Calculate cost based on vehicle type
    const costBreakdown = this.calculateCostBreakdown(
      distance,
      quantityKg,
      vehicleType,
      fuelPrice
    );

    return {
      distance_km: distance.km,
      duration_minutes: distance.duration,
      vehicle_type: vehicleType,
      total_cost: costBreakdown.total,
      cost_per_kg: Math.round(costBreakdown.total / quantityKg),
      breakdown: costBreakdown,
      fuel_price_per_liter: fuelPrice
    };
  }

  async getDistance(from, to) {
    // Check cache first
    const cacheKey = `${from.latitude},${from.longitude}_${to.latitude},${to.longitude}`;
    if (this.distanceCache.has(cacheKey)) {
      return this.distanceCache.get(cacheKey);
    }

    try {
      // Call Google Maps Distance Matrix API
      const response = await axios.get('https://maps.googleapis.com/maps/api/distancematrix/json', {
        params: {
          origins: `${from.latitude},${from.longitude}`,
          destinations: `${to.latitude},${to.longitude}`,
          mode: 'driving',
          key: this.googleMapsApiKey
        }
      });

      if (response.data.status !== 'OK') {
        throw new Error(`Maps API error: ${response.data.status}`);
      }

      const element = response.data.rows[0].elements[0];

      if (element.status !== 'OK') {
        // Fallback to straight-line distance
        return this.calculateStraightLineDistance(from, to);
      }

      const result = {
        km: Math.round(element.distance.value / 1000),
        meters: element.distance.value,
        duration: Math.round(element.duration.value / 60), // minutes
        text: element.distance.text
      };

      // Cache result
      this.distanceCache.set(cacheKey, result);

      return result;
    } catch (error) {
      console.error('Google Maps API error:', error);
      // Fallback to straight-line distance
      return this.calculateStraightLineDistance(from, to);
    }
  }

  calculateStraightLineDistance(from, to) {
    // Haversine formula
    const R = 6371; // Earth radius in km
    const dLat = this.toRad(to.latitude - from.latitude);
    const dLon = this.toRad(to.longitude - from.longitude);

    const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
              Math.cos(this.toRad(from.latitude)) * Math.cos(this.toRad(to.latitude)) *
              Math.sin(dLon / 2) * Math.sin(dLon / 2);

    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    const km = R * c;

    // Estimate driving time (average 40 km/h for rural roads)
    const duration = Math.round((km / 40) * 60);

    return {
      km: Math.round(km * 1.2), // Add 20% for road vs straight-line
      meters: Math.round(km * 1200),
      duration: duration,
      text: `${Math.round(km)} km (estimated)`
    };
  }

  toRad(degrees) {
    return degrees * (Math.PI / 180);
  }

  async getCurrentFuelPrice() {
    // Fetch from government or cached data
    const fuelPrice = await db.queryOne(`
      SELECT price_per_liter FROM fuel_prices
      WHERE fuel_type = 'diesel'
      ORDER BY effective_date DESC
      LIMIT 1
    `);

    return fuelPrice ? fuelPrice.price_per_liter : 280; // Default PKR 280/liter
  }

  calculateCostBreakdown(distance, quantityKg, vehicleType, fuelPricePerLiter) {
    // Vehicle-specific parameters
    const vehicleParams = {
      pickup: {
        capacity_kg: 1000,
        fuel_consumption_km_per_liter: 8,
        base_rate: 1000,
        per_km_rate: 15
      },
      truck: {
        capacity_kg: 5000,
        fuel_consumption_km_per_liter: 5,
        base_rate: 2500,
        per_km_rate: 25
      },
      trailer: {
        capacity_kg: 20000,
        fuel_consumption_km_per_liter: 3,
        base_rate: 5000,
        per_km_rate: 40
      }
    };

    const vehicle = vehicleParams[vehicleType] || vehicleParams.pickup;

    // Calculate costs
    const baseFee = vehicle.base_rate;
    const fuelCost = (distance.km / vehicle.fuel_consumption_km_per_liter) * fuelPricePerLiter;
    const driverCost = distance.km * 5; // PKR 5 per km for driver
    const maintenanceCost = distance.km * 3; // PKR 3 per km for maintenance
    const loadingCost = 200; // Flat loading fee
    const unloadingCost = 200; // Flat unloading fee

    // Total round trip (distance * 2)
    const totalFuelCost = fuelCost * 2;
    const totalDriverCost = driverCost * 2;
    const totalMaintenanceCost = maintenanceCost * 2;

    const total = baseFee + totalFuelCost + totalDriverCost + totalMaintenanceCost + loadingCost + unloadingCost;

    return {
      base_fee: baseFee,
      fuel_cost: Math.round(totalFuelCost),
      driver_cost: Math.round(totalDriverCost),
      maintenance_cost: Math.round(totalMaintenanceCost),
      loading_cost: loadingCost,
      unloading_cost: unloadingCost,
      total: Math.round(total)
    };
  }

  async updateFuelPrices(dieselPrice, petrolPrice) {
    // Update fuel prices (admin function)
    await db.insert('fuel_prices', {
      id: `fuel-${Date.now()}`,
      fuel_type: 'diesel',
      price_per_liter: dieselPrice,
      effective_date: new Date()
    });

    await db.insert('fuel_prices', {
      id: `fuel-${Date.now()}-p`,
      fuel_type: 'petrol',
      price_per_liter: petrolPrice,
      effective_date: new Date()
    });
  }
}
```

```javascript
// RouteOptimizerService.js
class RouteOptimizerService {
  constructor() {
    this.transportCostService = new TransportCostService();
  }

  async findOptimalMandi(farmLocation, cropType, quantityKg, maxDistance = 200) {
    // Get all mandis within maxDistance with prices for crop
    const mandis = await db.query(`
      SELECT
        m.id,
        m.name,
        m.district,
        m.location_gps,
        mp.modal_price_per_kg as price_per_kg
      FROM mandis m
      JOIN mandi_prices mp ON m.id = mp.mandi_id
      WHERE mp.crop_type = $1
        AND mp.price_date = (
          SELECT MAX(price_date) FROM mandi_prices WHERE mandi_id = m.id AND crop_type = $1
        )
        AND ST_DWithin(
          m.location_gps::geography,
          ST_MakePoint($2, $3)::geography,
          $4 * 1000
        )
    `, [cropType, farmLocation.longitude, farmLocation.latitude, maxDistance]);

    if (!mandis.length) {
      return {
        optimal_mandi: null,
        message: 'No mandis found within range with price data'
      };
    }

    // Calculate net profit for each mandi
    const analysis = await Promise.all(
      mandis.map(async (mandi) => {
        const transportCost = await this.transportCostService.calculateTransportCost(
          farmLocation,
          { latitude: mandi.location_gps.coordinates[1], longitude: mandi.location_gps.coordinates[0] },
          quantityKg
        );

        const grossRevenue = quantityKg * mandi.price_per_kg;
        const netRevenue = grossRevenue - transportCost.total_cost;
        const netPricePerKg = netRevenue / quantityKg;

        return {
          mandi_id: mandi.id,
          mandi_name: mandi.name,
          district: mandi.district,
          price_per_kg: mandi.price_per_kg,
          distance_km: transportCost.distance_km,
          transport_cost: transportCost.total_cost,
          gross_revenue: Math.round(grossRevenue),
          net_revenue: Math.round(netRevenue),
          net_price_per_kg: Math.round(netPricePerKg)
        };
      })
    );

    // Sort by net revenue (highest first)
    analysis.sort((a, b) => b.net_revenue - a.net_revenue);

    const optimal = analysis[0];
    const nearest = analysis.reduce((min, curr) =>
      curr.distance_km < min.distance_km ? curr : min
    );

    return {
      optimal_mandi: optimal,
      nearest_mandi: nearest,
      all_mandis: analysis,
      savings_vs_nearest: optimal.mandi_id !== nearest.mandi_id
        ? Math.round(optimal.net_revenue - nearest.net_revenue)
        : 0,
      recommendation: this.generateRecommendation(optimal, nearest)
    };
  }

  generateRecommendation(optimal, nearest) {
    if (optimal.mandi_id === nearest.mandi_id) {
      return {
        message: `${optimal.mandi_name} is both nearest and most profitable`,
        action: 'sell_here'
      };
    }

    const savings = optimal.net_revenue - nearest.net_revenue;
    const extraDistance = optimal.distance_km - nearest.distance_km;

    if (savings > 2000 || savings / nearest.net_revenue > 0.05) {
      // 5% improvement or 2000 PKR savings
      return {
        message: `Travel ${extraDistance} km extra to ${optimal.mandi_name} for PKR ${savings} more profit`,
        action: 'travel_further'
      };
    } else {
      return {
        message: `Minimal benefit (PKR ${savings}) from traveling further. Sell at ${nearest.mandi_name}.`,
        action: 'sell_nearest'
      };
    }
  }
}
```

```javascript
// SharedTransportService.js
class SharedTransportService {
  async createTransportRequest(farmerId, requestData) {
    const request = {
      id: `trans-req-${Date.now()}`,
      farmer_id: farmerId,
      origin_location: requestData.originLocation,
      destination_mandi_id: requestData.mandiId,
      crop_type: requestData.cropType,
      quantity_kg: requestData.quantityKg,
      preferred_date: requestData.preferredDate,
      flexible_dates: requestData.flexibleDates || false,
      status: 'active',
      created_at: new Date()
    };

    await db.insert('transport_sharing_requests', request);

    // Find potential matches
    const matches = await this.findMatches(request);

    if (matches.length > 0) {
      await this.notifyMatches(farmerId, matches);
    }

    return {
      request,
      potential_matches: matches
    };
  }

  async findMatches(request) {
    // Find other farmers going to same mandi around same time
    const matches = await db.query(`
      SELECT
        tsr.*,
        f.name as farmer_name,
        f.phone_number,
        ST_Distance(
          tsr.origin_location::geography,
          $1::geography
        ) / 1000 as distance_from_you_km
      FROM transport_sharing_requests tsr
      JOIN farmers f ON tsr.farmer_id = f.id
      WHERE tsr.destination_mandi_id = $2
        AND tsr.farmer_id != $3
        AND tsr.status = 'active'
        AND (
          tsr.preferred_date = $4
          OR (tsr.flexible_dates = TRUE AND ABS(EXTRACT(DAY FROM tsr.preferred_date - $4::date)) <= 2)
        )
        AND ST_DWithin(
          tsr.origin_location::geography,
          $1::geography,
          20000  -- Within 20 km
        )
      ORDER BY distance_from_you_km ASC
      LIMIT 5
    `, [
      `POINT(${request.origin_location.longitude} ${request.origin_location.latitude})`,
      request.destination_mandi_id,
      request.farmer_id,
      request.preferred_date
    ]);

    // Calculate cost savings for each match
    const enrichedMatches = await Promise.all(
      matches.map(async (match) => {
        const savings = await this.calculateSharedCostSavings(request, match);
        return {
          ...match,
          estimated_savings: savings
        };
      })
    );

    return enrichedMatches;
  }

  async calculateSharedCostSavings(request1, request2) {
    // Calculate individual costs

    const individualCost2 = 3000;

    // Shared cost (assume larger vehicle, split cost)

    const costPerFarmer = sharedCost / 2;

    const savings1 = individualCost1 - costPerFarmer;
    const savings2 = individualCost2 - costPerFarmer;

      your_savings: Math.round(savings1),
      their_savings: Math.round(savings2),
      total_savings: Math.round(savings1 + savings2),
      shared_cost: sharedCost,
      cost_per_farmer: costPerFarmer
    };
  }

  async notifyMatches(farmerId, matches) {

        data: {
          type: 'transport_match',
          farmer_id: farmerId
        }
      });
    }
  }
```

```javascript
// TransportProviderService.js
class TransportProviderService {
  async searchProviders(location, vehicleType = null) {
    let query = `
      SELECT
        tp.*,
        AVG(tpr.rating) as avg_rating,
        COUNT(tpr.id) as review_count
      FROM transport_providers tp
      LEFT JOIN transport_provider_reviews tpr ON tp.id = tpr.provider_id
      WHERE tp.active = TRUE
    `;

    const params = [];

    // Filter by vehicle type
    if (vehicleType) {
      query += ` AND $1 = ANY(tp.vehicle_types)`;
      params.push(vehicleType);
    }

    // Filter by service area
    if (location) {
      query += ` AND ST_DWithin(
        tp.service_area_center::geography,
        ST_MakePoint($${params.length + 1}, $${params.length + 2})::geography,
        tp.service_radius_km * 1000
      )`;
      params.push(location.longitude, location.latitude);
    }

    query += ` GROUP BY tp.id ORDER BY avg_rating DESC NULLS LAST`;

    return await db.query(query, params);
  }

  async getProviderProfile(providerId) {
    const provider = await db.queryOne(`
      SELECT
        tp.*,
        AVG(tpr.rating) as avg_rating,
        COUNT(tpr.id) as review_count
      FROM transport_providers tp
      LEFT JOIN transport_provider_reviews tpr ON tp.id = tpr.provider_id
      WHERE tp.id = $1
      GROUP BY tp.id
    `, [providerId]);

    const reviews = await db.query(`
      SELECT tpr.*, f.name as farmer_name
      FROM transport_provider_reviews tpr
      LEFT JOIN farmers f ON tpr.farmer_id = f.id
      WHERE tpr.provider_id = $1
      ORDER BY tpr.created_at DESC
      LIMIT 10
    `, [providerId]);

    return {
      ...provider,
      recent_reviews: reviews
    };
  }
}
```

```sql
-- Fuel prices
CREATE TABLE fuel_prices (
  id TEXT PRIMARY KEY,
  fuel_type TEXT NOT NULL,  -- 'diesel', 'petrol'
  price_per_liter DECIMAL(10, 2) NOT NULL,
  effective_date DATE NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_fuel_prices_type_date ON fuel_prices(fuel_type, effective_date DESC);

-- Transport sharing requests
CREATE TABLE transport_sharing_requests (
  id TEXT PRIMARY KEY,
  farmer_id TEXT NOT NULL,
  origin_location GEOGRAPHY(POINT) NOT NULL,
  destination_mandi_id TEXT NOT NULL,
  crop_type TEXT,
  quantity_kg DECIMAL(10, 2),
  preferred_date DATE NOT NULL,
  flexible_dates BOOLEAN DEFAULT FALSE,
  status TEXT DEFAULT 'active',  -- 'active', 'matched', 'completed', 'cancelled'
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (destination_mandi_id) REFERENCES mandis(id)
);

CREATE INDEX idx_transport_sharing_mandi_date ON transport_sharing_requests(destination_mandi_id, preferred_date);
CREATE INDEX idx_transport_sharing_location ON transport_sharing_requests USING GIST(origin_location);

-- Transport providers
CREATE TABLE transport_providers (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  phone_number TEXT NOT NULL,
  vehicle_types TEXT[],  -- ['pickup', 'truck', 'trailer']
  service_area_center GEOGRAPHY(POINT),
  service_radius_km INTEGER,
  base_rate DECIMAL(10, 2),
  per_km_rate DECIMAL(10, 2),
  active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Transport provider reviews
CREATE TABLE transport_provider_reviews (
  id TEXT PRIMARY KEY,
  farmer_id TEXT NOT NULL,
  provider_id TEXT NOT NULL,
  rating INTEGER CHECK (rating >= 1 AND rating <= 5),
  comment TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (provider_id) REFERENCES transport_providers(id)
);

-- Transport history
CREATE TABLE transport_history (
  id TEXT PRIMARY KEY,
  farmer_id TEXT NOT NULL,
  from_location GEOGRAPHY(POINT),
  to_mandi_id TEXT,
  distance_km DECIMAL(10, 2),
  actual_cost DECIMAL(10, 2),
  vehicle_type TEXT,
  provider_id TEXT,
  transport_date DATE,
  notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (to_mandi_id) REFERENCES mandis(id),
  FOREIGN KEY (provider_id) REFERENCES transport_providers(id)
);

CREATE INDEX idx_transport_history_farmer ON transport_history(farmer_id, transport_date DESC);
```
