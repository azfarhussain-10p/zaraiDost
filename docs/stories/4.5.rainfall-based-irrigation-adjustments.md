# Story 4.5: Rainfall-Based Irrigation Adjustments

## Status
Draft

## Story
**As a** farmer planning to irrigate,
**I want** automatic alerts when rainfall makes irrigation unnecessary,
**so that** I don't waste water and money.

## Acceptance Criteria
1. AC1: Automatic irrigation delay recommendation when rain forecasted (>10mm)
2. AC2: Post-rainfall assessment updates moisture predictions
3. AC3: Alert sent: "Delay irrigation - 15mm rain expected by 3 PM"
4. AC4: User can confirm or override rain-based recommendations
5. AC5: Water savings tracked and displayed to user
6. AC6: Rainfall accumulation tracked by field
7. AC7: Integration with weather alerts from Story 4.1

**Rainfall Monitoring and Alert Generation**

```javascript
// RainfallMonitorService.js
class RainfallMonitorService {
  async monitorUpcomingIrrigations() {
    // Get irrigations scheduled in next 48 hours
    const upcomingIrrigations = await db.query(`
      SELECT * FROM irrigation_schedule
      WHERE status = 'scheduled'
        AND scheduled_date BETWEEN NOW() AND NOW() + INTERVAL '48 hours'
    `);

    for (const irrigation of upcomingIrrigations) {
      await this.checkRainfallImpact(irrigation);
    }
  }

  async checkRainfallImpact(irrigationEvent) {
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [irrigationEvent.field_id]);
    const weatherForecast = await WeatherDataService.getHourlyForecast(field.latitude, field.longitude);

    // Check rainfall forecast for irrigation day
    const irrigationDate = new Date(irrigationEvent.scheduled_date);
    const relevantForecasts = weatherForecast.filter(f => {
      const forecastDate = new Date(f.forecast_datetime);
      return this.isSameDay(forecastDate, irrigationDate);
    });

    const totalRainfall = relevantForecasts.reduce((sum, f) => sum + (f.rainfall_mm || 0), 0);
    const avgProbability = relevantForecasts.reduce((sum, f) => sum + f.precipitation_probability, 0) / relevantForecasts.length;

    // Generate alert if significant rain forecasted
    if (totalRainfall > 10 && avgProbability > 70) {
      await this.generateDelayRecommendation(irrigationEvent, totalRainfall, relevantForecasts);
    }
  }

  async generateDelayRecommendation(irrigationEvent, rainfallAmount, forecasts) {
    // Find peak rainfall timing
    const peakRainfall = forecasts.reduce((max, f) =>
      f.rainfall_mm > max.rainfall_mm ? f : max
    , forecasts[0]);

    const peakTime = new Date(peakRainfall.forecast_datetime).toLocaleTimeString('en-US', {
      hour: 'numeric',
      minute: '2-digit'
    });

    const recommendation = {
      id: `rain-delay-${Date.now()}`,
      irrigation_event_id: irrigationEvent.id,
      field_id: irrigationEvent.field_id,
      recommendation_type: 'delay_for_rain',
      forecasted_rainfall_mm: rainfallAmount,
      rainfall_probability: peakRainfall.precipitation_probability,
      peak_rainfall_time: peakTime,
      water_saved_liters: irrigationEvent.water_volume_liters,
      cost_saved_pkr: this.calculateCostSavings(irrigationEvent),
      message: `Delay irrigation - ${rainfallAmount.toFixed(1)}mm rain expected by ${peakTime}`,
      status: 'pending',
      created_at: new Date()
    };

    await db.insert('irrigation_recommendations', recommendation);
    await this.sendDelayAlert(recommendation);

    return recommendation;
  }

  calculateCostSavings(irrigationEvent) {
    // If tube well, calculate electricity cost saved
    if (irrigationEvent.water_source === 'tube_well') {
      return irrigationEvent.duration_hours * 150;  // PKR 150/hour
    }
    return 0;  // Canal is free
  }

  async sendDelayAlert(recommendation) {
    // Send push notification with action buttons
    const notification = {
      title: 'Irrigation Delay Recommended',
      body: recommendation.message,
      data: {
        type: 'rain_delay_recommendation',
        recommendation_id: recommendation.id
      },
      actions: [
        { id: 'accept_delay', title: 'Delay Irrigation' },
        { id: 'override', title: 'Irrigate Anyway' }
      ]
    };

    await NotificationService.send(notification);
  }

  isSameDay(date1, date2) {
    return date1.toDateString() === date2.toDateString();
  }
}
```

**Post-Rainfall Assessment**

```javascript
// PostRainfallAssessmentService.js
class PostRainfallAssessmentService {
  async assessRainfallImpact(fieldId, cropSeasonId) {
    const field = await db.queryOne('SELECT * FROM fields WHERE id = $1', [fieldId]);
    const actualWeather = await WeatherDataService.getActualWeather(field.latitude, field.longitude);

    const rainfallToday = actualWeather.rainfall_mm || 0;

    if (rainfallToday > 5) {
      // Significant rainfall occurred
      await this.updateSoilMoisture(fieldId, cropSeasonId, rainfallToday);
      await this.recalculateIrrigationSchedule(fieldId, cropSeasonId);
      await this.trackWaterSavings(fieldId, rainfallToday);
      await this.sendPostRainfallNotification(fieldId, rainfallToday);
    }

    // Record rainfall event
    await this.recordRainfallEvent(fieldId, rainfallToday);
  }

  async updateSoilMoisture(fieldId, cropSeasonId, rainfallMm) {
    const moistureModel = new SoilMoisturePredictionModel(field, cropSeason, field.soil_type);

    // Calculate effective rainfall
    const effectiveRainfall = moistureModel.calculateEffectiveRainfall(rainfallMm);

    // Update current moisture prediction
    const currentMoisture = await moistureModel.calculateCurrentMoisture();
    const updatedMoisture = currentMoisture + effectiveRainfall;

    await db.insert('soil_moisture_observations', {
      id: `obs-rain-${Date.now()}`,
      field_id: fieldId,
      crop_season_id: cropSeasonId,
      observation_date: new Date(),
      observed_moisture_mm: updatedMoisture,
      source: 'rainfall_update',
      rainfall_mm: rainfallMm,
      effective_rainfall_mm: effectiveRainfall
    });

    return updatedMoisture;
  }

  async recalculateIrrigationSchedule(fieldId, cropSeasonId) {
    // Trigger schedule recalculation from Story 4.3
    const scheduleService = new IrrigationScheduleService();
    await scheduleService.recalculateSchedule(fieldId, cropSeasonId);
  }

  async trackWaterSavings(fieldId, rainfallMm) {
    // Check if irrigation was delayed/skipped due to rain
    const recentRecommendations = await db.query(`
      SELECT * FROM irrigation_recommendations
      WHERE field_id = $1
        AND recommendation_type = 'delay_for_rain'
        AND created_at >= NOW() - INTERVAL '48 hours'
        AND status = 'accepted'
    `, [fieldId]);

    for (const rec of recentRecommendations) {
      const irrigation = await db.queryOne(
        'SELECT * FROM irrigation_schedule WHERE id = $1',
        [rec.irrigation_event_id]
      );

      await db.insert('water_savings', {
        id: `savings-${Date.now()}`,
        field_id: fieldId,
        irrigation_event_id: rec.irrigation_event_id,
        savings_type: 'rainfall_substitution',
        water_saved_liters: irrigation.water_volume_liters,
        cost_saved_pkr: rec.cost_saved_pkr,
        rainfall_mm: rainfallMm,
        savings_date: new Date()
      });
    }
  }

  async sendPostRainfallNotification(fieldId, rainfallMm) {
    const notification = {
      title: 'Rainfall Recorded',
      body: `${rainfallMm.toFixed(1)}mm rain received. Irrigation schedule updated.`,
      data: {
        type: 'post_rainfall_update',
        field_id: fieldId
      }
    };

    await NotificationService.send(notification);
  }

  async recordRainfallEvent(fieldId, rainfallMm) {
    await db.insert('rainfall_events', {
      id: `rain-${Date.now()}`,
      field_id: fieldId,
      event_date: new Date(),
      rainfall_mm: rainfallMm,
      data_source: 'IBM Weather API'
    });
  }
}
```

### Data Model Requirements

```sql
CREATE TABLE irrigation_recommendations (
  id TEXT PRIMARY KEY,
  irrigation_event_id TEXT NOT NULL,
  field_id TEXT NOT NULL,
  recommendation_type TEXT NOT NULL,  -- 'delay_for_rain', 'skip_for_rain'
  forecasted_rainfall_mm REAL,
  rainfall_probability INTEGER,
  peak_rainfall_time TEXT,
  water_saved_liters REAL,
  cost_saved_pkr REAL,
  message TEXT,
  status TEXT DEFAULT 'pending',      -- 'pending', 'accepted', 'rejected'
  user_response TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  responded_at DATETIME,
  FOREIGN KEY (irrigation_event_id) REFERENCES irrigation_schedule(id)
);

CREATE TABLE water_savings (
  id TEXT PRIMARY KEY,
  field_id TEXT NOT NULL,
  irrigation_event_id TEXT,
  savings_type TEXT NOT NULL,         -- 'rainfall_substitution', 'optimization', 'method_efficiency'
  water_saved_liters REAL NOT NULL,
  cost_saved_pkr REAL DEFAULT 0,
  rainfall_mm REAL,
  savings_date DATE NOT NULL,
  notes TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (field_id) REFERENCES fields(id)
);

CREATE TABLE rainfall_events (
  id TEXT PRIMARY KEY,
  field_id TEXT NOT NULL,
  event_date DATE NOT NULL,
  rainfall_mm REAL NOT NULL,
  effective_rainfall_mm REAL,
  data_source TEXT,                   -- 'IBM Weather API', 'user_input'
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (field_id) REFERENCES fields(id),
  UNIQUE(field_id, event_date)
);

CREATE INDEX idx_water_savings_field_date ON water_savings(field_id, savings_date DESC);
CREATE INDEX idx_rainfall_events_field_date ON rainfall_events(field_id, event_date DESC);
```
