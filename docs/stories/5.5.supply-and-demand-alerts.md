# Story 5.5: Supply and Demand Alerts

## Status

## Story
**As a** farmer growing seasonal crops,
**I want** alerts about market oversupply or high demand,
**so that** I can adjust my selling strategy.

```javascript
// AgriStatisticsService.js
const axios = require('axios');
const cheerio = require('cheerio');

class AgriStatisticsService {
  constructor() {
    this.baseURL = process.env.AGRI_STATS_API_URL;
    this.dataCache = new Map();
  }

  async fetchAreaUnderCultivation(cropType, season, year) {
    // Try API first
    try {
      const data = await this.fetchFromAPI(cropType, season, year);
      if (data) return data;
    } catch (error) {
      console.log('API failed, trying web scraping:', error.message);
    }

    // Fallback to web scraping
    try {
      const data = await this.scrapeAreaData(cropType, season, year);
      return data;
    } catch (error) {
      console.error('Failed to fetch area data:', error);
      return null;
    }
  }

  async fetchFromAPI(cropType, season, year) {
    // Example: Pakistan Ministry of Agriculture API
    const response = await axios.get(`${this.baseURL}/area-statistics`, {
      params: {
        crop: cropType,
        season: season, // rabi, kharif
        year: year
      }
    });

    return this.transformAPIResponse(response.data);
  }

  async scrapeAreaData(cropType, season, year) {
    // Scrape from government agriculture department websites
    // This is a fallback when APIs are unavailable
    const url = `https://www.mnfsr.gov.pk/statistics.aspx`;
    const response = await axios.get(url);
    const $ = cheerio.load(response.data);

    // Extract data from tables (implementation depends on site structure)
    const areaData = this.parseAreaTable($, cropType, season);

    return areaData;
  }

  transformAPIResponse(data) {
    // Transform to standard format
    return {
      crop_type: data.commodity,
      season: data.season,
      year: data.year,
      total_area_hectares: data.area,
      provinces: {
        punjab: data.punjab_area,
        sindh: data.sindh_area,
        kpk: data.kpk_area,
        balochistan: data.balochistan_area
      },
      updated_at: new Date()
    };
  }

  async storeAreaData(areaData) {
    await db.insert('crop_area_statistics', {
      id: `area-${Date.now()}`,
      crop_type: areaData.crop_type,
      season: areaData.season,
      year: areaData.year,
      total_area_hectares: areaData.total_area_hectares,
      punjab_area: areaData.provinces.punjab,
      sindh_area: areaData.provinces.sindh,
      kpk_area: areaData.provinces.kpk,
      balochistan_area: areaData.provinces.balochistan,
      source: 'government',
      created_at: new Date()
    });
  }

  async getHistoricalAreaAverage(cropType, years = 5) {
    const historicalData = await db.query(`
      SELECT AVG(total_area_hectares) as avg_area
      FROM crop_area_statistics
      WHERE crop_type = $1
        AND year >= $2
    `, [cropType, new Date().getFullYear() - years]);

    return historicalData[0]?.avg_area || null;
  }
}
```

**Supply Forecast Service**

```javascript
// SupplyForecastService.js
class SupplyForecastService {
  constructor() {
    this.agriStatsService = new AgriStatisticsService();
  }

  async calculateSupplyIndex(cropType, season, year) {
    // Get current year area
    const currentArea = await this.agriStatsService.fetchAreaUnderCultivation(
      cropType,
      season,
      year
    );

    if (!currentArea) {
      throw new Error(`No area data available for ${cropType} ${season} ${year}`);
    }

    // Get 5-year average
    const avgArea = await this.agriStatsService.getHistoricalAreaAverage(cropType, 5);

    if (!avgArea) {
      return {
        supply_index: 100,
        status: 'normal',
        message: 'Insufficient historical data for comparison'
      };
    }

    // Calculate supply index (100 = normal, >100 = oversupply, <100 = shortage)
    const supplyIndex = (currentArea.total_area_hectares / avgArea) * 100;

    // Classify supply situation
    let status, message, price_impact_percent;

    if (supplyIndex > 120) {
      status = 'oversupply';
      price_impact_percent = -((supplyIndex - 100) / 2); // Rough estimate: 20% oversupply = -10% price
      message = `${Math.round(supplyIndex - 100)}% more ${cropType} planted than average. Expect price drop around ${Math.abs(Math.round(price_impact_percent))}%.`;
    } else if (supplyIndex > 110) {
      status = 'slight_oversupply';
      price_impact_percent = -((supplyIndex - 100) / 3);
      message = `${Math.round(supplyIndex - 100)}% more ${cropType} planted. Prices may soften slightly.`;
    } else if (supplyIndex < 80) {
      status = 'shortage';
      price_impact_percent = ((100 - supplyIndex) / 2);
      message = `${Math.round(100 - supplyIndex)}% less ${cropType} planted. Expect price rise around ${Math.round(price_impact_percent)}%.`;
    } else if (supplyIndex < 90) {
      status = 'slight_shortage';
      price_impact_percent = ((100 - supplyIndex) / 3);
      message = `${Math.round(100 - supplyIndex)}% less ${cropType} planted. Prices may firm up slightly.`;
    } else {
      status = 'normal';
      price_impact_percent = 0;
      message = `${cropType} planting is near average. Expect normal prices.`;
    }

    return {
      crop_type: cropType,
      season,
      year,
      supply_index: Math.round(supplyIndex),
      status,
      message,
      price_impact_percent: Math.round(price_impact_percent),
      current_area: currentArea.total_area_hectares,
      average_area: avgArea
    };
  }

  async calculateRegionalSupply(cropType, season, year) {
    const areaData = await this.agriStatsService.fetchAreaUnderCultivation(
      cropType,
      season,
      year
    );

    if (!areaData) return null;

    // Calculate regional supply indices
    const regions = ['punjab', 'sindh', 'kpk', 'balochistan'];
    const regionalSupply = {};

    for (const region of regions) {
      const currentRegionArea = areaData.provinces[region];
      const avgRegionArea = await this.getRegionalAverageArea(cropType, region, 5);

      if (avgRegionArea) {
        regionalSupply[region] = {
          current_area: currentRegionArea,
          average_area: avgRegionArea,
          supply_index: Math.round((currentRegionArea / avgRegionArea) * 100),
          status: this.classifySupplyStatus((currentRegionArea / avgRegionArea) * 100)
        };
      }
    }

    return regionalSupply;
  }

  async getRegionalAverageArea(cropType, region, years = 5) {
    const avg = await db.queryOne(`
      SELECT AVG(${region}_area) as avg_area
      FROM crop_area_statistics
      WHERE crop_type = $1
        AND year >= $2
    `, [cropType, new Date().getFullYear() - years]);

    return avg?.avg_area || null;
  }

  classifySupplyStatus(supplyIndex) {
    if (supplyIndex > 120) return 'oversupply';
    if (supplyIndex > 110) return 'slight_oversupply';
    if (supplyIndex < 80) return 'shortage';
    if (supplyIndex < 90) return 'slight_shortage';
    return 'normal';
  }

  async forecastHarvestTiming(cropType, season) {
    // Typical harvest months for Pakistani crops
    const harvestSchedule = {
      wheat: { season: 'rabi', months: [4, 5] }, // April-May
      rice: { season: 'kharif', months: [10, 11] }, // October-November
      cotton: { season: 'kharif', months: [9, 10, 11] }, // September-November
      sugarcane: { season: 'perennial', months: [11, 12, 1, 2] } // November-February
    };

    const schedule = harvestSchedule[cropType];
    if (!schedule) return null;

    // Current date
    const now = new Date();
    const currentMonth = now.getMonth() + 1;

    // Calculate days until harvest
    let daysUntilHarvest = 0;
    const firstHarvestMonth = schedule.months[0];

    if (currentMonth < firstHarvestMonth) {
      daysUntilHarvest = (firstHarvestMonth - currentMonth) * 30;
    } else if (currentMonth > schedule.months[schedule.months.length - 1]) {
      // Next year
      daysUntilHarvest = (12 - currentMonth + firstHarvestMonth) * 30;
    } else {
      // Currently in harvest season
      daysUntilHarvest = 0;
    }

    return {
      crop_type: cropType,
      harvest_months: schedule.months,
      days_until_harvest: daysUntilHarvest,
      in_harvest_season: daysUntilHarvest === 0
    };
  }
}
```

**Demand Analysis Service**

```javascript
// DemandAnalysisService.js
class DemandAnalysisService {
  async analyzeDemand(cropType) {
    // Analyze multiple demand factors
    const factors = {
      historical_consumption: await this.getHistoricalConsumption(cropType),
      seasonal_demand: await this.getSeasonalDemand(cropType),
      industrial_demand: await this.getIndustrialDemand(cropType),
      export_demand: await this.getExportDemand(cropType)
    };

    // Calculate overall demand index
    const demandIndex = this.calculateDemandIndex(factors);

    return {
      crop_type: cropType,
      demand_index: demandIndex.index,
      status: demandIndex.status,
      message: demandIndex.message,
      factors: factors
    };
  }

  async getHistoricalConsumption(cropType) {
    // Query historical consumption patterns
    const consumption = await db.query(`
      SELECT
        year,
        AVG(consumption_tons) as avg_consumption
      FROM crop_consumption_statistics
      WHERE crop_type = $1
      GROUP BY year
      ORDER BY year DESC
      LIMIT 5
    `, [cropType]);

    if (!consumption.length) return null;

    // Calculate trend (increasing/decreasing)
    const recent = consumption[0].avg_consumption;
    const older = consumption[consumption.length - 1].avg_consumption;
    const trend = ((recent - older) / older) * 100;

    return {
      recent_consumption: recent,
      trend_percent: Math.round(trend),
      trend: trend > 5 ? 'increasing' : trend < -5 ? 'decreasing' : 'stable'
    };
  }

  async getSeasonalDemand(cropType) {
    // Check for seasonal demand spikes
    const currentMonth = new Date().getMonth() + 1;

    // Ramadan demand (example - varies by Islamic calendar)
    if (currentMonth >= 3 && currentMonth <= 4) {
      return {
        event: 'Ramadan',
        impact: 'high',
        description: 'Food demand increases 20-30% during Ramadan'
      };
    }

    // Eid demand
    if (currentMonth === 5 || currentMonth === 8) {
      return {
        event: 'Eid',
        impact: 'medium',
        description: 'Increased demand for meat, dairy, and vegetables during Eid'
      };
    }

    return null;
  }

  async getIndustrialDemand(cropType) {
    // Industrial demand (flour mills, textile, sugar mills)
    const industrialCrops = {
      wheat: 'Flour mills',
      cotton: 'Textile industry',
      sugarcane: 'Sugar mills'
    };

    if (industrialCrops[cropType]) {
      return {
        industry: industrialCrops[cropType],
        demand_level: 'consistent',
        description: `Strong industrial demand from ${industrialCrops[cropType]}`
      };
    }

    return null;
  }

  async getExportDemand(cropType) {
    // Query export statistics
    const exports = await db.query(`
      SELECT
        year,
        export_volume_tons,
        export_value_usd
      FROM export_statistics
      WHERE crop_type = $1
      ORDER BY year DESC
      LIMIT 3
    `, [cropType]);

    if (!exports.length) return null;

    const recentExports = exports[0].export_volume_tons;
    const avgExports = exports.reduce((sum, e) => sum + e.export_volume_tons, 0) / exports.length;
    const trend = ((recentExports - avgExports) / avgExports) * 100;

    return {
      recent_exports: recentExports,
      trend_percent: Math.round(trend),
      trend: trend > 10 ? 'strong' : trend > 0 ? 'moderate' : 'weak',
      export_value: exports[0].export_value_usd
    };
  }

  calculateDemandIndex(factors) {
    let score = 100; // Base index

    // Adjust for consumption trend
    if (factors.historical_consumption) {
      score += factors.historical_consumption.trend_percent / 2;
    }

    // Adjust for seasonal demand
    if (factors.seasonal_demand) {
      if (factors.seasonal_demand.impact === 'high') score += 20;
      if (factors.seasonal_demand.impact === 'medium') score += 10;
    }

    // Adjust for export demand
    if (factors.export_demand) {
      score += factors.export_demand.trend_percent / 3;
    }

    let status, message;

    if (score > 120) {
      status = 'high_demand';
      message = 'Strong demand conditions - favorable for selling';
    } else if (score > 110) {
      status = 'moderate_demand';
      message = 'Moderate demand - normal selling conditions';
    } else if (score < 90) {
      status = 'weak_demand';
      message = 'Weak demand - prices may be pressured';
    } else {
      status = 'normal';
      message = 'Normal demand conditions';
    }

    return {
      index: Math.round(score),
      status,
      message
    };
  }
}
```

**Supply/Demand Alert Service**

```javascript
// SupplyDemandAlertService.js
const cron = require('node-cron');

class SupplyDemandAlertService {
  constructor() {
    this.supplyForecastService = new SupplyForecastService();
    this.demandAnalysisService = new DemandAnalysisService();
    this.notificationService = new NotificationService();
  }

  start() {
    // Check supply/demand weekly
    cron.schedule('0 10 * * 1', async () => { // Monday 10 AM
      await this.checkSupplyDemandAlerts();
    });
  }

  async checkSupplyDemandAlerts() {
    const crops = ['wheat', 'rice', 'cotton', 'sugarcane'];
    const currentYear = new Date().getFullYear();

    for (const crop of crops) {
      // Check supply
      const supplyForecast = await this.supplyForecastService.calculateSupplyIndex(
        crop,
        this.getCurrentSeason(crop),
        currentYear
      );

      // Check demand
      const demandAnalysis = await this.demandAnalysisService.analyzeDemand(crop);

      // Check harvest timing
      const harvestTiming = await this.supplyForecastService.forecastHarvestTiming(crop);

      // Send alerts if within 2 weeks of harvest
      if (harvestTiming && harvestTiming.days_until_harvest <= 14) {
        await this.sendSupplyDemandAlert(crop, supplyForecast, demandAnalysis);
      }
    }
  }

  async sendSupplyDemandAlert(cropType, supply, demand) {
    // Get farmers growing this crop
    const farmers = await db.query(`
      SELECT DISTINCT f.id, f.name, f.phone_number
      FROM farmers f
      JOIN fields fi ON f.id = fi.farmer_id
      JOIN crop_seasons cs ON fi.id = cs.field_id
      WHERE cs.crop_type = $1
        AND cs.status = 'active'
    `, [cropType]);

    let alertMessage, alertType;

    // Determine alert type
    if (supply.status === 'oversupply') {
      alertType = 'oversupply';
      alertMessage = `MARKET ALERT: ${supply.message}`;
    } else if (supply.status === 'shortage' || demand.status === 'high_demand') {
      alertType = 'high_demand';
      alertMessage = `OPPORTUNITY: ${demand.message}`;
    } else {
      return; // No alert needed for normal conditions
    }

    // Send to all farmers
    for (const farmer of farmers) {
      await this.notificationService.sendPushNotification(farmer.id, {
        title: `${cropType.toUpperCase()} Market Alert`,
        body: alertMessage,
        data: {
          type: 'supply_demand_alert',
          alert_type: alertType,
          crop_type: cropType
        }
      });

      console.log(`Sent ${alertType} alert to ${farmer.name} for ${cropType}`);
    }

    // Log alert
    await this.logAlert(cropType, alertType, alertMessage, farmers.length);
  }

  getCurrentSeason(cropType) {
    const seasons = {
      wheat: 'rabi',
      rice: 'kharif',
      cotton: 'kharif',
      sugarcane: 'perennial'
    };
    return seasons[cropType] || 'kharif';
  }

  async logAlert(cropType, alertType, message, farmerCount) {
    await db.insert('supply_demand_alerts_log', {
      id: `alert-${Date.now()}`,
      crop_type: cropType,
      alert_type: alertType,
      message: message,
      farmers_notified: farmerCount,
      created_at: new Date()
    });
  }
}
```
