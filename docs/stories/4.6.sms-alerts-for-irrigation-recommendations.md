# Story 4.6: SMS Alerts for Irrigation Recommendations

## Status

## Story
**As a** farmer who doesn't always open the app,
**I want** SMS alerts for important irrigation decisions,
**so that** I don't miss critical timing.

**SMS Gateway Service**

```javascript
// SMSGatewayService.js
class SMSGatewayService {
  constructor() {
    this.primaryGateway = new TwilioGateway();
    this.backupGateway = new LocalPakistaniGateway();
  }

  async sendSMS(phoneNumber, message, alertType) {
    // Check rate limit
    const canSend = await this.checkRateLimit(phoneNumber);
    if (!canSend) {
      console.log(`Rate limit exceeded for ${phoneNumber}`);
      return { success: false, reason: 'rate_limit_exceeded' };
    }

    // Format phone number (Pakistan: +92)
    const formattedNumber = this.formatPhoneNumber(phoneNumber);

    try {
      // Try primary gateway (Twilio)
      const result = await this.primaryGateway.send(formattedNumber, message);
      await this.logSMSDelivery(phoneNumber, message, alertType, result, 'twilio');
      return result;
    } catch (error) {
      console.error('Primary gateway failed:', error);

      // Fallback to backup gateway
      try {
        const result = await this.backupGateway.send(formattedNumber, message);
        await this.logSMSDelivery(phoneNumber, message, alertType, result, 'local');
        return result;
      } catch (backupError) {
        console.error('Backup gateway failed:', backupError);
        await this.logSMSDelivery(phoneNumber, message, alertType, { success: false }, 'failed');
        return { success: false, error: backupError };
      }
    }
  }

  async checkRateLimit(phoneNumber) {
    const today = new Date();
    today.setHours(0, 0, 0, 0);

    const count = await db.queryOne(`
      SELECT COUNT(*) as sms_count
      FROM sms_delivery_log
      WHERE phone_number = $1
        AND sent_at >= $2
        AND status = 'sent'
    `, [phoneNumber, today]);

    return count.sms_count < 2;  // Max 2 per day
  }

  formatPhoneNumber(phoneNumber) {
    // Ensure +92 country code for Pakistan
    let formatted = phoneNumber.replace(/\D/g, '');  // Remove non-digits

    if (formatted.startsWith('0')) {
      formatted = '92' + formatted.substring(1);
    }
    if (!formatted.startsWith('92')) {
      formatted = '92' + formatted;
    }

    return '+' + formatted;
  }

  async logSMSDelivery(phoneNumber, message, alertType, result, gateway) {
    await db.insert('sms_delivery_log', {
      id: `sms-${Date.now()}`,
      phone_number: phoneNumber,
      message_content: message,
      alert_type: alertType,
      gateway_used: gateway,
      status: result.success ? 'sent' : 'failed',
      delivery_id: result.messageId || null,
      error_message: result.error || null,
      sent_at: new Date()
    });
  }
}

// Twilio implementation
class TwilioGateway {
  constructor() {
    this.client = require('twilio')(
      process.env.TWILIO_ACCOUNT_SID,
      process.env.TWILIO_AUTH_TOKEN
    );
    this.fromNumber = process.env.TWILIO_PHONE_NUMBER;
  }

  async send(phoneNumber, message) {
    try {
      const result = await this.client.messages.create({
        body: message,
        from: this.fromNumber,
        to: phoneNumber
      });

      return {
        success: true,
        messageId: result.sid,
        status: result.status
      };
    } catch (error) {
      throw error;
    }
  }
}
```

```javascript
        ur: `${data.fieldName}: Nahr ka paani ${data.day} ko ${data.time} baje milega. Paani dene ki tayyari karein.`,
        pa: `${data.fieldName}: Nahar da pani ${data.day} nu ${data.time} vajje milega. Pani deen di tayyari karo.`,

    return this.truncateMessage(template, 160);

  truncateMessage(message, maxLength) {
    if (message.length <= maxLength) return message;

    // Truncate and add ellipsis
    return message.substring(0, maxLength - 3) + '...';
  }
```

**SMS Opt-in/Opt-out Management**

```javascript
// SMSPreferenceService.js
class SMSPreferenceService {
  async enableSMS(userId, phoneNumber, alertTypes = ['all']) {
    // Send OTP for verification
    const otp = this.generateOTP();
    await this.sendOTP(phoneNumber, otp);

    // Store pending verification
    await db.insert('sms_verifications', {
      id: `verify-${Date.now()}`,
      user_id: userId,
      phone_number: phoneNumber,
      otp_code: otp,
      verified: false,
      expires_at: new Date(Date.now() + 15 * 60 * 1000)  // 15 minutes
    });

    return { otp_sent: true };
  }

  async verifyOTP(userId, otp) {
    const verification = await db.queryOne(`
      SELECT * FROM sms_verifications
      WHERE user_id = $1 AND otp_code = $2
        AND verified = false
        AND expires_at > NOW()
      ORDER BY created_at DESC
      LIMIT 1
    `, [userId, otp]);

    if (!verification) {
      return { verified: false, error: 'Invalid or expired OTP' };
    }

    // Mark as verified
    await db.update('sms_verifications', { id: verification.id }, {
      verified: true,
      verified_at: new Date()
    });

    // Enable SMS preferences
    await db.insert('sms_preferences', {
      user_id: userId,
      phone_number: verification.phone_number,
      sms_enabled: true,
      irrigation_alerts: true,
      rain_alerts: true,
      moisture_alerts: true,
      canal_alerts: true,
      consent_given_at: new Date()
    });

    // Send confirmation SMS
    await this.sendConfirmationSMS(verification.phone_number);

    return { verified: true };
  }

  async disableSMS(userId) {
    await db.update('sms_preferences', { user_id: userId }, {
      sms_enabled: false,
      disabled_at: new Date()
    });
  }

  generateOTP() {
    return Math.floor(100000 + Math.random() * 900000).toString();  // 6-digit OTP
  }

  async sendOTP(phoneNumber, otp) {
    const message = `Your Zarai Dost verification code is: ${otp}. Valid for 15 minutes.`;
    await new SMSGatewayService().sendSMS(phoneNumber, message, 'otp_verification');
  }

  async sendConfirmationSMS(phoneNumber) {
    const message = `SMS alerts activated! You'll receive irrigation reminders. Reply STOP to unsubscribe.`;
    await new SMSGatewayService().sendSMS(phoneNumber, message, 'confirmation');
  }
}
```

### Data Model Requirements

```sql
CREATE TABLE sms_preferences (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  phone_number TEXT NOT NULL,
  sms_enabled BOOLEAN DEFAULT TRUE,
  irrigation_alerts BOOLEAN DEFAULT TRUE,
  rain_alerts BOOLEAN DEFAULT TRUE,
  moisture_alerts BOOLEAN DEFAULT TRUE,
  canal_alerts BOOLEAN DEFAULT TRUE,
  language_preference TEXT DEFAULT 'en',  -- 'en', 'ur', 'pa', 'sd'
  consent_given_at DATETIME,
  disabled_at DATETIME,
  FOREIGN KEY (user_id) REFERENCES farmers(id)
);

CREATE TABLE sms_delivery_log (
  id TEXT PRIMARY KEY,
  user_id TEXT,
  phone_number TEXT NOT NULL,
  message_content TEXT NOT NULL,
  alert_type TEXT NOT NULL,
  gateway_used TEXT,                    -- 'twilio', 'local', 'aws_sns'
  status TEXT NOT NULL,                 -- 'sent', 'delivered', 'failed', 'expired'
  delivery_id TEXT,
  error_message TEXT,
  sent_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  delivered_at DATETIME
);

CREATE TABLE sms_verifications (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  phone_number TEXT NOT NULL,
  otp_code TEXT NOT NULL,
  verified BOOLEAN DEFAULT FALSE,
  verified_at DATETIME,
  expires_at DATETIME NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_sms_delivery_log_phone_date ON sms_delivery_log(phone_number, sent_at DESC);
CREATE INDEX idx_sms_verifications_user ON sms_verifications(user_id, created_at DESC);
```
