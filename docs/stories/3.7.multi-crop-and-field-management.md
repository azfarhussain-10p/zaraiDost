## Story
**As a** farmer growing multiple crops in different fields,
**I want** to organize my diagnoses by field and crop,
**so that** I can manage each crop's health separately.

```sql
  variety TEXT,  -- e.g., "Punjab 11", "Basmati"
  planting_date DATE NOT NULL,
  expected_harvest_date DATE,
  actual_harvest_date DATE,
  growth_stage TEXT,  -- germination, vegetative, flowering, maturity, harvest

  notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (field_id) REFERENCES fields(id)
);

CREATE INDEX idx_crop_seasons_field ON crop_seasons(field_id);
CREATE INDEX idx_crop_seasons_status ON crop_seasons(status);
```

```javascript
// FieldManagementService.js
class FieldManagementService {
  async createField(farmerId, fieldData) {
    const field = {
      id: `field-${Date.now()}-${Math.random()}`,
      farmer_id: farmerId,
      name: fieldData.name,
      size_acres: fieldData.sizeAcres,
      soil_type: fieldData.soilType,
      location_gps: fieldData.location,  // { latitude, longitude }
      irrigation_type: fieldData.irrigationType,
      created_at: new Date(),
      updated_at: new Date()
    };

    await db.insert('fields', field);
    return field;
  }

  async getFieldsForFarmer(farmerId) {
    const fields = await db.query(`
      SELECT
        f.*,
        COUNT(DISTINCT cs.id) as total_crops,
        COUNT(DISTINCT hc.id) as total_health_checks,
        MAX(fhm.health_score) as latest_health_score
      FROM fields f
      LEFT JOIN crop_seasons cs ON f.id = cs.field_id
      LEFT JOIN health_checks hc ON f.id = hc.field_id
      LEFT JOIN field_health_metrics fhm ON f.id = fhm.field_id
      WHERE f.farmer_id = $1 AND f.active = TRUE
      GROUP BY f.id
      ORDER BY f.created_at DESC
    `, [farmerId]);

    return fields;
  }

  async getActiveCropsForField(fieldId) {
    const activeCrops = await db.query(`
      SELECT * FROM crop_seasons
      WHERE field_id = $1 AND status = 'active'
      ORDER BY planting_date DESC
    `, [fieldId]);

    return activeCrops;
  }

  async getCropRotationHistory(fieldId) {
    const rotations = await db.query(`
      SELECT
        crop_type,
        variety,
        planting_date,
        actual_harvest_date,
        yield_kg_per_acre,
        status
      FROM crop_seasons
      WHERE field_id = $1
      ORDER BY planting_date DESC
    `, [fieldId]);

    return rotations;
  }

  async detectFieldFromLocation(farmerId, location) {
    // Find field closest to captured image location
    const fields = await db.query(`
      SELECT
        id,
        name,
        ST_Distance(
          location_gps,
          ST_MakePoint($2, $3)::geography
        ) as distance_meters
      FROM fields
      WHERE farmer_id = $1 AND active = TRUE
      HAVING distance_meters < 1000  -- Within 1km
      ORDER BY distance_meters ASC
      LIMIT 1
    `, [farmerId, location.longitude, location.latitude]);

    return fields[0] || null;
  }
}
```

**Crop Season Management**

```javascript
// CropSeasonService.js
class CropSeasonService {
  async plantCrop(fieldId, cropData) {
    const cropSeason = {
      id: `crop-season-${Date.now()}-${Math.random()}`,
      field_id: fieldId,
      crop_type: cropData.cropType,
      variety: cropData.variety,
      planting_date: cropData.plantingDate,
      expected_harvest_date: this.calculateHarvestDate(cropData.cropType, cropData.plantingDate),
      growth_stage: 'germination',
      status: 'active',
      created_at: new Date()
    };

    await db.insert('crop_seasons', cropSeason);

    // Update growth stage automatically based on days since planting
    this.scheduleGrowthStageUpdates(cropSeason.id);

    return cropSeason;
  }

  calculateHarvestDate(cropType, plantingDate) {
    // Typical growing periods for Pakistani crops
    const growthPeriods = {
      wheat: 120,  // days
      rice: 150,
      cotton: 180,
      sugarcane: 365,
      corn: 90
    };

    const period = growthPeriods[cropType] || 120;
    const harvestDate = new Date(plantingDate);
    harvestDate.setDate(harvestDate.getDate() + period);

    return harvestDate;
  }

  async updateGrowthStage(cropSeasonId) {
    const cropSeason = await db.query(
      'SELECT * FROM crop_seasons WHERE id = $1',
      [cropSeasonId]
    );

    if (!cropSeason[0]) return;

    const daysSincePlanting = this.getDaysSince(cropSeason[0].planting_date);
    const newStage = this.determineGrowthStage(cropSeason[0].crop_type, daysSincePlanting);

    if (newStage !== cropSeason[0].growth_stage) {
      await db.update('crop_seasons', { id: cropSeasonId }, { growth_stage: newStage });
    }
  }

  determineGrowthStage(cropType, daysSincePlanting) {
    // Simplified growth stage determination
    if (daysSincePlanting < 15) return 'germination';
    if (daysSincePlanting < 60) return 'vegetative';
    if (daysSincePlanting < 90) return 'flowering';
    if (daysSincePlanting < 120) return 'maturity';
    return 'harvest';
  }

  getDaysSince(date) {
    const now = new Date();
    const past = new Date(date);
    return Math.floor((now - past) / (1000 * 60 * 60 * 24));
  }

  async suggestCropRotation(fieldId) {
    const history = await FieldManagementService.getCropRotationHistory(fieldId);

    // Simple rotation rules (expand based on agronomic principles)
    const lastCrop = history[0]?.crop_type;

    const rotationSuggestions = {
      wheat: ['rice', 'cotton', 'vegetables'],
      rice: ['wheat', 'pulses'],
      cotton: ['wheat', 'sugarcane'],
      sugarcane: ['wheat', 'rice'],
      corn: ['wheat', 'vegetables']
    };

    return rotationSuggestions[lastCrop] || ['wheat', 'rice', 'cotton'];
  }
}
```

**Field Health Score Calculation**

```javascript
// FieldHealthScoreCalculator.js
class FieldHealthScoreCalculator {
  async calculateFieldHealthScore(fieldId, cropSeasonId) {
    // Get recent health data
    const recentHealthChecks = await db.query(`
      SELECT * FROM health_checks
      WHERE field_id = $1 AND crop_season_id = $2
        AND created_at >= NOW() - INTERVAL '30 days'
      ORDER BY created_at DESC
    `, [fieldId, cropSeasonId]);

    let score = 100;  // Start with perfect score

    // Deduct points for each health issue
    recentHealthChecks.forEach(hc => {
      const deduction = this.getDeductionForIssue(hc);
      score -= deduction;
    });

    // Ensure score is between 0-100
    score = Math.max(0, Math.min(100, score));

    // Save metric
    await db.insert('field_health_metrics', {
      id: `metric-${Date.now()}`,
      field_id: fieldId,
      crop_season_id: cropSeasonId,
      metric_date: new Date(),
      health_score: score,
      disease_count: recentHealthChecks.length
    });

    return score;
  }

  getDeductionForIssue(healthCheck) {
    // Deduct points based on severity and resolution status
    const severityDeductions = {
      mild: 5,
      moderate: 10,
      severe: 20
    };

    const onDeviceResult = JSON.parse(healthCheck.on_device_result || '{}');
    const severity = onDeviceResult.topPrediction?.severity || 'moderate';

    let deduction = severityDeductions[severity] || 10;

    // Reduce deduction if issue is resolved
    if (healthCheck.resolution_status === 'resolved') {
      deduction = deduction * 0.5;
    }

    return deduction;
  }

  getHealthScoreLabel(score) {
    if (score >= 90) return { label: 'Excellent', color: '#4CAF50' };
    if (score >= 75) return { label: 'Good', color: '#8BC34A' };
    if (score >= 60) return { label: 'Fair', color: '#FFC107' };
    if (score >= 40) return { label: 'Poor', color: '#FF9800' };
    return { label: 'Critical', color: '#F44336' };
  }
}
```

**Cross-Field Bulk Operations**

```javascript
// BulkOperationsService.js
class BulkOperationsService {
  async getAllIssuesByCropType(farmerId, cropType) {
    const issues = await db.query(`
      SELECT
        hc.*,
        f.name as field_name,
        cs.variety,
        d.name_en as disease_name
      FROM health_checks hc
      JOIN crop_seasons cs ON hc.crop_season_id = cs.id
      JOIN fields f ON hc.field_id = f.id
      LEFT JOIN diseases d ON hc.top_disease_id = d.id
      WHERE f.farmer_id = $1 AND cs.crop_type = $2
      ORDER BY hc.created_at DESC
    `, [farmerId, cropType]);

    return issues;
  }

  async compareFieldPerformance(farmerId, cropType) {
    const comparison = await db.query(`
      SELECT
        f.id,
        f.name,
        COUNT(DISTINCT cs.id) as seasons_count,
        AVG(cs.yield_kg_per_acre) as avg_yield,
        COUNT(DISTINCT hc.id) as total_health_issues,
        AVG(fhm.health_score) as avg_health_score
      FROM fields f
      LEFT JOIN crop_seasons cs ON f.id = cs.field_id AND cs.crop_type = $2
      LEFT JOIN health_checks hc ON cs.id = hc.crop_season_id
      LEFT JOIN field_health_metrics fhm ON f.id = fhm.field_id
      WHERE f.farmer_id = $1 AND f.active = TRUE
      GROUP BY f.id
      ORDER BY avg_health_score DESC NULLS LAST
    `, [farmerId, cropType]);

    return comparison;
  }

  async getBestFieldForCrop(farmerId, cropType) {
    const comparison = await this.compareFieldPerformance(farmerId, cropType);

    // Find field with highest avg yield and fewest health issues
    const best = comparison.reduce((best, current) => {
      if (!best) return current;

      const currentScore = (current.avg_yield || 0) * 0.7 + (current.avg_health_score || 50) * 0.3;
      const bestScore = (best.avg_yield || 0) * 0.7 + (best.avg_health_score || 50) * 0.3;

      return currentScore > bestScore ? current : best;
    }, null);

    return best;
  }
}
```
