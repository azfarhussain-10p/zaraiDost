**I want** to connect with verified buyers,
**so that** I can get better prices and avoid middlemen.

```javascript
// BuyerSearchService.js
class BuyerSearchService {
  async searchBuyers(filters, location = null) {
    let query = `
      SELECT
        b.*,
        bv.verification_status,
        AVG(br.rating) as avg_rating,
        COUNT(DISTINCT br.id) as review_count,
        COUNT(DISTINCT bt.id) as transaction_count
      FROM buyers b
      LEFT JOIN buyer_verification bv ON b.id = bv.buyer_id
      LEFT JOIN buyer_reviews br ON b.id = br.buyer_id
      LEFT JOIN buyer_transactions bt ON b.id = bt.buyer_id
      WHERE 1=1
    `;

    const params = [];
    let paramIndex = 1;

    // Filter by crop type
    if (filters.cropType) {
      query += ` AND EXISTS (
        SELECT 1 FROM buyer_crops bc
        WHERE bc.buyer_id = b.id AND bc.crop_type = $${paramIndex}
      )`;
      params.push(filters.cropType);
      paramIndex++;
    }

    // Filter by buyer type
    if (filters.buyerType) {
      query += ` AND b.buyer_type = $${paramIndex}`;
      params.push(filters.buyerType);
      paramIndex++;
    }

    // Filter by payment terms
    if (filters.paymentTerms) {
      query += ` AND b.payment_terms ? $${paramIndex}`; // JSONB contains
      params.push(filters.paymentTerms);
      paramIndex++;
    }

    // Filter by verification status
    if (filters.verifiedOnly) {
      query += ` AND b.is_verified = TRUE`;
    }

    // Filter by location proximity
    if (location) {
      query += ` AND ST_DWithin(
        b.location_gps::geography,
        ST_MakePoint($${paramIndex}, $${paramIndex + 1})::geography,
        $${paramIndex + 2} * 1000
      )`;
      params.push(location.longitude, location.latitude, filters.maxDistance || 200);
      paramIndex += 3;
    }

    query += ` GROUP BY b.id, bv.verification_status`;

    // Sort by rating, then proximity
    if (location) {
      query += `, ST_Distance(
        b.location_gps::geography,
        ST_MakePoint($${params.length - 2}, $${params.length - 1})::geography
      )`;
    }

    query += ` ORDER BY avg_rating DESC NULLS LAST, transaction_count DESC`;
    query += ` LIMIT 50`;

    const buyers = await db.query(query, params);

    return buyers;
  }

  async getBuyerProfile(buyerId) {
    const buyer = await db.queryOne(`
      SELECT
        b.*,
        bv.verification_status,
        AVG(br.rating) as avg_rating,
        COUNT(DISTINCT br.id) as review_count
      FROM buyers b
      LEFT JOIN buyer_verification bv ON b.id = bv.buyer_id
      LEFT JOIN buyer_reviews br ON b.id = br.buyer_id
      WHERE b.id = $1
      GROUP BY b.id, bv.verification_status
    `, [buyerId]);

    // Get crops interested in
    const crops = await db.query(`
      SELECT crop_type FROM buyer_crops WHERE buyer_id = $1
    `, [buyerId]);

    // Get recent reviews
    const reviews = await db.query(`
      SELECT
        br.*,
        f.name as farmer_name
      FROM buyer_reviews br
      LEFT JOIN farmers f ON br.farmer_id = f.id
      WHERE br.buyer_id = $1
      ORDER BY br.created_at DESC
      LIMIT 10
    `, [buyerId]);

    return {
      ...buyer,
      crops_interested: crops.map(c => c.crop_type),
      recent_reviews: reviews
    };
  }
}
```

```javascript
class SellingInquiryService {
  async createInquiry(farmerId, inquiryData) {
    // Create inquiry record
    const inquiry = {
      id: `inq-${Date.now()}`,
      farmer_id: farmerId,
      crop_type: inquiryData.cropType,
      quantity_kg: inquiryData.quantityKg,
      expected_price_per_kg: inquiryData.expectedPrice,
      location: inquiryData.location,
      available_date: inquiryData.availableDate,
      additional_notes: inquiryData.notes,
      status: 'sent',
      expires_at: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000), // 7 days
      created_at: new Date()
    };

    await db.insert('selling_inquiries', inquiry);

    // Send to selected buyers
    for (const buyerId of inquiryData.buyerIds) {
      await this.sendInquiryToBuyer(inquiry.id, buyerId);
    }

    return inquiry;
  }

  async sendInquiryToBuyer(inquiryId, buyerId) {

    await this.notifyBuyer(buyerId, inquiryId);

      status: response.action, // 'interested', 'offer_made', 'declined'

    await this.notifyFarmerOfResponse(inquiry.farmer_id, inquiryId, buyerId);
  }

  async notifyFarmerOfResponse(farmerId, inquiryId, buyerId) {
    // Send push notification to farmer

    await notificationService.sendPushNotification(farmerId, {

      data: {
        type: 'inquiry_response',
        inquiry_id: inquiryId,
        buyer_id: buyerId
      }
    });
  }

    return await db.query(`
      SELECT
        si.*,
        COUNT(DISTINCT ib.buyer_id) as buyers_contacted,
        COUNT(DISTINCT CASE WHEN ib.status = 'interested' THEN ib.buyer_id END) as interested_buyers
      FROM selling_inquiries si
      LEFT JOIN inquiry_buyers ib ON si.id = ib.inquiry_id
      WHERE si.farmer_id = $1
      GROUP BY si.id
      ORDER BY si.created_at DESC
    `, [farmerId]);
  }
```

```javascript
// BuyerReviewService.js
class BuyerReviewService {
  async submitReview(farmerId, buyerId, reviewData) {
    // Check if farmer has transacted with buyer
    const hasTransacted = await this.hasTransactionHistory(farmerId, buyerId);

    if (!hasTransacted) {
      throw new Error('You can only review buyers you have transacted with');
    }

    // Check for duplicate review
    const existingReview = await db.queryOne(`
      SELECT id FROM buyer_reviews
      WHERE farmer_id = $1 AND buyer_id = $2
    `, [farmerId, buyerId]);

    if (existingReview) {
      throw new Error('You have already reviewed this buyer');
    }

    // Create review
    const review = {
      id: `review-${Date.now()}`,
      farmer_id: farmerId,
      buyer_id: buyerId,
      rating: reviewData.rating, // 1-5
      price_fairness: reviewData.priceFairness, // 1-5
      payment_reliability: reviewData.paymentReliability, // 1-5
      professionalism: reviewData.professionalism, // 1-5
      comment: reviewData.comment,
      would_recommend: reviewData.wouldRecommend,
      created_at: new Date()
    };

    await db.insert('buyer_reviews', review);

    // Update buyer's average rating
    await this.updateBuyerRating(buyerId);

    return review;
  }

  async hasTransactionHistory(farmerId, buyerId) {
    const transaction = await db.queryOne(`
      SELECT id FROM buyer_transactions
      WHERE farmer_id = $1 AND buyer_id = $2
    `, [farmerId, buyerId]);

    return !!transaction;
  }

  async updateBuyerRating(buyerId) {
    const avgRating = await db.queryOne(`
      SELECT AVG(rating) as avg_rating
      FROM buyer_reviews
      WHERE buyer_id = $1
    `, [buyerId]);

    await db.update('buyers', { id: buyerId }, {
      average_rating: avgRating.avg_rating
    });
  }

  async getReviewsForBuyer(buyerId, limit = 10) {
    return await db.query(`
      SELECT
        br.*,
        f.name as farmer_name,
        f.location as farmer_location
      FROM buyer_reviews br
      LEFT JOIN farmers f ON br.farmer_id = f.id
      WHERE br.buyer_id = $1
      ORDER BY br.created_at DESC
      LIMIT $2
    `, [buyerId, limit]);
  }

  async flagReviewForModeration(reviewId, reason) {
    await db.update('buyer_reviews', { id: reviewId }, {
      flagged: true,
      flag_reason: reason,
      flagged_at: new Date()
    });

    // Notify moderator
    console.log(`Review ${reviewId} flagged for moderation: ${reason}`);
  }
}
```

```javascript
// MessagingService.js
const socketIo = require('socket.io');

class MessagingService {
  constructor(server) {
    this.io = socketIo(server);
    this.setupSocketHandlers();
  }

  setupSocketHandlers() {
    this.io.on('connection', (socket) => {
      console.log('User connected:', socket.id);

      // Join conversation room
      socket.on('join_conversation', async ({ farmerId, buyerId }) => {
        const roomId = this.getConversationRoom(farmerId, buyerId);
        socket.join(roomId);
        console.log(`User joined room: ${roomId}`);
      });

      // Send message
      socket.on('send_message', async (data) => {
        const message = await this.saveMessage(data);
        const roomId = this.getConversationRoom(data.farmerId, data.buyerId);

        // Broadcast to room
        this.io.to(roomId).emit('new_message', message);

        // Send push notification if recipient offline
        await this.sendMessageNotification(data);
      });

      socket.on('disconnect', () => {
        console.log('User disconnected:', socket.id);
      });
    });
  }

  getConversationRoom(farmerId, buyerId) {
    // Create unique room ID for farmer-buyer pair
    return `conv_${[farmerId, buyerId].sort().join('_')}`;
  }

  async saveMessage(data) {
    const message = {
      id: `msg-${Date.now()}`,
      farmer_id: data.farmerId,
      buyer_id: data.buyerId,
      sender_type: data.senderType, // 'farmer' or 'buyer'
      message_text: data.text,
      created_at: new Date()
    };

    await db.insert('messages', message);
    return message;
  }

  async sendMessageNotification(data) {
    // Check if recipient is online
    const recipientId = data.senderType === 'farmer' ? data.buyerId : data.farmerId;
    const recipientType = data.senderType === 'farmer' ? 'buyer' : 'farmer';

    // Send push notification
    if (recipientType === 'farmer') {
      await notificationService.sendPushNotification(recipientId, {
        title: 'New Message',
        body: `Message from buyer: ${data.text.substring(0, 50)}...`,
        data: {
          type: 'new_message',
          buyer_id: data.buyerId
        }
      });
    }
  }

  async getConversationHistory(farmerId, buyerId, limit = 50) {
    return await db.query(`
      SELECT * FROM messages
      WHERE farmer_id = $1 AND buyer_id = $2
      ORDER BY created_at DESC
      LIMIT $3
    `, [farmerId, buyerId, limit]);
  }
}
```

```sql
CREATE TABLE buyers (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  buyer_type TEXT NOT NULL,  -- 'trader', 'wholesaler', 'exporter', 'processor'
  phone_number TEXT NOT NULL,
  email TEXT,
  location_gps GEOGRAPHY(POINT),
  district TEXT,
  province TEXT,
  payment_terms JSONB,  -- {cash: true, credit: true, advance: false, credit_days: 30}
  business_description TEXT,
  is_verified BOOLEAN DEFAULT FALSE,
  verified_at TIMESTAMP,
  average_rating DECIMAL(3, 2),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_buyers_location ON buyers USING GIST(location_gps);
CREATE INDEX idx_buyers_verified ON buyers(is_verified);

CREATE TABLE buyer_crops (

  buyer_id TEXT NOT NULL,
  crop_type TEXT NOT NULL,
  FOREIGN KEY (buyer_id) REFERENCES buyers(id),
  UNIQUE(buyer_id, crop_type)
);

CREATE INDEX idx_buyer_crops_buyer ON buyer_crops(buyer_id);
CREATE INDEX idx_buyer_crops_crop ON buyer_crops(crop_type);

  verified_by TEXT,

  FOREIGN KEY (buyer_id) REFERENCES buyers(id)

  response_notes TEXT,
  sent_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  responded_at TIMESTAMP,
  FOREIGN KEY (inquiry_id) REFERENCES selling_inquiries(id),
  FOREIGN KEY (buyer_id) REFERENCES buyers(id)

-- Buyer reviews

  buyer_id TEXT NOT NULL,

  flagged BOOLEAN DEFAULT FALSE,
  flag_reason TEXT,
  flagged_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (buyer_id) REFERENCES buyers(id),
  UNIQUE(farmer_id, buyer_id)
);

CREATE INDEX idx_buyer_reviews_buyer ON buyer_reviews(buyer_id, created_at DESC);

  farmer_id TEXT NOT NULL,
  buyer_id TEXT NOT NULL,
  sender_type TEXT NOT NULL,  -- 'farmer' or 'buyer'
  message_text TEXT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_messages_conversation ON messages(farmer_id, buyer_id, created_at DESC);
```
