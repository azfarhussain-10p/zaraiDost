## Story
**As a** farmer with harvested crops,
**I want** recommendations on the best time to sell,
**so that** I can maximize my profit.

```javascript
// SellingTimeOptimizer.js
class SellingTimeOptimizer {
  constructor() {
    this.pricePredictionService = new PricePredictionService();
    this.degradationService = new DegradationService();
    this.storageCostService = new StorageCostService();
  }

  async generateRecommendation(userId, cropSeasonId) {
    // Get crop details
    const cropSeason = await this.getCropSeason(cropSeasonId);
    const userProfile = await this.getUserProfile(userId);

    // Get price predictions
    const predictions = await this.pricePredictionService.getPredictions(
      cropSeason.crop_type,
      cropSeason.nearest_mandi_id
    );

    // Get current price
    const currentPrice = await this.getCurrentPrice(
      cropSeason.crop_type,
      cropSeason.nearest_mandi_id
    );

    // Calculate net profit for each day in next 90 days
    const profitTimeline = await this.calculateProfitTimeline(
      cropSeason,
      userProfile,
      currentPrice,
      predictions
    );

    // Find optimal selling date
    const optimal = this.findOptimalDate(profitTimeline);

    // Generate recommendation
    const recommendation = this.generateRecommendationText(
      optimal,
      cropSeason,
      currentPrice
    );

    // Calculate risk
    const risk = this.assessRisk(optimal, predictions, cropSeason);

    // Store recommendation
    await this.storeRecommendation(userId, cropSeasonId, recommendation, optimal);

    return {
      recommendation: recommendation.action,
      message: recommendation.message,
      optimal_date: optimal.date,
      days_to_wait: optimal.days,
      profit_difference: optimal.profit_gain,
      current_price: currentPrice,
      expected_price: optimal.expected_price,
      risk_level: risk.level,
      risk_factors: risk.factors,
      profit_timeline: profitTimeline
    };
  }

  async calculateProfitTimeline(cropSeason, userProfile, currentPrice, predictions) {
    const timeline = [];
    const quantity = cropSeason.quantity_kg || 1000; // Default 1000 kg

    for (let day = 0; day <= 90; day++) {
      const date = new Date();
      date.setDate(date.getDate() + day);

      // Get predicted price for this day
      const predictedPrice = this.interpolatePrediction(day, predictions, currentPrice);

      // Calculate degradation loss
      const degradation = await this.degradationService.calculateDegradation(
        cropSeason.crop_type,
        userProfile.storage_type,
        day
      );

      // Calculate storage cost
      const storageCost = await this.storageCostService.calculateCost(
        userProfile.storage_type,
        day,
        quantity
      );

      // Calculate transport cost (one-time)
      const transportCost = await this.calculateTransportCost(
        userProfile.location,
        cropSeason.nearest_mandi_id,
        quantity
      );

      // Calculate net profit
      const effectiveQuantity = quantity * (1 - degradation.loss_percent / 100);
      const revenue = effectiveQuantity * predictedPrice;
      const netProfit = revenue - storageCost - transportCost;

      timeline.push({
        day,
        date: date.toISOString().split('T')[0],
        predicted_price: predictedPrice,
        degradation_percent: degradation.loss_percent,
        effective_quantity_kg: effectiveQuantity,
        storage_cost: storageCost,
        transport_cost: transportCost,
        gross_revenue: revenue,
        net_profit: netProfit
      });
    }

    return timeline;
  }

  interpolatePrediction(day, predictions, currentPrice) {
    // Interpolate price between prediction points
    if (day === 0) return currentPrice;

    // Find surrounding predictions
    const pred30 = predictions.find(p => p.days_ahead === 30);
    const pred60 = predictions.find(p => p.days_ahead === 60);
    const pred90 = predictions.find(p => p.days_ahead === 90);

    if (day <= 30) {
      // Interpolate between current and 30-day
      const ratio = day / 30;
      return currentPrice + (pred30.predicted_price - currentPrice) * ratio;
    } else if (day <= 60) {
      // Interpolate between 30-day and 60-day
      const ratio = (day - 30) / 30;
      return pred30.predicted_price + (pred60.predicted_price - pred30.predicted_price) * ratio;
    } else {
      // Interpolate between 60-day and 90-day
      const ratio = (day - 60) / 30;
      return pred60.predicted_price + (pred90.predicted_price - pred60.predicted_price) * ratio;
    }
  }

  findOptimalDate(profitTimeline) {
    // Find date with maximum net profit
    const optimal = profitTimeline.reduce((best, current) => {
      return current.net_profit > best.net_profit ? current : best;
    }, profitTimeline[0]);

    const sellNowProfit = profitTimeline[0].net_profit;
    const profitGain = optimal.net_profit - sellNowProfit;
    const profitGainPercent = (profitGain / sellNowProfit) * 100;

    return {
      date: optimal.date,
      days: optimal.day,
      expected_price: optimal.predicted_price,
      net_profit: optimal.net_profit,
      profit_gain: profitGain,
      profit_gain_percent: profitGainPercent,
      degradation_percent: optimal.degradation_percent,
      storage_cost: optimal.storage_cost
    };
  }

  generateRecommendationText(optimal, cropSeason, currentPrice) {
    const profitThreshold = 5; // 5% minimum profit gain to recommend waiting
    const urgencyThreshold = 10; // 10% degradation = urgent

    let action, message, urgency;

    if (optimal.days === 0) {
      action = 'sell_now';
      message = `Sell now - prices are at peak or will decline`;
      urgency = 'normal';
    } else if (optimal.degradation_percent > urgencyThreshold) {
      action = 'sell_urgently';
      message = `Sell urgently - crop degradation (${Math.round(optimal.degradation_percent)}%) outweighs price gains`;
      urgency = 'high';
    } else if (optimal.profit_gain_percent < profitThreshold) {
      action = 'sell_now';
      message = `Sell now - minimal profit gain (${Math.round(optimal.profit_gain_percent)}%) from waiting`;
      urgency = 'normal';
    } else {
      action = 'wait';
      message = `Wait ${optimal.days} days - expected ${Math.round(optimal.profit_gain_percent)}% more profit (PKR ${Math.round(optimal.profit_gain)})`;
      urgency = 'low';
    }

    return { action, message, urgency };
  }

  assessRisk(optimal, predictions, cropSeason) {
    // Calculate risk based on prediction confidence and crop perishability
    const prediction = predictions.find(p => p.days_ahead === 30);
    const confidence = prediction ? prediction.confidence.score : 50;

    // Perishability risk
    const perishability = this.getCropPerishability(cropSeason.crop_type);

    // Combined risk score (0-100, higher = more risky)
    const riskScore = ((100 - confidence) * 0.6) + (perishability * 0.4);

    let level, factors;

    if (riskScore < 30) {
      level = 'low';
      factors = ['High price prediction confidence', 'Low crop perishability'];
    } else if (riskScore < 60) {
      level = 'medium';
      factors = ['Moderate price uncertainty', 'Some degradation risk'];
    } else {
      level = 'high';
      factors = ['Low price prediction confidence', 'High crop perishability'];
    }

    return {
      level,
      score: Math.round(riskScore),
      factors,
      best_case: optimal.net_profit,
      worst_case: optimal.net_profit * 0.8 // Assume 20% downside
    };
  }

  getCropPerishability(cropType) {
    // Perishability score (0-100, higher = more perishable)
    const scores = {
      'vegetables': 90,
      'fruits': 85,
      'wheat': 10,
      'rice': 15,
      'cotton': 20,
      'sugarcane': 50
    };
    return scores[cropType] || 50;
  }

  async calculateTransportCost(userLocation, mandiId, quantityKg) {
    // Get mandi location
    const mandi = await db.queryOne('SELECT location_gps FROM mandis WHERE id = $1', [mandiId]);

    // Calculate distance
    const distance = this.calculateDistance(userLocation, mandi.location_gps);

    // Transport cost formula (PKR)
    const baseRate = 500; // Base fee
    const perKmRate = 20; // Per km
    const perKgRate = 0.5; // Per kg

    return baseRate + (distance * perKmRate) + (quantityKg * perKgRate);
  }

  calculateDistance(from, to) {
    // Haversine formula (simplified)
    const R = 6371; // Earth radius in km
    const dLat = this.toRad(to.latitude - from.latitude);
    const dLon = this.toRad(to.longitude - from.longitude);

    const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
              Math.cos(this.toRad(from.latitude)) * Math.cos(this.toRad(to.latitude)) *
              Math.sin(dLon / 2) * Math.sin(dLon / 2);

    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    return R * c;
  }

  toRad(degrees) {
    return degrees * (Math.PI / 180);
  }

  async storeRecommendation(userId, cropSeasonId, recommendation, optimal) {
    await db.insert('selling_recommendations', {
      id: `rec-${Date.now()}`,
      user_id: userId,
      crop_season_id: cropSeasonId,
      recommendation_action: recommendation.action,
      recommendation_message: recommendation.message,
      optimal_date: optimal.date,
      days_to_wait: optimal.days,
      expected_profit_gain: optimal.profit_gain,
      risk_level: optimal.risk_level,
      created_at: new Date()
    });
  }
}
```

```javascript
// SellingWindowAlertService.js
const cron = require('node-cron');

class SellingWindowAlertService {
  constructor() {
    this.optimizer = new SellingTimeOptimizer();
    this.notificationService = new NotificationService();
    this.smsService = new SMSGatewayService();
  }

  start() {
    // Check daily at 7 AM for selling window opportunities
    cron.schedule('0 7 * * *', async () => {
      console.log('Checking selling windows...');
      await this.checkSellingWindows();
    });
  }

  async checkSellingWindows() {
    // Get all active crop seasons with recommendations
    const recommendations = await db.query(`
      SELECT
        sr.*,
        cs.crop_type,
        cs.quantity_kg,
        f.name as farmer_name,
        f.phone_number,
        f.language_preference
      FROM selling_recommendations sr
      JOIN crop_seasons cs ON sr.crop_season_id = cs.id
      JOIN farmers f ON sr.user_id = f.id
      WHERE sr.optimal_date >= CURRENT_DATE
        AND sr.alert_sent = FALSE
        AND sr.optimal_date <= CURRENT_DATE + INTERVAL '7 days'
    `);

    for (const rec of recommendations) {
      const daysUntilOptimal = this.calculateDaysUntil(rec.optimal_date);

      if (daysUntilOptimal <= 2) {
        // Optimal window approaching (within 2 days)
        await this.sendSellingWindowAlert(rec);

        // Mark alert as sent
        await db.update('selling_recommendations', { id: rec.id }, {
          alert_sent: true,
          alert_sent_at: new Date()
        });
      }
    }
  }

  async sendSellingWindowAlert(recommendation) {
    const { user_id, crop_type, optimal_date, expected_profit_gain, farmer_name, phone_number, language_preference } = recommendation;

    const daysUntil = this.calculateDaysUntil(optimal_date);

    // Send push notification
    const pushMessage = {
      title: 'Optimal Selling Time Approaching!',
      body: `Your ${crop_type} should be sold in ${daysUntil} days for maximum profit (PKR ${Math.round(expected_profit_gain)} extra).`,
      data: {
        type: 'selling_window',
        crop_season_id: recommendation.crop_season_id
      }
    };

    await this.notificationService.sendPushNotification(user_id, pushMessage);

    // Send SMS if enabled
    const smsPreferences = await this.getSMSPreferences(user_id);
    if (smsPreferences && smsPreferences.market_alerts) {
      const smsMessage = this.generateSMSMessage(recommendation, daysUntil, language_preference);
      await this.smsService.sendSMS(phone_number, smsMessage, 'selling_window');
    }

    console.log(`Sent selling window alert to ${farmer_name} for ${crop_type}`);
  }

  generateSMSMessage(recommendation, daysUntil, language) {
    const templates = {
      en: `ZARAI DOST: Sell your ${recommendation.crop_type} in ${daysUntil} days for PKR ${Math.round(recommendation.expected_profit_gain)} more profit. Check app for details.`,
      ur: `ZARAI DOST: Apni ${recommendation.crop_type} ${daysUntil} din mein bechein, PKR ${Math.round(recommendation.expected_profit_gain)} ziada munafa. App check karein.`,
      pa: `ZARAI DOST: Apni ${recommendation.crop_type} ${daysUntil} din vich vecho, PKR ${Math.round(recommendation.expected_profit_gain)} vaddha munafa. App vekho.`
    };

    return templates[language] || templates.en;
  }

  calculateDaysUntil(targetDate) {
    const today = new Date();
    const target = new Date(targetDate);
    const diffTime = target - today;
    return Math.ceil(diffTime / (1000 * 60 * 60 * 24));
  }

  async getSMSPreferences(userId) {
    return await db.queryOne(`
      SELECT * FROM sms_preferences
      WHERE user_id = $1 AND sms_enabled = TRUE
    `, [userId]);
  }
}
```

```javascript
// ActualSaleFeedbackService.js
class ActualSaleFeedbackService {
  async recordSale(userId, cropSeasonId, saleData) {
    // Store actual sale information
    const saleRecord = {
      id: `sale-${Date.now()}`,
      user_id: userId,
      crop_season_id: cropSeasonId,
      sale_date: saleData.saleDate,
      quantity_sold_kg: saleData.quantityKg,
      actual_price_per_kg: saleData.pricePerKg,
      mandi_id: saleData.mandiId,
      total_revenue: saleData.quantityKg * saleData.pricePerKg,
      transport_cost: saleData.transportCost || 0,
      storage_cost_paid: saleData.storageCost || 0,
      net_profit: (saleData.quantityKg * saleData.pricePerKg) - (saleData.transportCost || 0) - (saleData.storageCost || 0),
      created_at: new Date()
    };

    await db.insert('selling_history', saleRecord);

    // Compare with recommendation
    await this.evaluateRecommendation(cropSeasonId, saleRecord);

    // Update crop season status
    await db.update('crop_seasons', { id: cropSeasonId }, {
      status: 'harvested',
      actual_harvest_date: saleData.saleDate
    });

    return saleRecord;
  }

  async evaluateRecommendation(cropSeasonId, actualSale) {
    // Get the recommendation for this crop
    const recommendation = await db.queryOne(`
      SELECT * FROM selling_recommendations
      WHERE crop_season_id = $1
      ORDER BY created_at DESC
      LIMIT 1
    `, [cropSeasonId]);

    if (!recommendation) return;

    // Calculate if user followed recommendation
    const daysDiff = this.calculateDaysBetween(actualSale.sale_date, recommendation.optimal_date);
    const followedRecommendation = Math.abs(daysDiff) <= 7; // Within 1 week = followed

    // Get predicted price for sale date
    const predictedPrice = await this.getPredictedPriceForDate(
      cropSeasonId,
      actualSale.sale_date
    );

    // Calculate accuracy
    const priceAccuracy = predictedPrice
      ? 100 - Math.abs((actualSale.actual_price_per_kg - predictedPrice) / predictedPrice * 100)
      : null;

    // Update recommendation with actual results
    await db.update('selling_recommendations', { id: recommendation.id }, {
      actual_sale_date: actualSale.sale_date,
      actual_price_per_kg: actualSale.actual_price_per_kg,
      actual_net_profit: actualSale.net_profit,
      followed_recommendation: followedRecommendation,
      price_prediction_accuracy: priceAccuracy,
      evaluation_completed_at: new Date()
    });

    // Calculate profit difference
    const profitDifference = followedRecommendation
      ? actualSale.net_profit - (actualSale.net_profit / (1 + recommendation.expected_profit_gain / 100))
      : null;

    return {
      followed_recommendation: followedRecommendation,
      price_accuracy: priceAccuracy,
      profit_difference: profitDifference
    };
  }

  calculateDaysBetween(date1, date2) {
    const d1 = new Date(date1);
    const d2 = new Date(date2);
    const diffTime = d2 - d1;
    return Math.round(diffTime / (1000 * 60 * 60 * 24));
  }

  async getPredictedPriceForDate(cropSeasonId, date) {
    // Query predicted price for this date (if exists)
    const prediction = await db.queryOne(`
      SELECT predicted_price
      FROM price_predictions pp
      JOIN crop_seasons cs ON pp.crop_type = cs.crop_type
      WHERE cs.id = $1
        AND pp.target_date = $2
      ORDER BY pp.prediction_date DESC
      LIMIT 1
    `, [cropSeasonId, date]);

    return prediction ? prediction.predicted_price : null;
  }

  async getRecommendationAccuracyReport(cropType = null) {
    // Generate report on recommendation accuracy
    const query = `
      SELECT
        COUNT(*) as total_recommendations,
        SUM(CASE WHEN followed_recommendation THEN 1 ELSE 0 END) as followed_count,
        AVG(price_prediction_accuracy) as avg_price_accuracy,
        AVG(actual_net_profit - expected_profit_gain) as avg_profit_difference
      FROM selling_recommendations sr
      JOIN crop_seasons cs ON sr.crop_season_id = cs.id
      WHERE sr.evaluation_completed_at IS NOT NULL
        ${cropType ? 'AND cs.crop_type = $1' : ''}
    `;

    const params = cropType ? [cropType] : [];
    const report = await db.queryOne(query, params);

    return {
      total_recommendations: report.total_recommendations,
      followed_percent: (report.followed_count / report.total_recommendations) * 100,
      avg_price_accuracy: Math.round(report.avg_price_accuracy),
      avg_profit_difference: Math.round(report.avg_profit_difference)
    };
  }
}
```

```sql
-- Selling recommendations
CREATE TABLE selling_recommendations (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  crop_season_id TEXT NOT NULL,
  recommendation_action TEXT NOT NULL, -- 'sell_now', 'wait', 'sell_urgently'
  recommendation_message TEXT,
  optimal_date DATE,
  days_to_wait INTEGER,
  expected_profit_gain DECIMAL(10, 2),
  risk_level TEXT, -- 'low', 'medium', 'high'
  alert_sent BOOLEAN DEFAULT FALSE,
  alert_sent_at TIMESTAMP,
  actual_sale_date DATE,
  actual_price_per_kg DECIMAL(10, 2),
  actual_net_profit DECIMAL(10, 2),
  followed_recommendation BOOLEAN,
  price_prediction_accuracy DECIMAL(5, 2),
  evaluation_completed_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (crop_season_id) REFERENCES crop_seasons(id)
);

CREATE INDEX idx_selling_recommendations_optimal_date ON selling_recommendations(optimal_date, alert_sent);
CREATE INDEX idx_selling_recommendations_user ON selling_recommendations(user_id, created_at DESC);

-- Selling history
CREATE TABLE selling_history (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  crop_season_id TEXT NOT NULL,
  sale_date DATE NOT NULL,
  quantity_sold_kg DECIMAL(10, 2) NOT NULL,
  actual_price_per_kg DECIMAL(10, 2) NOT NULL,
  mandi_id TEXT,
  total_revenue DECIMAL(10, 2),
  transport_cost DECIMAL(10, 2),
  storage_cost_paid DECIMAL(10, 2),
  net_profit DECIMAL(10, 2),
  notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (crop_season_id) REFERENCES crop_seasons(id),
  FOREIGN KEY (mandi_id) REFERENCES mandis(id)
);

CREATE INDEX idx_selling_history_user ON selling_history(user_id, sale_date DESC);
CREATE INDEX idx_selling_history_crop ON selling_history(crop_season_id);
```
